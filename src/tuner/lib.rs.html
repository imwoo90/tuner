34pub fn prompt_input(query: &str) -> Result<String, String> {
35    print!("{}", query);
36    std::io::stdout().flush().map_err(|e| e.to_string())?;
37    let mut input = String::new();
38    std::io::stdin().read_line(&mut input).map_err(|e| e.to_string())?;
39    Ok(input.trim().to_string())
40}

42pub fn run_setup_wizard(paths: &workspace::paths::DuctorPaths) -> Result<(), String> {
43    println!("🤖 [tuner] Setup Wizard");
44    workspace::init::init_workspace(paths)?;
45    
46    let token = loop {
47        let t = prompt_input("Enter your Telegram Bot Token: ")?;
48        if !t.is_empty() && t != "YOUR_BOT_TOKEN_HERE" {
49            break t;
50        }
51        println!("❌ Token cannot be empty. Please try again.");
52    };
53
54    let content = std::fs::read_to_string(&paths.config_path()).map_err(|e| e.to_string())?;
55    let mut config_val: serde_json::Value = serde_json::from_str(&content).map_err(|e| e.to_string())?;
56    
57    if let Some(obj) = config_val.as_object_mut() {
58        obj.insert("telegram_token".to_string(), serde_json::Value::String(token));
59        obj.insert("allowed_user_ids".to_string(), serde_json::Value::Array(Vec::new()));
60    }
61
62    let config_content = serde_json::to_string_pretty(&config_val).map_err(|e| e.to_string())?;
63    std::fs::write(&paths.config_path(), config_content).map_err(|e| e.to_string())?;
64    println!("✅ Configuration saved successfully to {:?}", paths.config_path());
65
66    let config = config::CliConfig::load_from_file(&paths.config_path())?;
67
68    let install_sys = prompt_input("Do you want to install tuner as a systemd user service? (y/n): ")?;
69    if install_sys.to_lowercase().starts_with('y') {
70        setup::install_systemd_service(&config)?;
71    }
72    
73    println!("🎉 Setup complete! Please start the tuner bot and send a message (e.g. /start) to register as owner.");
74    println!("📌 Note: If you invite the bot to a group chat/supergroup, you MUST promote it to Administrator (specifically enabling message reading) to bypass Telegram's default Privacy Mode.");
75    Ok(())
76}

78pub async fn spawn_worker(current_exe: &std::path::Path, name: &str) -> Result<tokio::process::Child, String> {
79    tokio::process::Command::new(current_exe)
80        .arg("--worker")
81        .arg(name)
82        .env_remove("TELEGRAM_TOKEN")
83        .kill_on_drop(true)
84        .spawn()
85        .map_err(|e| format!("Failed to spawn worker '{}': {}", name, e))
86}

88pub async fn check_restart_marker(marker: &std::path::Path, workers: &mut [(String, tokio::process::Child)]) {
89    if marker.exists() {
90        let _ = std::fs::remove_file(marker);
91        println!("🤖 [tuner] Master detected restart request. Terminating all workers...");
92        for (name, child) in workers.iter_mut() {
93            println!("🤖 [tuner] Killing worker: {}", name);
94            let _ = child.kill().await;
95        }
96        std::process::exit(42);
97    }
98}

100pub async fn monitor_workers(workers: &mut [(String, tokio::process::Child)], current_exe: &std::path::Path) -> Result<(), String> {
101    let mut exit_requested = None;
102    for (name, child) in workers.iter_mut() {
103        match child.try_wait() {
104            Ok(Some(status)) => {
105                println!("🤖 [tuner] Worker '{}' exited with status: {}", name, status);
106                if let Some(code) = status.code() {
107                    if code == 42 {
108                        exit_requested = Some(name.clone());
109                        break;
110                    }
111                }
112                println!("🤖 [tuner] Restarting worker: {}", name);
113                *child = spawn_worker(current_exe, name).await?;
114            }
115            Ok(None) => {}
116            Err(e) => {
117                eprintln!("🤖 [tuner] Error checking status of worker '{}': {}", name, e);
118            }
119        }
120    }
121
122    if let Some(ref requester) = exit_requested {
123        println!("🤖 [tuner] Worker '{}' requested restart. Exiting master...", requester);
124        for (name, child) in workers.iter_mut() {
125            if name != requester {
126                let _ = child.kill().await;
127            }
128        }
129        std::process::exit(42);
130    }
131    Ok(())
132}

134pub async fn run_master_mode(config: config::CliConfig) -> Result<(), String> {
135    println!("🤖 [tuner] Starting in Master Mode (Supervisor)...");
136    let current_exe = std::env::current_exe()
137        .map_err(|e| format!("Failed to resolve current binary path: {}", e))?;
138
139    let mut workers = Vec::new();
140    for profile in &config.profiles {
141        println!("🤖 [tuner] Spawning worker for profile: {}", profile.name);
142        let child = spawn_worker(&current_exe, &profile.name).await?;
143        workers.push((profile.name.clone(), child));
144    }
145
146    let home = std::env::var("HOME").unwrap_or_else(|_| ".".to_string());
147    let marker = std::path::PathBuf::from(home).join(".tuner/restart-requested");
148    let mut interval = tokio::time::interval(tokio::time::Duration::from_secs(1));
149
150    loop {
151        interval.tick().await;
152        check_restart_marker(&marker, &mut workers).await;
153        monitor_workers(&mut workers, &current_exe).await?;
154    }
155}

157pub async fn handle_early_args(args: &[String]) -> Result<Option<()>, String> {
158    if args.contains(&"--version".to_string()) || args.contains(&"-V".to_string()) {
159        println!("tuner {}", env!("CARGO_PKG_VERSION"));
160        return Ok(Some(()));
161    }
162    if args.contains(&"--upgrade".to_string()) {
163        upgrade::run_cli_upgrade().await?;
164        return Ok(Some(()));
165    }
166    Ok(None)
167}

169pub async fn handle_daemon_mode(args: &[String], config: &config::CliConfig) -> Result<Option<()>, String> {
170    if args.contains(&"--install-systemd".to_string()) {
171        setup::install_systemd_service(config)?;
172        return Ok(Some(()));
173    }
174    if args.contains(&"--supervisor".to_string()) {
175        let current_exe = std::env::current_exe()
176            .map_err(|e| format!("Failed to resolve current binary path: {}", e))?;
177        let filtered_args: Vec<String> = args.iter().cloned().skip(1).filter(|arg| arg != "--supervisor").collect();
178        let supervisor = supervisor::Supervisor::with_args(current_exe, filtered_args);
179        supervisor.run().await?;
180        return Ok(Some(()));
181    }
182    Ok(None)
183}

185pub async fn run() -> Result<(), String> {
186    let args: Vec<String> = std::env::args().collect();
187
188    if let Some(()) = handle_early_args(&args).await? {
189        return Ok(());
190    }
191
192    let worker_profile = args.iter().position(|arg| arg == "--worker")
193        .and_then(|pos| args.get(pos + 1).cloned());
194
195    let tuner_home = std::path::PathBuf::from(std::env::var("HOME").unwrap_or_else(|_| ".".to_string())).join(".tuner");
196    let paths = workspace::paths::resolve_paths(Some(tuner_home), None, None, worker_profile.clone());
197
198    if args.contains(&"--setup".to_string()) {
199        return run_setup_wizard(&paths);
200    }
201
202    let _ = setup::load_env_file(&paths.env_file());
203    workspace::init::init_workspace(&paths)?;
204
205    let mut config = config::CliConfig::load_from_file(&paths.config_path())?;
206    config.working_dir = paths.workspace().clone();
207
208    if let Some(ref profile_name) = worker_profile {
209        setup::override_profile_config(&mut config, profile_name, &paths)?;
210        let profile_config_path = paths.profile_home().join("config").join("config.json");
211        if profile_config_path.is_file() {
212            config.merge_profile_file(&profile_config_path)?;
213        }
214    }
215
216    let startup_lang = config.language.as_deref().unwrap_or("en");
217    i18n::init(startup_lang);
218
219    if let Some(()) = handle_daemon_mode(&args, &config).await? {
220        return Ok(());
221    }
222
223    if let Some(ref p) = worker_profile {
224        println!("🤖 [tuner] Starting Telegram bot worker for '{}'...", p);
225        telegram::run_bot(config, paths).await
226    } else {
227        run_master_mode(config).await
228    }
