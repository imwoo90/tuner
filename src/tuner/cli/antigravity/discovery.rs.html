1//! # Antigravity Model Discovery
2//!
3//! This module implements dynamic discovery of models from the `agy models` CLI command.
4//! It executes the CLI with a timeout and parses the stdout stream.
5
6//! 
7//! ## Search Tags
8//! #discovery

10use std::process::Stdio;
11use std::time::Duration;
12use tokio::process::Command;
13use tokio::time::timeout;
14
15/// Parse `agy models` stdout into a vector of model display names.
16pub fn parse_models(output: &str) -> Vec<String> {
17    let mut models = Vec::new();
18    for line in output.lines() {
19        let name = line.trim();
20        if name.is_empty() {
21            continue;
22        }
23        // A usage/help banner means the command was rejected - treat as failure.
24        if name.starts_with("Usage:") || name.starts_with("Flags:") || name.starts_with("Available subcommands:") {
25            return Vec::new();
26        }
27        models.push(name.to_string());
28    }
29    models
30}
31
32/// Execute the `agy models` command and return discovered model display names.
33///
34/// Returns an empty vector on timeout, command missing, or error.
35pub async fn discover_models(cmd_name: &str) -> Vec<String> {
36    let mut child = match Command::new(cmd_name)
37        .arg("models")
38        .env_remove("CODEX_SANDBOX_NETWORK_DISABLED")
39        .stdout(Stdio::piped())
40        .stderr(Stdio::piped())
41        .spawn()
42    {
43        Ok(c) => c,
44        Err(_) => return Vec::new(),
45    };
46
47    // wait() takes &mut self, preserving ownership of child
48    let wait_res = timeout(Duration::from_secs(15), child.wait()).await;
49    match wait_res {
50        Ok(Ok(status)) if status.success() => {
51            let mut output = String::new();
52            if let Some(mut stdout) = child.stdout.take() {
53                use tokio::io::AsyncReadExt;
54                let _ = stdout.read_to_string(&mut output).await;
55            }
56            parse_models(&output)
57        }
58        _ => {
59            let _ = child.kill().await;
60            Vec::new()
61        }
62    }
63}

65#[cfg(test)]
66mod tests {
67    use super::*;
68
69    const SAMPLE_OUTPUT: &str = "gemini-3.6-flash-high\ngemini-3.1-pro-low\nclaude-sonnet-4-6\n";
70
71    #[test]
72    fn test_parse_models_keeps_display_names() {
73        let parsed = parse_models(SAMPLE_OUTPUT);
74        assert_eq!(parsed.len(), 3);
75        assert_eq!(parsed[0], "gemini-3.6-flash-high");
76        assert_eq!(parsed[1], "gemini-3.1-pro-low");
77        assert_eq!(parsed[2], "claude-sonnet-4-6");
78    }
79
80    #[test]
81    fn test_parse_models_skips_blank_lines() {
82        let parsed = parse_models("\ngemini-3.6-flash-high\n\n");
83        assert_eq!(parsed.len(), 1);
84        assert_eq!(parsed[0], "gemini-3.6-flash-high");
85    }
86
87    #[test]
88    fn test_parse_models_rejects_usage_banner() {
89        let parsed = parse_models("Usage: agy models [flags]\n\nList available models");
90        assert!(parsed.is_empty());
91    }
92
93    #[tokio::test]
94    async fn test_discover_models_returns_empty_when_agy_missing() {
95        let models = discover_models("nonexistent_agy_command_xyz").await;
96        assert!(models.is_empty());
97    }
98}
