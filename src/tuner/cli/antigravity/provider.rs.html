17    pub fn build_env(&self) -> HashMap<String, String> {
18        let mut env: HashMap<_, _> = std::env::vars().filter(|(k, _)| k != "CODEX_SANDBOX_NETWORK_DISABLED").collect();
19        env.insert("TUNER_AGENT_NAME".into(), "main".into());
20        env.insert("TUNER_CHAT_ID".into(), self.config.chat_id.to_string());
21        if let Some(tid) = self.config.topic_id { env.insert("TUNER_TOPIC_ID".into(), tid.to_string()); }
22        env.insert("TUNER_TRANSPORT".into(), self.config.transport.clone());
23        let home = std::env::var("HOME").unwrap_or_else(|_| ".".into());
24        let tuner_home = PathBuf::from(home).join(".tuner");
25        let shared_memory_path = tuner_home.join("SHAREDMEMORY.md");
26        env.insert("TUNER_HOME".into(), tuner_home.to_string_lossy().into());
27        env.insert("TUNER_SHARED_MEMORY_PATH".into(), shared_memory_path.to_string_lossy().into());
28        env
29    }

31    async fn ensure_interactive_session(&self, sid: &str, prompt: &str, _ws: &Path, env: &HashMap<String, String>) -> Result<bool, String> {
32        let ws = self.agy_workspace();
33
34        let mut holders = self.sessions.holders.lock().await;
35        let is_running = holders.get_mut(sid)
36            .map(|h| h.child.try_wait().map(|s| s.is_none()).unwrap_or(false))
37            .unwrap_or(false);
38
39        if is_running {
40            return Ok(false);
41        }
42        holders.remove(sid);
43
44        let mut args = vec!["--add-dir".into(), ws.to_string_lossy().into(), "--conversation".into(), sid.into()];
45        if let Some(m) = &self.config.model { if m != "antigravity-default" { args.extend(["--model".into(), m.clone()]); } }
46        if let Some(ef) = &self.config.effort { if !ef.is_empty() { args.extend(["--effort".into(), ef.clone()]); } }
47        if self.config.permission_mode == "bypassPermissions" { args.push("--dangerously-skip-permissions".into()); }
48        args.extend(["--prompt-interactive".into(), prompt.into()]);
49
50        let holder = super::pty_spawner::spawn_session(&ws, "agy", &args, env)?;
51        holders.insert(sid.to_string(), holder);
52        Ok(true)
53    }

55    async fn run_in_pty_session(
56        &self,
57        session_id: &str,
58        prompt: &str,
59        agy_ws: &Path,
60        env: &HashMap<String, String>,
61    ) -> Result<(String, String, std::process::ExitStatus), String> {
62        let sid_str = session_id.to_string();
63        let sessions = self.sessions.clone();
64
65        sessions.set_running(&sid_str, true).await;
66
67        let res = async {
68            let newly_spawned = self.ensure_interactive_session(session_id, prompt, agy_ws, env).await?;
69
70            let p = events::agy_state_root(Some(env)).join("brain").join(session_id)
71                .join(".system_generated/logs/transcript_full.jsonl");
72            let sz = std::fs::metadata(&p).map(|m| m.len()).unwrap_or(0);
73
74            if !newly_spawned {
75                let clean_prompt = prompt.replace('\n', " ").replace('\r', " ");
76                let input_prompt = format!("{}\r", clean_prompt);
77                let written = self.sessions.write_to_session(session_id, &input_prompt).await?;
78                if !written {
79                    return Err(format!("Failed to write to session {}", session_id));
80                }
81            }
82
83            super::polling::wait_for_log_completion(&self.sessions, session_id, Some(p), sz).await?;
84
85            use std::os::unix::process::ExitStatusExt;
86            Ok((String::new(), String::new(), std::process::ExitStatus::from_raw(0)))
87        }.await;
88
89        sessions.set_running(&sid_str, false).await;
90        if res.is_err() {
91            sessions.terminate(&sid_str).await;
92        }
93        if !sessions.is_active(&sid_str).await {
94            sessions.set_ask_active(&sid_str, false).await;
95        }
96        res
97    }

99    async fn run_oneshot(&self, args: &[String], env: &HashMap<String, String>, ws: &Path) -> Result<(String, String, std::process::ExitStatus), String> {
100        let mut cmd = Command::new("agy");
101        cmd.args(&args[1..]).current_dir(ws).envs(env).stdin(Stdio::null()).stdout(Stdio::piped()).stderr(Stdio::piped()).kill_on_drop(true);
102        #[cfg(unix)] cmd.process_group(0);
103        let child = cmd.spawn().map_err(|e| format!("Spawn failed: {}", e))?;
104        match tokio::time::timeout(tokio::time::Duration::from_secs(300), child.wait_with_output()).await {
105            Ok(Ok(o)) => Ok((String::from_utf8_lossy(&o.stdout).into(), String::from_utf8_lossy(&o.stderr).into(), o.status)),
106            Ok(Err(e)) => Err(format!("Wait failed: {}", e)),
107            Err(_) => Err("Timeout".to_string()),
108        }
109    }
110}

112fn parse_oneshot_result(stdout: &str, resume: Option<&str>, ws: &Path, env: &HashMap<String, String>) -> (Option<String>, String, bool) {
113    let parsed: Option<serde_json::Value> = serde_json::from_str(stdout.trim()).ok();
114    let sid = resume.map(String::from)
115        .or_else(|| parsed.as_ref()?.get("conversation_id")?.as_str().filter(|s| !s.is_empty()).map(String::from))
116        .or_else(|| Some(events::resolve_brain_dir(ws, Some(env))?.file_name()?.to_string_lossy().into_owned()));
117    let brain = sid.as_ref().map(|s| events::agy_state_root(Some(env)).join("brain").join(s)).or_else(|| events::resolve_brain_dir(ws, Some(env)));
118    let text = parsed.as_ref().and_then(|j| j.get("response").or_else(|| j.get("result")).or_else(|| j.get("error"))?.as_str().filter(|s| !s.is_empty()).map(String::from))
119        .unwrap_or_else(|| events::read_transcript_answer(ws, Some(env), brain.as_deref()).unwrap_or_else(|| events::parse_antigravity_json(stdout)));
120    let is_err = parsed.as_ref().and_then(|j| j.get("status")?.as_str()) == Some("ERROR");
121    (sid, text, is_err)
122}

124#[async_trait]
125impl AgentProvider for AntigravityCli {
126    async fn send(
127        &self,
128        prompt: &str,
129        resume_session: Option<&str>,
130        continue_session: bool,
131        _workspace: PathBuf,
132    ) -> Result<CliResponse, String> {
133        let env = self.build_env();
134        let agy_ws = self.agy_workspace();
135        crate::cli::antigravity::trust::trust_workspace_in_settings(&agy_ws, None);
136
137        let (stdout, stderr, status) = if let Some(session_id) = resume_session {
138            self.run_in_pty_session(session_id, prompt, &agy_ws, &env).await?
139        } else {
140            let cmd_args = self.build_command(prompt, resume_session, continue_session);
141            self.run_oneshot(&cmd_args, &env, &agy_ws).await?
142        };
143
144        let (final_sid, result_text, is_err) = parse_oneshot_result(&stdout, resume_session, &agy_ws, &env);
145
146        if resume_session.is_none() {
147            if let Some(ref fid) = final_sid {
148                self.ensure_interactive_session(fid, prompt, &agy_ws, &env).await?;
149            }
150        }
151
152        Ok(CliResponse {
153            session_id: final_sid,
154            result: result_text,
155            is_error: !status.success() || is_err,
156            returncode: status.code(),
157            stderr,
158        })
159    }

161    async fn send_streaming<'a>(
162        &'a self,
163        prompt: &str,
164        resume_session: Option<&str>,
165        continue_session: bool,
166        workspace: PathBuf,
167    ) -> Result<BoxStream<'a, StreamEvent>, String> {
168        let (tx, rx) = tokio::sync::mpsc::unbounded_channel();
169        let self_arc = std::sync::Arc::new(self.clone());
170        let prompt_string = prompt.to_string();
171        let resume_id = resume_session.map(|s| s.to_string());
172        let workspace_path = workspace.clone();
173
174        let env = self.build_env();
175        let initial_size = resume_id.as_ref().map(|sid| {
176            let p = events::agy_state_root(Some(&env)).join("brain").join(sid)
177                .join(".system_generated/logs/transcript_full.jsonl");
178            std::fs::metadata(p).map(|m| m.len()).unwrap_or(0)
179        }).unwrap_or(0);
180
181        let rid = resume_id.clone();
182        let oneshot_handle = tokio::spawn(async move {
183            self_arc.send(&prompt_string, rid.as_deref(), continue_session, workspace_path).await
184        });
185
186        super::polling::spawn_log_polling(oneshot_handle, tx, self.agy_workspace(), self.build_env(), Some(initial_size), resume_id);
187
188        let stream = futures::stream::unfold(rx, |mut rx| async move {
189            rx.recv().await.map(|event| (event, rx))
190        });
191        Ok(Box::pin(stream))
192    }
193}
