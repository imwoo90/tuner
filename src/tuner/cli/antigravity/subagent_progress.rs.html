16/// Reads up to the trailing `max_bytes` UTF-8 lines of `path` aligned to a newline boundary.
17pub fn read_tail_utf8(path: &Path, max_bytes: u64) -> Option<String> {
18    use std::io::{Read, Seek, SeekFrom};
19    let mut f = std::fs::File::open(path).ok()?;
20    let len = f.metadata().ok()?.len();
21    let start = len.saturating_sub(max_bytes);
22    f.seek(SeekFrom::Start(start)).ok()?;
23    let mut buf = Vec::new();
24    f.read_to_end(&mut buf).ok()?;
25    let slice = if start > 0 {
26        buf.iter().position(|&b| b == b'\n').map(|i| &buf[i + 1..]).unwrap_or(&buf[..])
27    } else {
28        &buf[..]
29    };
30    Some(String::from_utf8_lossy(slice).into_owned())
31}

33fn is_safe_cid(cid: &str) -> bool {
34    !cid.is_empty() && cid != "." && !cid.contains('/') && !cid.contains('\\') && !cid.contains("..")
35}

37fn now_secs() -> u64 {
38    SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_secs()).unwrap_or(0)
39}

41fn mtime_secs(path: &Path) -> Option<u64> {
42    std::fs::metadata(path).ok()?.modified().ok()?.duration_since(UNIX_EPOCH).ok().map(|d| d.as_secs())
43}

45fn is_swarm_killed(m: &serde_json::Value) -> bool {
46    m.get("killed").and_then(|v| v.as_bool()).unwrap_or(false)
47        || m.get("desired_state").and_then(|v| v.as_str()) == Some("KILLED")
48        || m.get("phase").and_then(|v| v.as_str()) == Some("KILLED")
49        || m.get("state").and_then(|v| v.as_str()) == Some("killed")
50}

52fn collect_subagent_jsons(gen_dir: &Path, out: &mut Vec<(String, String, u64)>) {
53    let Ok(entries) = std::fs::read_dir(gen_dir.join("subagents")) else { return };
54    for entry in entries.flatten() {
55        let p = entry.path();
56        if p.extension().and_then(|e| e.to_str()) != Some("json") {
57            continue;
58        }
59        let Ok(raw) = std::fs::read_to_string(&p) else { continue };
60        let Ok(val) = serde_json::from_str::<serde_json::Value>(&raw) else { continue };
61        if val.get("state").and_then(|v| v.as_str()) != Some("SUBAGENT_STATE_ALIVE") {
62            continue;
63        }
64        let cid = val.get("conversationId").or_else(|| val.get("conversation_id")).and_then(|v| v.as_str()).unwrap_or("");
65        if is_safe_cid(cid) {
66            let role = val
67                .pointer("/subagentDescriptor/role")
68                .or_else(|| val.get("role"))
69                .or_else(|| val.pointer("/subagentDescriptor/typeName"))
70                .or_else(|| val.get("type"))
71                .or_else(|| val.get("name"))
72                .and_then(|v| v.as_str())
73                .filter(|s| !s.is_empty())
74                .unwrap_or("subagent");
75            out.push((cid.to_string(), role.to_string(), mtime_secs(&p).unwrap_or(0)));
76        }
77    }
78}

80fn collect_swarm_json(gen_dir: &Path, out: &mut Vec<(String, String, u64)>) {
81    let mp = gen_dir.join("swarm").join("members.json");
82    let Ok(raw) = std::fs::read_to_string(&mp) else { return };
83    let Ok(val) = serde_json::from_str::<serde_json::Value>(&raw) else { return };
84    let Some(members) = val.get("members").and_then(|v| v.as_object()) else { return };
85    let mm = mtime_secs(&mp).unwrap_or(0);
86    for (k, m) in members {
87        if is_swarm_killed(m) {
88            continue;
89        }
90        let st = m.get("state").or_else(|| m.get("phase")).and_then(|v| v.as_str()).unwrap_or("");
91        if matches!(st, "waiting" | "stopped") {
92            continue;
93        }
94        if let Some(cid) = m.get("conversation_id").or_else(|| m.get("conversationId")).and_then(|v| v.as_str()).filter(|s| is_safe_cid(s)) {
95            let name = m.get("name").and_then(|v| v.as_str()).unwrap_or(k.as_str());
96            out.push((cid.to_string(), format!("🐝 {name}"), mm));
97        }
98    }
99}

101fn collect_child_entries(gen_dir: &Path, out: &mut Vec<(String, String, u64)>) {
102    collect_subagent_jsons(gen_dir, out);
103    collect_swarm_json(gen_dir, out);
104}

106/// Discovers alive descendant subagent and active swarm conversations `(cid, label, desc_mtime)`.
107pub fn discover_active_descendants(
108    brain_root: &Path,
109    root_sid: &str,
110    max_nodes: usize,
111) -> Vec<(String, String, u64)> {
112    if !is_safe_cid(root_sid) || !brain_root.is_dir() {
113        return Vec::new();
114    }
115    let mut visited = HashSet::from([root_sid.to_string()]);
116    let mut queue = vec![root_sid.to_string()];
117    let mut result = Vec::new();
118    while let Some(cur) = queue.pop() {
119        if result.len() >= max_nodes {
120            break;
121        }
122        let gen_dir = brain_root.join(&cur).join(".system_generated");
123        let mut children = Vec::new();
124        collect_child_entries(&gen_dir, &mut children);
125        for (cid, label, dm) in children {
126            if result.len() >= max_nodes || !visited.insert(cid.clone()) {
127                continue;
128            }
129            queue.push(cid.clone());
130            result.push((cid, label, dm));
131        }
132    }
133    result
134}

136/// Counts descendants under `root_sid` that have active/recent (`<20m`) subagent or swarm work.
137pub fn count_active_descendants(brain_root: &Path, root_sid: &str) -> usize {
138    let now = now_secs();
139    discover_active_descendants(brain_root, root_sid, 24)
140        .into_iter()
141        .filter(|(cid, _, dm)| {
142            let tp = brain_root.join(cid).join(".system_generated/logs/transcript_full.jsonl");
143            let tm = mtime_secs(&tp).unwrap_or(0);
144            now.saturating_sub((*dm).max(tm)) < 20 * 60
145        })
146        .count()
147}

149/// Stateful incremental byte-offset tracker for descendant subagent/swarm transcripts.
150#[derive(Default)]
151pub struct SubagentProgressTracker {
152    offsets: HashMap<String, u64>,
153    cards: BTreeMap<String, String>,
154}
155
156impl SubagentProgressTracker {
157    /// Creates an empty tracker.
158    pub fn new() -> Self {
159        Self::default()
160    }
161
162    /// Clears rendered progress cards when the parent conversation starts or finishes a turn.
163    pub fn reset(&mut self) {
164        self.cards.clear();
165    }
166
167    /// Polls alive descendant transcripts and returns a combined Markdown card if any child advanced.
168    pub fn poll_delta(&mut self, brain_root: &Path, root_sid: &str) -> Option<String> {
169        let active = discover_active_descendants(brain_root, root_sid, 16);
170        let live_ids: HashSet<&str> = active.iter().map(|(c, _, _)| c.as_str()).collect();
171        self.cards.retain(|k, _| live_ids.contains(k.as_str()));
172        if active.is_empty() {
173            return None;
174        }
175        let now = now_secs();
176        let mut changed = false;
177        for (cid, label, _) in &active {
178            let tp = brain_root.join(cid).join(".system_generated/logs/transcript_full.jsonl");
179            let Ok(meta) = std::fs::metadata(&tp) else { continue };
180            let prev = match self.offsets.get(cid).copied() {
181                Some(o) => o,
182                None => {
183                    let tm = mtime_secs(&tp).unwrap_or(0);
184                    let init = if now.saturating_sub(tm) > 120 { meta.len() } else { 0 };
185                    self.offsets.insert(cid.clone(), init);
186                    init
187                }
188            };
189            if meta.len() <= prev {
190                self.offsets.insert(cid.clone(), meta.len());
191                continue;
192            }
193            let delta = AntigravityLogParser::new().parse_log_delta_structured(&tp, Some(prev));
194            self.offsets.insert(cid.clone(), delta.new_size);
195            if let Some(fmt) = delta.formatted.filter(|s| !s.trim().is_empty()) {
196                let short: String = cid.chars().take(8).collect();
197                self.cards.insert(cid.clone(), format!("🤖 **[Subagent: {label} `#{short}`]**\n{fmt}"));
198                changed = true;
199            }
200        }
201        (changed && !self.cards.is_empty()).then(|| self.cards.values().cloned().collect::<Vec<_>>().join("\n\n"))
202    }
203}
