1//! # OpenAI Codex Warm & Ephemeral Process Spawner (`spawner.rs`)
2//!
3//! ## Overview
4//! Spawns `codex app-server --stdio` processes in dedicated process groups and executes
5//! the JSON-RPC `initialize` + `thread/resume` or `thread/start` handshake.
6//!
7//! ## Search Tags
8//! #codex-spawner, #json-rpc-handshake, #app-server

23pub(crate) async fn wait_rpc_response(
24    stdout: &mut Lines<BufReader<ChildStdout>>,
25    expected_id: u64,
26) -> Result<serde_json::Value, String> {
27    let dl = tokio::time::Instant::now() + std::time::Duration::from_secs(15);
28    while let Ok(Ok(Some(line))) = tokio::time::timeout_at(dl, stdout.next_line()).await {
29        if let Ok(val) = serde_json::from_str::<serde_json::Value>(line.trim())
30            && val.get("id").and_then(|v| v.as_u64()) == Some(expected_id)
31        {
32            return Ok(val);
33        }
34    }
35    Err("Codex app-server closed stdout or timed out during RPC handshake".to_string())
36}

38pub(crate) fn extract_thread_id(resp: &serde_json::Value) -> Option<String> {
39    resp.get("result")?
40        .get("thread")?
41        .get("id")?
42        .as_str()
43        .map(str::to_string)
44}

46async fn send_codex_initialize(
47    stdin: &mut ChildStdin,
48    stdout: &mut Lines<BufReader<ChildStdout>>,
49) -> Result<(), String> {
50    write_json_stdin(
51        stdin,
52        &json!({
53            "jsonrpc": "2.0", "id": 1, "method": "initialize",
54            "params": {
55                "clientInfo": { "name": "tuner", "version": env!("CARGO_PKG_VERSION") },
56                "capabilities": {}
57            }
58        }),
59    )
60    .await?;
61    let _ = wait_rpc_response(stdout, 1).await?;
62    write_json_stdin(
63        stdin,
64        &json!({ "jsonrpc": "2.0", "method": "initialized", "params": {} }),
65    )
66    .await
67}

69async fn handshake_codex_thread(
70    cli: &AntigravityCli,
71    ws: &Path,
72    resume_sid: Option<&str>,
73    stdin: &mut ChildStdin,
74    stdout: &mut Lines<BufReader<ChildStdout>>,
75) -> Result<String, String> {
76    send_codex_initialize(stdin, stdout).await?;
77    let m = resolve_codex_model_effort(cli.config.model.as_deref(), cli.config.effort.as_deref())
78        .0
79        .unwrap_or_else(|| DEFAULT_CODEX_MODEL.to_string());
80    let (ap, sb) = if cli.config.permission_mode == "bypassPermissions" {
81        ("never", "danger-full-access")
82    } else {
83        ("on-request", "workspace-write")
84    };
85    let cwd = ws.to_string_lossy();
86    if let Some(sid) = resume_sid.filter(|s| !s.is_empty() && !s.starts_with("mock-")) {
87        write_json_stdin(
88            stdin,
89            &json!({
90                "jsonrpc": "2.0", "id": 2, "method": "thread/resume",
91                "params": { "threadId": sid, "cwd": cwd, "model": m, "approvalPolicy": ap, "sandbox": sb }
92            }),
93        )
94        .await?;
95        if let Some(tid) = extract_thread_id(&wait_rpc_response(stdout, 2).await?) {
96            return Ok(tid);
97        }
98        if cli.oneshot {
99            return Err(format!("Codex oneshot failed to resume thread '{sid}'"));
100        }
101    }
102    write_json_stdin(
103        stdin,
104        &json!({
105            "jsonrpc": "2.0", "id": 3, "method": "thread/start",
106            "params": { "cwd": cwd, "model": m, "approvalPolicy": ap, "sandbox": sb }
107        }),
108    )
109    .await?;
110    let resp = wait_rpc_response(stdout, 3).await?;
111    extract_thread_id(&resp).ok_or_else(|| format!("Failed to start codex thread: {resp}"))
112}

114pub(crate) async fn spawn_warm_codex(
115    cli: &AntigravityCli,
116    ws: PathBuf,
117    resume_sid: Option<&str>,
118) -> Result<Arc<WarmCodexSession>, String> {
119    let args = build_app_server_command(cli);
120    let mut cmd = Command::new(&args[0]);
121    cmd.args(&args[1..])
122        .current_dir(&ws)
123        .envs(crate::messenger::web::provider_status::build_peer_env(cli))
124        .stdin(Stdio::piped())
125        .stdout(Stdio::piped())
126        .stderr(Stdio::null())
127        .kill_on_drop(true);
128    #[cfg(unix)]
129    cmd.process_group(0);
130    let mut child = cmd
131        .spawn()
132        .map_err(|e| format!("Failed to spawn warm codex app-server: {e}"))?;
133    let pid = child.id().unwrap_or(0);
134    let mut stdin = child
135        .stdin
136        .take()
137        .ok_or_else(|| "Failed to open codex stdin".to_string())?;
138    let mut stdout = BufReader::new(
139        child
140            .stdout
141            .take()
142            .ok_or_else(|| "Failed to open codex stdout".to_string())?,
143    )
144    .lines();
145    let thread_id = match handshake_codex_thread(cli, &ws, resume_sid, &mut stdin, &mut stdout).await
146    {
147        Ok(tid) => tid,
148        Err(e) => {
149            if pid > 0 {
150                kill_process_group(pid);
151            }
152            return Err(e);
153        }
154    };
155    Ok(Arc::new(WarmCodexSession {
156        pid,
157        thread_id,
158        workspace: ws,
159        activity: WarmActivityTracker::new(),
160        pending_ask: Mutex::new(None),
161        queued_ask_answers: Mutex::new(None),
162        active_turn_id: Mutex::new(None),
163        next_req_id: AtomicU64::new(10),
164        stdin: Mutex::new(stdin),
165        io: Mutex::new(CodexProcessIo { child, stdout }),
166        running: AtomicBool::new(false),
167        interrupted: AtomicBool::new(false),
168        interrupt_notify: Notify::new(),
169    }))
170}

172pub(crate) async fn prepare_codex_turn_rpc(
173    warm: &WarmCodexSession,
174    cli: &AntigravityCli,
175    prompt: &str,
176) -> (u64, serde_json::Value) {
177    if prompt.is_empty()
178        && let Some((rpc_id, q_ids)) = warm.pending_ask.lock().await.take()
179    {
180        let ans = warm.queued_ask_answers.lock().await.take().unwrap_or_default();
181        return (
182            0,
183            crate::cli::common::build_codex_ask_rpc_response(rpc_id, &q_ids, &ans),
184        );
185    }
186    let id = warm.next_req_id.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
187    let (m, eff) = resolve_codex_model_effort(cli.config.model.as_deref(), cli.config.effort.as_deref());
188    let req = json!({
189        "jsonrpc": "2.0", "id": id, "method": "turn/start",
190        "params": {
191            "threadId": warm.thread_id,
192            "input": [{ "type": "text", "text": cli.format_prompt(prompt) }],
193            "model": m.unwrap_or_else(|| DEFAULT_CODEX_MODEL.to_string()),
194            "effort": eff.unwrap_or_else(|| "medium".to_string()),
195        }
196    });
197    (id, req)
198}
