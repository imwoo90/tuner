3//! ## Overview

6//! agent deltas into `StreamEvent` items.

42/// Supported reasoning effort levels for the `codex` CLI.
43pub const CODEX_EFFORTS: &[&str] = &["minimal", "low", "medium", "high", "xhigh"];

45/// Returns the static fallback model list (with effort suffixes) for the `codex` provider.
46pub fn discover_models() -> Vec<String> {
47    CODEX_MODELS.iter().flat_map(|b| CODEX_EFFORTS.iter().map(move |e| format!("{}-{}", b, e))).collect()
48}

50/// Dynamically queries `codex app-server --stdio` `model/list` for the user's live `/model` options.
51pub async fn discover_models_live(env: &std::collections::HashMap<String, String>) -> Vec<String> {
52    let init = format!(
53        r#"{{"jsonrpc":"2.0","id":1,"method":"initialize","params":{{"clientInfo":{{"name":"tuner","version":"{}"}}}}}}"#,
54        env!("CARGO_PKG_VERSION")
55    );
56    let reqs = [
57        init.as_str(),
58        r#"{"jsonrpc":"2.0","method":"initialized","params":{}}"#,
59        r#"{"jsonrpc":"2.0","id":2,"method":"model/list","params":{}}"#,
60    ];
61    probe_cli_json_lines(&resolve_codex_bin(), &["app-server", "--stdio"], &reqs, env, 10)
62        .await.iter().find_map(|l| Some(parse_codex_models_response(l)).filter(|p| !p.is_empty()))
63        .unwrap_or_else(discover_models)
64}

69fn normalize_codex_model(raw: &str) -> &str {
70    match raw {
71        "astra" => "gpt-6-astra",
72        "sol" => "gpt-6.1-sol",
73        "terra" => "gpt-6.1-terra",
74        "luna" => "gpt-6-luna",
75        other => other,
76    }
77}

79/// Returns `true` when the given Codex model supports reasoning effort configuration.
80pub fn supports_codex_effort(model: &str) -> bool {
81    let low = model.trim().to_ascii_lowercase();
82    !low.is_empty() && low != "antigravity-default"
83}

85/// Resolves the canonical model and reasoning effort flags for the `codex` CLI.
86pub fn resolve_codex_model_effort(model: Option<&str>, effort: Option<&str>) -> (Option<String>, Option<String>) {
87    let mut suffix_eff = None;
88    let explicit_effortless = model.map(str::trim).is_some_and(|m| !m.is_empty() && !supports_codex_effort(m));
89    let resolved_model = model
90        .map(str::trim)
91        .filter(|s| !s.is_empty() && !s.eq_ignore_ascii_case("antigravity-default"))
92        .and_then(|raw| {
93            let low = raw.to_ascii_lowercase();
94            let (base, eff) = strip_effort_suffix(&low);
95            let c = normalize_codex_model(base);
96            let ok = crate::cli::codex::CODEX_MODELS.contains(&c)
97                || ((c.starts_with("gpt-5")
98                    || c.starts_with("gpt-6")
99                    || ["o1", "o3", "o4", "codex"].iter().any(|p| c.starts_with(p)))
100                    && !c.starts_with("gpt-oss"));
101            if ok {
102                suffix_eff = eff.map(str::to_string);
103                Some(c.to_string())
104            } else {
105                None
106            }
107        });
108    if explicit_effortless {
109        return (resolved_model, None);
110    }
111    let raw_eff = effort.map(str::trim).filter(|e| !e.is_empty()).map(str::to_lowercase).or(suffix_eff);
112    let resolved_eff = raw_eff.and_then(|e| match e.as_str() {
113        "minimal" | "low" | "medium" | "high" => Some(e),
114        "xhigh" | "max" | "ultra" => Some("xhigh".to_string()),
115        _ => None,
116    });
117    (resolved_model, resolved_eff)
118}

120/// Builds the persistent warm `codex app-server --stdio` command-line vector.
121pub fn build_app_server_command(cli: &AntigravityCli) -> Vec<String> {
122    let mut cmd = [
123        resolve_codex_bin().as_str(), "app-server", "--stdio",
124        "-c", "features.default_mode_request_user_input=true",
125        "-c", "suppress_unstable_features_warning=true",
126    ].into_iter().map(str::to_string).collect::<Vec<_>>();
127    if let Some(params) = cli.config.cli_parameters.get("codex") {
128        let mut iter = params.iter();
129        while let Some(a) = iter.next() {
130            if matches!(a.as_str(), "-c" | "--config" | "--enable" | "--disable") {
131                if let Some(v) = iter.next() { cmd.extend([a.clone(), v.clone()]); }
132            } else if a.starts_with("-c=") || a.starts_with("--config=") {
133                cmd.push(a.clone());
134            }
135        }
136    }
137    cmd
138}

140/// Builds the command-line invocation vector for a `codex` CLI turn.
141pub fn build_command(
142    cli: &AntigravityCli,
143    prompt: &str,
144    resume_session: Option<&str>,
145    continue_session: bool,
146) -> Vec<String> {
147    let mut cmd = vec![resolve_codex_bin(), "exec".to_string()];
148    if let Some(sid) = resume_session.filter(|s| !s.is_empty() && !s.starts_with("mock-")) {
149        cmd.extend(["resume".to_string(), sid.to_string()]);
150    } else if continue_session {
151        cmd.extend(["resume".to_string(), "--last".to_string()]);
152    }
153    cmd.extend(["--json".to_string(), "--skip-git-repo-check".to_string()]);
154    if cli.config.permission_mode == "bypassPermissions" {
155        cmd.push("--dangerously-bypass-approvals-and-sandbox".to_string());
156    }
157    let (m, eff) = resolve_codex_model_effort(cli.config.model.as_deref(), cli.config.effort.as_deref());
158    if let Some(m) = m { cmd.extend(["-m".to_string(), m]); }
159    if let Some(e) = eff { cmd.extend(["-c".to_string(), format!("model_reasoning_effort=\"{}\"", e)]); }
160    if let Some(p) = cli.config.cli_parameters.get("codex") { cmd.extend(p.iter().cloned()); }
161    cmd.push(cli.format_prompt(prompt));
162    cmd
163}
