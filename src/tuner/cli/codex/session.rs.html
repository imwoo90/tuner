30/// Persistent warm `codex app-server --stdio` session bound to a conversation topic.
31pub struct WarmCodexSession {
32    /// OS process ID of the warm `codex app-server` child.
33    pub pid: u32,
34    /// Active Codex thread UUID.
35    pub thread_id: String,
36    /// Workspace path where the warm process was started.
37    pub workspace: PathBuf,
38    /// 24-hour activity tracker matching Antigravity PTY session expiration.
39    pub activity: WarmActivityTracker,
40    pub(crate) pending_ask: Mutex<Option<(serde_json::Value, Vec<String>)>>,
41    pub(crate) queued_ask_answers: Mutex<Option<Vec<String>>>,
42    pub(crate) active_turn_id: Mutex<Option<String>>,
43    pub(crate) next_req_id: AtomicU64,
44    pub(crate) stdin: Mutex<ChildStdin>,
45    pub(crate) io: Mutex<CodexProcessIo>,
46    pub(crate) running: AtomicBool,
47    pub(crate) interrupted: AtomicBool,
48    pub(crate) interrupt_notify: Notify,
49}
50
51impl WarmCodexSession {
52    pub(crate) async fn is_alive(&self) -> bool {
53        self.io.try_lock().map(|mut io| io.child.try_wait().map(|s| s.is_none()).unwrap_or(false)).unwrap_or(true)
54    }
55
56    pub(crate) async fn write_rpc(&self, payload: &serde_json::Value) -> Result<(), String> {
57        self.activity.touch();
58        write_json_stdin(&mut *self.stdin.lock().await, payload).await
59    }
60
61    pub(crate) async fn send_turn_interrupt(&self, turn_id: &str) {
62        let id = self.next_req_id.fetch_add(1, Ordering::SeqCst);
63        let _ = self.write_rpc(&json!({
64            "jsonrpc": "2.0", "id": id, "method": "turn/interrupt",
65            "params": { "threadId": self.thread_id, "turnId": turn_id }
66        })).await;
67    }
68
69    pub(crate) fn kill_now(&self) {
70        if self.pid > 0 { kill_process_group(self.pid); }
71    }
72}

82async fn wait_rpc_response(stdout: &mut Lines<BufReader<ChildStdout>>, expected_id: u64) -> Result<serde_json::Value, String> {
83    let dl = tokio::time::Instant::now() + std::time::Duration::from_secs(15);
84    while let Ok(Ok(Some(line))) = tokio::time::timeout_at(dl, stdout.next_line()).await {
85        if let Ok(val) = serde_json::from_str::<serde_json::Value>(line.trim())
86            && val.get("id").and_then(|v| v.as_u64()) == Some(expected_id)
87        {
88            return Ok(val);
89        }
90    }
91    Err("Codex app-server closed stdout or timed out during RPC handshake".to_string())
92}

94fn extract_thread_id(resp: &serde_json::Value) -> Option<String> {
95    resp.get("result")?.get("thread")?.get("id")?.as_str().map(str::to_string)
96}

98async fn handshake_codex_thread(
99    cli: &AntigravityCli,
100    ws: &std::path::Path,
101    resume_sid: Option<&str>,
102    stdin: &mut ChildStdin,
103    stdout: &mut Lines<BufReader<ChildStdout>>,
104) -> Result<String, String> {
105    write_json_stdin(stdin, &json!({
106        "jsonrpc": "2.0", "id": 1, "method": "initialize",
107        "params": { "clientInfo": { "name": "tuner", "version": env!("CARGO_PKG_VERSION") }, "capabilities": {} }
108    })).await?;
109    let _ = wait_rpc_response(stdout, 1).await?;
110    write_json_stdin(stdin, &json!({ "jsonrpc": "2.0", "method": "initialized", "params": {} })).await?;
111    let m = resolve_codex_model_effort(cli.config.model.as_deref(), cli.config.effort.as_deref()).0
112        .unwrap_or_else(|| DEFAULT_CODEX_MODEL.to_string());
113    let (ap, sb) = if cli.config.permission_mode == "bypassPermissions" { ("never", "danger-full-access") } else { ("on-request", "workspace-write") };
114    let cwd = ws.to_string_lossy();
115    if let Some(sid) = resume_sid.filter(|s| !s.is_empty() && !s.starts_with("mock-")) {
116        write_json_stdin(stdin, &json!({
117            "jsonrpc": "2.0", "id": 2, "method": "thread/resume",
118            "params": { "threadId": sid, "cwd": cwd, "model": m, "approvalPolicy": ap, "sandbox": sb }
119        })).await?;
120        if let Some(tid) = extract_thread_id(&wait_rpc_response(stdout, 2).await?) { return Ok(tid); }
121    }
122    write_json_stdin(stdin, &json!({
123        "jsonrpc": "2.0", "id": 3, "method": "thread/start",
124        "params": { "cwd": cwd, "model": m, "approvalPolicy": ap, "sandbox": sb }
125    })).await?;
126    let resp = wait_rpc_response(stdout, 3).await?;
127    extract_thread_id(&resp).ok_or_else(|| format!("Failed to start codex thread: {}", resp))
128}

130async fn spawn_warm_codex(cli: &AntigravityCli, ws: PathBuf, resume_sid: Option<&str>) -> Result<Arc<WarmCodexSession>, String> {
131    let args = build_app_server_command(cli);
132    let mut cmd = Command::new(&args[0]);
133    cmd.args(&args[1..]).current_dir(&ws).envs(cli.build_env())
134        .stdin(Stdio::piped()).stdout(Stdio::piped()).stderr(Stdio::null()).kill_on_drop(true);
135    #[cfg(unix)]
136    cmd.process_group(0);
137    let mut child = cmd.spawn().map_err(|e| format!("Failed to spawn warm codex app-server: {}", e))?;
138    let pid = child.id().unwrap_or(0);
139    let mut stdin = child.stdin.take().ok_or_else(|| "Failed to open codex stdin".to_string())?;
140    let mut stdout = BufReader::new(child.stdout.take().ok_or_else(|| "Failed to open codex stdout".to_string())?).lines();
141    let thread_id = match handshake_codex_thread(cli, &ws, resume_sid, &mut stdin, &mut stdout).await {
142        Ok(tid) => tid,
143        Err(e) => {
144            if pid > 0 { kill_process_group(pid); }
145            return Err(e);
146        }
147    };
148    Ok(Arc::new(WarmCodexSession {
149        pid, thread_id, workspace: ws, activity: WarmActivityTracker::new(),
150        pending_ask: Mutex::new(None), queued_ask_answers: Mutex::new(None),
151        active_turn_id: Mutex::new(None), next_req_id: AtomicU64::new(10),
152        stdin: Mutex::new(stdin), io: Mutex::new(CodexProcessIo { child, stdout }),
153        running: AtomicBool::new(false), interrupted: AtomicBool::new(false), interrupt_notify: Notify::new(),
154    }))
155}

157/// Queues user answers for a pending `item/tool/requestUserInput` request on the topic's warm `codex` session.
158pub async fn submit_ask_answers(chat_id: i64, topic_id: Option<i64>, answers: &[String]) {
159    if let Some(w) = global_pool().sessions.lock().await.get(&(chat_id, topic_id)).cloned() {
160        *w.queued_ask_answers.lock().await = Some(answers.to_vec());
161    }
162}

164/// Reaps dead or 24-hour (`86400s`) idle warm `codex` processes from the pool.
165pub async fn cleanup_expired() {
166    let mut map = global_pool().sessions.lock().await;
167    for k in map.iter().filter(|(_, w)| !w.running.load(Ordering::SeqCst) && w.activity.is_expired()).map(|(k, _)| *k).collect::<Vec<_>>() {
168        if let Some(old) = map.remove(&k) { old.kill_now(); }
169    }
170}

172pub(crate) async fn get_or_spawn_warm_codex(
173    cli: &AntigravityCli,
174    workspace: PathBuf,
175    resume_sid: Option<&str>,
176    continue_session: bool,
177) -> Result<Arc<WarmCodexSession>, String> {
178    cleanup_expired().await;
179    let (ws, key) = (resolve_warm_workspace(cli, workspace), (cli.config.chat_id, cli.config.topic_id));
180    if let Some(ex) = global_pool().sessions.lock().await.get(&key).cloned() {
181        if (resume_sid.is_some_and(|r| ex.thread_id == r) || continue_session) && ex.workspace == ws && ex.is_alive().await {
182            ex.activity.touch();
183            return Ok(ex);
184        }
185        if let Some(old) = global_pool().sessions.lock().await.remove(&key) { old.kill_now(); }
186    }
187    let warm = spawn_warm_codex(cli, ws, resume_sid).await?;
188    if let Some(old) = global_pool().sessions.lock().await.insert(key, warm.clone()) { old.kill_now(); }
189    Ok(warm)
190}

192/// Sends a JSON-RPC `turn/interrupt` request to the active warm `codex` session for a topic.
193pub async fn interrupt_session(chat_id: i64, topic_id: Option<i64>) -> bool {
194    let Some(w) = global_pool().sessions.lock().await.get(&(chat_id, topic_id)).cloned() else { return false; };
195    if !w.running.load(Ordering::SeqCst) { return false; }
196    if let Some(tid) = w.active_turn_id.lock().await.clone() { w.send_turn_interrupt(&tid).await; }
197    w.interrupted.store(true, Ordering::SeqCst);
198    w.interrupt_notify.notify_one();
199    true
200}

202/// Terminates and removes the warm `codex` process for a specific topic or thread ID.
203pub async fn terminate_topic(chat_id: i64, topic_id: Option<i64>, session_id: Option<&str>) -> bool {
204    let (mut map, mut rem) = (global_pool().sessions.lock().await, false);
205    if let Some(w) = map.remove(&(chat_id, topic_id)) { w.kill_now(); rem = true; }
206    if let Some(sid) = session_id.filter(|s| !s.is_empty()) {
207        for k in map.iter().filter(|(_, w)| w.thread_id == sid).map(|(k, _)| *k).collect::<Vec<_>>() {
208            if let Some(w) = map.remove(&k) { w.kill_now(); rem = true; }
209        }
210    }
211    rem
212}

214/// Terminates all warm `codex` processes across all topics for a given `chat_id` (or all chats when `chat_id == 0`).
215pub async fn terminate_all_for_chat(chat_id: i64) {
216    let mut map = global_pool().sessions.lock().await;
217    for k in map.keys().filter(|(cid, _)| chat_id == 0 || *cid == chat_id).copied().collect::<Vec<_>>() {
218        if let Some(w) = map.remove(&k) { w.kill_now(); }
219    }
220}

222/// Terminates all warm `codex` processes across all topics.
223pub async fn terminate_all() { terminate_all_for_chat(0).await; }

225/// Returns the OS PID of the active warm `codex` process for `(chat_id, topic_id)`, if any.
226pub async fn active_pid(chat_id: i64, topic_id: Option<i64>) -> Option<u32> {
227    let warm = global_pool().sessions.lock().await.get(&(chat_id, topic_id)).cloned()?;
228    warm.is_alive().await.then_some(warm.pid)

231/// Returns `true` if a turn is currently running on the warm `codex` session for the topic or `session_id`.
232pub async fn is_running(chat_id: i64, topic_id: Option<i64>, session_id: Option<&str>) -> bool {
233    let map = global_pool().sessions.lock().await;
234    if chat_id == 0 && topic_id.is_none() && session_id.is_none() {
235        return map.values().any(|w| w.running.load(Ordering::SeqCst));
236    }
237    if let Some(w) = map.get(&(chat_id, topic_id)) { return w.running.load(Ordering::SeqCst); }
238    session_id.filter(|s| !s.is_empty()).is_some_and(|sid| {
239        map.values().any(|w| w.thread_id == sid && w.running.load(Ordering::SeqCst))
240    })
241}
