25pub(crate) fn adjust_ephemeral_runs(inc: bool) {
26    if inc { EPHEMERAL_RUNS.fetch_add(1, Ordering::SeqCst); } else { EPHEMERAL_RUNS.fetch_sub(1, Ordering::SeqCst); }
27}

37/// Queues user answers for a pending `AskUserQuestion` control request on the topic's warm `claude` session.
38pub async fn submit_ask_answers(chat_id: i64, topic_id: Option<i64>, answers: &[String]) {
39    if let Some(w) = global_pool().sessions.lock().await.get(&(chat_id, topic_id)).cloned() {
40        *w.queued_ask_answers.lock().await = Some(answers.to_vec());
41    }
42}

44/// Reaps dead or 24-hour (`86400s`) idle warm `claude` processes from the pool.
45pub async fn cleanup_expired() {
46    let mut map = global_pool().sessions.lock().await;
47    let mut dead = Vec::new();
48    for (&k, w) in map.iter() {
49        if !w.running.load(Ordering::SeqCst) && (w.activity.is_expired() || !w.is_alive().await) { dead.push(k); }
50    }
51    for k in dead { if let Some(old) = map.remove(&k) { old.kill_now(); } }
52}

54pub(crate) async fn acquire_claude_session(
55    cli: &AntigravityCli,
56    workspace: PathBuf,
57    resume_sid: Option<&str>,
58    continue_session: bool,
59    answering_ask: bool,
60) -> Result<(Arc<WarmClaudeSession>, bool), String> {
61    cleanup_expired().await;
62    let ws = resolve_warm_workspace(cli, workspace);
63    crate::cli::common::trust_peer_workspace("claude", &ws, None);
64    crate::workspace::skills_helpers::sync_claude_workspace_bridges(&ws);
65    if cli.oneshot {
66        return spawn_warm_claude(cli, ws, resume_sid, continue_session).map(|w| (w, true));
67    }
68    let key = (cli.config.chat_id, cli.config.topic_id);
69    if let Some(ex) = global_pool().sessions.lock().await.get(&key).cloned() {
70        let cur_s = ex.session_id.lock().await.clone();
71        let sid_ok = resume_sid.is_some_and(|r| cur_s.as_deref() == Some(r)) || continue_session;
72        let busy = ex.running.load(Ordering::SeqCst)
73            || ex.pending_ask.lock().await.is_some()
74            || ex.queued_ask_answers.lock().await.is_some();
75        if busy && (!sid_ok || ex.workspace != ws) {
76            return spawn_warm_claude(cli, ws, resume_sid, continue_session).map(|w| (w, true));
77        }
78    }
79    get_or_spawn_warm_session(cli, ws, resume_sid, continue_session, answering_ask).await.map(|w| (w, false))
80}

82pub(crate) async fn get_or_spawn_warm_session(
83    cli: &AntigravityCli, ws: PathBuf, resume_sid: Option<&str>, continue_session: bool, answering_ask: bool,
84) -> Result<Arc<WarmClaudeSession>, String> {
85    let (req_m, req_e) = resolve_claude_model_effort(cli.config.model.as_deref(), cli.config.effort.as_deref());
86    let (key, mut eff_res) = ((cli.config.chat_id, cli.config.topic_id), resume_sid.map(str::to_string));
87    let mut map = global_pool().sessions.lock().await;
88    if let Some(ex) = map.get(&key).cloned() {
89        let (cur_s, cur_e) = (ex.session_id.lock().await.clone(), ex.effort.lock().await.clone());
90        let has_ask = ex.pending_ask.lock().await.is_some();
91        let ask = answering_ask && (has_ask || ex.queued_ask_answers.lock().await.is_some());
92        let sid_ok = resume_sid.is_some_and(|r| cur_s.as_deref() == Some(r)) || continue_session;
93        if (!has_ask || answering_ask) && sid_ok && (ask || req_e == cur_e) && ex.workspace == ws && ex.is_alive().await {
94            if !ask && let Some(ref m) = req_m && *ex.model.lock().await != Some(m.clone())
95                && ex.write_json_line(&model_ctrl(m)).await.is_ok()
96            {
97                *ex.model.lock().await = Some(m.clone());
98            }
99            ex.activity.touch();
100            return Ok(ex);
101        }
102        if sid_ok && eff_res.is_none() { eff_res = cur_s; }
103        if let Some(old) = map.remove(&key) { old.kill_now(); }
104    }
105    let warm = spawn_warm_claude(cli, ws, eff_res.as_deref(), continue_session)?;
106    map.insert(key, warm.clone());
107    Ok(warm)
108}

110/// Sends a soft JSONL `interrupt` control request (or cancels a pending ask) for a topic.
111pub async fn interrupt_session(chat_id: i64, topic_id: Option<i64>) -> bool {
112    let Some(w) = global_pool().sessions.lock().await.get(&(chat_id, topic_id)).cloned() else { return false; };
113    if w.pending_ask.lock().await.is_some() {
114        w.interrupted.store(true, Ordering::SeqCst);
115        w.interrupt_notify.notify_one();
116        return terminate_topic(chat_id, topic_id, None).await;
117    }
118    if !w.running.load(Ordering::SeqCst) {
119        return false;
120    }
121    let ctrl = serde_json::json!({ "type": "control_request", "request_id": format!("stop-{}", chat_id), "request": { "subtype": "interrupt" } });
122    let _ = w.write_json_line(&ctrl).await;
123    w.interrupted.store(true, Ordering::SeqCst);
124    w.interrupt_notify.notify_one();
125    true
126}

128async fn find_warm(
129    map: &HashMap<TopicKey, Arc<WarmClaudeSession>>,
130    chat_id: i64,
131    topic_id: Option<i64>,
132    session_id: Option<&str>,
133) -> Option<Arc<WarmClaudeSession>> {
134    if (chat_id != 0 || topic_id.is_some())
135        && let Some(w) = map.get(&(chat_id, topic_id)).cloned()
136    {
137        return Some(w);
138    }
139    let sid = session_id.filter(|s| !s.is_empty())?;
140    for w in map.values() {
141        if w.session_id.lock().await.as_deref() == Some(sid) { return Some(w.clone()); }
142    }
143    None
144}

146/// Dynamically switches the active model of a warm `claude` session via `set_model` control_request.
147pub async fn set_model_live(chat_id: i64, topic_id: Option<i64>, session_id: Option<&str>, model: &str) -> bool {
148    let (Some(canonical), _) = resolve_claude_model_effort(Some(model), None) else { return false; };
149    let warm = { let map = global_pool().sessions.lock().await; find_warm(&map, chat_id, topic_id, session_id).await };
150    let Some(w) = warm else { return false; };
151    if w.running.load(Ordering::SeqCst) || w.pending_ask.lock().await.is_some() || w.queued_ask_answers.lock().await.is_some() {
152        return false;
153    }
154    if !super::supports_claude_effort(&canonical) && w.effort.lock().await.is_some() {
155        return terminate_topic(chat_id, topic_id, session_id).await;
156    }
157    let ok = w.write_json_line(&model_ctrl(&canonical)).await.is_ok();
158    if ok { *w.model.lock().await = Some(canonical); }
159    ok
160}

162/// Terminates and removes the warm `claude` process for a specific topic or session ID.
163pub async fn terminate_topic(chat_id: i64, topic_id: Option<i64>, session_id: Option<&str>) -> bool {
164    let (mut map, mut rem) = (global_pool().sessions.lock().await, false);
165    if let Some(w) = map.remove(&(chat_id, topic_id)) { w.kill_now(); rem = true; }
166    if let Some(sid) = session_id.filter(|s| !s.is_empty()) {
167        let mut keys = Vec::new();
168        for (k, w) in map.iter() {
169            if w.session_id.lock().await.as_deref() == Some(sid) { keys.push(*k); }
170        }
171        for k in keys { if let Some(w) = map.remove(&k) { w.kill_now(); rem = true; } }
172    }
173    rem
174}

176/// Terminates all warm `claude` processes across all topics for a given `chat_id` (or all chats when `chat_id == 0`).
177pub async fn terminate_all_for_chat(chat_id: i64) {
178    let mut map = global_pool().sessions.lock().await;
179    for k in map.keys().filter(|(cid, _)| chat_id == 0 || *cid == chat_id).copied().collect::<Vec<_>>() {
180        if let Some(w) = map.remove(&k) { w.kill_now(); }
181    }
182}

184/// Terminates all warm `claude` processes across all topics.
185pub async fn terminate_all() { terminate_all_for_chat(0).await; }

187/// Returns the OS PID of the active warm `claude` process for `(chat_id, topic_id)`, if any.
188pub async fn active_pid(chat_id: i64, topic_id: Option<i64>) -> Option<u32> {
189    let warm = global_pool().sessions.lock().await.get(&(chat_id, topic_id)).cloned()?;
190    warm.is_alive().await.then_some(warm.pid)
191}

193/// Returns the live session identifier of the topic's warm `claude` process, if any.
194pub async fn active_session_id(chat_id: i64, topic_id: Option<i64>) -> Option<String> {
195    let warm = global_pool().sessions.lock().await.get(&(chat_id, topic_id)).cloned()?;
196    if !warm.is_alive().await { return None; }
197    warm.session_id.lock().await.clone().filter(|s| !s.is_empty())
198}

200/// Returns `true` if a turn is currently running on the warm `claude` session for the topic or `session_id`.
201pub async fn is_running(chat_id: i64, topic_id: Option<i64>, session_id: Option<&str>) -> bool {
202    let map = global_pool().sessions.lock().await;
203    if chat_id == 0 && topic_id.is_none() && session_id.is_none() {
204        return EPHEMERAL_RUNS.load(Ordering::SeqCst) > 0 || map.values().any(|w| w.running.load(Ordering::SeqCst));
205    }
206    find_warm(&map, chat_id, topic_id, session_id).await.is_some_and(|w| w.running.load(Ordering::SeqCst))
207}

209/// Returns `true` if a live non-expired warm `claude` session is awaiting `AskUserQuestion` input.
210pub async fn has_pending_ask(chat_id: i64, topic_id: Option<i64>, session_id: Option<&str>) -> bool {
211    let warm = {
212        let map = global_pool().sessions.lock().await;
213        if chat_id != 0 || topic_id.is_some() {
214            map.get(&(chat_id, topic_id)).cloned()
215        } else {
216            find_warm(&map, 0, None, session_id).await
217        }
218    };
219    let Some(w) = warm else { return false; };
220    !w.activity.is_expired() && w.is_alive().await && w.pending_ask.lock().await.is_some()
221}
