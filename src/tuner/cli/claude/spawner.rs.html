22    pub(crate) child: Child,
23    pub(crate) stdout: Lines<BufReader<ChildStdout>>,

26impl ClaudeProcessIo {
27    pub(crate) async fn drain_after_interrupt(&mut self) {
28        let deadline = tokio::time::Instant::now() + tokio::time::Duration::from_millis(2500);
29        while let Ok(Ok(Some(line))) = tokio::time::timeout_at(deadline, self.stdout.next_line()).await {
30            if line.contains("\"type\":\"result\"") {
31                break;
32            }
33        }
34    }
35}

37/// Persistent warm `claude` process bound to a `(chat_id, topic_id)` conversation.
38pub struct WarmClaudeSession {
39    /// OS process ID of the warm `claude` child.
40    pub pid: u32,
41    /// Active Claude session UUID once discovered.
42    pub session_id: Mutex<Option<String>>,
43    /// Active canonical model on this warm process.
44    pub model: Mutex<Option<String>>,
45    /// Active reasoning effort level on this warm process.
46    pub effort: Mutex<Option<String>>,
47    /// Workspace path where the warm process was started.
48    pub workspace: PathBuf,
49    /// 24-hour activity tracker matching Antigravity PTY session expiration.
50    pub activity: WarmActivityTracker,
51    pub(crate) pending_ask: Mutex<Option<(String, serde_json::Value)>>,
52    pub(crate) queued_ask_answers: Mutex<Option<Vec<String>>>,
53    pub(crate) stdin: Mutex<ChildStdin>,
54    pub(crate) io: Mutex<ClaudeProcessIo>,
55    pub(crate) running: AtomicBool,
56    pub(crate) interrupted: AtomicBool,
57    pub(crate) interrupt_notify: Notify,
58}
59
60impl WarmClaudeSession {
61    pub(crate) async fn is_alive(&self) -> bool {
62        self.io.try_lock().map(|mut io| io.child.try_wait().map(|s| s.is_none()).unwrap_or(false)).unwrap_or(true)
63    }
64
65    pub(crate) async fn write_json_line(&self, payload: &serde_json::Value) -> Result<(), String> {
66        self.activity.touch();
67        write_json_stdin(&mut *self.stdin.lock().await, payload).await
68    }
69
70    pub(crate) fn kill_now(&self) {
71        if self.pid > 0 { kill_process_group(self.pid); }
72    }
73}

83pub(crate) fn spawn_warm_claude(
84    cli: &AntigravityCli,
85    ws: PathBuf,
86    resume: Option<&str>,
87    cont: bool,
88) -> Result<Arc<WarmClaudeSession>, String> {
89    let args = build_warm_command(cli, resume, cont);
90    let mut cmd = Command::new(&args[0]);
91    cmd.args(&args[1..])
92        .current_dir(&ws)
93        .envs(cli.build_env())
94        .stdin(Stdio::piped())
95        .stdout(Stdio::piped())
96        .stderr(Stdio::null())
97        .kill_on_drop(true);
98    #[cfg(unix)]
99    cmd.process_group(0);
100    let mut child = cmd
101        .spawn()
102        .map_err(|e| format!("Failed to spawn warm claude process: {e}"))?;
103    let pid = child.id().unwrap_or(0);
104    let stdin = child
105        .stdin
106        .take()
107        .ok_or_else(|| "Failed to open claude stdin".to_string())?;
108    let stdout = child
109        .stdout
110        .take()
111        .ok_or_else(|| "Failed to open claude stdout".to_string())?;
112    let (init_m, init_e) =
113        resolve_claude_model_effort(cli.config.model.as_deref(), cli.config.effort.as_deref());
114    Ok(Arc::new(WarmClaudeSession {
115        pid,
116        session_id: Mutex::new(resume.map(str::to_string)),
117        model: Mutex::new(init_m),
118        effort: Mutex::new(init_e),
119        workspace: ws,
120        activity: WarmActivityTracker::new(),
121        pending_ask: Mutex::new(None),
122        queued_ask_answers: Mutex::new(None),
123        stdin: Mutex::new(stdin),
124        io: Mutex::new(ClaudeProcessIo {
125            child,
126            stdout: BufReader::new(stdout).lines(),
127        }),
128        running: AtomicBool::new(false),
129        interrupted: AtomicBool::new(false),
130        interrupt_notify: Notify::new(),
131    }))
132}

134pub(crate) async fn prepare_claude_turn_payload(
135    warm: &WarmClaudeSession,
136    cli: &AntigravityCli,
137    prompt: &str,
138) -> serde_json::Value {
139    if prompt.is_empty()
140        && let Some((req_id, input)) = warm.pending_ask.lock().await.take()
141    {
142        let ans = warm.queued_ask_answers.lock().await.take().unwrap_or_default();
143        return crate::cli::common::build_claude_ask_control_response(&req_id, input, &ans);
144    }
145    serde_json::json!({
146        "type": "user",
147        "message": { "role": "user", "content": cli.format_prompt(prompt) }
148    })
149}
