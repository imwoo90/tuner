19async fn drain_after_interrupt(io: &mut ClaudeProcessIo) {
20    let deadline = tokio::time::Instant::now() + tokio::time::Duration::from_millis(2500);
21    while let Ok(Ok(Some(line))) = tokio::time::timeout_at(deadline, io.stdout.next_line()).await {
22        if line.contains("\"type\":\"result\"") {
23            break;
24        }
25    }
26}

28async fn handle_claude_control_request(
29    warm: &WarmClaudeSession,
30    line: &str,
31    parser: &mut ClaudeStreamParser,
32    tx: &tokio::sync::mpsc::UnboundedSender<StreamEvent>,
33) -> bool {
34    let Ok(val) = serde_json::from_str::<serde_json::Value>(line.trim()) else { return false; };
35    if val.get("type").and_then(|v| v.as_str()) != Some("control_request") {
36        return false;
37    }
38    let req_id = val.get("request_id").and_then(|v| v.as_str()).unwrap_or("").to_string();
39    let req = val.get("request").unwrap_or(&serde_json::Value::Null);
40    if req.get("subtype").and_then(|v| v.as_str()) == Some("can_use_tool") {
41        let tool = req.get("tool_name").and_then(|v| v.as_str()).unwrap_or("");
42        let input = req.get("input").cloned().unwrap_or_else(|| serde_json::json!({}));
43        if tool == "AskUserQuestion" {
44            let questions = crate::cli::common::parse_claude_ask_questions(&input);
45            if !questions.is_empty() {
46                *warm.pending_ask.lock().await = Some((req_id, input));
47                let _ = tx.send(StreamEvent::AskQuestion(questions));
48                let resp = CliResponse {
49                    session_id: parser.session_id.clone(),
50                    result: String::new(),
51                    is_error: false,
52                    returncode: Some(0),
53                    stderr: String::new(),
54                };
55                parser.final_result = Some(resp.clone());
56                let _ = tx.send(StreamEvent::Result(resp));
57                return true;
58            }
59        }
60        let allow = serde_json::json!({
61            "type": "control_response",
62            "response": {
63                "subtype": "success",
64                "request_id": req_id,
65                "response": { "behavior": "allow", "updatedInput": input }
66            }
67        });
68        let _ = warm.write_json_line(&allow).await;
69    }
70    true
71}

73async fn read_claude_turn_stream(
74    warm: &Arc<WarmClaudeSession>,
75    io: &mut ClaudeProcessIo,
76    parser: &mut ClaudeStreamParser,
77    tx: &tokio::sync::mpsc::UnboundedSender<StreamEvent>,
78) {
79    loop {
80        if warm.interrupted.load(Ordering::SeqCst) {
81            break;
82        }
83        tokio::select! {
84            biased;
85            _ = warm.interrupt_notify.notified() => break,
86            _ = tx.closed() => {
87                let ctrl = serde_json::json!({
88                    "type": "control_request",
89                    "request_id": "stop-drop",
90                    "request": { "subtype": "interrupt" }
91                });
92                let _ = warm.write_json_line(&ctrl).await;
93                warm.interrupted.store(true, Ordering::SeqCst);
94                break;
95            }
96            line_res = io.stdout.next_line() => {
97                let Ok(Some(line)) = line_res else { break; };
98                if handle_claude_control_request(warm, &line, parser, tx).await {
99                    if parser.final_result.is_some() { break; }
100                    continue;
101                }
102                for ev in parser.parse_line(&line) { let _ = tx.send(ev); }
103                if let Some(ref sid) = parser.session_id {
104                    *warm.session_id.lock().await = Some(sid.clone());
105                }
106                if parser.final_result.is_some() { break; }
107            }
108        }
109    }
110}

112async fn finalize_claude_turn(
113    cli: &AntigravityCli,
114    warm: &WarmClaudeSession,
115    io: &mut ClaudeProcessIo,
116    parser: ClaudeStreamParser,
117    resume_sid: Option<String>,
118    tx: &tokio::sync::mpsc::UnboundedSender<StreamEvent>,
119) {
120    let _ = tokio::time::timeout(std::time::Duration::ZERO, warm.interrupt_notify.notified()).await;
121    if warm.interrupted.swap(false, Ordering::SeqCst) {
122        let _ = tx.send(StreamEvent::Result(CliResponse {
123            session_id: parser.session_id.or(resume_sid),
124            result: String::new(), is_error: true, returncode: Some(130),
125            stderr: "Interrupted by user (/stop)".to_string(),
126        }));
127        drain_after_interrupt(io).await;
128    } else if parser.final_result.is_none() {
129        let _ = terminate_topic(cli.config.chat_id, cli.config.topic_id, None).await;
130        let _ = tx.send(StreamEvent::Result(CliResponse {
131            session_id: parser.session_id.or(resume_sid),
132            result: parser.assistant_text, is_error: true, returncode: Some(1),
133            stderr: "Warm claude process exited unexpectedly".to_string(),
134        }));
135    }
136}

138async fn prepare_claude_turn_payload(warm: &WarmClaudeSession, cli: &AntigravityCli, prompt: &str) -> serde_json::Value {
139    if prompt.is_empty()
140        && let Some((req_id, input)) = warm.pending_ask.lock().await.take()
141    {
142        let ans = warm.queued_ask_answers.lock().await.take().unwrap_or_default();
143        return crate::cli::common::build_claude_ask_control_response(&req_id, input, &ans);
144    }
145    serde_json::json!({
146        "type": "user",
147        "message": { "role": "user", "content": cli.format_prompt(prompt) }
148    })
149}

151async fn execute_warm_claude_turn(
152    cli: AntigravityCli,
153    prompt: String,
154    workspace: PathBuf,
155    resume_sid: Option<String>,
156    continue_session: bool,
157    tx: tokio::sync::mpsc::UnboundedSender<StreamEvent>,
158) {
159    let warm = match get_or_spawn_warm_session(&cli, workspace, resume_sid.as_deref(), continue_session).await {
160        Ok(w) => w,
161        Err(msg) => {
162            let _ = tx.send(StreamEvent::Result(CliResponse {
163                session_id: resume_sid, result: msg.clone(), is_error: true, returncode: Some(1), stderr: msg,
164            }));
165            return;
166        }
167    };
168    let mut io = warm.io.lock().await;
169    let _ = tokio::time::timeout(std::time::Duration::ZERO, warm.interrupt_notify.notified()).await;
170    warm.interrupted.store(false, Ordering::SeqCst);
171    warm.running.store(true, Ordering::SeqCst);
172    let outbound = prepare_claude_turn_payload(&warm, &cli, &prompt).await;
173    if let Err(err) = warm.write_json_line(&outbound).await {
174        warm.running.store(false, Ordering::SeqCst);
175        let _ = terminate_topic(cli.config.chat_id, cli.config.topic_id, None).await;
176        let _ = tx.send(StreamEvent::Result(CliResponse {
177            session_id: resume_sid, result: err.clone(), is_error: true, returncode: Some(1), stderr: err,
178        }));
179        return;
180    }
181    let mut parser = ClaudeStreamParser { session_id: resume_sid.clone(), ..Default::default() };
182    read_claude_turn_stream(&warm, &mut io, &mut parser, &tx).await;
183    warm.running.store(false, Ordering::SeqCst);
184    finalize_claude_turn(&cli, &warm, &mut io, parser, resume_sid, &tx).await;
185}

187/// Streams a `claude` CLI turn using the topic's persistent warm process.
188pub async fn send_streaming<'a>(
189    cli: &'a AntigravityCli,
190    prompt: &str,
191    resume_session: Option<&str>,
192    continue_session: bool,
193    workspace: PathBuf,
194) -> Result<BoxStream<'a, StreamEvent>, String> {
195    let (tx, rx) = tokio::sync::mpsc::unbounded_channel();
196    tokio::spawn(execute_warm_claude_turn(
197        cli.clone(), prompt.to_string(), workspace, resume_session.map(str::to_string), continue_session, tx,
198    ));
199    Ok(Box::pin(futures::stream::unfold(rx, |mut rx| async move { rx.recv().await.map(|ev| (ev, rx)) })))
200}

202/// Runs a `claude` CLI turn to completion on the persistent warm process and returns `CliResponse`.
203pub async fn send(
204    cli: &AntigravityCli,
205    prompt: &str,
206    resume_session: Option<&str>,
207    continue_session: bool,
208    workspace: PathBuf,
209) -> Result<CliResponse, String> {
210    use futures::StreamExt;
211    let mut stream = send_streaming(cli, prompt, resume_session, continue_session, workspace).await?;
212    let mut last_result = None;
213    while let Some(ev) = stream.next().await {
214        if let StreamEvent::Result(resp) = ev { last_result = Some(resp); }
215    }
216    last_result.ok_or_else(|| "Claude CLI stream ended without a result".to_string())
217}
