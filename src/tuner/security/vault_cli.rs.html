1//! # Zero-Knowledge Credential Vault CLI Runner
2//!
3//! ## Overview
4//! Executes `--vault` CLI subcommands (`list`, `exec`, `session`) for AI agents and scripts.
5//! Ensures plaintext passwords never enter LLM transcripts by injecting credentials strictly
6//! through child process environment variables and scrubbing output streams.

16/// Ephemeral Web Vault session record persisted across worker restarts (15-minute TTL).
17#[derive(Clone, Debug, serde::Serialize, serde::Deserialize)]
18pub struct VaultWebSession {
19    /// Creation timestamp in Unix seconds.
20    pub created_at_secs: u64,
21    /// Target profile name.
22    pub profile: String,
23    /// Target profile vault directory.
24    pub vault_dir: PathBuf,
25}
26
27/// Loads persisted `vault_sessions.json` from `vault_dir`.
28pub fn load_persisted_sessions(vault_dir: &Path) -> HashMap<String, VaultWebSession> {
29    let f = vault_dir.join("vault_sessions.json");
30    std::fs::read_to_string(f)
31        .ok()
32        .and_then(|raw| serde_json::from_str(&raw).ok())
33        .unwrap_or_default()
34}
35
36/// Saves `vault_sessions.json` into `vault_dir`.
37pub fn save_persisted_sessions(vault_dir: &Path, map: &HashMap<String, VaultWebSession>) {
38    let _ = std::fs::create_dir_all(vault_dir);
39    if let Ok(s) = serde_json::to_string(map) {
40        let _ = std::fs::write(vault_dir.join("vault_sessions.json"), s);
41    }
42}

44/// Detects the active profile name from CLI args, `TUNER_PROFILE`, or current working directory.
45pub fn resolve_cli_profile(args: &[String]) -> String {
46    if let Some(pos) = args.iter().position(|a| a == "--profile")
47        && let Some(val) = args.get(pos + 1)
48        && !val.trim().is_empty()
49    {
50        return val.trim().to_string();
51    }
52    if let Ok(env_prof) = std::env::var("TUNER_PROFILE")
53        && !env_prof.trim().is_empty()
54    {
55        return env_prof.trim().to_string();
56    }
57    if let Ok(cwd) = std::env::current_dir()
58        && let Some(detected) = extract_profile_from_path(&cwd)
59    {
60        return detected;
61    }
62    "default".to_string()
63}

65/// Extracts `<profile>` if `path` contains `.../profiles/<profile>/...`.
66pub fn extract_profile_from_path(path: &Path) -> Option<String> {
67    let comps: Vec<String> = path
68        .components()
69        .map(|c| c.as_os_str().to_string_lossy().to_string())
70        .collect();
71    for i in 0..comps.len() {
72        if comps[i] == "profiles"
73            && let Some(next) = comps.get(i + 1)
74            && !next.is_empty()
75        {
76            return Some(next.clone());
77        }
78    }
79    None
80}

82/// Redacts sensitive secret strings from command output text.
83pub fn redact_secrets(text: &str, secrets: &[&str]) -> String {
84    let mut out = text.to_string();
85    for secret in secrets {
86        let trimmed = secret.trim();
87        if trimmed.len() >= 3 {
88            out = out.replace(trimmed, "***REDACTED***");
89        }
90    }
91    out
92}

94fn open_profile_store(profile: &str) -> VaultStore {
95    let paths = resolve_paths(None, None, None, Some(profile.to_string()));
96    VaultStore::new(paths.vault_dir())
97}

99fn print_vault_help() {
100    println!(
101        "Usage:
102  tuner --vault list [--profile <name>]
103  tuner --vault exec <site_key> [--profile <name>] -- <command> [args...]
104
105Environment variables injected during `exec`:
106  TUNER_VAULT_SITE   Normalized site key (e.g. github)
107  TUNER_VAULT_TITLE  Service display name
108  TUNER_VAULT_URL    Login or service URL
109  TUNER_VAULT_USER   Account username / email
110  TUNER_VAULT_PASS   Account password (auto-redacted from stdout/stderr)"
111    );
112}

114fn handle_vault_list(profile: &str) -> Result<(), String> {
115    let store = open_profile_store(profile);
116    let masked = store.list_masked()?;
117    if masked.is_empty() {
118        println!("🔐 [Vault: {}] No site credentials registered yet. Run `/vault` in Telegram to add credentials.", profile);
119        return Ok(());
120    }
121    println!("🔐 [Vault: {}] {} registered site(s):", profile, masked.len());
122    for item in &masked {
123        let url_str = if item.url.is_empty() { "-" } else { &item.url };
124        println!(
125            "  • [{}] {} ({}) | user: {}",
126            item.site_key, item.title, url_str, item.masked_username
127        );
128    }
129    Ok(())
130}

132fn handle_vault_exec(profile: &str, site_key: &str, cmd_args: &[String]) -> Result<(), String> {
133    if cmd_args.is_empty() {
134        return Err("Missing command after `--`. Usage: tuner --vault exec <site_key> -- <command> [args...]".to_string());
135    }
136    let store = open_profile_store(profile);
137    let entry = store
138        .get_entry(site_key)?
139        .ok_or_else(|| format!("Site '{}' not found in profile '{}' vault", site_key, profile))?;
140
141    let mut cmd = std::process::Command::new(&cmd_args[0]);
142    if cmd_args.len() > 1 {
143        cmd.args(&cmd_args[1..]);
144    }
145    cmd.env("TUNER_VAULT_SITE", &entry.site_key)
146        .env("TUNER_VAULT_TITLE", &entry.title)
147        .env("TUNER_VAULT_URL", &entry.url)
148        .env("TUNER_VAULT_USER", &entry.username)
149        .env("TUNER_VAULT_PASS", &entry.password);
150
151    let output = cmd
152        .output()
153        .map_err(|e| format!("Failed to execute '{}': {}", cmd_args[0], e))?;
154
155    let secrets = [entry.password.as_str()];
156    let stdout_str = String::from_utf8_lossy(&output.stdout);
157    let stderr_str = String::from_utf8_lossy(&output.stderr);
158    let clean_out = redact_secrets(&stdout_str, &secrets);
159    let clean_err = redact_secrets(&stderr_str, &secrets);
160
161    if !clean_out.is_empty() {
162        print!("{}", clean_out);
163    }
164    if !clean_err.is_empty() {
165        eprint!("{}", clean_err);
166    }
167    if !output.status.success() {
168        let code = output.status.code().unwrap_or(1);
169        return Err(format!("Command exited with code {}", code));
170    }
171    Ok(())
172}

174fn handle_vault_session(profile: &str) -> Result<(), String> {
175    let store = open_profile_store(profile);
176    let dir = store.dir().to_path_buf();
177    let tok = hex::encode(rand::random::<[u8; 16]>());
178    let mut map = load_persisted_sessions(&dir);
179    map.insert(
180        tok.clone(),
181        VaultWebSession {
182            created_at_secs: current_unix_secs(),
183            profile: profile.to_string(),
184            vault_dir: dir.clone(),
185        },
186    );
187    save_persisted_sessions(&dir, &map);
188    println!("{}", tok);
189    Ok(())
190}

192/// Entrypoint for `tuner --vault ...` CLI invocation.
193pub fn run_vault_cli(args: &[String]) -> Result<(), String> {
194    let vault_pos = match args.iter().position(|a| a == "--vault") {
195        Some(p) => p,
196        None => return Ok(()),
197    };
198    let profile = resolve_cli_profile(args);
199    let sub = args.get(vault_pos + 1).map(String::as_str).unwrap_or("help");
200
201    match sub {
202        "list" | "ls" => handle_vault_list(&profile),
203        "session" | "token" => handle_vault_session(&profile),
204        "exec" | "run" => {
205            let site = args
206                .get(vault_pos + 2)
207                .ok_or_else(|| "Missing <site_key>. Usage: tuner --vault exec <site_key> -- <cmd>".to_string())?;
208            let sep_pos = args
209                .iter()
210                .position(|a| a == "--")
211                .ok_or_else(|| "Missing `--` separator before command".to_string())?;
212            let cmd_slice = if sep_pos + 1 < args.len() {
213                &args[sep_pos + 1..]
214            } else {
215                &[]
216            };
217            handle_vault_exec(&profile, site, cmd_slice)
218        }
219        _ => {
220            print_vault_help();
221            Ok(())
222        }
223    }
