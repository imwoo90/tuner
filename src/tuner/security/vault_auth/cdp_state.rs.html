1//! # Playwright-Compatible `storage_state.json` Persistence (`cdp_state.rs`)
2//!
3//! ## Overview

5//! browser cookies plus page `localStorage` origins in Playwright's exact
6//! `{"cookies": [...], "origins": [...]}` schema (`chmod 0600`).

8//! ## Search Tags

18/// Summary of an exported Playwright `storage_state.json` snapshot.
19#[derive(Clone, Debug, Serialize, Deserialize)]
20pub struct ExportedStateSummary {
21    /// Total number of cookies persisted.
22    pub cookie_count: usize,
23    /// Total number of `localStorage` origins captured.
24    pub origin_count: usize,
25    /// Last active page URL at snapshot time.
26    pub final_url: String,
27}

29/// Validates that a user/CLI navigation URL uses strictly `http://` or `https://` (`file://` forbidden).
30///
31/// # Examples
32/// ```
33/// use tuner::security::vault_auth::cdp_state::validate_http_url;
34/// assert!(validate_http_url("https://www.aliexpress.com/").is_ok());
35/// assert!(validate_http_url("file:///etc/passwd").is_err());
36/// ```
37pub fn validate_http_url(raw: &str) -> Result<String, String> {
38    let trimmed = raw.trim();
39    let parsed = url::Url::parse(trimmed).map_err(|e| format!("Invalid URL '{trimmed}': {e}"))?;
40    match parsed.scheme() {
41        "http" | "https" => Ok(parsed.to_string()),
42        other => Err(format!(
43            "Forbidden URL scheme '{other}://': only http:// and https:// are permitted."
44        )),
45    }
46}

48/// Loads saved cookies from an existing Playwright `storage_state.json` into Chrome via CDP.
49pub async fn seed_existing_storage_state(browser_ws_url: &str, state_path: &Path) {
50    let Ok(raw) = std::fs::read_to_string(state_path) else {
51        return;
52    };
53    let Ok(parsed) = serde_json::from_str::<Value>(&raw) else {
54        return;
55    };
56    let Some(cookies) = parsed.get("cookies").and_then(|c| c.as_array()) else {
57        return;
58    };
59    if cookies.is_empty() {
60        return;
61    }
62    let mut cdp_cookies = Vec::with_capacity(cookies.len());
63    for c in cookies {
64        let Some(name) = c.get("name").and_then(Value::as_str) else {
65            continue;
66        };
67        let Some(val) = c.get("value").and_then(Value::as_str) else {
68            continue;
69        };
70        let Some(domain) = c.get("domain").and_then(Value::as_str) else {
71            continue;
72        };
73        let mut obj = json!({
74            "name": name,
75            "value": val,
76            "domain": domain,
77            "path": c.get("path").and_then(Value::as_str).unwrap_or("/"),
78            "secure": c.get("secure").and_then(Value::as_bool).unwrap_or(false),
79            "httpOnly": c.get("httpOnly").and_then(Value::as_bool).unwrap_or(false),
80        });
81        if let Some(exp) = c.get("expires").and_then(Value::as_f64)
82            && exp > 0.0
83            && let Some(map) = obj.as_object_mut()
84        {
85            map.insert("expires".to_string(), json!(exp));
86        }
87        if let Some(ss) = c.get("sameSite").and_then(Value::as_str)
88            && matches!(ss, "Strict" | "Lax" | "None")
89            && let Some(map) = obj.as_object_mut()
90        {
91            map.insert("sameSite".to_string(), json!(ss));
92        }
93        cdp_cookies.push(obj);
94    }
95    if !cdp_cookies.is_empty() {
96        let _ = cdp_call_ws(
97            browser_ws_url,
98            "Storage.setCookies",
99            json!({ "cookies": cdp_cookies }),
100        )
101        .await;
102    }
103}

105fn normalize_playwright_cookie(raw: &Value) -> Option<Value> {
106    let name = raw.get("name")?.as_str()?;
107    let value = raw.get("value")?.as_str()?;
108    let domain = raw.get("domain")?.as_str()?;
109    let path = raw.get("path").and_then(Value::as_str).unwrap_or("/");
110    let expires = raw.get("expires").and_then(Value::as_f64).unwrap_or(-1.0);
111    let http_only = raw.get("httpOnly").and_then(Value::as_bool).unwrap_or(false);
112    let secure = raw.get("secure").and_then(Value::as_bool).unwrap_or(false);
113    let same_site = match raw.get("sameSite").and_then(Value::as_str) {
114        Some("Strict") => "Strict",
115        Some("None") => "None",
116        _ => "Lax",
117    };
118    Some(json!({
119        "name": name,
120        "value": value,
121        "domain": domain,
122        "path": path,
123        "expires": if expires <= 0.0 { -1.0 } else { expires },
124        "httpOnly": http_only,
125        "secure": secure,
126        "sameSite": same_site,
127    }))
128}

130async fn collect_open_page_origins(cdp_port: u16) -> (Vec<Value>, String) {
131    let targets = list_page_targets(cdp_port).await;
132    let final_url = targets
133        .first()
134        .map(|t| t.url.clone())
135        .unwrap_or_default();
136    let mut seen = HashSet::new();
137    let mut origins = Vec::new();
138    let expr = r#"(() => {
139        try {
140            const origin = window.location.origin;
141            if (!origin || origin === 'null' || !origin.startsWith('http')) return null;
142            const ls = [];
143            for (let i = 0; i < window.localStorage.length; i++) {
144                const k = window.localStorage.key(i);
145                if (k !== null) ls.push({ name: k, value: String(window.localStorage.getItem(k) ?? '') });
146            }
147            return ls.length ? { origin, localStorage: ls } : null;
148        } catch (_) { return null; }
149    })()"#;
150
151    for t in targets {
152        if !t.url.starts_with("http://") && !t.url.starts_with("https://") {
153            continue;
154        }
155        let Ok(res) = cdp_call_ws(
156            &t.ws_url,
157            "Runtime.evaluate",
158            json!({ "expression": expr, "returnByValue": true }),
159        )
160        .await
161        else {
162            continue;
163        };
164        let val = &res["result"]["value"];
165        if let Some(origin) = val.get("origin").and_then(Value::as_str)
166            && seen.insert(origin.to_string())
167        {
168            origins.push(val.clone());
169        }
170    }
171    (origins, final_url)
172}

174/// Exports cookies and `localStorage` origins to `state_path` (`0600`) in Playwright format.
175pub async fn export_storage_state(
176    cdp_port: u16,
177    browser_ws_url: &str,
178    state_path: &Path,
179) -> Result<ExportedStateSummary, String> {
180    let raw_cookies = cdp_call_ws(browser_ws_url, "Storage.getCookies", json!({})).await?;
181    let cookies: Vec<Value> = raw_cookies
182        .get("cookies")
183        .and_then(Value::as_array)
184        .map(|arr| arr.iter().filter_map(normalize_playwright_cookie).collect())
185        .unwrap_or_default();
186
187    let (origins, final_url) = collect_open_page_origins(cdp_port).await;
188    let doc = json!({
189        "cookies": cookies,
190        "origins": origins,
191    });
192    if let Some(parent) = state_path.parent() {
193        let _ = std::fs::create_dir_all(parent);
194    }
195    let serialized = serde_json::to_string_pretty(&doc).map_err(|e| e.to_string())?;
196    let tmp = state_path.with_extension("json.tmp");
197    std::fs::write(&tmp, serialized)
198        .map_err(|e| format!("Failed to write storage_state tmp {:?}: {}", tmp, e))?;
199    set_mode_0600(&tmp);
200    std::fs::rename(&tmp, state_path)
201        .map_err(|e| format!("Failed to commit storage_state {:?}: {}", state_path, e))?;
202    set_mode_0600(state_path);
203
204    Ok(ExportedStateSummary {
205        cookie_count: cookies.len(),
206        origin_count: origins.len(),
207        final_url,
208    })
209}
