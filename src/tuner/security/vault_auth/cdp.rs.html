1//! # Pure Rust Chrome DevTools Protocol Controller (`cdp.rs`)
2//!
3//! ## Overview
4//! Drives the active MRU Chrome page target (`type == "page"`) over a persistent loopback CDP
5//! WebSocket session—preserving `Emulation.setDeviceMetricsOverride` (`760 × 980` desktop layout)
6//! and `Page.addScriptToEvaluateOnNewDocument` across navigations and popups while eliminating
7//! ephemeral socket churn, blocking `file://` URLs, and routing input via [`super::cdp_input`].
8//!
9//! ## Search Tags
10//! #cdp-controller, #persistent-cdp-session, #mru-tab-tracking, #screencast-jpeg

70async fn send_rpc_on_stream(
71    ws: &mut CdpWsStream,
72    req_id: u64,
73    method: &str,
74    params: Value,
75) -> Result<Value, String> {
76    let fut = async {
77        let req = json!({ "id": req_id, "method": method, "params": params });
78        ws.send(Message::Text(req.to_string()))
79            .await
80            .map_err(|e| format!("__BROKEN__:CDP send error: {e}"))?;
81        while let Some(msg) = ws.next().await {
82            let msg = msg.map_err(|e| format!("__BROKEN__:CDP recv error: {e}"))?;
83            match msg {
84                Message::Ping(p) => {
85                    let _ = ws.send(Message::Pong(p)).await;
86                }
87                Message::Close(_) => return Err(format!("__BROKEN__:CDP closed ({method})")),
88                Message::Text(txt) => {
89                    if let Ok(val) = serde_json::from_str::<Value>(&txt)
90                        && val.get("id").and_then(Value::as_u64) == Some(req_id)
91                    {
92                        if let Some(err) = val.get("error") {
93                            return Err(format!("CDP {method} error: {err}"));
94                        }
95                        return Ok(val.get("result").cloned().unwrap_or(Value::Null));
96                    }
97                }
98                _ => {}
99            }
100        }
101        Err(format!("__BROKEN__:CDP EOF ({method})"))
102    };
103    tokio::time::timeout(Duration::from_secs(6), fut)
104        .await
105        .map_err(|_| format!("__BROKEN__:CDP {method} timed out"))?
106}

108/// Sends a single one-off JSON-RPC command to a CDP WebSocket URL (used for browser/export queries).
109pub async fn cdp_call_ws(ws_url: &str, method: &str, params: Value) -> Result<Value, String> {
110    let (mut ws, _) = tokio::time::timeout(Duration::from_secs(5), connect_async(ws_url))
111        .await
112        .map_err(|_| format!("CDP connect timeout ({method})"))?
113        .map_err(|e| format!("CDP connect error: {e}"))?;
114    let res = send_rpc_on_stream(&mut ws, 1, method, params)
115        .await
116        .map_err(|e| e.trim_start_matches("__BROKEN__:").to_string());
117    let _ = ws.close(None).await;
118    res
119}

121async fn open_configured_page_conn(target: &CdpPageTarget) -> Result<ActivePageConn, String> {
122    let (mut ws, _) = tokio::time::timeout(Duration::from_secs(5), connect_async(&target.ws_url))
123        .await
124        .map_err(|_| "CDP page connect timeout".to_string())?
125        .map_err(|e| format!("CDP page connect error: {e}"))?;
126    let stealth_js = "Object.defineProperty(navigator, 'webdriver', {get: () => undefined});";
127    let _ = send_rpc_on_stream(&mut ws, 1, "Page.enable", json!({})).await;
128    let _ = send_rpc_on_stream(
129        &mut ws,
130        2,
131        "Page.addScriptToEvaluateOnNewDocument",
132        json!({ "source": stealth_js }),
133    )
134    .await;
135    let _ = send_rpc_on_stream(
136        &mut ws,
137        3,
138        "Runtime.evaluate",
139        json!({ "expression": stealth_js }),
140    )
141    .await;
142    let _ = send_rpc_on_stream(
143        &mut ws,
144        4,
145        "Emulation.setDeviceMetricsOverride",
146        json!({
147            "width": VIEWPORT_WIDTH,
148            "height": VIEWPORT_HEIGHT,
149            "deviceScaleFactor": 1,
150            "mobile": false,
151        }),
152    )
153    .await;
154    Ok(ActivePageConn {
155        target_id: target.id.clone(),
156        ws,
157        next_id: 10,
158    })
159}

161/// Queries `http://127.0.0.1:<cdp_port>/json/list` and returns user `"page"` targets in MRU order.
162pub async fn list_page_targets(cdp_port: u16) -> Vec<CdpPageTarget> {
163    let url = format!("http://127.0.0.1:{cdp_port}/json/list");
164    let Ok(client) = reqwest::Client::builder()
165        .timeout(Duration::from_secs(2))
166        .build()
167    else {
168        return Vec::new();
169    };
170    let Ok(resp) = client.get(&url).send().await else {
171        return Vec::new();
172    };
173    let Ok(all) = resp.json::<Vec<CdpPageTarget>>().await else {
174        return Vec::new();
175    };
176    let pages: Vec<CdpPageTarget> = all
177        .into_iter()
178        .filter(|t| t.target_type == "page" && !t.ws_url.is_empty())
179        .collect();
180    let non_internal: Vec<CdpPageTarget> = pages
181        .iter()
182        .filter(|t| !t.url.starts_with("chrome://") && !t.url.starts_with("devtools://"))
183        .cloned()
184        .collect();
185    if non_internal.is_empty() {
186        pages
187    } else {
188        non_internal
189    }
190}

217    /// Executes a CDP command over the persistent WebSocket bound to `target`.
218    pub async fn call_page(
219        &self,
220        target: &CdpPageTarget,
221        method: &str,
222        params: Value,
223    ) -> Result<Value, String> {
224        let mut guard = self.active_conn.lock().await;
225        if guard.as_ref().map(|c| c.target_id.as_str()) != Some(target.id.as_str()) {
226            *guard = Some(open_configured_page_conn(target).await?);
227        }
228        let conn = guard.as_mut().ok_or_else(|| "Missing CDP conn".to_string())?;
229        let req_id = conn.next_id;
230        conn.next_id = conn.next_id.saturating_add(1);
231        match send_rpc_on_stream(&mut conn.ws, req_id, method, params).await {
232            Ok(val) => Ok(val),
233            Err(err) => {
234                if err.starts_with("__BROKEN__:") {
235                    *guard = None;
236                }
237                Err(err.trim_start_matches("__BROKEN__:").to_string())
238            }
239        }
240    }
241
242    async fn resolve_active_page(&self) -> Result<(CdpPageTarget, usize), String> {
243        let pages = list_page_targets(self.cdp_port).await;
244        let count = pages.len();
245        let target = pages
246            .into_iter()
247            .next()
248            .ok_or_else(|| "No active browser page tab available".to_string())?;
249        if target.url.starts_with("file://") {
250            let _ = self
251                .call_page(&target, "Page.navigate", json!({ "url": "about:blank" }))
252                .await;
253            return Err("Blocked forbidden file:// navigation".to_string());
254        }
255        Ok((target, count))
256    }

258    /// Configures the initial tab (`760 × 980` + stealth) and navigates to `start_url`.
259    pub async fn navigate_initial(&self, start_url: &str) -> Result<(), String> {
260        let safe_url = validate_http_url(start_url)?;
261        let (target, _) = self.resolve_active_page().await?;
262        let _ = self
263            .call_page(&target, "Page.navigate", json!({ "url": safe_url }))
264            .await?;
265        Ok(())
266    }

268    /// Captures a `quality: 68` JPEG frame from the active MRU page tab.
269    pub async fn capture_frame(&self) -> Result<FrameSnapshot, String> {
270        let (target, tab_count) = self.resolve_active_page().await?;
271        let res = self
272            .call_page(
273                &target,
274                "Page.captureScreenshot",
275                json!({ "format": "jpeg", "quality": 68 }),
276            )
277            .await?;
278        let b64 = res
279            .get("data")
280            .and_then(Value::as_str)
281            .ok_or_else(|| "Missing screenshot base64 payload".to_string())?;
282        let jpeg_bytes = base64::engine::general_purpose::STANDARD
283            .decode(b64)
284            .map_err(|e| format!("Screenshot base64 decode error: {e}"))?;
285        Ok(FrameSnapshot {
286            jpeg_bytes,
287            url: target.url,
288            tab_count,
289        })
290    }

292    /// Executes a single user UI action JSON message from the remote viewer WebSocket.
293    pub async fn dispatch_action(&self, action: &Value) -> Result<Option<String>, String> {
294        let act = action.get("type").and_then(Value::as_str).unwrap_or("");
295        if act == "cycle_tab" {
296            return cycle_browser_tab(self.cdp_port).await;
297        }
298        let (target, _) = self.resolve_active_page().await?;
299        match act {
300            "mousedown" | "mousemove" | "mouseup" | "wheel" => {
301                dispatch_pointer_or_wheel(self, &target, act, action).await?;
302                Ok(None)
303            }
304            "text" | "vault_user" | "vault_pass" => {
305                dispatch_text_or_credential(
306                    self,
307                    &target,
308                    act,
309                    action,
310                    self.vault_username.as_deref(),
311                    self.vault_password.as_deref(),
312                )
313                .await
314            }
315            _ => dispatch_nav_or_key(self, &target, act, action).await,
316        }
317    }
