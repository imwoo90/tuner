1//! # Xvfb & Chrome Persistent Profile Process Runtime (`browser.rs`)
2//!
3//! ## Overview
4//! Launches an isolated `Xvfb -displayfd 1` X11 framebuffer and headed Chrome process group
5//! using [`super::browser_env`], awaits `DevToolsActivePort`, and enforces zero-orphan teardown
6//! via process-group signals and a safe parent-PID watchdog.
7//!
8//! ## Search Tags
9//! #xvfb-launcher, #chrome-cdp-spawner, #persistent-profile, #orphan-reaper
10

20/// Viewport width in CSS pixels optimized for mobile portrait thumb ergonomics while keeping desktop layout.
21pub const VIEWPORT_WIDTH: u32 = 760;
22/// Viewport height in CSS pixels for the remote authentication canvas.
23pub const VIEWPORT_HEIGHT: u32 = 980;

25/// Active `Xvfb` + Chrome child process handles and discovered CDP WebSocket endpoint.
26pub struct BrowserRuntime {
27    xvfb_child: Child,
28    xvfb_pid: Option<u32>,
29    chrome_child: Child,
30    chrome_pid: Option<u32>,
31    watchdog_child: Option<Child>,
32    /// Local loopback CDP HTTP port (`127.0.0.1:<cdp_port>`).
33    pub cdp_port: u16,
34    /// Browser-level CDP WebSocket URL (`ws://127.0.0.1:<cdp_port>/devtools/browser/<id>`).
35    pub browser_ws_url: String,
36}

38async fn spawn_xvfb() -> Result<(Child, String), String> {
39    let xvfb_bin = which_bin("Xvfb").ok_or_else(|| {
40        "Xvfb not found in PATH. Run `tuner --doctor --fix` or install `xvfb`.".to_string()
41    })?;
42    let mut cmd = Command::new(xvfb_bin);
43    cmd.args([
44        "-displayfd",
45        "1",
46        "-screen",
47        "0",
48        "800x1100x24",
49        "-nolisten",
50        "tcp",
51        "-ac",
52    ])
53    .stdout(Stdio::piped())
54    .stderr(Stdio::null())
55    .process_group(0)
56    .kill_on_drop(true);
57
58    let mut child = cmd
59        .spawn()
60        .map_err(|e| format!("Failed to spawn Xvfb: {e}"))?;
61    let stdout = child
62        .stdout
63        .take()
64        .ok_or_else(|| "Failed to capture Xvfb stdout".to_string())?;
65    let mut line = String::new();
66    let res =
67        tokio::time::timeout(Duration::from_secs(6), BufReader::new(stdout).read_line(&mut line))
68            .await;
69    if let Ok(Ok(n)) = res
70        && n > 0
71    {
72        let disp = line.trim();
73        if !disp.is_empty() && disp.chars().all(|c| c.is_ascii_digit()) {
74            return Ok((child, format!(":{disp}")));
75        }
76    }
77    let _ = child.kill().await;
78    Err("Timed out or failed waiting for Xvfb -displayfd readiness".to_string())
79}

81fn spawn_chrome_process(
82    chrome_bin: &Path,
83    display: &str,
84    profile_dir: &Path,
85    initial_url: &str,
86) -> Result<Child, String> {
87    let mut cmd = Command::new(chrome_bin);
88    cmd.env("DISPLAY", display)
89        .env("XDG_CACHE_HOME", profile_dir.join(".xdg_cache"))
90        .env("DBUS_SESSION_BUS_ADDRESS", "/dev/null")
91        .args([
92            "--disable-blink-features=AutomationControlled",
93            "--password-store=basic",
94            "--use-mock-keychain",
95            "--disable-dev-shm-usage",
96            "--no-first-run",
97            "--no-default-browser-check",
98            "--disable-search-engine-choice-screen",
99            "--disable-infobars",
100            "--disable-popup-blocking",
101            "--allow-file-access-from-files=false",
102            "--lang=ko-KR",
103            "--remote-debugging-port=0",
104            "--remote-debugging-address=127.0.0.1",
105            &format!("--window-size={},{}", VIEWPORT_WIDTH, VIEWPORT_HEIGHT + 90),
106            &format!("--user-data-dir={}", profile_dir.display()),
107        ]);
108    if needs_no_sandbox() {
109        cmd.arg("--no-sandbox");
110    }
111    cmd.arg(initial_url)
112        .stdin(Stdio::null())
113        .stdout(Stdio::null())
114        .stderr(Stdio::null())
115        .process_group(0)
116        .kill_on_drop(true)
117        .spawn()
118        .map_err(|e| format!("Failed to launch Chrome ({:?}): {}", chrome_bin, e))
119}

121async fn await_devtools_port(profile_dir: &Path) -> Result<(u16, String), String> {
122    let active_port_file = profile_dir.join("DevToolsActivePort");
123    let start = Instant::now();
124    while start.elapsed() <= Duration::from_secs(12) {
125        if let Ok(raw) = std::fs::read_to_string(&active_port_file) {
126            let mut lines = raw.lines();
127            if let (Some(p_str), Some(path_str)) = (lines.next(), lines.next())
128                && let Ok(port) = p_str.trim().parse::<u16>()
129                && port > 0
130                && path_str.trim().starts_with('/')
131            {
132                return Ok((port, path_str.trim().to_string()));
133            }
134        }
135        tokio::time::sleep(Duration::from_millis(120)).await;
136    }
137    Err("Timed out waiting for Chrome DevToolsActivePort".to_string())
138}

140fn spawn_parent_death_watchdog(pgids: &[u32]) -> Option<Child> {
141    let ppid = std::process::id();
142    let targets: Vec<String> = pgids.iter().map(|p| format!("-{p}")).collect();
143    if targets.is_empty() {
144        return None;
145    }
146    let grp_list = targets.join(" ");
147    let script = format!(
148        "while kill -0 {ppid} 2>/dev/null; do sleep 1; done; \
149         kill -TERM {grp_list} 2>/dev/null || true; sleep 1; \
150         kill -KILL {grp_list} 2>/dev/null || true"
151    );
152    Command::new("sh")
153        .args(["-c", &script])
154        .stdin(Stdio::null())
155        .stdout(Stdio::null())
156        .stderr(Stdio::null())
157        .kill_on_drop(true)
158        .spawn()
159        .ok()
160}

162impl BrowserRuntime {
163    /// Launches `Xvfb` + headed Chrome against `profile_dir` and awaits CDP readiness.
164    pub async fn launch(profile_dir: &Path, initial_url: &str) -> Result<Self, String> {
165        prepare_profile_dir(profile_dir)?;
166        let chrome_bin = resolve_browser_binary()?;
167        let (mut xvfb_child, display) = spawn_xvfb().await?;
168        let xvfb_pid = xvfb_child.id();
169
170        let mut chrome_child =
171            match spawn_chrome_process(&chrome_bin, &display, profile_dir, initial_url) {
172                Ok(c) => c,
173                Err(e) => {
174                    if let Some(pid) = xvfb_pid {
175                        kill_process_group(pid);
176                    }
177                    let _ = xvfb_child.kill().await;
178                    return Err(e);
179                }
180            };
181        let chrome_pid = chrome_child.id();
182
183        let (cdp_port, ws_path) = match await_devtools_port(profile_dir).await {
184            Ok(v) => v,
185            Err(e) => {
186                if let Some(pid) = chrome_pid {
187                    kill_process_group(pid);
188                }
189                if let Some(pid) = xvfb_pid {
190                    kill_process_group(pid);
191                }
192                let _ = chrome_child.kill().await;
193                let _ = xvfb_child.kill().await;
194                return Err(e);
195            }
196        };
197
198        let pgids: Vec<u32> = [chrome_pid, xvfb_pid].into_iter().flatten().collect();
199        let watchdog_child = spawn_parent_death_watchdog(&pgids);
200
201        Ok(Self {
202            xvfb_child,
203            xvfb_pid,
204            chrome_child,
205            chrome_pid,
206            watchdog_child,
207            cdp_port,
208            browser_ws_url: format!("ws://127.0.0.1:{cdp_port}{ws_path}"),
209        })
210    }
211
212    /// Terminates Chrome, `Xvfb`, and the parent-death watchdog cleanly.
213    pub async fn shutdown(&mut self) {
214        if let Some(mut wd) = self.watchdog_child.take() {
215            let _ = wd.kill().await;
216            let _ = wd.wait().await;
217        }
218        let _ = tokio::time::timeout(Duration::from_millis(1500), self.chrome_child.wait()).await;
219        if let Some(pid) = self.chrome_pid {
220            kill_process_group(pid);
221        }
222        let _ = self.chrome_child.kill().await;
223        let _ = self.chrome_child.wait().await;
224
225        if let Some(pid) = self.xvfb_pid {
226            kill_process_group(pid);
227        }
228        let _ = self.xvfb_child.kill().await;
229        let _ = self.xvfb_child.wait().await;
230    }
231}
