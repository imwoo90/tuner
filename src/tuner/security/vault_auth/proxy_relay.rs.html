1//! # Public `:8743` Viewer HTTP & WebSocket Reverse Proxy (`proxy_relay.rs`)
2//!
3//! ## Overview
4//! Serves the public token-gated `/auth/:token` HTML page and `/auth/:token/ws` binary JPEG
5//! WebSocket relay on the warm `:8743` [`crate::messenger::telegram::review_server`] router,
6//! forwarding strictly to verified `127.0.0.1` ports registered in [`super::proxy`].
7//!
8//! ## Search Tags
9//! #vault-auth-relay, #websocket-proxy, #review-server-auth
10

26/// Builds ONLY the public token-gated viewer routes merged into `:8743` `review_server`.
27pub fn auth_proxy_routes() -> Router {
28    Router::new()
29        .route("/auth/:token", get(handle_proxy_page))
30        .route("/auth/:token/ws", get(handle_proxy_ws))
31}
32
33async fn handle_proxy_page(Path(token): Path<String>) -> Response {
34    let Some(port) = resolve_route_port(&token).await else {
35        return (
36            StatusCode::NOT_FOUND,
37            "Remote auth session expired or completed.",
38        )
39            .into_response();
40    };
41    let url = format!("http://127.0.0.1:{port}/auth/{token}");
42    let Ok(client) = reqwest::Client::builder()
43        .redirect(reqwest::redirect::Policy::none())
44        .timeout(Duration::from_secs(5))
45        .build()
46    else {
47        return StatusCode::INTERNAL_SERVER_ERROR.into_response();
48    };
49    match client.get(&url).send().await {
50        Ok(resp) if resp.status().is_success() => {
51            let body = resp.text().await.unwrap_or_default();
52            let mut out = Html(body).into_response();
53            if let Ok(hv) = header::HeaderValue::from_str("no-store, no-cache, must-revalidate") {
54                out.headers_mut().insert(header::CACHE_CONTROL, hv);
55            }
56            out
57        }
58        Ok(resp) => StatusCode::from_u16(resp.status().as_u16())
59            .unwrap_or(StatusCode::BAD_GATEWAY)
60            .into_response(),
61        Err(_) => (StatusCode::BAD_GATEWAY, "Upstream auth server closed").into_response(),
62    }
63}
64
65async fn handle_proxy_ws(Path(token): Path<String>, ws: WebSocketUpgrade) -> Response {
66    let Some(port) = resolve_route_port(&token).await else {
67        return (StatusCode::NOT_FOUND, "Remote auth session expired").into_response();
68    };
69    ws.max_message_size(8 * 1024 * 1024)
70        .on_upgrade(move |client_ws| relay_websocket(client_ws, port, token))
71}
72
73async fn relay_websocket(client_ws: WebSocket, port: u16, token: String) {
74    let upstream_url = format!("ws://127.0.0.1:{port}/auth/{token}/ws");
75    let Ok((upstream_ws, _)) = tokio_tungstenite::connect_async(&upstream_url).await else {
76        return;
77    };
78    let (mut client_tx, mut client_rx) = client_ws.split();
79    let (mut up_tx, mut up_rx) = upstream_ws.split();
80
81    let c2u = async {
82        while let Some(Ok(msg)) = client_rx.next().await {
83            let mapped = match msg {
84                AxumWsMsg::Text(t) => TungMsg::Text(t.to_string()),
85                AxumWsMsg::Binary(b) => TungMsg::Binary(b.to_vec()),
86                AxumWsMsg::Ping(p) => TungMsg::Ping(p.to_vec()),
87                AxumWsMsg::Pong(p) => TungMsg::Pong(p.to_vec()),
88                AxumWsMsg::Close(_) => break,
89            };
90            if up_tx.send(mapped).await.is_err() {
91                break;
92            }
93        }
94    };
95
96    let u2c = async {
97        while let Some(Ok(msg)) = up_rx.next().await {
98            let mapped = match msg {
99                TungMsg::Text(t) => AxumWsMsg::Text(t),
100                TungMsg::Binary(b) => AxumWsMsg::Binary(b),
101                TungMsg::Ping(p) => AxumWsMsg::Ping(p),
102                TungMsg::Pong(p) => AxumWsMsg::Pong(p),
103                TungMsg::Close(_) => break,
104                TungMsg::Frame(_) => continue,
105            };
106            if client_tx.send(mapped).await.is_err() {
107                break;
108            }
109        }
110    };
111
112    tokio::select! {
113        _ = c2u => {}
114        _ = u2c => {}
115    }
116}
