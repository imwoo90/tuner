17/// Determine whether the local time of the given timezone is within the quiet hours window.
18pub fn is_within_quiet_hours(
19    now: &DateTime<Tz>,
20    quiet_start: NaiveTime,
21    quiet_end: NaiveTime,
22) -> bool {
23    if quiet_start == quiet_end {
24        return false; // same start/end means never quiet
25    }
26
27    let current_local_time = NaiveTime::from_hms_opt(
28        now.hour(),
29        now.minute(),
30        now.second(),
31    )
32    .unwrap_or_default();
33
34    if quiet_start < quiet_end {
35        current_local_time >= quiet_start && current_local_time < quiet_end
36    } else {
37        current_local_time >= quiet_start || current_local_time < quiet_end
38    }
39}

41/// Check if the agent's heartbeat response is equivalent to the acknowledge token.
42pub fn should_suppress_heartbeat(output: &str, ack_token: &str) -> bool {
43    let clean_output = output.trim();
44    let clean_ack = ack_token.trim();
45    clean_output == clean_ack
46}

53pub(crate) fn should_deliver_heartbeat_response(
54    chat_id: i64,
55    topic_id: Option<i64>,
56    start_sid: &str,
57    ack_token: &str,
58    resp: &crate::cli::CliResponse,
59) -> bool {
60    if resp.is_error {
61        eprintln!(
62            "⚠️ [tuner] Heartbeat CLI error for chat {} topic {:?}: {}",
63            chat_id, topic_id, resp.stderr
64        );
65        return false;
66    }
67    if !start_sid.is_empty()
68        && resp
69            .session_id
70            .as_deref()
71            .is_some_and(|s| !s.is_empty() && s != start_sid)
72    {
73        eprintln!(
74            "⚠️ [tuner] Heartbeat ignored mismatched session_id for chat {} topic {:?}",
75            chat_id, topic_id
76        );
77        return false;
78    }
79    !resp.result.trim().is_empty() && !should_suppress_heartbeat(&resp.result, ack_token)
80}

82pub(crate) fn reconcile_heartbeat_session(
83    latest: &mut crate::session::data::SessionData,
84    prov_key: &str,
85    start_sid: &str,
86    resp_sid: Option<&str>,
87    active: &mut bool,
88) -> bool {
89    let latest_sid = latest.get_session_id(prov_key);
90    let was_reset = (!start_sid.is_empty()
91        && (latest.get_past_session_ids(prov_key).iter().any(|p| p == start_sid)
92            || latest_sid != start_sid))
93        || (start_sid.is_empty() && !latest_sid.is_empty() && Some(latest_sid.as_str()) != resp_sid);
94    if was_reset {
95        return false;
96    }
97    *active = true;
98    if start_sid.is_empty()
99        && let Some(new_sid) = resp_sid.filter(|s| !s.is_empty())
100        && latest_sid != new_sid
101    {
102        latest.set_session_id(prov_key, new_sid);
103        return true;
104    }
105    false
106}
