79    /// Evaluates if the current tick falls into quiet hours and should be skipped.
80    pub fn should_skip_tick(&self, now: &DateTime<Tz>) -> bool {
81        if !self.heartbeat_enabled() {
82            return true;
83        }
84
85        if let (Some(start_h), Some(end_h)) = (
86            self.heartbeat_quiet_start(),
87            self.heartbeat_quiet_end(),
88        ) {
89            let start = NaiveTime::from_hms_opt(start_h % 24, 0, 0).unwrap_or_default();
90            let end = NaiveTime::from_hms_opt(end_h % 24, 0, 0).unwrap_or_default();
91            return super::quiet::is_within_quiet_hours(now, start, end);
92        }
93
94        false
95    }

97    /// Checks if a given chat/topic session has active running CLI processes or open `ask_question` prompts.
98    pub async fn is_session_busy(&self, chat_id: i64, topic_id: Option<i64>) -> bool {
99        if self.cli.is_topic_turn_in_flight(chat_id, topic_id)
100            || self.cli.sessions.boot_cancels.lock().await.contains_key(&(chat_id, topic_id))
101        {
102            return true;
103        }
104        let key = crate::session::key::SessionKey::telegram(chat_id, topic_id);
105        if let Ok(Some(sess)) = self.sessions.get_active(&key).await {
106            let prov = self.effective_provider(&sess);
107            let sid = sess.get_session_id(prov);
108            let opt_sid = (!sid.is_empty()).then_some(sid.as_str());
109            if !sid.is_empty() && self.cli.is_pty_busy_or_asking(&sid, chat_id, topic_id).await {
110                return true;
111            }
112            return match prov {
113                "claude" => {
114                    crate::cli::claude::is_running(chat_id, topic_id, opt_sid).await
115                        || crate::cli::claude::has_pending_ask(chat_id, topic_id, opt_sid).await
116                }
117                "codex" => {
118                    crate::cli::codex::is_running(chat_id, topic_id, opt_sid).await
119                        || crate::cli::codex::has_pending_ask(chat_id, topic_id, opt_sid).await
120                }
121                _ => !sid.is_empty() && self.cli.sessions.is_active(&sid).await,
122            };
123        }
124        false
125    }

127    fn effective_provider<'a>(&'a self, session: &'a SessionData) -> &'static str {
128        let raw = if session.provider.trim().is_empty() {
129            self.config.provider.as_str()
130        } else {
131            session.provider.as_str()
132        };
133        SessionData::canonical_provider(raw).unwrap_or("antigravity")
134    }

136    /// Returns `true` only when `session` already has a persisted conversation ID on its active
137    /// provider and has not exceeded idle, daily-reset, or message-count freshness limits.
138    pub(crate) fn has_fresh_active_session(&self, session: &SessionData) -> bool {
139        let eff_prov = self.effective_provider(session);
140        !session.get_session_id(eff_prov).is_empty()
141            && self.sessions.is_fresh_for_provider(session, eff_prov)
142    }

144    fn record_heartbeat_timestamp(&self, session: &SessionData) {
145        let mut map = self.last_heartbeat_at.lock().unwrap_or_else(|e| e.into_inner());
146        map.insert(session.session_key().storage_key(), Utc::now());
147    }
148
149    /// Evaluates if the session activity or last heartbeat is still within the cooldown threshold.
150    pub fn is_cooling_down(&self, session: &SessionData) -> bool {
151        let cooldown = chrono::Duration::minutes(self.heartbeat_interval_minutes());
152        let now = Utc::now();
153        if let Ok(last) = DateTime::parse_from_rfc3339(&session.last_active)
154            && now.signed_duration_since(last.with_timezone(&Utc)) < cooldown
155        {
156            return true;
157        }
158        let map = self.last_heartbeat_at.lock().unwrap_or_else(|e| e.into_inner());
159        map.get(&session.session_key().storage_key())
160            .is_some_and(|last_hb| now.signed_duration_since(*last_hb) < cooldown)
161    }

163    /// Starts the background interval check.
164    pub fn start(self: Arc<Self>) {
165        let interval_mins = self.heartbeat_interval_minutes() as u64;
166        let scheduler = self.clone();
167        tokio::spawn(async move {
168            let mut interval = tokio::time::interval(tokio::time::Duration::from_secs(interval_mins * 60));
169            interval.tick().await;
170
171            loop {
172                interval.tick().await;
173                let _ = scheduler.tick().await;
174            }
175        });
176    }

178    /// Perform a single tick check over all persisted sessions.
179    pub async fn tick(&self) -> Result<(), String> {
180        let user_tz = self.config.user_timezone.as_deref().unwrap_or("UTC");
181        let tz = user_tz.parse::<Tz>().unwrap_or(Tz::UTC);
182        let now_local = Utc::now().with_timezone(&tz);
183
184        if self.should_skip_tick(&now_local) {
185            return Ok(());
186        }
187
188        let all_sessions = self.sessions.load()?;
189        for (_, session) in all_sessions {
190            if session.transport != "tg" || !self.has_fresh_active_session(&session) {
191                continue;
192            }
193
194            if self.is_cooling_down(&session) || self.is_session_busy(session.chat_id, session.topic_id).await {
195                continue;
196            }
197
198            let _ = self.run_heartbeat_for_chat(&session).await;
199        }
200
201        Ok(())
202    }

208    pub(crate) async fn finish_heartbeat_turn(
209        &self,
210        cur: &SessionData,
211        eff_prov: &str,
212        start_sid: &str,
213        resp: &crate::cli::CliResponse,
214    ) {
215        self.record_heartbeat_timestamp(cur);
216        let ack = self.heartbeat_ack_token();
217        if !super::should_deliver_heartbeat_response(cur.chat_id, cur.topic_id, start_sid, &ack, resp) {
218            return;
219        }
220        let prov_key = SessionData::canonical_provider(eff_prov).unwrap_or("antigravity");
221        let fallback = start_sid.is_empty().then_some(cur);
222        let mut active = false;
223        let _ = self
224            .sessions
225            .mutate_session(&cur.session_key(), fallback, |latest| {
226                super::reconcile_heartbeat_session(
227                    latest,
228                    prov_key,
229                    start_sid,
230                    resp.session_id.as_deref(),
231                    &mut active,
232                )
233            })
234            .await;
235        if active {
236            crate::upgrade::record_user_activity();
237            let eff_transport = if self.config.transport == "web" { "web" } else { cur.transport.as_str() };
238            let mut env = crate::bus::adapters::from_heartbeat(cur.chat_id, &resp.result, cur.topic_id, Some(eff_transport));
239            self.bus.submit(&mut env).await;
240        }
241    }

243    async fn run_heartbeat_for_chat(&self, session: &SessionData) -> Result<(), String> {
244        let lock = self.sessions.lock_pool.get((session.chat_id, session.topic_id));
245        let Ok(_topic_guard) = lock.try_lock() else { return Ok(()); };
246        if self.is_session_busy(session.chat_id, session.topic_id).await {
247            return Ok(());
248        }
249        let Some(cur) = self.sessions.get_active(&session.session_key()).await.ok().flatten() else {
250            return Ok(());
251        };
252        if !self.has_fresh_active_session(&cur) || self.is_cooling_down(&cur) {
253            return Ok(());
254        }
255        self.record_heartbeat_timestamp(&cur);
256        let _bg_guard = crate::upgrade::begin_background_run();
257        let _turn_guard = self.cli.begin_topic_turn(cur.chat_id, cur.topic_id);
258        let prompt = "System self-check: are there any outstanding issues or background alerts to report? Answer 'HEARTBEAT_OK' if everything is running smoothly.";
259        let eff_prov = self.effective_provider(&cur);
260        if !matches!(eff_prov, "claude" | "codex") {
261            crate::cli::antigravity::model_spec::ensure_live_catalog_fresh().await;
262        }
263        let sid = cur.get_session_id(eff_prov);
264        let cli = AntigravityCli::new_oneshot(self.build_heartbeat_cli_config(&cur));
265        match cli.send(prompt, Some(&sid), false, self.config.working_dir.clone()).await {
266            Ok(resp) => self.finish_heartbeat_turn(&cur, eff_prov, &sid, &resp).await,
267            Err(e) => eprintln!("⚠️ [tuner] Heartbeat send failed for chat {} topic {:?}: {e}", cur.chat_id, cur.topic_id),
268        }
269        Ok(())
270    }
