1//! # Cron Engine Event Loop
2//!
3//! ## Overview
4//! Continuously polls active cron jobs, evaluates timezone-adjusted upcoming runs, checks quiet hours,
5//! and triggers background agent runs via [`AntigravityCli`].
6//!
7//! ## Collaboration Graph
8//! - Ticked by [`CronScheduler::start`] event loop spawning background monitoring threads.
9//! - Queries [`super::manager::CronManager`] to retrieve active [`CronJob`] structures.
10//! - Utilizes [`AntigravityCli`] to run prompt instructions.
11//! - Dispatches results as envelopes to the central [`MessageBus`](crate::bus::bus::MessageBus).
12//!
13//! ## Key Types
14//! - [`CronScheduler`]: Controller holding configuration, client bindings, message bus adapters, and quiet-hour status.
15//!
16//! ## Search Tags
17//! #cron-engine, #tick-loop, #quiet-hours, #task-spawner, #time-evaluator

49    fn normalize_cron_expression(expr: &str) -> String {
50        if expr.split_whitespace().count() == 5 { format!("0 {}", expr) } else { expr.to_string() }
51    }
52
53    pub(crate) fn calculate_next_run(&self, job: &CronJob) -> Result<DateTime<Utc>, String> {
54        let norm = Self::normalize_cron_expression(&job.schedule);
55        let sched = Schedule::from_str(&norm).map_err(|e| e.to_string())?;
56        let tz_s = if job.timezone.is_empty() { "UTC" } else { &job.timezone };
57        let tz: Tz = tz_s.parse().map_err(|e| format!("Invalid timezone: {}", e))?;
58        let next = sched.upcoming(tz).next().ok_or_else(|| "No upcoming run".to_string())?;
59        Ok(next.with_timezone(&Utc))
60    }

62    pub fn start(self: Arc<Self>) {
63        let scheduler = self.clone();
64        tokio::spawn(async move {
65            let mut last_mtime: Option<SystemTime> = None;
66            let mut next_run_times: HashMap<String, DateTime<Utc>> = HashMap::new();
67            let mut loaded_jobs_configs: HashMap<String, (String, String)> = HashMap::new();
68            let mut check_interval = tokio::time::interval(tokio::time::Duration::from_secs(5));
69            println!("🤖 [tuner] CronScheduler started");
70            loop {
71                check_interval.tick().await;
72                let _ = scheduler.tick(&mut last_mtime, &mut next_run_times, &mut loaded_jobs_configs).await;
73            }
74        });
75    }
76
77    async fn tick(
78        &self,
79        last_mtime: &mut Option<SystemTime>,
80        next_run_times: &mut HashMap<String, DateTime<Utc>>,
81        loaded_jobs_configs: &mut HashMap<String, (String, String)>,
82    ) -> Result<(), String> {
83        let _ = self.reload_if_changed(last_mtime, next_run_times, loaded_jobs_configs).await;
84        let _ = self.run_due_jobs(next_run_times).await;
85        Ok(())
86    }
87
88    async fn reload_if_changed(
89        &self,
90        last_mtime: &mut Option<SystemTime>,
91        next_run_times: &mut HashMap<String, DateTime<Utc>>,
92        loaded_jobs_configs: &mut HashMap<String, (String, String)>,
93    ) -> Result<(), String> {
94        let current_mtime = std::fs::metadata(self.manager.jobs_path()).ok().and_then(|m| m.modified().ok());
95        if current_mtime != *last_mtime {
96            *last_mtime = current_mtime;
97            println!("🤖 [tuner] CronScheduler: reloading jobs");
98            if let Ok(jobs) = self.manager.list_jobs().await {
99                let mut updated_times = HashMap::new();
100                let mut updated_configs = HashMap::new();
101                for job in jobs {
102                    if job.enabled {
103                        let config_key = (job.schedule.clone(), job.timezone.clone());
104                        let has_changed = match loaded_jobs_configs.get(&job.id) {
105                            Some(old_cfg) => old_cfg != &config_key,
106                            None => true,
107                        };
108
109                        if let Ok(next) = self.calculate_next_run(&job) {
110                            let next_run = if has_changed {
111                                next
112                            } else {
113                                next_run_times.get(&job.id).filter(|&&e| e > Utc::now()).copied().unwrap_or(next)
114                            };
115                            updated_times.insert(job.id.clone(), next_run);
116                            updated_configs.insert(job.id, config_key);
117                        }
118                    }
119                }
120                *next_run_times = updated_times;
121                *loaded_jobs_configs = updated_configs;
122            }
123        }
124        Ok(())
125    }

127    async fn run_due_jobs(&self, next_run_times: &mut HashMap<String, DateTime<Utc>>) -> Result<(), String> {
128        let now = Utc::now();
129        let mut jobs_to_run = Vec::new();
130        for (job_id, next_run) in next_run_times.iter() {
131            if now >= *next_run { jobs_to_run.push(job_id.clone()); }
132        }
133        for job_id in jobs_to_run {
134            if let Ok(Some(job)) = self.manager.get_job(&job_id).await {
135                let self_clone = Arc::new(Self::new(self.config.clone(), self.manager.clone(), self.cli.clone(), self.bus.clone()));
136                let job_for_task = job.clone();
137                tokio::spawn(async move { let _ = self_clone.execute_job(job_for_task).await; });
138                if let Ok(next_run) = self.calculate_next_run(&job) {
139                    next_run_times.insert(job_id, next_run);
140                } else {
141                    next_run_times.remove(&job_id);
142                }
143            }
144        }
145        Ok(())
146    }

148    fn check_quiet_hours(&self, job: &CronJob) -> bool {
149        self.check_quiet_hours_at(job, Utc::now())
150    }

152    pub fn check_quiet_hours_at(&self, job: &CronJob, now: DateTime<Utc>) -> bool {
153        if let (Some(qs), Some(qe)) = (job.quiet_start, job.quiet_end) {
154            let tz_str = if job.timezone.is_empty() { "UTC" } else { &job.timezone };
155            if let Ok(tz) = tz_str.parse::<Tz>() {
156                let now_local = now.with_timezone(&tz);
157                let start = NaiveTime::from_hms_opt(qs, 0, 0).unwrap();
158                let end = NaiveTime::from_hms_opt(qe, 0, 0).unwrap();
159                return crate::heartbeat::quiet::is_within_quiet_hours(&now_local, start, end);
160            }
161        }
162        false
163    }

184    async fn run_cli_command(&self, job: &CronJob, workspace: std::path::PathBuf, cli: &AntigravityCli) -> Result<(), String> {
185        let job_id = job.id.clone();
186        let job_title = job.title.clone();
187        let chat_id = job.chat_id;
188        let topic_id = job.topic_id;
189        let enriched = self.get_enriched_prompt(&job.agent_instruction, &job.task_folder);
190        let res = cli.send(&enriched, None, false, workspace).await;
191        match res {
192            Ok(resp) => {
193                let status = if resp.is_error {
194                    format!("error:exit_{}", resp.returncode.unwrap_or(1))
195                } else {
196                    "success".to_string()
197                };
198                self.manager.update_run_status(&job_id, &status).await?;
199                if !job.silent_on_success || resp.is_error {
200                    self.submit_cron_result(&job_title, &resp.result, &status, chat_id, topic_id).await;
201                }
202            }
203            Err(e) => {
204                let err_status = format!("error:{}", e);
205                self.manager.update_run_status(&job_id, &err_status).await?;
206                self.submit_cron_result(&job_title, "", &err_status, chat_id, topic_id).await;
207            }
208        }
209        Ok(())
210    }

212    async fn execute_job(&self, job: CronJob) -> Result<(), String> {
213        if self.check_quiet_hours(&job) {
214            println!("🤖 [tuner] Cron job {} skipped due to quiet hours", job.title);
215            return Ok(());
216        }
217        println!("🤖 [tuner] Cron job executing: {}", job.title);
218        let ctr = self.config.working_dir.join("cron_tasks");
219        let workspace = ctr.join(&job.task_folder);
220        if !crate::security::paths::is_path_safe(&workspace, &[ctr]) {
221            let _ = self.manager.update_run_status(&job.id, "error:path_outside_allowed_roots").await;
222            return Err("Cron task folder outside allowed roots".into());
223        }
224        if !workspace.is_dir() {
225            let err_msg = format!("Cron task folder missing: {}", workspace.display());
226            eprintln!("❌ [tuner] {}", err_msg);
227            self.manager.update_run_status(&job.id, "error:folder_missing").await?;
228            return Err(err_msg);
229        }
230
231        let mut jc = (*self.config).clone();
232        jc.chat_id = job.chat_id;
233        jc.topic_id = job.topic_id;
234        jc.transport = job.transport.clone();
235        if let Some(m) = &job.model { jc.model = Some(m.clone()); }
236        if let Some(p) = &job.provider { jc.provider = p.clone(); }
237        if !job.cli_parameters.is_empty() {
238            jc.cli_parameters.insert("antigravity".to_string(), job.cli_parameters.clone());
239        }
240        if let Some(r) = &job.reasoning_effort {
241            jc.cli_parameters.entry("antigravity".to_string()).or_default()
242                .extend(vec!["-c".to_string(), format!("model_reasoning_effort={}", r)]);
243        }
244
245        let job_cli = AntigravityCli::new(jc);
246        self.run_cli_command(&job, workspace, &job_cli).await
247    }
