1//! # Cron Scheduler Trigger and Evaluation Engine
2//!
3//! ## Overview
4//! Handles the cron evaluation tick loop, job file modification watching, timezone-adjusted next run calculations, and task dispatch logic.
5//!
6//! ## Search Tags
7//! #cron-trigger, #tick-loop, #next-run-evaluator
8
9use std::collections::HashMap;
10use std::str::FromStr;
11use std::sync::Arc;
12use std::time::SystemTime;
13use chrono::{DateTime, Utc};
14use chrono_tz::Tz;
15use cron::Schedule;
16
17use super::CronScheduler;
18use crate::cron::manager::CronJob;

20impl CronScheduler {
21    fn normalize_cron_expression(expr: &str) -> String {
22        if expr.split_whitespace().count() == 5 { format!("0 {}", expr) } else { expr.to_string() }
23    }
24
25    pub fn calculate_next_run(&self, job: &CronJob) -> Result<DateTime<Utc>, String> {
26        let norm = Self::normalize_cron_expression(&job.schedule);
27        let sched = Schedule::from_str(&norm).map_err(|e| e.to_string())?;
28        let tz_s = if job.timezone.is_empty() { "UTC" } else { &job.timezone };
29        let tz: Tz = tz_s.parse().map_err(|e| format!("Invalid timezone: {}", e))?;
30        let next = sched.upcoming(tz).next().ok_or_else(|| "No upcoming run".to_string())?;
31        Ok(next.with_timezone(&Utc))
32    }
33
34    pub fn start(self: Arc<Self>) {
35        let scheduler = self.clone();
36        tokio::spawn(async move {
37            let mut last_mtime: Option<SystemTime> = None;
38            let mut next_run_times: HashMap<String, DateTime<Utc>> = HashMap::new();
39            let mut loaded_jobs_configs: HashMap<String, (String, String)> = HashMap::new();
40            let mut check_interval = tokio::time::interval(tokio::time::Duration::from_secs(5));
41            println!("🤖 [tuner] CronScheduler started");
42            loop {
43                check_interval.tick().await;
44                let _ = scheduler.tick(&mut last_mtime, &mut next_run_times, &mut loaded_jobs_configs).await;
45            }
46        });
47    }
48
49    pub(super) async fn tick(
50        &self,
51        last_mtime: &mut Option<SystemTime>,
52        next_run_times: &mut HashMap<String, DateTime<Utc>>,
53        loaded_jobs_configs: &mut HashMap<String, (String, String)>,
54    ) -> Result<(), String> {
55        let _ = self.reload_if_changed(last_mtime, next_run_times, loaded_jobs_configs).await;
56        let _ = self.run_due_jobs(next_run_times).await;
57        Ok(())
58    }
59
60    pub(super) async fn reload_if_changed(
61        &self,
62        last_mtime: &mut Option<SystemTime>,
63        next_run_times: &mut HashMap<String, DateTime<Utc>>,
64        loaded_jobs_configs: &mut HashMap<String, (String, String)>,
65    ) -> Result<(), String> {
66        let current_mtime = std::fs::metadata(self.manager.jobs_path()).ok().and_then(|m| m.modified().ok());
67        if current_mtime != *last_mtime {
68            *last_mtime = current_mtime;
69            println!("🤖 [tuner] CronScheduler: reloading jobs");
70            if let Ok(jobs) = self.manager.list_jobs().await {
71                let mut updated_times = HashMap::new();
72                let mut updated_configs = HashMap::new();
73                for job in jobs {
74                    if job.enabled {
75                        let config_key = (job.schedule.clone(), job.timezone.clone());
76                        let has_changed = match loaded_jobs_configs.get(&job.id) {
77                            Some(old_cfg) => old_cfg != &config_key,
78                            None => true,
79                        };
80
81                        if let Ok(next) = self.calculate_next_run(&job) {
82                            let next_run = if has_changed {
83                                next
84                            } else {
85                                next_run_times.get(&job.id).filter(|&&e| e > Utc::now()).copied().unwrap_or(next)
86                            };
87                            updated_times.insert(job.id.clone(), next_run);
88                            updated_configs.insert(job.id, config_key);
89                        }
90                    }
91                }
92                *next_run_times = updated_times;
93                *loaded_jobs_configs = updated_configs;
94            }
95        }
96        Ok(())
97    }
98
99    pub(super) async fn run_due_jobs(&self, next_run_times: &mut HashMap<String, DateTime<Utc>>) -> Result<(), String> {
100        let now = Utc::now();
101        let mut jobs_to_run = Vec::new();
102        for (job_id, next_run) in next_run_times.iter() {
103            if now >= *next_run { jobs_to_run.push(job_id.clone()); }
104        }
105        for job_id in jobs_to_run {
106            if let Ok(Some(job)) = self.manager.get_job(&job_id).await {
107                let self_clone = Arc::new(Self::new(self.config.clone(), self.manager.clone(), self.cli.clone(), self.bus.clone()));
108                let job_for_task = job.clone();
109                tokio::spawn(async move { let _ = self_clone.execute_job(job_for_task).await; });
110                if let Ok(next_run) = self.calculate_next_run(&job) {
111                    next_run_times.insert(job_id, next_run);
112                } else {
113                    next_run_times.remove(&job_id);
114                }
115            }
116        }
117        Ok(())
118    }
