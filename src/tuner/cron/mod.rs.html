25/// Computes the next timezone-aware execution timestamp for a 5- or 6-field cron expression.
26pub fn calculate_job_next_run(schedule: &str, timezone: &str) -> Result<DateTime<Tz>, String> {
27    let norm = if schedule.split_whitespace().count() == 5 {
28        format!("0 {}", schedule)
29    } else {
30        schedule.to_string()
31    };
32    let sched = Schedule::from_str(&norm).map_err(|e| e.to_string())?;
33    let tz_s = if timezone.is_empty() { "UTC" } else { timezone };
34    let tz: Tz = tz_s.parse().map_err(|e| format!("Invalid timezone: {}", e))?;
35    sched
36        .upcoming(tz)
37        .next()
38        .ok_or_else(|| "No upcoming run".to_string())
39}

41pub(crate) fn build_job_cli_config(
42    base: &crate::config::CliConfig,
43    jobs_path: &Path,
44    job: &CronJob,
45    workspace: std::path::PathBuf,
46) -> crate::config::CliConfig {
47    let mut jc = base.clone();
48    if let Some(prof_dir) = jobs_path.parent() {
49        let prof_cfg = prof_dir.join("config").join("config.json");
50        if prof_cfg.is_file() {
51            let _ = jc.merge_profile_file(&prof_cfg);
52        }
53    }
54    jc.working_dir = workspace;
55    jc.chat_id = job.chat_id;
56    jc.topic_id = job.topic_id;
57    jc.transport = job.transport.clone();
58    if let Some(p) = job.provider.as_deref().map(str::trim).filter(|s| !s.is_empty()) {
59        jc.provider = crate::session::data::SessionData::canonical_provider(p)
60            .unwrap_or("antigravity")
61            .to_string();
62    } else {
63        jc.provider = crate::session::data::SessionData::effective_provider(&jc.provider, "antigravity")
64            .to_string();
65    }
66    let eff_prov = jc.provider.clone();
67    if !job.cli_parameters.is_empty() {
68        jc.cli_parameters.insert(eff_prov.clone(), job.cli_parameters.clone());
69    }
70    apply_job_model_effort(&mut jc, &eff_prov, job);
71    jc
72}

74fn apply_job_model_effort(jc: &mut crate::config::CliConfig, eff_prov: &str, job: &CronJob) {
75    use crate::session::data::SessionData;
76    let job_m = job.model.as_deref().map(str::trim).filter(|s| !s.is_empty());
77    let raw_m = job_m.or(jc.model.as_deref());
78    let job_suf = job_m
79        .filter(|m| SessionData::is_model_compatible(m, eff_prov))
80        .and_then(|m| crate::cli::common::strip_effort_suffix(m).1)
81        .map(str::to_string);
82    let suf_eff = raw_m
83        .filter(|m| SessionData::is_model_compatible(m, eff_prov))
84        .and_then(|m| crate::cli::common::strip_effort_suffix(m).1)
85        .map(str::to_string);
86    let resolved_m = SessionData::default_model_for_provider(eff_prov, raw_m);
87    jc.model = Some(resolved_m.clone());
88    if let Some(r) = job.effort.as_deref().map(str::trim).filter(|s| !s.is_empty()) {
89        jc.effort = Some(r.to_string());
90    } else if let Some(se) = job_suf {
91        jc.effort = Some(se);
92    } else if jc.effort.is_none() {
93        jc.effort = suf_eff;
94    }
95    if !crate::messenger::telegram::reply_directives::session_supports_effort(eff_prov, &resolved_m) {
96        jc.effort = None;
97    } else if let Some(ref eff) = jc.effort {
98        jc.effort = crate::messenger::telegram::reply_directives::clamp_session_effort(
99            eff_prov,
100            &resolved_m,
101            eff,
102        );
103    }
104}
