13/// Find the directory where translation locales are stored.
14pub fn find_locales_dir() -> PathBuf {
15    if let Ok(val) = std::env::var("TUNER_I18N_DIR") {
16        let p = PathBuf::from(val);
17        if p.exists() {
18            return p;
19        }
20    }
21    for p in &[
22        "src/i18n/locales",
23        "i18n/locales",
24        "../src/i18n/locales",
25    ] {
26        let pb = PathBuf::from(p);
27        if pb.exists() {
28            return pb;
29        }
30    }
31    PathBuf::from("src/i18n/locales")
32}

34/// Recursively flattens a nested TOML value into a dot-notation key map.
35pub fn flatten(value: &toml::Value, prefix: &str, map: &mut HashMap<String, String>) {
36    match value {
37        toml::Value::Table(table) => {
38            for (k, v) in table {
39                let full_key = if prefix.is_empty() {
40                    k.clone()
41                } else {
42                    format!("{}.{}", prefix, k)
43                };
44                flatten(v, &full_key, map);
45            }
46        }
47        toml::Value::String(s) => {
48            map.insert(prefix.to_string(), s.clone());
49        }
50        _ => {
51            let s = match value {
52                toml::Value::Integer(i) => i.to_string(),
53                toml::Value::Float(f) => f.to_string(),
54                toml::Value::Boolean(b) => b.to_string(),
55                toml::Value::Datetime(dt) => dt.to_string(),
56                _ => value.to_string(),
57            };
58            map.insert(prefix.to_string(), s);
59        }
60    }
61}

63/// Load and parse a TOML file into a flat key-value map.
64///
65/// # Examples
66/// ```no_run
67/// use std::path::Path;
68/// let result = tuner::i18n::loader::load_toml(Path::new("en/chat.toml"));
69/// ```
70pub fn load_toml(path: &Path) -> anyhow::Result<HashMap<String, String>> {
71    let mut map = HashMap::new();
72    let content = std::fs::read_to_string(path)?;
73    let value = content.parse::<toml::Value>()?;
74    flatten(&value, "", &mut map);
75    Ok(map)
76}

78/// Check whether a language identifier is safe for filesystem lookup.
79fn is_safe_lang(lang: &str) -> bool {
80    if lang.is_empty() {
81        return false;
82    }
83    lang.chars().all(|c| c.is_ascii_alphanumeric() || c == '-' || c == '_')
84}

86fn embedded_chat_toml(lang: &str) -> Option<&'static str> {
87    match lang {
88        "en" => Some(include_str!("locales/en/chat.toml")),
89        "de" => Some(include_str!("locales/de/chat.toml")),
90        "nl" => Some(include_str!("locales/nl/chat.toml")),
91        "es" => Some(include_str!("locales/es/chat.toml")),
92        "fr" => Some(include_str!("locales/fr/chat.toml")),
93        "id" => Some(include_str!("locales/id/chat.toml")),
94        "pt" => Some(include_str!("locales/pt/chat.toml")),
95        "ru" => Some(include_str!("locales/ru/chat.toml")),
96        "ko" => Some(include_str!("locales/ko/chat.toml")),
97        _ => None,
98    }
99}

101/// Load the chat translation catalog for a language, falling back to embedded assets when running standalone.
102pub fn load_language(root: &Path, lang: &str) -> HashMap<String, String> {
103    if !is_safe_lang(lang) {
104        return HashMap::new();
105    }
106    let chat_path = root.join(lang).join("chat.toml");
107    if chat_path.is_file() {
108        return load_toml(&chat_path).unwrap_or_default();
109    }
110    if !root.is_dir()
111        && let Some(raw) = embedded_chat_toml(lang)
112            && let Ok(val) = raw.parse::<toml::Value>() {
113                let mut map = HashMap::new();
114                flatten(&val, "", &mut map);
115                return map;
116            }
117    HashMap::new()
118}

120/// Represents a collection of localized chat strings and their English fallbacks.
121#[derive(Clone, Debug)]
122pub struct TranslationStore {
123    /// Language.
124    pub language: String,
125    /// En chat.
126    pub en_chat: HashMap<String, String>,
127    /// Primary chat.
128    pub primary_chat: Option<HashMap<String, String>>,
129}
130
131impl TranslationStore {
132    #[cfg(test)]
133    /// Create a new TranslationStore using the default locales path.
134    pub fn new(language: &str) -> Self {
135        let root = find_locales_dir();
136        Self::new_with_root(language, &root)
137    }
138
139    /// Create a new TranslationStore with a specific locales path.
140    pub fn new_with_root(language: &str, root: &Path) -> Self {
141        let en_chat = load_language(root, "en");
142        let language = if is_safe_lang(language) { language } else { "en" };
143        let primary_chat = if language == "en" {
144            None
145        } else {
146            Some(load_language(root, language))
147        };
148        Self {
149            language: language.to_string(),
150            en_chat,
151            primary_chat,
152        }
153    }
154
155    /// Retrieve a chat translation.
156    pub fn chat(&self, key: &str, args: &[(&str, &str)]) -> String {
157        self.resolve(self.primary_chat.as_ref(), &self.en_chat, key, args)
158    }
159
160    fn resolve(
161        &self,
162        primary: Option<&HashMap<String, String>>,
163        fallback: &HashMap<String, String>,
164        key: &str,
165        args: &[(&str, &str)],
166    ) -> String {
167        let raw = primary
168            .and_then(|m| m.get(key))
169            .filter(|s| !s.is_empty())
170            .or_else(|| fallback.get(key));
171        match raw {
172            Some(raw_str) => format_string(key, raw_str, args),
173            None => format!("[MISSING: {}]", key),
174        }
175    }
176}

178/// Format a translation string by replacing placeholders like `{name}` or `{first.name}`.
179/// Performs best-effort substitution when some arguments are missing.
180pub fn format_string(_key: &str, raw: &str, args: &[(&str, &str)]) -> String {
181    use regex::Regex;
182    use std::sync::OnceLock;
183    static RE: OnceLock<Option<Regex>> = OnceLock::new();
184    if args.is_empty() {
185        return raw.to_string();
186    }
187    let Some(re) = RE.get_or_init(|| Regex::new(r"\{([\w.-]+)\}").ok()) else {
188        return raw.to_string();
189    };
190
191    let result = re.replace_all(raw, |caps: &regex::Captures| {
192        let ph_name = &caps[1];
193        if let Some((_, val)) = args.iter().find(|(k, _)| *k == ph_name) {
194            val.to_string()
195        } else {
196            caps[0].to_string()
197        }
198    });
199
200    result.into_owned()
201}
