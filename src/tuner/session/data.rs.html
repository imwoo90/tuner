73#[derive(Clone, Debug, Serialize, Deserialize)]
74/// Represents `SessionData` (session data).
75pub struct SessionData {
76    #[serde(default = "default_transport")]
77    /// Transport.
78    pub transport: String,
79    /// Chat id.
80    pub chat_id: i64,
81    /// Topic id.
82    pub topic_id: Option<i64>,
83    /// Topic name.
84    pub topic_name: Option<String>,
85    #[serde(default = "default_provider")]
86    /// Provider.
87    pub provider: String,
88    #[serde(default = "default_model")]
89    /// Model.
90    pub model: String,
91    #[serde(default)]
92    /// Effort.
93    pub effort: Option<String>,
94    #[serde(default = "default_now_iso")]
95    /// Created at.
96    pub created_at: String,
97    #[serde(default = "default_now_iso")]
98    /// Last active.
99    pub last_active: String,
100    #[serde(default)]
101    /// Provider sessions.
102    pub provider_sessions: HashMap<String, ProviderSessionData>,
103    #[serde(default)]
104    /// Language.
105    pub language: Option<String>,
106    #[serde(default)]
107    /// Last progress msg id.
108    pub last_progress_msg_id: Option<i32>,
109    #[serde(default)]
110    /// Pending attachments.
111    pub pending_attachments: Vec<String>,
112    #[serde(default)]
113    /// Last handover summary.
114    pub last_handover_summary: Option<String>,
115
116    // Legacy fields used for migration
117    #[serde(skip_serializing, default)]
118    /// Session id.
119    pub session_id: Option<String>,
120    #[serde(skip_serializing, default)]
121    /// Message count.
122    pub message_count: Option<i64>,
123    #[serde(skip_serializing, default)]
124    /// Total cost usd.
125    pub total_cost_usd: Option<f64>,
126    #[serde(skip_serializing, default)]
127    /// Total tokens.
128    pub total_tokens: Option<i64>,
129}
130
131fn default_transport() -> String { "tg".to_string() }
132fn default_provider() -> String { "claude".to_string() }
133fn default_model() -> String { "opus".to_string() }
134fn default_now_iso() -> String { Utc::now().to_rfc3339() }
135
136impl SessionData {
137    /// Creates a new instance.
138    pub fn new(chat_id: i64, transport: String, topic_id: Option<i64>, provider: String, model: String) -> Self {
139        let now = default_now_iso();
140        Self {
141            transport,
142            chat_id,
143            topic_id,
144            topic_name: None,
145            provider,
146            model,
147            effort: None,
148            created_at: now.clone(),
149            last_active: now,
150            provider_sessions: HashMap::new(),
151            language: None,
152            last_progress_msg_id: None,
153            pending_attachments: Vec::new(),
154            last_handover_summary: None,
155            session_id: None,
156            message_count: None,
157            total_cost_usd: None,
158            total_tokens: None,
159        }
160    }
161
162    /// Executes `session_key` (session key).
163    pub fn session_key(&self) -> SessionKey {
164        SessionKey {
165            transport: self.transport.clone(),
166            chat_id: self.chat_id,
167            topic_id: self.topic_id,
168        }
169    }
170
171    /// Executes `get_session_id` (get session id).
172    pub fn get_session_id(&self, provider: &str) -> String {
173        self.provider_sessions.get(provider)
174            .map(|ps| ps.session_id.clone())
175            .unwrap_or_default()
176    }
177
178    /// Executes `set_session_id` (set session id).
179    pub fn set_session_id(&mut self, provider: &str, session_id: &str) {
180        let ps = self.provider_sessions.entry(provider.to_string()).or_default();
181        if !ps.session_id.is_empty() && ps.session_id != session_id
182            && !ps.past_session_ids.contains(&ps.session_id) {
183                ps.past_session_ids.push(ps.session_id.clone());
184            }
185        ps.session_id = session_id.to_string();
186    }
187
188    /// Executes `get_past_session_ids` (get past session ids).
189    pub fn get_past_session_ids(&self, provider: &str) -> Vec<String> {
190        self.provider_sessions.get(provider)
191            .map(|ps| ps.past_session_ids.clone())
192            .unwrap_or_default()
193    }
194
195    /// Executes `get_all_session_ids` (get all session ids).
196    pub fn get_all_session_ids(&self, provider: &str) -> Vec<String> {
197        if let Some(ps) = self.provider_sessions.get(provider) {
198            let mut list = ps.past_session_ids.clone();
199            if !ps.session_id.is_empty() && !list.contains(&ps.session_id) {
200                list.push(ps.session_id.clone());
201            }
202            list
203        } else {
204            Vec::new()
205        }
206    }
207
208    /// Executes `migrate_legacy_metrics` (migrate legacy metrics).
209    pub fn migrate_legacy_metrics(&mut self) {
210        let has_legacy = self.session_id.is_some()
211            || self.message_count.is_some()
212            || self.total_cost_usd.is_some()
213            || self.total_tokens.is_some();
214
215        if has_legacy && !self.provider.is_empty() {
216            let ps = self.provider_sessions.entry(self.provider.clone()).or_default();
217            if let Some(ref sid) = self.session_id {
218                ps.session_id = sid.clone();
219            }
220            if let Some(cnt) = self.message_count {
221                ps.message_count = cnt;
222            }
223            if let Some(cost) = self.total_cost_usd {
224                ps.total_cost_usd = cost;
225            }
226            if let Some(tok) = self.total_tokens {
227                ps.total_tokens = tok;
228            }
229        }
230    }
231
232    /// Executes `clear_provider_session` (clear provider session).
233    pub fn clear_provider_session(&mut self, provider: &str) {
234        if let Some(ps) = self.provider_sessions.get_mut(provider) {
235            if !ps.session_id.is_empty() && !ps.past_session_ids.contains(&ps.session_id) {
236                ps.past_session_ids.push(ps.session_id.clone());
237            }
238            ps.session_id = String::new();
239        }
240    }
