13/// Executes `is_zone2_py_file` (is zone2 py file).
14pub fn is_zone2_py_file(entry: &Path, src: &Path, root_src: &Path) -> bool {
15    if entry.extension().and_then(|s| s.to_str()) != Some("py") {
16        return false;
17    }
18    if let Ok(rel_dir) = src.strip_prefix(root_src) {
19        let rel_str = rel_dir.to_string_lossy().replace("\\", "/");
20        let zone2_py_dirs = [
21            "workspace/tools/cron_tools",
22            "workspace/tools/webhook_tools",
23            "workspace/tools/media_tools",
24            "workspace/skills/memory-system/scripts",
25            "workspace/skills/skill-creator/scripts",
26        ];
27        return zone2_py_dirs.contains(&rel_str.as_str());
28    }
29    false
30}

32fn is_bundled_system_doc(name: &str, src: &Path, root_src: &Path) -> bool {
33    if let Ok(rel_dir) = src.strip_prefix(root_src) {
34        let rel_str = rel_dir.to_string_lossy().replace("\\", "/");
35        if name == "SKILL.md"
36            && (rel_str == "workspace/skills/memory-system"
37                || rel_str == "workspace/skills/skill-creator")
38        {
39            return true;
40        }
41        if name == "TASK_DESCRIPTION.md" && rel_str == "workspace/cron_tasks/memory-consolidator" {
42            return true;
43        }
44    }
45    false
46}

48/// Executes `walk_and_copy` (walk and copy).
49pub fn walk_and_copy(src: &Path, dst: &Path, root_src: &Path) -> Result<(), String> {
50    if let Err(e) = std::fs::create_dir_all(dst) {
51        return Err(format!("Failed to create dir: {}", e));
52    }
53    let skip_dirs = [".venv", ".git", ".mypy_cache", "__pycache__", "node_modules"];
54
55    let entries = match std::fs::read_dir(src) {
56        Ok(e) => e,
57        Err(e) => return Err(format!("Failed to read dir: {}", e)),
58    };
59
60    for entry in entries.flatten() {
61        let path = entry.path();
62        let name = match path.file_name().and_then(|s| s.to_str()) {
63            Some(n) => n,
64            None => continue,
65        };
66        if name.starts_with('.') || skip_dirs.contains(&name) {
67            continue;
68        }
69        if name == "RULES.md" && (src == root_src || src == root_src.join("workspace")) {
70            continue;
71        }
72        let target = dst.join(name);
73        handle_entry(&path, &target, name, src, root_src)?;
74    }
75    Ok(())
76}
77
78fn handle_entry(path: &Path, target: &Path, name: &str, src: &Path, root_src: &Path) -> Result<(), String> {
79    if path.is_dir() {
80        if !target.is_symlink() {
81            walk_and_copy(path, target, root_src)?;
82        }
83    } else if name == "RULES.md" || name == "AGENTS.md" || is_bundled_system_doc(name, src, root_src) {
84        if target.is_symlink() || target.is_file() {
85            let _ = std::fs::remove_file(target);
86        }
87        let _ = std::fs::copy(path, target);
88    } else if is_zone2_py_file(path, src, root_src) {
89        if target.is_file() {
90            let same_content = match (std::fs::read(path), std::fs::read(target)) {
91                (Ok(a), Ok(b)) => a == b,
92                _ => false,
93            };
94            if !same_content {
95                let backup = target.with_extension("py.bak");
96                let _ = std::fs::rename(target, &backup);
97            }
98        }
99        if target.is_symlink() {
100            let _ = std::fs::remove_file(target);
101        }
102        let _ = std::fs::copy(path, target);
103    } else if !target.exists() {
104        let _ = std::fs::copy(path, target);
105    }
106    Ok(())
107}

116/// Executes `smart_merge_config` (smart merge config).
117pub fn smart_merge_config(paths: &TunerPaths) -> Result<(), String> {
118    let example_path = paths.config_example_path();
119    if !example_path.is_file() {
120        return Ok(());
121    }
122    let defaults_content = match std::fs::read_to_string(&example_path) {
123        Ok(c) => c,
124        Err(_) => return Ok(()),
125    };
126    let defaults: serde_json::Value = match serde_json::from_str(&defaults_content) {
127        Ok(v) => v,
128        Err(_) => return Ok(()),
129    };
130
131    let config_path = paths.config_path();
132    if !config_path.is_file() {
133        if let Some(parent) = config_path.parent() {
134            let _ = std::fs::create_dir_all(parent);
135        }
136        if let Ok(pretty) = serde_json::to_string_pretty(&defaults) {
137            let _ = std::fs::write(&config_path, pretty);
138        }
139    }
140    Ok(())
141}

143/// Executes `smart_merge_profile_config` (smart merge profile config).
144pub fn smart_merge_profile_config(paths: &TunerPaths) -> Result<(), String> {
145    let example_path = paths.home_defaults.join("config").join("config.json");
146    if !example_path.is_file() {
147        return Ok(());
148    }
149    let defaults_content = std::fs::read_to_string(&example_path).map_err(|e| e.to_string())?;
150    let defaults: serde_json::Value = serde_json::from_str(&defaults_content).map_err(|e| e.to_string())?;
151
152    let config_path = paths.profile_home().join("config").join("config.json");
153    if !config_path.is_file() {
154        if let Some(parent) = config_path.parent() {
155            let _ = std::fs::create_dir_all(parent);
156        }
157        if let Ok(pretty) = serde_json::to_string_pretty(&defaults) {
158            let _ = std::fs::write(&config_path, pretty);
159        }
160    }
161    Ok(())
162}
