30/// Serialized session summary returned to the Web Dashboard sidebar and chat header.
31#[derive(Clone, Debug, Serialize)]
32pub struct SessionSummary {
33    /// Composite storage key (`tg:<cid>[:<tid>]` or `web:0:<tid>`).
34    pub session_key: String,
35    /// Transport identifier (`tg` or `web`).
36    pub transport: String,
37    /// Telegram chat ID (`0` for Web-native topics).
38    pub chat_id: i64,
39    /// Forum or Web topic ID.
40    pub topic_id: Option<i64>,
41    /// Optional human-readable topic title.
42    pub topic_name: Option<String>,
43    /// Effective CLI provider (`antigravity`, `claude`, `codex`).
44    pub provider: String,
45    /// Effective canonical model identifier.
46    pub model: String,
47    /// Clamped reasoning effort level, if supported by the active model.
48    pub effort: Option<String>,
49    /// Active provider session ID, if initialized.
50    pub session_id: String,
51    /// Turn count on the active provider session.
52    pub message_count: i64,
53    /// Queued attachment paths waiting for the next user prompt.
54    pub pending_attachments: Vec<String>,
55    /// RFC3339 timestamp of most recent activity.
56    pub last_active: String,
57    /// True when a foreground or background turn is executing.
58    pub busy: bool,
59    /// True when waiting for an interactive `AskQuestion` response.
60    pub asking: bool,
61    /// True when pinned to the top of the sidebar.
62    pub pinned: bool,
63}

65/// Checks whether `sess` currently has an in-flight turn or active `AskQuestion` state.
66pub async fn check_busy_and_asking(ctx: &WebAppCtx, sess: &SessionData) -> (bool, bool) {
67    let prov = SessionData::effective_provider(&sess.provider, &ctx.config.provider);
68    let sid = sess.get_session_id(prov);
69    let (cid, tid) = (sess.chat_id, sess.topic_id);
70    let asking = !sid.is_empty()
71        && ctx.cli.sessions.is_ask_active(&sid).await
72        && ((cfg!(test) && sid.starts_with("mock-"))
73            || ctx.cli.is_ask_backing_alive(prov, cid, tid, &sid).await);
74    let busy = asking
75        || ctx.cli.is_topic_turn_in_flight(cid, tid)
76        || ctx.cli.sessions.boot_cancels.lock().await.contains_key(&(cid, tid))
77        || (!sid.is_empty()
78            && (ctx.cli.sessions.is_running(&sid).await
79                || ctx.cli.sessions.is_async_running(&sid).await))
80        || crate::cli::claude::is_running(cid, tid, (!sid.is_empty()).then_some(sid.as_str())).await
81        || crate::cli::codex::is_running(cid, tid, (!sid.is_empty()).then_some(sid.as_str())).await;
82    (busy, asking)
83}

85/// Builds a [`SessionSummary`] from persisted [`SessionData`] and live pool state.
86pub async fn build_session_summary(ctx: &WebAppCtx, sess: &SessionData) -> SessionSummary {
87    let (prov, model, effort) = SessionData::resolve_effective_target(Some(sess), &ctx.config);
88    let sid = sess.get_session_id(&prov);
89    let count = sess.get_message_count(&prov);
90    let (busy, asking) = check_busy_and_asking(ctx, sess).await;
91    SessionSummary {
92        session_key: sess.session_key().storage_key(),
93        transport: sess.transport.clone(),
94        chat_id: sess.chat_id,
95        topic_id: sess.topic_id,
96        topic_name: sess.topic_name.clone(),
97        provider: prov,
98        model,
99        effort,
100        session_id: sid,
101        message_count: count,
102        pending_attachments: sess.pending_attachments.clone(),
103        last_active: sess.last_active.clone(),
104        busy,
105        asking,
106        pinned: sess.pinned,
107    }
108}

110/// Lists all persisted Telegram & Web sessions plus global warm pool counts.
111pub async fn handle_list_sessions(
112    ext: Option<Extension<Arc<WebAppCtx>>>,
113    headers: HeaderMap,
114) -> Response {
115    let ctx = match resolve_ctx(ext) {
116        Ok(c) => c,
117        Err(r) => return *r,
118    };
119    if let Err(r) = require_auth(&ctx, &headers).await {
120        return *r;
121    }
122    let mut all = ctx.sessions.list_all().await.unwrap_or_default();
123    all.sort_by(|a, b| b.last_active.cmp(&a.last_active));
124    let mut summaries = Vec::with_capacity(all.len());
125    for s in &all {
126        summaries.push(build_session_summary(&ctx, s).await);
127    }
128    let agy = ctx.cli.sessions.active_count().await;
129    let claude = crate::cli::claude::active_count().await;
130    let codex = crate::cli::codex::active_count().await;
131    Json(serde_json::json!({
132        "sessions": summaries,
133        "warm_pools": { "agy": agy, "claude": claude, "codex": codex, "total_active": agy + claude + codex }
134    }))
135    .into_response()
136}

160/// Creates a new `web:0:<tid>` or real Telegram `tg:<cid>:<tid>` topic (`POST /dashboard/api/sessions`).
161pub async fn handle_create_session(
162    ext: Option<Extension<Arc<WebAppCtx>>>,
163    headers: HeaderMap,
164    Json(payload): Json<CreateSessionPayload>,
165) -> Response {
166    let ctx = match resolve_ctx(ext) {
167        Ok(c) => c,
168        Err(r) => return *r,
169    };
170    if let Err(r) = require_mutation_auth(&ctx, &headers).await {
171        return *r;
172    }
173    match super::tg_topics::create_session_from_payload(&ctx, payload).await {
174        Ok(sess) => {
175            emit_session_updated(&ctx, &sess, false, false).await;
176            (StatusCode::CREATED, Json(build_session_summary(&ctx, &sess).await)).into_response()
177        }
178        Err(resp) => *resp,
179    }
180}

199fn validate_model_and_effort(
200    eff_prov: &str,
201    model_opt: Option<&str>,
202    effort_opt: Option<&str>,
203) -> Result<(Option<String>, Option<String>), String> {
204    let target_model = match model_opt.map(str::trim).filter(|s| !s.is_empty()) {
205        Some(raw) => Some(
206            SessionData::canonicalize_model_for_provider(eff_prov, raw)
207                .ok_or_else(|| format!("Model `{raw}` is incompatible with `{eff_prov}`"))?,
208        ),
209        None => None,
210    };
211    let clean_eff = match effort_opt.map(str::trim).filter(|s| !s.is_empty()) {
212        Some(raw) => {
213            let low = raw.to_ascii_lowercase();
214            if !is_valid_effort_level(&low) {
215                return Err(format!("Invalid effort level `{raw}`"));
216            }
217            Some(low)
218        }
219        None => None,
220    };
221    Ok((target_model, clean_eff))
222}

224async fn apply_model_and_effort_patch(
225    ctx: &WebAppCtx,
226    key: &SessionKey,
227    sess: &mut SessionData,
228    payload: &PatchConfigPayload,
229) -> Result<(), String> {
230    let eff_prov = SessionData::effective_provider(&sess.provider, &ctx.config.provider).to_string();
231    let (target_model, clean_eff) =
232        validate_model_and_effort(&eff_prov, payload.model.as_deref(), payload.effort.as_deref())?;
233    let name_opt = payload.topic_name.as_deref().map(str::trim).filter(|s| !s.is_empty());
234    if let Ok(Some(updated)) = ctx
235        .sessions
236        .mutate_session(key, Some(sess), |cur| {
237            let cp = SessionData::effective_provider(&cur.provider, &ctx.config.provider).to_string();
238            cur.provider = cp.clone();
239            if let Some(ref m) = target_model {
240                cur.model = m.clone();
241            }
242            if let Some(ref eff) = clean_eff {
243                cur.effort = clamp_session_effort(&cp, &cur.model, eff);
244            } else if target_model.is_some() && !session_supports_effort(&cp, &cur.model) {
245                cur.effort = None;
246            }
247            if let Some(tname) = name_opt {
248                cur.topic_name = Some(tname.to_string());
249            }
250            if let Some(pin) = payload.pinned {
251                cur.pinned = pin;
252            }
253            true
254        })
255        .await
256    {
257        *sess = updated;
258    }
259    apply_model_to_active_pty(&ctx.cli, &ctx.sessions, sess).await;
260    Ok(())
261}

263/// Updates session provider, model, effort, or topic name (`PATCH /dashboard/api/sessions/config`).
264pub async fn handle_patch_session_config(
265    ext: Option<Extension<Arc<WebAppCtx>>>,
266    headers: HeaderMap,
267    Json(payload): Json<PatchConfigPayload>,
268) -> Response {
269    let ctx = match resolve_ctx(ext) {
270        Ok(c) => c,
271        Err(r) => return *r,
272    };
273    if let Err(r) = require_mutation_auth(&ctx, &headers).await {
274        return *r;
275    }
276    let Ok(key) = payload.session_key.parse::<SessionKey>() else {
277        return json_err(StatusCode::BAD_REQUEST, "Invalid session_key");
278    };
279    let Ok(Some(mut sess)) = ctx.sessions.get_active(&key).await else {
280        return json_err(StatusCode::NOT_FOUND, "Session not found");
281    };
282    if let Some(ref raw_p) = payload.provider {
283        let Some(target_p) = SessionData::canonical_provider(raw_p) else {
284            return json_err(StatusCode::BAD_REQUEST, "Unknown provider");
285        };
286        let cur_p = SessionData::effective_provider(&sess.provider, &ctx.config.provider);
287        if cur_p != target_p {
288            let _ = switch_session_provider(target_p, &ctx.config, &ctx.sessions, &ctx.cli, &mut sess).await;
289            if key.transport == "tg" && let Some(bot) = ctx.bot_clone().await {
290                update_chat_commands_for_provider(&bot, teloxide::types::ChatId(key.chat_id), target_p).await;
291            }
292        }
293    }
294    if let Err(msg) = apply_model_and_effort_patch(&ctx, &key, &mut sess, &payload).await {
295        return json_err(StatusCode::BAD_REQUEST, &msg);
296    }
297    if key.transport == "tg"
298        && let Some(tname) = payload.topic_name.as_deref().map(str::trim).filter(|s| !s.is_empty())
299    {
300        super::tg_topics::sync_remote_topic_rename(&ctx, &key, tname).await;
301    }
302    let (busy, asking) = check_busy_and_asking(&ctx, &sess).await;
303    emit_session_updated(&ctx, &sess, busy, asking).await;
304    Json(build_session_summary(&ctx, &sess).await).into_response()
305}
