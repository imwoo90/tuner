23#[derive(Serialize, Deserialize)]
24struct TunnelStateFile {
25    pid: u32,
26    url: String,

36/// Persists or clears the active Quick Tunnel URL (`0600`) for CLI `--dashboard` discovery.
37pub fn publish_dashboard_tunnel_url(vault_dir: &Path, url: Option<&str>) {
38    let dst = vault_dir.join(TUNNEL_STATE_FILE);
39    match url {
40        Some(u) if !u.trim().is_empty() => {
41            let _ = std::fs::create_dir_all(vault_dir);
42            let tmp = vault_dir.join(".dashboard_tunnel.json.tmp");
43            let payload = TunnelStateFile {
44                pid: std::process::id(),
45                url: u.trim().to_string(),
46            };
47            if let Ok(raw) = serde_json::to_string(&payload)
48                && std::fs::write(&tmp, raw).is_ok()
49            {
50                set_mode_0600(&tmp);
51                let _ = std::fs::rename(&tmp, &dst);
52                set_mode_0600(&dst);
53            }
54        }
55        _ => {
56            let _ = std::fs::remove_file(&dst);
57        }
58    }
59}

61/// Loads an active Quick Tunnel URL from `dashboard_tunnel.json` if its owner PID is alive.
62pub fn load_dashboard_tunnel_url(vault_dir: &Path) -> Option<String> {
63    let raw = std::fs::read_to_string(vault_dir.join(TUNNEL_STATE_FILE)).ok()?;
64    let st: TunnelStateFile = serde_json::from_str(&raw).ok()?;
65    if st.pid == 0 || st.url.is_empty() {
66        return None;
67    }
68    #[cfg(target_os = "linux")]
69    if !Path::new(&format!("/proc/{}", st.pid)).exists() {
70        return None;
71    }
72    Some(st.url)
73}

75/// Parses `--dashboard [--profile <name>] [--set-pin <pin>]` CLI arguments.
76pub fn parse_dashboard_cli_args(args: &[String]) -> Result<(String, Option<String>), String> {
77    let mut profile = "default".to_string();
78    let mut set_pin = None;
79    let mut i = 1;
80    while i < args.len() {
81        match args[i].as_str() {
82            "--dashboard" => i += 1,
83            "--profile" => {
84                let val = args
85                    .get(i + 1)
86                    .filter(|s| !s.trim().is_empty() && !s.starts_with('-'))
87                    .ok_or_else(|| "Missing value for --profile".to_string())?;
88                profile = val.clone();
89                i += 2;
90            }
91            "--set-pin" => {
92                let val = args
93                    .get(i + 1)
94                    .filter(|s| !s.is_empty())
95                    .ok_or_else(|| "Missing value for --set-pin".to_string())?;
96                set_pin = Some(val.clone());
97                i += 2;
98            }
99            other => {
100                return Err(format!(
101                    "Unknown option '{other}' for --dashboard. Usage: tuner --dashboard [--profile <name>] [--set-pin <pin>]"
102                ));
103            }
104        }
105    }
106    Ok((profile, set_pin))
107}

109/// Executes `tuner --dashboard [--profile <name>] [--set-pin <pin>]`.
110pub async fn run_dashboard_cli(args: &[String]) -> Result<(), String> {
111    let (profile, set_pin) = parse_dashboard_cli_args(args)?;
112    let paths = resolve_paths(None, None, None, Some(profile.clone()));
113    let _ = crate::setup::load_env_file(&paths.env_file());
114    let vault_dir = paths.vault_dir();
115    let store = WebAuthStore::new(vault_dir.clone());
116    if let Some(ref pin) = set_pin {
117        store.set_pin(pin).await?;
118        println!("🔐 [tuner] Owner dashboard PIN updated for profile '{profile}'.");
119    }
120    let token = store.issue_magic_token().await?;
121    let port = load_tunnel_endpoint(&vault_dir).map(|e| e.port).unwrap_or(8743);
122    let local_url = format!("http://127.0.0.1:{port}/dashboard/magic/{token}");
123    let ts_url = match check_tailscale_status() {
124        (ServiceStatus::Active, Some(ip)) => Some(format!("http://{ip}:{port}/dashboard/magic/{token}")),
125        _ => None,
126    };
127    let tunnel_url = load_dashboard_tunnel_url(&vault_dir)
128        .map(|base| format!("{}/dashboard/magic/{token}", base.trim_end_matches('/')));
129    println!("🖥️ [tuner] Web Dashboard Magic Link (single-use, 10m TTL, profile: {profile}):");
130    println!("  • Local     : {local_url}");
131    if let Some(ref u) = ts_url {
132        println!("  • Tailscale : {u}");
133    }
134    if let Some(ref u) = tunnel_url {
135        println!("  • Tunnel    : {u}");
136    }
137    Ok(())
138}

140fn resolve_dashboard_target(config: &CliConfig) -> (String, PathBuf, u16) {
141    if let Some(ctx) = super::global_web_ctx() {
142        let port = load_tunnel_endpoint(&ctx.vault_dir).map(|e| e.port).unwrap_or(8743);
143        return (ctx.profile.clone(), ctx.vault_dir.clone(), port);
144    }
145    if let Some((prof, vdir)) = extract_profile_from_working_dir(&config.working_dir) {
146        let port = load_tunnel_endpoint(&vdir).map(|e| e.port).unwrap_or(8743);
147        return (prof, vdir, port);
148    }
149    let paths = resolve_paths(None, None, None, Some("default".to_string()));
150    let vdir = paths.vault_dir();
151    let port = load_tunnel_endpoint(&vdir).map(|e| e.port).unwrap_or(8743);
152    ("default".to_string(), vdir, port)
153}

155fn format_dashboard_html_and_kb(
156    profile: &str,
157    local_url: &str,
158    ts_url: Option<&str>,
159    tunnel_url: Option<&str>,
160) -> (String, Vec<Vec<InlineKeyboardButton>>) {
161    let mut lines = vec![
162        format!("🖥️ <b>[Tuner Web Dashboard]</b> (프로필: <code>{}</code>)", html_escape::encode_text(profile)),
163        String::new(),
164        "• 아래 링크는 <b>1회용(10분 유효)</b> 자동 로그인 Magic Link입니다.".to_string(),
165        format!("• <b>Local</b>: <code>{}</code>", html_escape::encode_text(local_url)),
166    ];
167    if let Some(u) = ts_url {
168        lines.push(format!("• <b>Tailscale</b>: <code>{}</code>", html_escape::encode_text(u)));
169    }
170    if let Some(u) = tunnel_url {
171        lines.push(format!("• <b>Tunnel</b>: <code>{}</code>", html_escape::encode_text(u)));
172    }
173    let mut rows = Vec::new();
174    if let Some(u) = tunnel_url
175        && u.starts_with("https://")
176        && let Ok(parsed) = url::Url::parse(u)
177    {
178        rows.push(vec![InlineKeyboardButton::url("🖥️ Tuner Web Dashboard 열기", parsed)]);
179    }
180    (lines.join("\n"), rows)
181}

183async fn send_dashboard_card(
184    bot: &Bot,
185    msg: &Message,
186    text: String,
187    rows: Vec<Vec<InlineKeyboardButton>>,
188) {
189    let no_preview = teloxide::types::LinkPreviewOptions {
190        is_disabled: true,
191        url: None,
192        prefer_small_media: false,
193        prefer_large_media: false,
194        show_above_text: false,
195    };
196    let mut req = bot
197        .send_message(msg.chat.id, text.clone())
198        .parse_mode(ParseMode::Html)
199        .link_preview_options(no_preview.clone());
200    if let Some(tid) = msg.thread_id {
201        req = req.message_thread_id(tid);
202    }
203    if !rows.is_empty() {
204        if req.reply_markup(InlineKeyboardMarkup::new(rows)).await.is_ok() {
205            return;
206        }
207        let mut fb = bot
208            .send_message(msg.chat.id, text)
209            .parse_mode(ParseMode::Html)
210            .link_preview_options(no_preview);
211        if let Some(tid) = msg.thread_id {
212            fb = fb.message_thread_id(tid);
213        }
214        let _ = fb.await;
215    } else {
216        let _ = req.await;
217    }
218}

220/// Issues a fresh single-use Magic Link and sends the Dashboard card to Telegram.
221pub async fn handle_telegram_dashboard_command(
222    bot: &Bot,
223    msg: &Message,
224    config: &CliConfig,
225) -> Result<(), teloxide::RequestError> {
226    let (profile, vault_dir, port) = resolve_dashboard_target(config);
227    let store = WebAuthStore::new(vault_dir.clone());
228    let Ok(token) = store.issue_magic_token().await else {
229        let _ = crate::messenger::telegram::commands::send_reply(bot, msg, "❌ Failed to issue dashboard magic link.").await;
230        return Ok(());
231    };
232    let tunnel_base = crate::messenger::telegram::review::global_review_manager()
233        .current_tunnel_url()
234        .await
235        .or_else(|| load_dashboard_tunnel_url(&vault_dir));
236    let local_url = format!("http://127.0.0.1:{port}/dashboard/magic/{token}");
237    let ts_url = match check_tailscale_status() {
238        (ServiceStatus::Active, Some(ip)) => Some(format!("http://{ip}:{port}/dashboard/magic/{token}")),
239        _ => None,
240    };
241    let tunnel_url = tunnel_base.map(|b| format!("{}/dashboard/magic/{token}", b.trim_end_matches('/')));
242    let (text, rows) = format_dashboard_html_and_kb(&profile, &local_url, ts_url.as_deref(), tunnel_url.as_deref());
243    send_dashboard_card(bot, msg, text, rows).await;
244    Ok(())
