14fn en_secretary_prompt(topic_hint: &str) -> String {
15    format!(
16        "You are Woo-Tuner ([우튜너]), the user's real-time Front-Desk Voice Secretary in front of the Tuner multi-agent workspace. {topic_hint}\n\
17         Rules:\n\
18         1. Speak naturally and concisely (1-3 short sentences). Never read Markdown symbols like `**` or tables aloud.\n\
19         2. Answer quick checks via `list_topics`, `get_topic_context`, or `list_available_models`.\n\
20         3. Delegate coding, builds, or deep research via `delegate_to_topic` (pass `provider`/`model`/`effort` whenever requested, e.g. `gemini-4-hydrogen`, `claude`, `codex`; for parallel work, call `delegate_to_topic` multiple times in one turn with distinct `create_new_topic_title`s).\n\
21         4. On `[BATCH_COMPLETION_BRIEFING]` or `[BACKGROUND_EVENT]`, synthesize all completed topics together toward the user's goal instead of reading each room verbatim; read `AskQuestion` options clearly and submit answers via `answer_topic_question`.\n\
22         5. Ignore unintelligible ambient noise or distant chatter not addressed to you."
23    )
24}

49fn build_secretary_prompt(lang: &str, active_topic: Option<&str>) -> String {
50    let topic_hint = active_topic
51        .filter(|s| !s.is_empty())
52        .map(|s| format!("Currently open topic on user's screen: `{s}`."))
53        .unwrap_or_else(|| "User is currently on a blank Draft screen (no topic selected yet).".to_string());
54    if lang.eq_ignore_ascii_case("auto") {
55        auto_secretary_prompt(&topic_hint)
56    } else if lang.starts_with("en") {
57        en_secretary_prompt(&topic_hint)
58    } else {
59        ko_secretary_prompt(&topic_hint)
60    }
61}

74fn build_speech_config(lang: &str) -> Value {
75    if lang.eq_ignore_ascii_case("auto") {
76        json!({ "voiceConfig": { "prebuiltVoiceConfig": { "voiceName": "Kore" } } })
77    } else {
78        let code = if lang.starts_with("en") { "en-US" } else { "ko-KR" };
79        json!({
80            "languageCode": code,
81            "voiceConfig": { "prebuiltVoiceConfig": { "voiceName": "Kore" } }
82        })
83    }
84}

86/// Builds the initial `BidiGenerateContent` `setup` JSON message.
87pub fn build_live_setup_frame(
88    lang: &str,
89    active_topic: Option<&str>,
90    resume_handle: Option<&str>,
91) -> String {
92    let resumption = match resume_handle.filter(|h| !h.is_empty()) {
93        Some(h) => json!({ "handle": h }),
94        None => json!({}),
95    };
96    json!({
97        "setup": {
98            "model": resolve_voice_model(),
99            "generationConfig": {
100                "responseModalities": ["AUDIO"],
101                "speechConfig": build_speech_config(lang)
102            },
103            "realtimeInputConfig": realtime_vad_config(),
104            "inputAudioTranscription": {},
105            "outputAudioTranscription": {},
106            "contextWindowCompression": { "slidingWindow": {} },
107            "sessionResumption": resumption,
108            "systemInstruction": {
109                "parts": [{ "text": build_secretary_prompt(lang, active_topic) }]
110            },
111            "tools": [{
112                "functionDeclarations": voice_tool_declarations()
113            }]
114        }
115    })
116    .to_string()
117}

119/// Parsed action emitted from a single upstream Gemini Live WebSocket JSON frame.
120#[derive(Debug)]
121pub enum UpstreamAction {
122    /// Upstream `setupComplete` handshake succeeded.
123    Ready,
124    /// User barged in; browser must flush queued audio playback immediately.
125    Interrupted,
126    /// Base64-encoded 24 kHz 16-bit mono PCM audio chunk.
127    AudioPcm24(String),
128    /// Incremental user speech transcription.
129    UserTranscript(String),
130    /// Incremental assistant speech transcription.
131    BotTranscript(String),
132    /// Model turn completed.
133    TurnComplete,
134    /// Function call requested by Gemini Live `(id, name, args)`.
135    ToolCalls(Vec<(String, String, Value)>),
136    /// Updated `sessionResumption` token handle.
137    ResumeHandle(String),
138    /// Upstream `goAway` rotation warning; reconnect using stored `ResumeHandle`.
139    GoAway,
140    /// Upstream error message reported by Gemini Live.
141    Error(String),
142}
143
144/// Maps a simple [`UpstreamAction`] directly to a browser WebSocket JSON payload when applicable.
145pub fn action_to_client_frame(act: &UpstreamAction) -> Option<Value> {
146    match act {
147        UpstreamAction::Ready => {
148            let m = resolve_voice_model().trim_start_matches("models/").to_string();
149            Some(json!({ "type": "ready", "model": m }))
150        }
151        UpstreamAction::Interrupted => Some(json!({ "type": "interrupted" })),
152        UpstreamAction::AudioPcm24(b) => Some(json!({ "type": "audio", "data": b })),
153        UpstreamAction::UserTranscript(t) => Some(json!({ "type": "user_transcript", "text": t })),
154        UpstreamAction::BotTranscript(t) => Some(json!({ "type": "bot_transcript", "text": t })),
155        UpstreamAction::TurnComplete => Some(json!({ "type": "turn_complete" })),
156        UpstreamAction::Error(e) => Some(json!({ "type": "error", "error": e })),
157        _ => None,
158    }
159}

161fn parse_server_content(sc: &Value, out: &mut Vec<UpstreamAction>) {
162    if sc.get("interrupted").and_then(Value::as_bool) == Some(true) {
163        out.push(UpstreamAction::Interrupted);
164    }
165    if let Some(parts) = sc.pointer("/modelTurn/parts").and_then(Value::as_array) {
166        for p in parts {
167            if let Some(b64) = p.pointer("/inlineData/data").and_then(Value::as_str).filter(|s| !s.is_empty()) {
168                out.push(UpstreamAction::AudioPcm24(b64.to_string()));
169            }
170        }
171    }
172    if let Some(ut) = sc.pointer("/inputTranscription/text").and_then(Value::as_str).filter(|s| !s.is_empty()) {
173        out.push(UpstreamAction::UserTranscript(ut.to_string()));
174    }
175    if let Some(bt) = sc.pointer("/outputTranscription/text").and_then(Value::as_str).filter(|s| !s.is_empty()) {
176        out.push(UpstreamAction::BotTranscript(bt.to_string()));
177    }
178    if sc.get("turnComplete").and_then(Value::as_bool) == Some(true) {
179        out.push(UpstreamAction::TurnComplete);
180    }
181}

183/// Parses an upstream Gemini Live JSON text frame into zero or more [`UpstreamAction`] items.
184pub fn parse_gemini_server_frame(raw: &str) -> Vec<UpstreamAction> {
185    let Ok(v) = serde_json::from_str::<Value>(raw) else {
186        return Vec::new();
187    };
188    let mut out = Vec::new();
189    if v.get("setupComplete").is_some() {
190        out.push(UpstreamAction::Ready);
191    }
192    if let Some(sc) = v.get("serverContent") {
193        parse_server_content(sc, &mut out);
194    }
195    if let Some(fcs) = v.pointer("/toolCall/functionCalls").and_then(Value::as_array) {
196        let mut calls = Vec::new();
197        for fc in fcs {
198            let id = fc.get("id").and_then(Value::as_str).unwrap_or("").to_string();
199            let name = fc.get("name").and_then(Value::as_str).unwrap_or("").to_string();
200            let args = fc.get("args").cloned().unwrap_or_else(|| json!({}));
201            if !name.is_empty() {
202                calls.push((id, name, args));
203            }
204        }
205        if !calls.is_empty() {
206            out.push(UpstreamAction::ToolCalls(calls));
207        }
208    }
209    if let Some(sru) = v.get("sessionResumptionUpdate")
210        && sru.get("resumable").and_then(Value::as_bool) == Some(true)
211        && let Some(h) = sru.get("newHandle").and_then(Value::as_str).filter(|s| !s.is_empty())
212    {
213        out.push(UpstreamAction::ResumeHandle(h.to_string()));
214    }
215    if let Some(err_obj) = v.get("error") {
216        let msg = err_obj.get("message").and_then(Value::as_str).unwrap_or("Gemini Live error");
217        out.push(UpstreamAction::Error(msg.to_string()));
218    }
219    if v.get("goAway").is_some() {
220        out.push(UpstreamAction::GoAway);
221    }
222    out
