27/// Query parameters for `GET /dashboard/api/models`.
28#[derive(Deserialize)]
29pub struct ModelsQuery {
30    /// Provider name (`antigravity`, `claude`, `codex`).
31    pub provider: Option<String>,
32}
33
34const WEB_CORE_SESSION_COMMANDS: &[(&str, &str, &str)] = &[
35    ("new", "Start a fresh conversation session", "builtin"),
36    ("compact", "Condense conversation history to free context", "workflow"),
37    ("stop", "Cancel active CLI processes in chat", "builtin"),
38];
39
40fn build_slash_catalog(prov: &str) -> Vec<serde_json::Value> {
41    let mut cmds = Vec::new();
42    for &(cmd, desc, kind) in WEB_CORE_SESSION_COMMANDS {
43        cmds.push(serde_json::json!({ "command": cmd, "description": desc, "kind": kind }));
44    }
45    for &(cmd, desc) in workflow_commands_for_provider(prov) {
46        if !cmds.iter().any(|c| c.get("command").and_then(|v| v.as_str()) == Some(cmd)) {
47            cmds.push(serde_json::json!({ "command": cmd, "description": desc, "kind": "workflow" }));
48        }
49    }
50    cmds
51}

53/// Lists discovered models, effort levels, and slash commands (`GET /dashboard/api/models`).
54pub async fn handle_list_models(
55    ext: Option<Extension<Arc<WebAppCtx>>>,
56    headers: HeaderMap,
57    Query(q): Query<ModelsQuery>,
58) -> Response {
59    let ctx = match resolve_ctx(ext) {
60        Ok(c) => c,
61        Err(r) => return *r,
62    };
63    if let Err(r) = require_auth(&ctx, &headers).await {
64        return *r;
65    }
66    let prov = SessionData::effective_provider(q.provider.as_deref().unwrap_or(""), &ctx.config.provider);
67    let raw_models = discover_models_with_default(&ctx.cli, prov).await;
68    let grouped = group_discovered_models(&raw_models);
69    let entries: Vec<serde_json::Value> = grouped
70        .into_iter()
71        .map(|(model, mut efforts)| {
72            if efforts.is_empty() && session_supports_effort(prov, &model) {
73                efforts = match prov {
74                    "claude" => ["low", "medium", "high", "xhigh", "max"].map(str::to_string).to_vec(),
75                    "codex" => ["minimal", "low", "medium", "high", "xhigh"].map(str::to_string).to_vec(),
76                    _ => crate::cli::antigravity::AntigravityCli::supported_efforts_for_model(&model),
77                };
78            }
79            serde_json::json!({ "model": model, "efforts": efforts })
80        })
81        .collect();
82    let commands = build_slash_catalog(prov);
83    Json(serde_json::json!({ "provider": prov, "models": entries, "commands": commands })).into_response()
84}
