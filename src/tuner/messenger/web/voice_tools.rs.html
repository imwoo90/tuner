25async fn resolve_key_or_default(
26    ctx: &WebAppCtx,
27    explicit: Option<&str>,
28    current_topic: Option<&str>,
29    prefer_asking: bool,
30) -> Option<SessionKey> {
31    if let Some(raw) = explicit.map(str::trim).filter(|s| !s.is_empty())
32        && let Ok(k) = raw.parse::<SessionKey>()
33    {
34        return Some(k);
35    }
36    if prefer_asking {
37        let all = ctx.sessions.list_all().await.unwrap_or_default();
38        for s in &all {
39            if super::api_sessions::check_busy_and_asking(ctx, s).await.1 {
40                return Some(s.session_key());
41            }
42        }
43    }
44    if let Some(cur) = current_topic.map(str::trim).filter(|s| !s.is_empty())
45        && let Ok(k) = cur.parse::<SessionKey>()
46    {
47        return Some(k);
48    }
49    ctx.sessions.list_all().await.ok().and_then(|mut v| {
50        v.sort_by(|a, b| b.last_active.cmp(&a.last_active));
51        v.into_iter().next().map(|s| s.session_key())
52    })
53}

55async fn exec_list_topics(ctx: &WebAppCtx) -> Value {
56    let mut all = ctx.sessions.list_all().await.unwrap_or_default();
57    all.sort_by(|a, b| b.last_active.cmp(&a.last_active));
58    let mut items = Vec::new();
59    for s in all.iter().take(20) {
60        let sum = build_session_summary(ctx, s).await;
61        items.push(json!({
62            "session_key": sum.session_key,
63            "title": sum.topic_name.unwrap_or_else(|| format!("{} #{:?}", sum.transport, sum.topic_id)),
64            "transport": sum.transport,
65            "provider": sum.provider,
66            "model": sum.model,
67            "busy": sum.busy,
68            "asking": sum.asking,
69            "last_active": sum.last_active
70        }));
71    }
72    json!({ "count": items.len(), "topics": items })
73}

75async fn exec_get_context(ctx: &WebAppCtx, args: &Value, cur: Option<&str>) -> Value {
76    let sk_arg = args.get("session_key").and_then(Value::as_str);
77    let Some(key) = resolve_key_or_default(ctx, sk_arg, cur, false).await else {
78        return json!({ "error": "No active topics exist yet." });
79    };
80    let Ok(Some(sess)) = ctx.sessions.get_active(&key).await else {
81        return json!({ "error": "Topic not found." });
82    };
83    let prov = SessionData::effective_provider(&sess.provider, &ctx.config.provider);
84    let sid = sess.get_session_id(prov);
85    let max_n = args.get("max_messages").and_then(Value::as_u64).unwrap_or(6).clamp(1, 15) as usize;
86    let recent: Vec<Value> = collect_history_messages(ctx, &sess, prov, &sid)
87        .into_iter()
88        .rev()
89        .take(max_n)
90        .collect::<Vec<_>>()
91        .into_iter()
92        .rev()
93        .map(|m| {
94            let txt: String = m["text"].as_str().unwrap_or("").chars().take(600).collect();
95            json!({ "sender": m["sender"], "text": txt, "timestamp": m["timestamp"] })
96        })
97        .collect();
98    let ask = build_active_ask_view(ctx, &sess, prov, &sid).await;
99    let stream_tail = ctx
100        .events
101        .get_live_stream(&key.storage_key())
102        .await
103        .map(|s| s.chars().rev().take(500).collect::<Vec<_>>().into_iter().rev().collect::<String>());
104    let summary = build_session_summary(ctx, &sess).await;
105    json!({ "session": summary, "active_ask": ask, "live_stream_tail": stream_tail, "recent_messages": recent })
106}

112async fn create_voice_topic(
113    ctx: &Arc<WebAppCtx>,
114    title: &str,
115    tr: Option<String>,
116    raw_spec: (Option<&str>, Option<&str>, Option<&str>),
117) -> Result<(SessionKey, String, String, Option<String>), String> {
118    let (p, m, e) = resolve_spoken_model_spec(ctx, &ctx.config.provider, raw_spec.0, raw_spec.1, raw_spec.2).await?;
119    let req = CreateSessionPayload {
120        transport: tr,
121        chat_id: None,
122        silent_greeting: true,
123        topic_name: Some(title.to_string()),
124        provider: Some(p.clone()),
125        model: Some(m.clone()),
126        effort: e.clone(),
127    };
128    let sess = super::tg_topics::create_session_from_payload(ctx, req)
129        .await
130        .map_err(|_| "Failed to create topic".to_string())?;
131    emit_session_updated(ctx, &sess, false, false).await;
132    Ok((sess.session_key(), p, m, e))
133}

135async fn exec_delegate(ctx: &Arc<WebAppCtx>, args: &Value, cur: &mut Option<String>) -> (Value, Option<String>) {
136    let Some(prompt) = opt_str(args, "prompt").map(str::to_string) else {
137        return (json!({ "error": "prompt is required" }), None);
138    };
139    let tr = opt_str(args, "transport").map(str::to_string);
140    let raw_spec = (opt_str(args, "provider"), opt_str(args, "model"), opt_str(args, "effort"));
141
142    let resolved = if let Some(title) = opt_str(args, "create_new_topic_title") {
143        create_voice_topic(ctx, title, tr, raw_spec).await
144    } else if let Some(k) = resolve_key_or_default(ctx, opt_str(args, "session_key"), cur.as_deref(), false).await {
145        apply_voice_spec_to_session(ctx, &k, raw_spec.0, raw_spec.1, raw_spec.2, None)
146            .await
147            .map(|(p, m, e)| (k, p, m, e))
148    } else {
149        let auto_title: String = prompt.chars().take(28).collect();
150        create_voice_topic(ctx, &auto_title, Some("web".to_string()), raw_spec).await
151    };
152    let (target_key, eff_prov, eff_model, eff_effort) = match resolved {
153        Ok(v) => v,
154        Err(e) => return (json!({ "error": e }), None),
155    };
156
157    let sk = target_key.storage_key();
158    *cur = Some(sk.clone());
159    let ctx_bg = ctx.clone();
160    tokio::spawn(async move {
161        execute_web_turn(ctx_bg, target_key, prompt, Vec::new()).await;
162    });
163    let out = json!({
164        "delegated": true,
165        "session_key": sk,
166        "provider": eff_prov,
167        "model": eff_model,
168        "effort": eff_effort
169    });
170    (out, Some(sk))
171}

173async fn exec_answer_ask(ctx: &Arc<WebAppCtx>, args: &Value, cur: Option<&str>) -> Value {
174    let sk_arg = args.get("session_key").and_then(Value::as_str);
175    let Some(key) = resolve_key_or_default(ctx, sk_arg, cur, true).await else {
176        return json!({ "error": "No topic with an open AskQuestion found." });
177    };
178    let skip = args.get("skip").and_then(Value::as_bool).unwrap_or(false);
179    let selected_indices = args.get("option_index").and_then(Value::as_u64).map(|i| vec![i as usize]).unwrap_or_default();
180    let payload = AskAnswerPayload {
181        session_key: key.storage_key(),
182        action: if skip { "skip".to_string() } else { "answer".to_string() },
183        answer: args.get("answer").and_then(Value::as_str).map(str::to_string),
184        selected_indices,
185    };
186    let resp = process_validated_ask(ctx.clone(), key.clone(), payload).await;
187    json!({ "ok": resp.status().is_success(), "session_key": key.storage_key() })
188}

190async fn exec_stop_topic(ctx: &Arc<WebAppCtx>, args: &Value, cur: Option<&str>) -> Value {
191    let sk_arg = args.get("session_key").and_then(Value::as_str);
192    let Some(key) = resolve_key_or_default(ctx, sk_arg, cur, false).await else {
193        return json!({ "error": "No active topic found." });
194    };
195    let a = ctx.cli.sessions.interrupt(key.chat_id, key.topic_id).await;
196    let b = crate::cli::claude::interrupt_session(key.chat_id, key.topic_id).await;
197    let c = crate::cli::codex::interrupt_session(key.chat_id, key.topic_id).await;
198    if let Ok(Some(sess)) = ctx.sessions.get_active(&key).await {
199        emit_session_updated(ctx, &sess, false, false).await;
200    }
201    json!({ "ok": true, "session_key": key.storage_key(), "interrupted": a || b || c })
202}

204/// Executes a single Gemini Live tool call and returns `(tool_output_json, optional_focus_topic_key)`.
205pub async fn execute_voice_tool(
206    ctx: &Arc<WebAppCtx>,
207    current_topic: &mut Option<String>,
208    name: &str,
209    args: &Value,
210) -> (Value, Option<String>) {
211    match name {
212        "list_topics" => (exec_list_topics(ctx).await, None),
213        "get_topic_context" => (exec_get_context(ctx, args, current_topic.as_deref()).await, None),
214        "list_available_models" => (exec_list_available_models(ctx, args).await, None),
215        "delegate_to_topic" => exec_delegate(ctx, args, current_topic).await,
216        "configure_topic" => {
217            let sk_arg = args.get("session_key").and_then(Value::as_str);
218            match resolve_key_or_default(ctx, sk_arg, current_topic.as_deref(), false).await {
219                Some(k) => (exec_configure_topic(ctx, &k, args).await, None),
220                None => (json!({ "error": "No active topic found." }), None),
221            }
222        }
223        "answer_topic_question" => (exec_answer_ask(ctx, args, current_topic.as_deref()).await, None),
224        "stop_topic" => (exec_stop_topic(ctx, args, current_topic.as_deref()).await, None),
225        "switch_active_topic" => {
226            if let Some(sk) = args.get("session_key").and_then(Value::as_str).map(str::trim).filter(|s| !s.is_empty()) {
227                *current_topic = Some(sk.to_string());
228                (json!({ "ok": true, "focused": sk }), Some(sk.to_string()))
229            } else {
230                (json!({ "error": "session_key is required" }), None)
231            }
232        }
233        other => (json!({ "error": format!("Unknown tool `{other}`") }), None),
234    }
