23const COOKIE_NAME: &str = "tuner_dash_sid";
24const CSRF_HEADER: &str = "x-tuner-csrf";

26/// Extracts the `tuner_dash_sid` cookie value from HTTP request headers.
27pub fn extract_session_cookie(headers: &HeaderMap) -> Option<String> {
28    let raw = headers.get(header::COOKIE)?.to_str().ok()?;
29    for part in raw.split(';') {
30        if let Some((k, v)) = part.trim().split_once('=')
31            && k.trim() == COOKIE_NAME
32            && !v.trim().is_empty()
33        {
34            return Some(v.trim().to_string());
35        }
36    }
37    None
38}

40fn is_https_request(headers: &HeaderMap) -> bool {
41    headers
42        .get("x-forwarded-proto")
43        .and_then(|v| v.to_str().ok())
44        .is_some_and(|p| p.trim().eq_ignore_ascii_case("https"))
45}

47fn build_session_cookie(sid: &str, max_age_secs: i64, secure: bool) -> String {
48    let mut s = format!("{COOKIE_NAME}={sid}; HttpOnly; SameSite=Lax; Path=/; Max-Age={max_age_secs}");
49    if secure {
50        s.push_str("; Secure");
51    }
52    s
53}

55fn extract_client_ip(headers: &HeaderMap, peer: Option<SocketAddr>) -> IpAddr {
56    let peer_ip = peer.map(|s| s.ip());
57    if peer_ip.is_some_and(|ip| ip.is_loopback())
58        && headers.contains_key("cf-ray")
59        && let Some(ip) = headers
60            .get("cf-connecting-ip")
61            .and_then(|v| v.to_str().ok())
62            .and_then(|s| s.trim().parse::<IpAddr>().ok())
63    {
64        return ip;
65    }
66    peer_ip.unwrap_or(IpAddr::V4(Ipv4Addr::LOCALHOST))
67}

69fn is_preview_bot_or_prefetch(headers: &HeaderMap) -> bool {
70    for hdr in ["sec-purpose", "purpose", "x-moz"] {
71        if headers
72            .get(hdr)
73            .and_then(|v| v.to_str().ok())
74            .is_some_and(|v| v.to_ascii_lowercase().contains("prefetch"))
75        {
76            return true;
77        }
78    }
79    if let Some(ua) = headers.get(header::USER_AGENT).and_then(|v| v.to_str().ok()) {
80        let low = ua.to_ascii_lowercase();
81        if [
82            "telegrambot",
83            "slackbot",
84            "discordbot",
85            "whatsapp",
86            "twitterbot",
87            "facebookexternalhit",
88            "kakaotalk-scrap",
89        ]
90        .iter()
91        .any(|b| low.contains(b))
92        {
93            return true;
94        }
95    }
96    false
97}

99/// Authenticates a read-only (`GET`) API request via `tuner_dash_sid` cookie.
100pub async fn require_auth(
101    ctx: &Arc<WebAppCtx>,
102    headers: &HeaderMap,
103) -> Result<AuthSession, Box<Response>> {
104    let Some(sid) = extract_session_cookie(headers) else {
105        return Err(Box::new(json_err(StatusCode::UNAUTHORIZED, "Authentication required")));
106    };
107    ctx.auth
108        .verify_session(&sid)
109        .await
110        .ok_or_else(|| Box::new(json_err(StatusCode::UNAUTHORIZED, "Invalid or expired session")))
111}

113/// Authenticates a mutating API request via both `tuner_dash_sid` cookie and `x-tuner-csrf`.
114pub async fn require_mutation_auth(
115    ctx: &Arc<WebAppCtx>,
116    headers: &HeaderMap,
117) -> Result<AuthSession, Box<Response>> {
118    let sess = require_auth(ctx, headers).await?;
119    let Some(csrf) = headers.get(CSRF_HEADER).and_then(|v| v.to_str().ok()) else {
120        return Err(Box::new(json_err(StatusCode::FORBIDDEN, "Missing CSRF token")));
121    };
122    if !constant_time_eq(csrf.as_bytes(), sess.csrf_token.as_bytes()) {
123        return Err(Box::new(json_err(StatusCode::FORBIDDEN, "Invalid CSRF token")));
124    }
125    Ok(sess)
126}

133/// Exchanges a single-use Magic Link token (`GET /dashboard/magic/:token`) for a session cookie.
134pub async fn handle_magic_exchange(
135    ext: Option<Extension<Arc<WebAppCtx>>>,
136    Path(token): Path<String>,
137    headers: HeaderMap,
138) -> Response {
139    if is_preview_bot_or_prefetch(&headers) {
140        return StatusCode::NO_CONTENT.into_response();
141    }
142    let ctx = match resolve_ctx(ext) {
143        Ok(c) => c,
144        Err(r) => return *r,
145    };
146    match ctx.auth.consume_magic_token(&token).await {
147        Ok(Some(sess)) => {
148            let cookie = build_session_cookie(&sess.sid, 30 * 86_400, is_https_request(&headers));
149            let mut res = (StatusCode::SEE_OTHER, [(header::LOCATION, "/dashboard")]).into_response();
150            if let Ok(hv) = header::HeaderValue::from_str(&cookie) {
151                res.headers_mut().insert(header::SET_COOKIE, hv);
152            }
153            res
154        }
155        _ => (
156            StatusCode::UNAUTHORIZED,
157            Html(
158                "<!doctype html><html><body style='background:#FAF9F7;color:#18181B;font-family:-apple-system,sans-serif;padding:36px;line-height:1.6;'><h2>매직 링크가 만료되었거나 이미 사용되었습니다 · Magic Link Expired or Already Used</h2><p>텔레그램에서 <code>/dashboard</code> 를 입력하거나 터미널에서 <code>tuner --dashboard</code> 로 새 링크를 발급해 주세요.</p><p><a href='/dashboard/login' style='color:#18181B;font-weight:600;'>로그인 화면으로 이동 · Go to Sign In →</a></p></body></html>",
159            ),
160        )
161            .into_response(),
162    }
163}

172/// Verifies owner PIN (`POST /dashboard/api/auth/login`) with per-IP 5-strike lockout.
173pub async fn handle_login(
174    ext: Option<Extension<Arc<WebAppCtx>>>,
175    peer: Option<ConnectInfo<SocketAddr>>,
176    headers: HeaderMap,
177    Json(payload): Json<PinPayload>,
178) -> Response {
179    let ctx = match resolve_ctx(ext) {
180        Ok(c) => c,
181        Err(r) => return *r,
182    };
183    let ip = extract_client_ip(&headers, peer.map(|c| c.0));
184    if let Err(retry_after) = ctx.auth.check_login_rate_limit(ip).await {
185        return (
186            StatusCode::TOO_MANY_REQUESTS,
187            Json(serde_json::json!({
188                "error": format!("Too many failed login attempts. Retry in {}s.", retry_after),
189                "retry_after": retry_after
190            })),
191        )
192            .into_response();
193    }
194    if !ctx.auth.has_pin().await {
195        return json_err(StatusCode::FORBIDDEN, "No owner PIN configured. Use a Magic Link first.");
196    }
197    if !ctx.auth.verify_pin(&payload.pin).await.unwrap_or(false) {
198        ctx.auth.record_login_failure(ip).await;
199        return json_err(StatusCode::UNAUTHORIZED, "Invalid PIN or passphrase");
200    }
201    ctx.auth.clear_login_failures(ip).await;
202    let Ok(sess) = ctx.auth.create_session().await else {
203        return json_err(StatusCode::INTERNAL_SERVER_ERROR, "Failed to create session");
204    };
205    let cookie = build_session_cookie(&sess.sid, 30 * 86_400, is_https_request(&headers));
206    let mut res = Json(serde_json::json!({
207        "authenticated": true,
208        "csrf_token": sess.csrf_token,
209        "profile": ctx.profile
210    }))
211    .into_response();
212    if let Ok(hv) = header::HeaderValue::from_str(&cookie) {
213        res.headers_mut().insert(header::SET_COOKIE, hv);
214    }
215    res
216}

218/// Returns current session status (`GET /dashboard/api/auth/me`).
219pub async fn handle_auth_me(
220    ext: Option<Extension<Arc<WebAppCtx>>>,
221    headers: HeaderMap,
222) -> Response {
223    let ctx = match resolve_ctx(ext) {
224        Ok(c) => c,
225        Err(r) => return *r,
226    };
227    let has_pin = ctx.auth.has_pin().await;
228    let tg_connected = ctx.bot.read().await.is_some();
229    let bot_username = super::api_telegram::resolve_bot_username(&ctx).await;
230    let lang = ctx.config.language.as_deref().unwrap_or("ko");
231    let (any_ready, def_prov, ready_providers) =
232        super::provider_status::resolve_ready_provider_summary(&ctx.tuner_home, &ctx.config.provider);
233    match require_auth(&ctx, &headers).await {
234        Ok(sess) => Json(serde_json::json!({
235            "authenticated": true,
236            "csrf_token": sess.csrf_token,
237            "has_pin": has_pin,
238            "profile": ctx.profile,
239            "language": lang,
240            "telegram_connected": tg_connected,
241            "bot_username": bot_username,
242            "default_provider": def_prov,
243            "default_model": ctx.config.model,
244            "needs_onboarding": !any_ready,
245            "ready_providers": ready_providers
246        }))
247        .into_response(),
248        Err(_) => (
249            StatusCode::UNAUTHORIZED,
250            Json(serde_json::json!({ "authenticated": false, "has_pin": has_pin, "language": lang })),
251        )
252            .into_response(),
253    }
254}

256/// Sets or updates the owner PIN (`POST /dashboard/api/auth/pin`).
257pub async fn handle_set_pin(
258    ext: Option<Extension<Arc<WebAppCtx>>>,
259    headers: HeaderMap,
260    Json(payload): Json<PinPayload>,
261) -> Response {
262    let ctx = match resolve_ctx(ext) {
263        Ok(c) => c,
264        Err(r) => return *r,
265    };
266    if let Err(r) = require_mutation_auth(&ctx, &headers).await {
267        return *r;
268    }
269    match ctx.auth.set_pin(&payload.pin).await {
270        Ok(()) => Json(serde_json::json!({ "ok": true, "has_pin": true })).into_response(),
271        Err(e) => json_err(StatusCode::BAD_REQUEST, &e),
272    }
273}

275fn clear_cookie_response(headers: &HeaderMap) -> Response {
276    let clear = build_session_cookie("", 0, is_https_request(headers));
277    let mut res = Json(serde_json::json!({ "ok": true })).into_response();
278    if let Ok(hv) = header::HeaderValue::from_str(&clear) {
279        res.headers_mut().insert(header::SET_COOKIE, hv);
280    }
281    res
282}

284/// Logs out the current session (`POST /dashboard/api/auth/logout`).
285pub async fn handle_logout(
286    ext: Option<Extension<Arc<WebAppCtx>>>,
287    headers: HeaderMap,
288) -> Response {
289    let ctx = match resolve_ctx(ext) {
290        Ok(c) => c,
291        Err(r) => return *r,
292    };
293    let sess = match require_mutation_auth(&ctx, &headers).await {
294        Ok(s) => s,
295        Err(r) => return *r,
296    };
297    let _ = ctx.auth.revoke_session(&sess.sid).await;
298    clear_cookie_response(&headers)
299}

301/// Revokes all active sessions and pending Magic Links (`POST /dashboard/api/auth/revoke_all`).
302pub async fn handle_revoke_all(
303    ext: Option<Extension<Arc<WebAppCtx>>>,
304    headers: HeaderMap,
305) -> Response {
306    let ctx = match resolve_ctx(ext) {
307        Ok(c) => c,
308        Err(r) => return *r,
309    };
310    if let Err(r) = require_mutation_auth(&ctx, &headers).await {
311        return *r;
312    }
313    let _ = ctx.auth.revoke_all().await;
314    clear_cookie_response(&headers)
