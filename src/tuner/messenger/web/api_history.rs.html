53/// Builds an [`ActiveAskView`] snapshot when `sess` has a live `AskQuestion` prompt.
54pub(crate) async fn build_active_ask_view(
55    ctx: &WebAppCtx,
56    sess: &SessionData,
57    prov: &str,
58    sid: &str,
59) -> Option<ActiveAskView> {
60    let alive = (cfg!(test) && sid.starts_with("mock-"))
61        || ctx
62            .cli
63            .is_ask_backing_alive(prov, sess.chat_id, sess.topic_id, sid)
64            .await;
65    if sid.is_empty() || !ctx.cli.sessions.is_ask_active(sid).await || !alive {
66        return None;
67    }
68    let st = ctx.cli.sessions.get_ask_state(sid).await?;
69    Some(ActiveAskView {
70        session_key: sess.session_key().storage_key(),
71        session_id: sid.to_string(),
72        provider: prov.to_string(),
73        questions: st.questions,
74        current_index: st.current_index,
75        answers: st.answers,
76    })
77}

79pub(crate) fn collect_history_messages(
80    ctx: &WebAppCtx,
81    sess: &SessionData,
82    prov: &str,
83    sid: &str,
84) -> Vec<serde_json::Value> {
85    let mut sids = sess.get_past_session_ids(prov);
86    if !sid.is_empty() && !sids.contains(&sid.to_string()) {
87        sids.push(sid.to_string());
88    }
89    let mut messages = Vec::new();
90    for id in &sids {
91        for entry in extract_topic_history(&ctx.working_dir, id) {
92            let files = filter_safe_preview_paths(ctx, &entry.text);
93            messages.push(serde_json::json!({
94                "session_id": id,
95                "timestamp": entry.timestamp,
96                "sender": entry.sender,
97                "text": entry.text,
98                "is_success": entry.is_success,
99                "error": entry.error,
100                "files": files
101            }));
102        }
103    }
104    messages
105}

107/// Returns message history, active `AskQuestion`, and live streaming buffer for a session.
108pub async fn handle_session_history(
109    ext: Option<Extension<Arc<WebAppCtx>>>,
110    headers: HeaderMap,
111    Query(q): Query<HistoryQuery>,
112) -> Response {
113    let ctx = match resolve_ctx(ext) {
114        Ok(c) => c,
115        Err(r) => return *r,
116    };
117    if let Err(r) = require_auth(&ctx, &headers).await {
118        return *r;
119    }
120    let Ok(key) = q.session_key.parse::<SessionKey>() else {
121        return json_err(StatusCode::BAD_REQUEST, "Invalid session_key");
122    };
123    let Ok(Some(sess)) = ctx.sessions.get_active(&key).await else {
124        return json_err(StatusCode::NOT_FOUND, "Session not found");
125    };
126    let prov = SessionData::effective_provider(&sess.provider, &ctx.config.provider);
127    let sid = sess.get_session_id(prov);
128    let messages = collect_history_messages(&ctx, &sess, prov, &sid);
129    let active_ask = build_active_ask_view(&ctx, &sess, prov, &sid).await;
130    let live_stream = ctx.events.get_live_stream(&q.session_key).await;
131    let summary = build_session_summary(&ctx, &sess).await;
132    Json(serde_json::json!({
133        "session": summary,
134        "messages": messages,
135        "active_ask": active_ask,
136        "live_stream": live_stream
137    }))
138    .into_response()
139}

141async fn read_multipart_upload(
142    mut multipart: Multipart,
143) -> (Option<String>, Option<(String, Vec<u8>)>) {
144    let mut session_key = None;
145    let mut first_file = None;
146    while let Ok(Some(field)) = multipart.next_field().await {
147        let name = field.name().unwrap_or("").to_string();
148        if name == "session_key" {
149            if let Ok(t) = field.text().await {
150                session_key = Some(t.trim().to_string());
151            }
152        } else if name == "file" && first_file.is_none() {
153            let fname = field.file_name().unwrap_or("upload.bin").to_string();
154            if let Ok(bytes) = field.bytes().await {
155                first_file = Some((fname, bytes.to_vec()));
156            }
157        }
158    }
159    (session_key, first_file)
160}

162/// Handles multipart file uploads (`POST /dashboard/api/sessions/upload`) into `telegram_files/`.
163pub async fn handle_upload_file(
164    ext: Option<Extension<Arc<WebAppCtx>>>,
165    headers: HeaderMap,
166    multipart: Multipart,
167) -> Response {
168    let ctx = match resolve_ctx(ext) {
169        Ok(c) => c,
170        Err(r) => return *r,
171    };
172    if let Err(r) = require_mutation_auth(&ctx, &headers).await {
173        return *r;
174    }
175    let (session_key, first_file) = read_multipart_upload(multipart).await;
176    let Some((raw_name, bytes)) = first_file else {
177        return json_err(StatusCode::BAD_REQUEST, "Missing file field");
178    };
179    let clean = sanitize_filename(&raw_name);
180    let safe_name = if clean.is_empty() { "upload.bin".to_string() } else { clean };
181    let stored_name = format!("web_{}_{}", chrono::Utc::now().timestamp_millis(), safe_name);
182    let dest_dir = ctx.working_dir.join("telegram_files");
183    if tokio::fs::create_dir_all(&dest_dir).await.is_err() {
184        return json_err(StatusCode::INTERNAL_SERVER_ERROR, "Failed to create upload dir");
185    }
186    if tokio::fs::write(dest_dir.join(&stored_name), &bytes).await.is_err() {
187        return json_err(StatusCode::INTERNAL_SERVER_ERROR, "Failed to write uploaded file");
188    }
189    let rel_path = format!("telegram_files/{stored_name}");
190    let pending = attach_uploaded_file(&ctx, session_key.as_deref(), &rel_path).await;
191    Json(serde_json::json!({
192        "path": rel_path,
193        "filename": stored_name,
194        "size_bytes": bytes.len(),
195        "pending_attachments": pending
196    }))
197    .into_response()
198}

200async fn attach_uploaded_file(
201    ctx: &Arc<WebAppCtx>,
202    sk_opt: Option<&str>,
203    rel_path: &str,
204) -> Vec<String> {
205    let Some(sk) = sk_opt.and_then(|s| s.parse::<SessionKey>().ok()) else {
206        return vec![rel_path.to_string()];
207    };
208    let Ok(Some(updated)) = ctx
209        .sessions
210        .mutate_session(&sk, None, |cur| {
211            if !cur.pending_attachments.contains(&rel_path.to_string()) {
212                cur.pending_attachments.push(rel_path.to_string());
213                return true;
214            }
215            false
216        })
217        .await
218    else {
219        return vec![rel_path.to_string()];
220    };
221    let (busy, asking) = check_busy_and_asking(ctx, &updated).await;
222    emit_session_updated(ctx, &updated, busy, asking).await;
223    updated.pending_attachments
224}
