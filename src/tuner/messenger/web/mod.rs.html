1//! # Responsive Web UI & PWA Control Plane (`web/mod.rs`)
2//!
3//! ## Overview
4//! Hosts the embedded pure-Rust Web Dashboard (`:8743/dashboard`), unified session & chat
5//! REST/WebSocket APIs, PBKDF2 + Magic Link auth, subagent tree, and operational panels.
6//!
7//! ## Submodules
8//! - [`api_ask`] (`api_ask.rs`): Interactive `AskQuestion` answers.
9//! - [`api_ask_helpers`] (`api_ask_helpers.rs`): PTY helpers for `AskQuestion`.
10//! - [`api_chat`] (`api_chat.rs`): Prompt send, `/stop`, `/new`, and prune.
11//! - [`api_files`] (`api_files.rs`): Sandboxed `/file` preview and `/vault`.
12//! - [`api_history`] (`api_history.rs`): Chat history and multipart upload.
13//! - [`api_hitl`] (`api_hitl.rs`): HITL remote browser status and launcher.
14//! - [`api_models`] (`api_models.rs`): Model, effort, and slash catalog.
15//! - [`api_ops`] (`api_ops.rs`): `/usage` and `/cron` endpoints.
16//! - [`api_providers`] (`api_providers.rs`): CLI install and OAuth login API.
17//! - [`api_sessions`] (`api_sessions.rs`): Session CRUD and config patch.
18//! - [`api_telegram`] (`api_telegram.rs`): Telegram bot setup and hot-connect API.
19//! - [`assets_http`] (`assets_http.rs`): Embedded static PWA assets.
20//! - [`auth`] (`auth.rs`): Magic Links and PBKDF2 PIN logic.
21//! - [`auth_http`] (`auth_http.rs`): Auth endpoints and CSRF guards.
22//! - [`auth_store`] (`auth_store.rs`): `0600` auth persistence and lockouts.
23//! - [`cli_dash`] (`cli_dash.rs`): `--dashboard` CLI and `/dashboard` command.
24//! - [`events`] (`events.rs`): `WebEventHub` broadcast and stream buffer.
25//! - [`events_bridge`] (`events_bridge.rs`): Telegram/Cron Web event hooks.
26//! - [`provider_install`] (`provider_install.rs`): Background CLI installer.
27//! - [`provider_oauth`] (`provider_oauth.rs`): Safe PTY CLI OAuth bridge.
28//! - [`provider_oauth_watch`] (`provider_oauth_watch.rs`): Web-in-Web OAuth auto-harvester.
29//! - [`provider_status`] (`provider_status.rs`): CLI readiness and `.env` store.
30//! - [`subagent_tree`] (`subagent_tree.rs`): Warm PID and subagent tree.
31//! - [`tg_topics`] (`tg_topics.rs`): Telegram forum topic CRUD sync.
32//! - [`turn_boot`] (`turn_boot.rs`): Session init and handover boot.
33//! - [`turn_runner`] (`turn_runner.rs`): Lock-pooled Web turn runner.
34//! - [`turn_stream`] (`turn_stream.rs`): Live `StreamEvent` pump.
35//! - [`ws`] (`ws.rs`): `/dashboard/api/ws` WebSocket handler.
36//!
37//! ## Search Tags
38//! #web-dashboard, #pwa-ui, #control-plane, #axum-router

94/// Shared control-plane state powering all `/dashboard` HTTP and WebSocket endpoints.
95pub struct WebAppCtx {
96    /// Active profile name (`default`, etc.).
97    pub profile: String,
98    /// Tuner home directory (`~/.tuner`).
99    pub tuner_home: PathBuf,
100    /// Per-profile encrypted vault directory.
101    pub vault_dir: PathBuf,
102    /// Active workspace root directory.
103    pub working_dir: PathBuf,
104    /// Loaded profile configuration.
105    pub config: Arc<CliConfig>,
106    /// Shared session manager and `LockPool`.
107    pub sessions: Arc<SessionManager>,
108    /// Shared multi-provider CLI orchestrator.
109    pub cli: Arc<AntigravityCli>,
110    /// Shared cron manager.
111    pub cron_manager: Arc<CronManager>,
112    /// Forum topic name cache.
113    pub topic_cache: Arc<TopicNameCache>,
114    /// Shared internal message bus.
115    pub bus: Arc<MessageBus>,
116    /// Optional attached Telegram bot handle (`None` in Web-Standalone mode).
117    pub bot: RwLock<Option<Bot>>,
118    /// Real-time WebSocket event broadcast hub.
119    pub events: Arc<WebEventHub>,
120    /// Persistent `0600` Web authentication store.
121    pub auth: Arc<WebAuthStore>,
122}

124impl WebAppCtx {
125    /// Attaches a validated Telegram `Bot` handle after startup initialization succeeds.
126    pub async fn set_bot(&self, bot: Bot) {
127        *self.bot.write().await = Some(bot);
128    }
129
130    /// Clones the attached Telegram `Bot` handle if Co-Resident mode is active.
131    pub async fn bot_clone(&self) -> Option<Bot> {
132        self.bot.read().await.clone()
133    }
134}

138/// Registers the global [`WebAppCtx`] instance for `:8743` and Telegram hooks.

143/// Returns the global [`WebAppCtx`] instance once initialized.
144pub fn global_web_ctx() -> Option<Arc<WebAppCtx>> {
145    GLOBAL_WEB_CTX.get().cloned()

148pub(crate) fn resolve_ctx(ext: Option<Extension<Arc<WebAppCtx>>>) -> Result<Arc<WebAppCtx>, Box<Response>> {
149    if let Some(Extension(c)) = ext {
150        return Ok(c);
151    }
152    global_web_ctx().ok_or_else(|| {
153        Box::new(auth_http::json_err(StatusCode::SERVICE_UNAVAILABLE, "Dashboard context is still initializing"))
154    })
155}

157/// MessageBus [`Transport`] adapter delivering cron & heartbeat results to the Web Dashboard.
158pub struct WebTransport {
159    ctx: Arc<WebAppCtx>,
160}
161
162impl WebTransport {
163    /// Creates a new `WebTransport` bound to `ctx`.
164    pub fn new(ctx: Arc<WebAppCtx>) -> Self {
165        Self { ctx }
166    }
167
168    async fn pick_fallback_web_key(&self, topic_hint: Option<i64>) -> SessionKey {
169        if let Some(tid) = topic_hint.filter(|&t| t > 0) {
170            return SessionKey::web(tid);
171        }
172        let all = self.ctx.sessions.list_all().await.unwrap_or_default();
173        if let Some(s) = all.iter().find(|s| s.transport == "web") {
174            return s.session_key();
175        }
176        if let Some(s) = all.first() {
177            return s.session_key();
178        }
179        SessionKey::web(1)
180    }
181
182    async fn resolve_target_key(&self, envelope: &Envelope, is_broadcast: bool) -> Option<SessionKey> {
183        let has_bot = self.ctx.bot.read().await.is_some();
184        if envelope.transport == "web" {
185            if envelope.chat_id != 0 {
186                let tg_key = SessionKey::telegram(envelope.chat_id, envelope.topic_id);
187                if self.ctx.sessions.get_active(&tg_key).await.ok().flatten().is_some() {
188                    return Some(tg_key);
189                }
190            }
191            return Some(self.pick_fallback_web_key(envelope.topic_id).await);
192        }
193        if is_broadcast && has_bot {
194            return None;
195        }
196        if envelope.chat_id != 0 {
197            let tg_key = SessionKey::telegram(envelope.chat_id, envelope.topic_id);
198            if self.ctx.sessions.get_active(&tg_key).await.ok().flatten().is_some() {
199                return Some(tg_key);
200            }
201        }
202        Some(self.pick_fallback_web_key(envelope.topic_id).await)
203    }
204
205    async fn deliver_resolved(&self, key: SessionKey, envelope: &Envelope) -> Result<(), String> {
206        let clean_text = strip_bus_fallback_banner(&envelope.result_text);
207        if let Ok(Some(sess)) = self.ctx.sessions.get_active(&key).await {
208            let prov = SessionData::effective_provider(&sess.provider, &self.ctx.config.provider);
209            let mut sid = sess.get_session_id(prov);
210            if sid.is_empty() {
211                sid = sess.get_past_session_ids(prov).last().cloned().unwrap_or_default();
212            }
213            if !sid.is_empty() {
214                crate::messenger::telegram::history::log_telegram_message(
215                    &self.ctx.working_dir,
216                    &sid,
217                    (sess.topic_id, sess.topic_name.as_deref()),
218                    "bot",
219                    None,
220                    clean_text,
221                    (!envelope.is_error, envelope.is_error.then_some(envelope.status.as_str())),
222                );
223            }
224        }
225        let files = api_files::filter_safe_preview_paths(&self.ctx, clean_text);
226        self.ctx
227            .events
228            .publish(WebEvent::TurnResult {
229                session_key: key.storage_key(),
230                session_id: None,
231                text: clean_text.to_string(),
232                is_error: envelope.is_error,
233                files,
234                timestamp: chrono::Utc::now().to_rfc3339(),
235            })
236            .await;
237        Ok(())
238    }
239}

241fn strip_bus_fallback_banner(text: &str) -> &str {
242    const PREFIX: &str = "**Delivery fallback**\n\nTarget transport '";
243    if text.starts_with(PREFIX)
244        && let Some((_, rest)) = text.split_once("\n\n---\n")
245    {
246        return rest;
247    }
248    text
249}

251#[async_trait::async_trait]
252impl Transport for WebTransport {
253    fn transport_name(&self) -> &str {
254        "web"
255    }
256
257    async fn deliver(&self, envelope: &Envelope) -> Result<(), String> {
258        if let Some(key) = self.resolve_target_key(envelope, false).await {
259            self.deliver_resolved(key, envelope).await?;
260        }
261        Ok(())
262    }
263
264    async fn deliver_broadcast(&self, envelope: &Envelope) -> Result<(), String> {
265        if let Some(key) = self.resolve_target_key(envelope, true).await {
266            self.deliver_resolved(key, envelope).await?;
267        }
268        Ok(())
269    }
270}

272fn attach_auth_routes(r: Router) -> Router {
273    r.route("/dashboard", get(assets_http::handle_dashboard_page))
274        .route("/dashboard/login", get(assets_http::handle_login_page))
275        .route("/dashboard/magic/:token", get(auth_http::handle_magic_exchange))
276        .route("/dashboard/manifest.webmanifest", get(assets_http::handle_manifest))
277        .route("/dashboard/sw.js", get(assets_http::handle_sw))
278        .route("/dashboard/icon.svg", get(assets_http::handle_icon))
279        .route("/dashboard/assets/:file", get(assets_http::handle_asset))
280        .route("/dashboard/api/auth/me", get(auth_http::handle_auth_me))
281        .route("/dashboard/api/auth/login", post(auth_http::handle_login))
282        .route("/dashboard/api/auth/pin", post(auth_http::handle_set_pin))
283        .route("/dashboard/api/auth/logout", post(auth_http::handle_logout))
284        .route("/dashboard/api/auth/revoke_all", post(auth_http::handle_revoke_all))
285        .route("/dashboard/api/ws", get(ws::handle_ws))
286}

288fn attach_session_routes(r: Router) -> Router {
289    r.route(
290        "/dashboard/api/sessions",
291        get(api_sessions::handle_list_sessions)
292            .post(api_sessions::handle_create_session)
293            .delete(api_chat::handle_delete_session),
294    )
295        .route("/dashboard/api/sessions/config", patch(api_sessions::handle_patch_session_config))
296        .route("/dashboard/api/sessions/history", get(api_history::handle_session_history))
297        .route("/dashboard/api/sessions/upload", post(api_history::handle_upload_file).layer(DefaultBodyLimit::max(32 * 1024 * 1024)))
298        .route("/dashboard/api/sessions/send", post(api_chat::handle_send_prompt))
299        .route("/dashboard/api/sessions/ask", post(api_ask::handle_answer_ask))
300        .route("/dashboard/api/sessions/stop", post(api_chat::handle_stop_session))
301        .route("/dashboard/api/sessions/new", post(api_chat::handle_new_session))
302        .route("/dashboard/api/sessions/prune", post(api_chat::handle_prune_sessions))
303        .route("/dashboard/api/models", get(api_models::handle_list_models))
304}

306fn attach_ops_routes(r: Router) -> Router {
307    r.route("/dashboard/api/subagents", get(subagent_tree::handle_get_subagents))
308        .route("/dashboard/api/usage", get(api_ops::handle_get_usage))
309        .route("/dashboard/api/cron", get(api_ops::handle_list_cron))
310        .route("/dashboard/api/cron/:id/toggle", post(api_ops::handle_toggle_cron))
311        .route("/dashboard/api/cron/:id/run", post(api_ops::handle_run_cron))
312        .route("/dashboard/api/file", get(api_files::handle_get_file))
313        .route("/dashboard/api/vault/entries", get(api_files::handle_vault_list).post(api_files::handle_vault_upsert))
314        .route("/dashboard/api/vault/entries/:site_key", delete(api_files::handle_vault_delete))
315        .route("/dashboard/api/hitl", get(api_hitl::handle_hitl_status))
316        .route("/dashboard/api/hitl/start", post(api_hitl::handle_hitl_start))
317        .merge(api_providers::provider_routes())
318        .merge(api_telegram::telegram_routes())
319}

321/// Builds the `/dashboard` and `/dashboard/api/*` Axum router.
322pub fn dashboard_routes() -> Router {
323    attach_ops_routes(attach_session_routes(attach_auth_routes(Router::new())))
324}

326/// Builds an isolated `/dashboard` router with an explicit [`WebAppCtx`] layer (for tests).
327pub fn build_router_with_ctx(ctx: Arc<WebAppCtx>) -> Router {
328    dashboard_routes().layer(Extension(ctx))
