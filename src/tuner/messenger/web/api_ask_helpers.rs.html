18pub(crate) fn build_multi_keystrokes_from_indices(
19    options: &[String],
20    selected_indices: &[usize],
21    prev_ans: &str,
22) -> (String, Vec<String>) {
23    let mut prev_set = Vec::new();
24    if !prev_ans.is_empty() {
25        let parts: Vec<&str> = prev_ans.split(", ").map(str::trim).collect();
26        for (i, opt) in options.iter().enumerate() {
27            if parts.contains(&opt.as_str()) {
28                prev_set.push(i);
29            }
30        }
31    }
32    let mut toggled = Vec::new();
33    let mut labels = Vec::new();
34    for (i, opt) in options.iter().enumerate() {
35        let is_sel = selected_indices.contains(&i);
36        if is_sel {
37            labels.push(opt.clone());
38        }
39        if is_sel != prev_set.contains(&i) {
40            toggled.push(i);
41        }
42    }
43    toggled.sort_unstable();
44    let mut ks = String::new();
45    let mut last = 0;
46    for idx in toggled {
47        ks.push_str(&"j".repeat(idx - last));
48        ks.push(' ');
49        last = idx;
50    }
51    ks.push('\r');
52    (ks, labels)
53}

55async fn send_agy_writein_keystrokes(
56    ctx: &WebAppCtx,
57    sid: &str,
58    q: &crate::cli::AskQuestionData,
59    prev_text: &str,
60    txt: &str,
61) {
62    use crate::messenger::telegram::commands_registry::escape_non_workflow_slash;
63    let w_idx = find_write_in_index(&q.options);
64    let _ = ctx.cli.sessions.write_to_session(sid, &format!("{}", w_idx + 1)).await;
65    tokio::time::sleep(std::time::Duration::from_millis(120)).await;
66    if !prev_text.is_empty() {
67        let prev_safe = escape_non_workflow_slash(prev_text);
68        let bs = "\x7F".repeat(prev_safe.chars().count());
69        let _ = ctx.cli.sessions.write_to_session(sid, &bs).await;
70        tokio::time::sleep(std::time::Duration::from_millis(120)).await;
71    }
72    let safe_txt = escape_non_workflow_slash(txt);
73    let _ = ctx.cli.sessions.write_to_session(sid, &format!("{safe_txt}\r")).await;
74    if q.is_multi_select {
75        tokio::time::sleep(std::time::Duration::from_millis(120)).await;
76        let _ = ctx.cli.sessions.write_to_session(sid, "\r").await;
77    }
78}

80async fn apply_single_or_writein(
81    ctx: &WebAppCtx,
82    is_agy: bool,
83    sid: &str,
84    state: &mut AskState,
85    q: &crate::cli::AskQuestionData,
86    payload: &AskAnswerPayload,
87) -> Result<String, &'static str> {
88    let ans = if let Some(&idx) = payload.selected_indices.first() {
89        let label = q.options.get(idx).cloned().ok_or("Option index out of bounds")?;
90        if is_agy {
91            clear_previous_write_in_if_any(sid, &ctx.cli, state, true).await;
92            let _ = ctx.cli.sessions.write_to_session(sid, &format!("{}", idx + 1)).await;
93        }
94        label
95    } else if let Some(txt) = payload.answer.as_deref().map(str::trim).filter(|s| !s.is_empty()) {
96        let matched_idx = (!q.is_multi_select && !state.waiting_for_write_in)
97            .then(|| crate::messenger::telegram::formatting::find_best_option(txt, &q.options))
98            .flatten()
99            .filter(|&i| {
100                let low = q.options[i].to_lowercase();
101                !low.contains("write-in") && !q.options[i].contains("직접 입력")
102            });
103        if let Some(idx) = matched_idx {
104            if is_agy {
105                clear_previous_write_in_if_any(sid, &ctx.cli, state, true).await;
106                let _ = ctx.cli.sessions.write_to_session(sid, &format!("{}", idx + 1)).await;
107            }
108            q.options[idx].clone()
109        } else {
110            if is_agy {
111                let prev = state.answers.get(state.current_index).map(String::as_str).unwrap_or("");
112                send_agy_writein_keystrokes(ctx, sid, q, prev, txt).await;
113            }
114            txt.to_string()
115        }
116    } else {
117        return Err("Missing selected_indices or answer");
118    };
119    if state.current_index < state.answers.len() {
120        state.answers[state.current_index] = ans.clone();
121    }
122    Ok(ans)
123}

125async fn apply_ask_prev(
126    ctx: &WebAppCtx,
127    is_agy: bool,
128    sid: &str,
129    state: &mut AskState,
130) -> Result<String, &'static str> {
131    if state.current_index == 0 {
132        return Err("Already at first question");
133    }
134    if is_agy {
135        let _ = ctx.cli.sessions.write_to_session(sid, "\x1B[D").await;
136    }
137    state.current_index -= 1;
138    let pq = &state.questions[state.current_index];
139    state.current_bitmap = get_bitmap_for_question(pq, &state.answers, state.current_index);
140    state.waiting_for_write_in = false;
141    Ok("prev".to_string())
142}

144async fn apply_ask_skip(
145    ctx: &WebAppCtx,
146    is_agy: bool,
147    sid: &str,
148    state: &mut AskState,
149) -> Result<String, &'static str> {
150    if is_agy {
151        clear_previous_write_in_if_any(sid, &ctx.cli, state, true).await;
152        let _ = ctx.cli.sessions.write_to_session(sid, "\x1B").await;
153    }
154    if state.current_index < state.answers.len() {
155        state.answers[state.current_index] = "User Skipped".to_string();
156    }
157    Ok("User Skipped".to_string())
158}

160async fn apply_ask_submit(
161    ctx: &WebAppCtx,
162    is_agy: bool,
163    sid: &str,
164    state: &mut AskState,
165    q: &crate::cli::AskQuestionData,
166    selected_indices: &[usize],
167) -> Result<String, &'static str> {
168    let prev = state.answers.get(state.current_index).map(String::as_str).unwrap_or("");
169    let (ks, opts) = build_multi_keystrokes_from_indices(&q.options, selected_indices, prev);
170    let joined = if opts.is_empty() { "None".to_string() } else { opts.join(", ") };
171    if is_agy {
172        clear_previous_write_in_if_any(sid, &ctx.cli, state, true).await;
173        let _ = ctx.cli.sessions.write_to_session(sid, &ks).await;
174    }
175    if state.current_index < state.answers.len() {
176        state.answers[state.current_index] = joined.clone();
177    }
178    Ok(joined)
179}

181async fn is_ask_tui_prompt_rendered(
182    ctx: &WebAppCtx,
183    sid: &str,
184    q_idx: usize,
185    q_total: usize,
186) -> bool {
187    let slice = {
188        let holders = ctx.cli.sessions.holders.lock().await;
189        let Some(h) = holders.get(sid) else {
190            return true;
191        };
192        let off = h.turn_start_offset;
193        let raw = h.output.lock().await;
194        let start = off.min(raw.len()).max(raw.len().saturating_sub(65_536));
195        raw[start..].to_vec()
196    };
197    let clean = crate::cli::antigravity::pty_spawner::strip_ansi(&String::from_utf8_lossy(&slice));
198    let marker = format!("Question {}/{}:", q_idx + 1, q_total);
199    clean.contains(&marker)
200        || (q_idx == 0 && clean.contains("Question") && (clean.contains("esc Skip") || clean.contains("Navigate ·")))
201}

203async fn wait_for_agy_ask_tui_ready(ctx: &WebAppCtx, sid: &str, q_idx: usize, q_total: usize) {
204    if cfg!(test) || sid.starts_with("mock-") || !ctx.cli.sessions.is_active(sid).await {
205        return;
206    }
207    let deadline = std::time::Instant::now() + std::time::Duration::from_millis(4500);
208    while std::time::Instant::now() < deadline {
209        if is_ask_tui_prompt_rendered(ctx, sid, q_idx, q_total).await {
210            tokio::time::sleep(std::time::Duration::from_millis(120)).await;
211            return;
212        }
213        tokio::time::sleep(std::time::Duration::from_millis(60)).await;
214    }
215}

217pub(crate) async fn apply_ask_step_input(
218    ctx: &WebAppCtx,
219    sess: &SessionData,
220    sid: &str,
221    state: &mut AskState,
222    payload: &AskAnswerPayload,
223) -> Result<String, &'static str> {
224    let q = state.questions.get(state.current_index).cloned().ok_or("Invalid question index")?;
225    let is_agy = SessionData::effective_provider(&sess.provider, &ctx.config.provider) == "antigravity";
226    if is_agy {
227        wait_for_agy_ask_tui_ready(ctx, sid, state.current_index, state.questions.len()).await;
228    }
229    let res = match payload.action.as_str() {
230        "prev" => apply_ask_prev(ctx, is_agy, sid, state).await,
231        "skip" => apply_ask_skip(ctx, is_agy, sid, state).await,
232        "submit" => apply_ask_submit(ctx, is_agy, sid, state, &q, &payload.selected_indices).await,
233        _ => apply_single_or_writein(ctx, is_agy, sid, state, &q, payload).await,
234    };
235    if is_agy && !cfg!(test) && res.is_ok() {
236        tokio::time::sleep(std::time::Duration::from_millis(150)).await;
237    }
238    res
