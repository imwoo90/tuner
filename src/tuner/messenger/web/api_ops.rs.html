32/// Query parameters for `GET /dashboard/api/usage`.
33#[derive(Deserialize)]
34pub struct UsageQuery {
35    /// Optional target session key for per-session Claude/Codex token stats.
36    #[serde(default)]
37    pub session_key: String,
38    /// When `1`, fetches all 3 providers concurrently.
39    #[serde(default)]
40    pub all: u8,
41}

43async fn query_claude_json(sess: &SessionData) -> serde_json::Value {
44    let home = std::env::var("HOME").unwrap_or_else(|_| "/root".to_string());
45    let creds = PathBuf::from(&home).join(".claude/.credentials.json");
46    let base = std::env::var("ANTHROPIC_BASE_URL").unwrap_or_else(|_| "https://api.anthropic.com".to_string());
47    let (sub_type, quota) = fetch_claude_oauth_quota(&creds, &base).await;
48    let sid = sess.get_session_id("claude");
49    let (inp, outp, cread, cwrite, turns) = read_claude_session_stats(&sid);
50    let auth_label = sub_type.map(|s| format!("Claude OAuth ({s})")).unwrap_or_else(|| "API Key / Local".to_string());
51    serde_json::json!({
52        "auth_label": auth_label,
53        "quota": quota,
54        "turns": turns,
55        "input_tokens": inp,
56        "output_tokens": outp,
57        "cache_read": cread,
58        "cache_write": cwrite
59    })
60}

62async fn query_codex_json(sess: &SessionData) -> serde_json::Value {
63    let (mut acc, mut grp) = query_codex_app_server_rate_limits().await;
64    let sid = sess.get_session_id("codex");
65    let (inp, cached, outp, reas, total, ctx_win, fb_plan, fb_grp) = read_codex_session_telemetry(&sid).await;
66    if acc.is_none() { acc = fb_plan; }
67    if grp.is_none() { grp = fb_grp; }
68    serde_json::json!({
69        "auth_label": acc.unwrap_or_else(|| "API Key / Custom".to_string()),
70        "quota": grp,
71        "input_tokens": inp,
72        "cached_input": cached,
73        "output_tokens": outp,
74        "reasoning_tokens": reas,
75        "total_tokens": total,
76        "context_window": ctx_win
77    })
78}

80/// Returns structured multi-provider quota & token telemetry (`GET /dashboard/api/usage`).
81pub async fn handle_get_usage(
82    ext: Option<Extension<Arc<WebAppCtx>>>,
83    headers: HeaderMap,
84    Query(q): Query<UsageQuery>,
85) -> Response {
86    let ctx = match resolve_ctx(ext) {
87        Ok(c) => c,
88        Err(r) => return *r,
89    };
90    if let Err(r) = require_auth(&ctx, &headers).await {
91        return *r;
92    }
93    let sess = if let Ok(key) = q.session_key.parse::<SessionKey>() {
94        ctx.sessions.get_active(&key).await.ok().flatten().unwrap_or_default()
95    } else {
96        SessionData::default()
97    };
98    let active_prov = SessionData::effective_provider(&sess.provider, &ctx.config.provider);
99    let fetch_all = q.all == 1;
100    let (agy_res, cl_val, cx_val) = tokio::join!(
101        async {
102            if fetch_all || active_prov == "antigravity" {
103                Some(query_antigravity_quota_with_fallback(&ctx.sessions, &ctx.cli, &sess, sess.chat_id, sess.topic_id).await)
104            } else {
105                None
106            }
107        },
108        async { if fetch_all || active_prov == "claude" { Some(query_claude_json(&sess).await) } else { None } },
109        async { if fetch_all || active_prov == "codex" { Some(query_codex_json(&sess).await) } else { None } }
110    );
111    let agy_val = agy_res.map(|res| match res {
112        Ok(rep) => serde_json::json!({ "account": rep.account, "gemini": rep.gemini, "claude": rep.claude }),
113        Err(e) => serde_json::json!({ "error": e }),
114    });
115    Json(serde_json::json!({
116        "active_provider": active_prov,
117        "antigravity": agy_val,
118        "claude": cl_val,
119        "codex": cx_val
120    }))
121    .into_response()
122}

124/// Lists all cron jobs with computed next local run times (`GET /dashboard/api/cron`).
125pub async fn handle_list_cron(
126    ext: Option<Extension<Arc<WebAppCtx>>>,
127    headers: HeaderMap,
128) -> Response {
129    let ctx = match resolve_ctx(ext) {
130        Ok(c) => c,
131        Err(r) => return *r,
132    };
133    if let Err(r) = require_auth(&ctx, &headers).await {
134        return *r;
135    }
136    let jobs = ctx.cron_manager.list_jobs().await.unwrap_or_default();
137    let default_tz = ctx.config.user_timezone.as_deref().unwrap_or("UTC");
138    let list: Vec<serde_json::Value> = jobs
139        .into_iter()
140        .map(|j| {
141            let eff_tz = if j.timezone.is_empty() { default_tz } else { &j.timezone };
142            let next_run = calculate_job_next_run(&j.schedule, eff_tz)
143                .ok()
144                .map(|dt| dt.format("%Y-%m-%d %H:%M %Z").to_string());
145            serde_json::json!({
146                "id": j.id,
147                "title": j.title,
148                "schedule": j.schedule,
149                "timezone": eff_tz,
150                "enabled": j.enabled,
151                "last_run_at": j.last_run_at,
152                "last_run_status": j.last_run_status,
153                "next_run": next_run,
154                "provider": j.provider,
155                "model": j.model
156            })
157        })
158        .collect();
159    Json(serde_json::json!({ "jobs": list })).into_response()
160}

162/// Toggles a cron job's `enabled` state (`POST /dashboard/api/cron/:id/toggle`).
163pub async fn handle_toggle_cron(
164    ext: Option<Extension<Arc<WebAppCtx>>>,
165    headers: HeaderMap,
166    AxumPath(id): AxumPath<String>,
167) -> Response {
168    let ctx = match resolve_ctx(ext) {
169        Ok(c) => c,
170        Err(r) => return *r,
171    };
172    if let Err(r) = require_mutation_auth(&ctx, &headers).await {
173        return *r;
174    }
175    let Ok(Some(job)) = ctx.cron_manager.get_job(&id).await else {
176        return json_err(StatusCode::NOT_FOUND, "Cron job not found");
177    };
178    let next_enabled = !job.enabled;
179    if let Err(e) = ctx.cron_manager.set_enabled(&id, next_enabled).await {
180        return json_err(StatusCode::INTERNAL_SERVER_ERROR, &e);
181    }
182    ctx.events
183        .publish(WebEvent::CronUpdated {
184            job_id: id.clone(),
185            enabled: next_enabled,
186            last_run_status: job.last_run_status.clone(),
187        })
188        .await;
189    Json(serde_json::json!({ "ok": true, "id": id, "enabled": next_enabled })).into_response()
190}

200async fn resolve_cron_exec_target(
201    ctx: &WebAppCtx,
202    job: &crate::cron::manager::CronJob,
203    session_key: Option<&str>,
204) -> (Arc<crate::config::CliConfig>, i64, Option<i64>) {
205    let has_bot = ctx.bot.read().await.is_some();
206    let parsed_sk = session_key.and_then(|s| s.trim().parse::<SessionKey>().ok());
207    if !has_bot
208        || (job.chat_id == 0 && parsed_sk.as_ref().is_some_and(|k| k.transport == "web"))
209    {
210        let mut web_cfg = (*ctx.config).clone();
211        web_cfg.transport = "web".to_string();
212        let tid = parsed_sk
213            .as_ref()
214            .filter(|k| k.transport == "web")
215            .and_then(|k| k.topic_id)
216            .or(Some(1));
217        return (Arc::new(web_cfg), 0, tid);
218    }
219    if let Some(ref k) = parsed_sk
220        && k.transport == "tg"
221        && k.chat_id != 0
222    {
223        return (ctx.config.clone(), k.chat_id, k.topic_id);
224    }
225    let fallback_cid = ctx
226        .config
227        .allowed_group_ids
228        .first()
229        .or_else(|| ctx.config.allowed_user_ids.first())
230        .copied()
231        .unwrap_or(0);
232    if fallback_cid == 0 && job.chat_id == 0 {
233        let mut web_cfg = (*ctx.config).clone();
234        web_cfg.transport = "web".to_string();
235        return (Arc::new(web_cfg), 0, Some(1));
236    }
237    (ctx.config.clone(), fallback_cid, None)
238}

240/// Triggers immediate cron job execution (`POST /dashboard/api/cron/:id/run`).
241pub async fn handle_run_cron(
242    ext: Option<Extension<Arc<WebAppCtx>>>,
243    headers: HeaderMap,
244    AxumPath(id): AxumPath<String>,
245    body: Option<Json<RunCronPayload>>,
246) -> Response {
247    let ctx = match resolve_ctx(ext) {
248        Ok(c) => c,
249        Err(r) => return *r,
250    };
251    if let Err(r) = require_mutation_auth(&ctx, &headers).await {
252        return *r;
253    }
254    let Ok(Some(job)) = ctx.cron_manager.get_job(&id).await else {
255        return json_err(StatusCode::NOT_FOUND, "Cron job not found");
256    };
257    let req_sk = body.as_ref().and_then(|b| b.session_key.as_deref());
258    let (sched_cfg, caller_cid, caller_tid) = resolve_cron_exec_target(&ctx, &job, req_sk).await;
259    let scheduler = CronScheduler::new(sched_cfg, ctx.cron_manager.clone(), ctx.bus.clone());
260    let job_id = job.id.clone();
261    let enabled = job.enabled;
262    let events = ctx.events.clone();
263    let mgr = ctx.cron_manager.clone();
264    tokio::spawn(async move {
265        let _ = scheduler.execute_job_now(job, caller_cid, caller_tid).await;
266        let status = mgr.get_job(&job_id).await.ok().flatten().and_then(|j| j.last_run_status);
267        events
268            .publish(WebEvent::CronUpdated { job_id, enabled, last_run_status: status })
269            .await;
270    });
271    (StatusCode::ACCEPTED, Json(serde_json::json!({ "ok": true, "id": id }))).into_response()
