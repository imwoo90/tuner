15/// Resolves the configured `GEMINI_API_KEY` from `$TUNER_HOME/.env` or process environment.
16pub fn resolve_gemini_api_key(tuner_home: &Path) -> Option<String> {
17    let dotenv = read_dotenv_map(&tuner_home.join(".env"));
18    dotenv
19        .get("GEMINI_API_KEY")
20        .cloned()
21        .filter(|s| !s.trim().is_empty())
22        .or_else(|| std::env::var("GEMINI_API_KEY").ok().filter(|s| !s.trim().is_empty()))
23}

25/// Returns the configured Gemini Live model identifier (defaults to `models/gemini-3.8-live`).
26pub fn resolve_voice_model() -> String {
27    let raw = std::env::var("TUNER_VOICE_MODEL")
28        .ok()
29        .filter(|s| !s.trim().is_empty())
30        .unwrap_or_else(|| "models/gemini-3.8-live".to_string());
31    if raw.starts_with("models/") {
32        raw
33    } else {
34        format!("models/{raw}")
35    }
36}

38const DEFAULT_GEMINI_LIVE_URL: &str =
39    "wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1beta.GenerativeService.BidiGenerateContent";

41/// Builds the upstream Gemini Live WebSocket URL with `?key=` attached.
42pub fn build_upstream_url(api_key: &str) -> String {
43    let base = std::env::var("TUNER_GEMINI_LIVE_WS_URL")
44        .ok()
45        .filter(|s| !s.trim().is_empty())
46        .unwrap_or_else(|| DEFAULT_GEMINI_LIVE_URL.to_string());
47    let sep = if base.contains('?') { '&' } else { '?' };
48    format!("{base}{sep}key={api_key}")
49}

51/// Translates a browser `/dashboard/api/voice/ws` JSON frame into an upstream Gemini Live JSON string.
52pub fn translate_browser_msg(raw: &str, current_topic: &mut Option<String>) -> Option<String> {
53    let v: Value = serde_json::from_str(raw).ok()?;
54    match v.get("type").and_then(Value::as_str)? {
55        "audio" => {
56            let b64 = v.get("data").and_then(Value::as_str)?.trim();
57            (!b64.is_empty()).then(|| {
58                json!({ "realtimeInput": { "audio": { "mimeType": "audio/pcm;rate=16000", "data": b64 } } }).to_string()
59            })
60        }
61        "text" => {
62            let txt = v.get("text").and_then(Value::as_str)?.trim();
63            (!txt.is_empty()).then(|| json!({ "realtimeInput": { "text": txt } }).to_string())
64        }
65        "active_topic" => {
66            *current_topic = v
67                .get("session_key")
68                .and_then(Value::as_str)
69                .map(str::trim)
70                .filter(|s| !s.is_empty())
71                .map(str::to_string);
72            None
73        }
74        _ => None,
75    }
76}

185/// Returns the Gemini Live `functionDeclarations` array for the Front-Desk Voice Secretary.
186pub fn voice_tool_declarations() -> Value {
187    let mut all = read_tool_decls();
188    all.extend(delegation_decls());
189    all.extend(session_mutation_decls());
190    all.extend(control_decls());
191    Value::Array(all)
192}
