24fn canonicalize_spoken_provider(raw: &str) -> Option<&'static str> {
25    let clean = raw.trim().to_ascii_lowercase();
26    if let Some(p) = SessionData::canonical_provider(&clean) {
27        return Some(p);
28    }
29    if clean.contains("클로드") || clean.contains("앤트로픽") || clean.contains("claude") {
30        Some("claude")
31    } else if clean.contains("코덱스") || clean.contains("오픈ai") || clean.contains("codex") || clean == "openai" {
32        Some("codex")
33    } else if clean.contains("안티그래비티") || clean.contains("제미나이") || clean.contains("antigravity") {
34        Some("antigravity")
35    } else {
36        None
37    }
38}

40fn map_spoken_model_token(raw: &str) -> String {
41    let mut s = raw.trim().to_ascii_lowercase().replace(['_', ' '], "-");
42    while s.contains("--") {
43        s = s.replace("--", "-");
44    }
45    if s.contains("하이드로젠") || s.contains("hydrogen") {
46        return "gemini-4-hydrogen".to_string();
47    }
48    if s.contains("네온") || s.contains("neon") {
49        return "gemini-4-neon".to_string();
50    }
51    if s.contains("오케스트레이터") || s.contains("orchestrator") {
52        return if s.contains("아르곤") || s.contains("argon") {
53            "gemini-pro-orchestrator-argon".to_string()
54        } else {
55            "gemini-pro-orchestrator".to_string()
56        };
57    }
58    if s.contains("아르곤") || s.contains("argon") {
59        return "gemini-4-argon".to_string();
60    }
61    if s.contains("카본") || s.contains("carbon") {
62        return "gemini-4-carbon".to_string();
63    }
64    if s.contains("3.8") || (s.contains("플래시") && s.contains("3")) {
65        return "gemini-3.8-flash".to_string();
66    }
67    if s.contains("소넷") || s == "sonnet" {
68        return "sonnet".to_string();
69    }
70    if s.contains("오퍼스") || s == "opus" {
71        return "opus".to_string();
72    }
73    if s.contains("하이쿠") || s == "haiku" {
74        return "haiku".to_string();
75    }
76    if s.contains("아스트라") || s == "astra" {
77        return "astra".to_string();
78    }
79    if s.contains("루나") || s == "luna" {
80        return "luna".to_string();
81    }
82    s.replace("gemini4-", "gemini-4-").replace("gemini3.", "gemini-3.")
83}

85fn infer_provider_for_model(base_prov: &str, slug: &str) -> &'static str {
86    let cur = SessionData::canonical_provider(base_prov).unwrap_or("antigravity");
87    if SessionData::canonicalize_model_for_provider(cur, slug).is_some() {
88        return cur;
89    }
90    for cand in ["antigravity", "claude", "codex"] {
91        if SessionData::canonicalize_model_for_provider(cand, slug).is_some() {
92            return cand;
93        }
94    }
95    cur
96}

98/// Resolves optional spoken `(provider, model, effort)` inputs into validated canonical values.
99pub async fn resolve_spoken_model_spec(
100    ctx: &WebAppCtx,
101    fallback_prov: &str,
102    raw_prov: Option<&str>,
103    raw_model: Option<&str>,
104    raw_effort: Option<&str>,
105) -> Result<(String, String, Option<String>), String> {
106    crate::cli::antigravity::model_spec::ensure_live_catalog_fresh().await;
107    let explicit_prov = match raw_prov.map(str::trim).filter(|s| !s.is_empty()) {
108        Some(rp) => Some(
109            canonicalize_spoken_provider(rp)
110                .ok_or_else(|| format!("Unknown provider `{rp}` (use antigravity, claude, or codex)"))?,
111        ),
112        None => None,
113    };
114    let model_slug = raw_model
115        .map(str::trim)
116        .filter(|s| !s.is_empty())
117        .map(map_spoken_model_token);
118
119    let prov = match (explicit_prov, model_slug.as_deref()) {
120        (Some(p), _) => p,
121        (None, Some(m)) => infer_provider_for_model(fallback_prov, m),
122        (None, None) => SessionData::effective_provider(fallback_prov, &ctx.config.provider),
123    };
124
125    let model = match model_slug.as_deref() {
126        Some(m) => SessionData::canonicalize_model_for_provider(prov, m)
127            .ok_or_else(|| format!("Model `{m}` is not supported on provider `{prov}`"))?,
128        None => SessionData::default_model_for_provider(prov, ctx.config.model.as_deref()),
129    };
130
131    let eff_seed = raw_effort
132        .map(str::trim)
133        .filter(|s| !s.is_empty())
134        .or(ctx.config.effort.as_deref());
135    let effort = match eff_seed {
136        Some(e) => {
137            let low = e.to_ascii_lowercase();
138            if raw_effort.is_some() && !is_valid_effort_level(&low) {
139                return Err(format!("Invalid effort level `{e}`"));
140            }
141            clamp_session_effort(prov, &model, &low)
142        }
143        None => None,
144    };
145    Ok((prov.to_string(), model, effort))
146}

148async fn persist_voice_spec_mutation(
149    ctx: &Arc<WebAppCtx>,
150    key: &SessionKey,
151    sess: &mut SessionData,
152    cur_p: &str,
153    spec: (&str, &str, Option<&str>),
154    has_raw_eff_or_mod: bool,
155    clean_title: Option<&str>,
156) {
157    let (target_p, target_m, target_e) = spec;
158    if let Ok(Some(updated)) = ctx
159        .sessions
160        .mutate_session(key, Some(sess), |cur| {
161            cur.provider = target_p.to_string();
162            cur.model = target_m.to_string();
163            if has_raw_eff_or_mod || target_p != cur_p {
164                cur.effort = if session_supports_effort(target_p, target_m) {
165                    target_e.map(str::to_string).or_else(|| {
166                        cur.effort.as_deref().and_then(|prev| clamp_session_effort(target_p, target_m, prev))
167                    })
168                } else {
169                    None
170                };
171            }
172            if let Some(t) = clean_title {
173                cur.topic_name = Some(t.to_string());
174            }
175            true
176        })
177        .await
178    {
179        *sess = updated;
180    }
181    apply_model_to_active_pty(&ctx.cli, &ctx.sessions, sess).await;
182    if let Some(t) = clean_title {
183        super::tg_topics::sync_remote_topic_rename(ctx, key, t).await;
184    }
185    let (busy, asking) = check_busy_and_asking(ctx, sess).await;
186    emit_session_updated(ctx, sess, busy, asking).await;
187}

189/// Applies an optional spoken `(provider, model, effort, topic_name)` update to an existing topic.
190pub async fn apply_voice_spec_to_session(
191    ctx: &Arc<WebAppCtx>,
192    key: &SessionKey,
193    raw_prov: Option<&str>,
194    raw_model: Option<&str>,
195    raw_effort: Option<&str>,
196    new_title: Option<&str>,
197) -> Result<(String, String, Option<String>), String> {
198    let mut sess = ctx
199        .sessions
200        .get_active(key)
201        .await
202        .ok()
203        .flatten()
204        .ok_or_else(|| "Topic not found".to_string())?;
205    let cur_p = SessionData::effective_provider(&sess.provider, &ctx.config.provider).to_string();
206    let has_spec = raw_prov.is_some() || raw_model.is_some() || raw_effort.is_some();
207    if !has_spec && new_title.is_none() {
208        return Ok(SessionData::resolve_effective_target(Some(&sess), &ctx.config));
209    }
210    let (target_p, target_m, target_e) = if has_spec {
211        let seed_m = raw_model.or(if raw_prov.is_none() { Some(sess.model.as_str()) } else { None });
212        resolve_spoken_model_spec(ctx, &cur_p, raw_prov, seed_m, raw_effort).await?
213    } else {
214        SessionData::resolve_effective_target(Some(&sess), &ctx.config)
215    };
216    if target_p != cur_p {
217        let _ = switch_session_provider(&target_p, &ctx.config, &ctx.sessions, &ctx.cli, &mut sess).await;
218    }
219    let clean_title = new_title.map(str::trim).filter(|s| !s.is_empty());
220    persist_voice_spec_mutation(
221        ctx,
222        key,
223        &mut sess,
224        &cur_p,
225        (&target_p, &target_m, target_e.as_deref()),
226        raw_effort.is_some() || raw_model.is_some(),
227        clean_title,
228    )
229    .await;
230    let (_, _, final_eff) = SessionData::resolve_effective_target(Some(&sess), &ctx.config);
231    Ok((target_p, target_m, final_eff))
232}

234/// Lists discovered models and supported effort levels (`list_available_models`).
235pub async fn exec_list_available_models(ctx: &WebAppCtx, args: &Value) -> Value {
236    let filter_p = args
237        .get("provider")
238        .and_then(Value::as_str)
239        .and_then(canonicalize_spoken_provider);
240    let providers: Vec<&str> = match filter_p {
241        Some(p) => vec![p],
242        None => vec!["antigravity", "claude", "codex"],
243    };
244    let mut catalog = serde_json::Map::new();
245    for prov in providers {
246        let raw = discover_models_with_default(&ctx.cli, prov).await;
247        let models: Vec<Value> = group_discovered_models(&raw)
248            .into_iter()
249            .map(|(m, effs)| json!({ "model": m, "efforts": effs }))
250            .collect();
251        catalog.insert(prov.to_string(), Value::Array(models));
252    }
253    let (_, def_m, def_e) = SessionData::resolve_effective_target(None, &ctx.config);
254    json!({ "default_model": def_m, "default_effort": def_e, "providers": catalog })
255}

257/// Executes `configure_topic` tool (`provider` / `model` / `effort` / `topic_name` mutation).
258pub async fn exec_configure_topic(ctx: &Arc<WebAppCtx>, key: &SessionKey, args: &Value) -> Value {
259    let rp = args.get("provider").and_then(Value::as_str);
260    let rm = args.get("model").and_then(Value::as_str);
261    let re = args.get("effort").and_then(Value::as_str);
262    let rt = args.get("topic_name").and_then(Value::as_str);
263    match apply_voice_spec_to_session(ctx, key, rp, rm, re, rt).await {
264        Ok(_) => match ctx.sessions.get_active(key).await.ok().flatten() {
265            Some(s) => json!({ "ok": true, "topic": build_session_summary(ctx, &s).await }),
266            None => json!({ "ok": true, "session_key": key.storage_key() }),
267        },
268        Err(e) => json!({ "error": e }),
269    }
