1//! # Subagent, Native Swarm & Background Task Tree Inspector (`subagent_tree.rs`)

3//! ## Overview
4//! Queries live warm CLI pool counts and recursively traverses Antigravity
5//! `<appDataDir>/brain/<cid>/.system_generated/` artifacts (`subagents/*.json`,
6//! `swarm/members.json`, `tasks/*.log`, `logs/transcript.jsonl`) to render a structured
7//! live execution tree for `/dashboard/api/subagents`.

105fn is_safe_cid(cid: &str) -> bool {
106    !cid.is_empty() && cid != "." && !cid.contains('/') && !cid.contains('\\') && !cid.contains("..")
107}

109fn read_subagents(gen_dir: &Path) -> Vec<SubagentDescriptorView> {
110    let mut out = Vec::new();
111    let Ok(entries) = std::fs::read_dir(gen_dir.join("subagents")) else {
112        return out;
113    };
114    for entry in entries.flatten() {
115        let p = entry.path();
116        if p.extension().and_then(|e| e.to_str()) != Some("json") {
117            continue;
118        }
119        let Ok(raw) = std::fs::read_to_string(&p) else { continue };
120        let Ok(val) = serde_json::from_str::<serde_json::Value>(&raw) else { continue };
121        let cid = val.get("conversationId").or_else(|| val.get("conversation_id")).and_then(|v| v.as_str()).unwrap_or("").to_string();
122        if !is_safe_cid(&cid) {
123            continue;
124        }
125        let role = val
126            .pointer("/subagentDescriptor/role")
127            .or_else(|| val.get("role"))
128            .or_else(|| val.pointer("/subagentDescriptor/typeName"))
129            .or_else(|| val.get("type"))
130            .or_else(|| val.get("name"))
131            .and_then(|v| v.as_str())
132            .filter(|s| !s.is_empty())
133            .unwrap_or("subagent")
134            .to_string();
135        let state = val.get("state").and_then(|v| v.as_str()).unwrap_or("UNKNOWN").to_string();
136        let modified_epoch = std::fs::metadata(&p).ok().and_then(|m| m.modified().ok()).and_then(|t| t.duration_since(std::time::UNIX_EPOCH).ok()).map(|d| d.as_secs()).unwrap_or(0);
137        out.push(SubagentDescriptorView { conversation_id: cid, role, state, modified_epoch });
138    }
139    out
140}

142fn read_swarm_members(gen_dir: &Path) -> Vec<SwarmMemberView> {
143    let mut out = Vec::new();
144    let Ok(raw) = std::fs::read_to_string(gen_dir.join("swarm").join("members.json")) else {
145        return out;
146    };
147    let Ok(val) = serde_json::from_str::<serde_json::Value>(&raw) else { return out };
148    let Some(members) = val.get("members").and_then(|v| v.as_object()) else { return out };
149    for (k, m) in members {
150        let name = m.get("name").and_then(|v| v.as_str()).unwrap_or(k.as_str()).to_string();
151        let model = m.get("model").and_then(|v| v.as_str()).unwrap_or("").to_string();
152        let state = m.get("state").or_else(|| m.get("phase")).and_then(|v| v.as_str()).unwrap_or("unknown").to_string();
153        let cid = m.get("conversation_id").or_else(|| m.get("conversationId")).and_then(|v| v.as_str()).filter(|s| is_safe_cid(s)).map(str::to_string);
154        out.push(SwarmMemberView { name, model, state, conversation_id: cid });
155    }
156    out
157}

159fn read_tasks(gen_dir: &Path) -> Vec<TaskLogView> {
160    let mut out = Vec::new();
161    let Ok(entries) = std::fs::read_dir(gen_dir.join("tasks")) else {
162        return out;
163    };
164    for entry in entries.flatten() {
165        let p = entry.path();
166        if !p.is_file() {
167            continue;
168        }
169        let task_id = p.file_stem().and_then(|s| s.to_str()).unwrap_or("task").to_string();
170        let raw = read_tail_utf8(&p, 8192).unwrap_or_default();
171        let lines: Vec<&str> = raw.lines().collect();
172        let tail = lines[lines.len().saturating_sub(6)..].join("\n");
173        let clipped: String = tail.chars().rev().take(600).collect::<Vec<_>>().into_iter().rev().collect();
174        out.push(TaskLogView { task_id, tail: clipped });
175    }
176    out
177}

179fn read_latest_step(gen_dir: &Path) -> Option<LatestStepView> {
180    let logs = gen_dir.join("logs");
181    let compact = logs.join("transcript.jsonl");
182    let path = if compact.is_file() { compact } else { logs.join("transcript_full.jsonl") };
183    let raw = read_tail_utf8(&path, 32 * 1024)?;
184    let last_line = raw.lines().rev().find(|l| !l.trim().is_empty())?;
185    let val: serde_json::Value = serde_json::from_str(last_line).ok()?;
186    let tools = val
187        .get("tool_calls")
188        .and_then(|v| v.as_array())
189        .map(|arr| arr.iter().filter_map(|t| t.get("name").and_then(|n| n.as_str()).map(str::to_string)).collect())
190        .unwrap_or_default();
191    Some(LatestStepView {
192        step_index: val.get("step_index").and_then(|v| v.as_u64()),
193        r#type: val.get("type").and_then(|v| v.as_str()).unwrap_or("").to_string(),
194        status: val.get("status").and_then(|v| v.as_str()).unwrap_or("").to_string(),
195        created_at: val.get("created_at").and_then(|v| v.as_str()).map(str::to_string),
196        tools,
197    })
198}

200/// Traverses `brain_root/<root_cid>/.system_generated/` recursively up to `max_nodes`.
201pub fn collect_agent_tree_nodes(brain_root: &Path, root_cid: &str, max_nodes: usize) -> Vec<AgentTreeNode> {
202    if !is_safe_cid(root_cid) || !brain_root.is_dir() {
203        return Vec::new();
204    }
205    let mut visited = HashSet::new();
206    let mut queue = vec![(root_cid.to_string(), true, None::<String>)];
207    let mut nodes = Vec::new();
208    while let Some((cid, is_root, label)) = queue.pop() {
209        if nodes.len() >= max_nodes || !is_safe_cid(&cid) || !visited.insert(cid.clone()) {
210            continue;
211        }
212        let gen_dir = brain_root.join(&cid).join(".system_generated");
213        if !gen_dir.is_dir() && !is_root {
214            continue;
215        }
216        let subagents = read_subagents(&gen_dir);
217        let swarm_members = read_swarm_members(&gen_dir);
218        for s in &subagents {
219            queue.push((s.conversation_id.clone(), false, Some(s.role.clone())));
220        }
221        for m in &swarm_members {
222            if let Some(ref child_cid) = m.conversation_id {
223                queue.push((child_cid.clone(), false, Some(format!("🐝 {}", m.name))));
224            }
225        }
226        nodes.push(AgentTreeNode {
227            conversation_id: cid,
228            is_root,
229            label,
230            subagents,
231            swarm_members,
232            tasks: read_tasks(&gen_dir),
233            latest_step: read_latest_step(&gen_dir),
234        });
235    }
236    nodes
237}

239async fn collect_session_tree(ctx: &WebAppCtx, key: &SessionKey) -> (serde_json::Value, Vec<AgentTreeNode>) {
240    let Ok(Some(sess)) = ctx.sessions.get_active(key).await else {
241        return (serde_json::Value::Null, Vec::new());
242    };
243    let agy_sid = sess.get_session_id("antigravity");
244    let agy_pid = if !agy_sid.is_empty() {
245        ctx.cli.sessions.holders.lock().await.get(&agy_sid).and_then(|h| h.child.id())
246    } else {
247        None
248    };
249    let cl_pid = crate::cli::claude::active_pid(key.chat_id, key.topic_id).await;
250    let cx_pid = crate::cli::codex::active_pid(key.chat_id, key.topic_id).await;
251    let pids = serde_json::json!({ "agy": agy_pid, "claude": cl_pid, "codex": cx_pid });
252    let nodes = if !agy_sid.is_empty() {
253        let root = agy_state_root(Some(&ctx.cli.build_env())).join("brain");
254        collect_agent_tree_nodes(&root, &agy_sid, 32)
255    } else {
256        Vec::new()
257    };
258    (pids, nodes)
259}

261/// Returns warm pool stats, session PIDs, and recursive subagent/swarm/task tree JSON.
262pub async fn handle_get_subagents(
263    ext: Option<Extension<Arc<WebAppCtx>>>,
264    headers: HeaderMap,
265    Query(q): Query<SubagentsQuery>,
266) -> Response {
267    let ctx = match resolve_ctx(ext) {
268        Ok(c) => c,
269        Err(r) => return *r,
270    };
271    if let Err(r) = require_auth(&ctx, &headers).await {
272        return *r;
273    }
274    let (agy_cnt, cl_cnt, cx_cnt) = (
275        ctx.cli.sessions.active_count().await,
276        crate::cli::claude::active_count().await,
277        crate::cli::codex::active_count().await,
278    );
279    let (session_pids, nodes) = if !q.session_key.trim().is_empty() {
280        let Ok(key) = q.session_key.parse::<SessionKey>() else {
281            return json_err(StatusCode::BAD_REQUEST, "Invalid session_key");
282        };
283        collect_session_tree(&ctx, &key).await
284    } else {
285        (serde_json::Value::Null, Vec::new())
286    };
287    Json(serde_json::json!({
288        "warm_pools": { "agy": agy_cnt, "claude": cl_cnt, "codex": cx_cnt, "total_active": agy_cnt + cl_cnt + cx_cnt },
289        "session_pids": session_pids,
290        "nodes": nodes
291    }))
292    .into_response()
