1//! # Telegram Ingress Authentication and Migration
2//!
3//! Validates user authorization, handles first-time owner self-registration,
4//! group access control filtering, and Telegram forum topic creation/migration events.
5
6use teloxide::prelude::*;
7use crate::config::CliConfig;
8use crate::session::manager::SessionManager;
9use super::topic_cache::{self, TopicNameCache};

11pub(crate) fn auto_register_owner(from_id: i64) {
12    let home = std::env::var("HOME").unwrap_or_else(|_| ".".to_string());
13    let path = std::path::PathBuf::from(&home).join(".tuner/config/config.json");
14    if let Ok(c) = std::fs::read_to_string(&path) {
15        if let Ok(mut val) = serde_json::from_str::<serde_json::Value>(&c) {
16            if let Some(obj) = val.as_object_mut() {
17                obj.insert("allowed_user_ids".to_string(), serde_json::json!([from_id]));
18                if let Ok(pretty) = serde_json::to_string_pretty(&val) {
19                    if std::fs::write(&path, pretty).is_ok() {
20                        let restart = std::path::PathBuf::from(&home).join(".tuner/restart-requested");
21                        let _ = std::fs::write(restart, "");
22                    }
23                }
24            }
25        }
26    }
27}

29pub(crate) async fn validate_and_auth_message(
30    bot: &Bot,
31    msg: &Message,
32    config: &CliConfig,
33    sessions: &SessionManager,
34    topic_cache: &TopicNameCache,
35) -> Result<bool, teloxide::RequestError> {
36    let from_id = msg.from.as_ref().map(|u| u.id.0 as i64).unwrap_or(0);
37    let chat_id = msg.chat.id.0;
38    if let Some(to_chat) = msg.migrate_to_chat_id() {
39        let _ = sessions.migrate_chat_id(chat_id, to_chat.0).await;
40        return Ok(false);
41    }
42    if topic_cache::handle_forum_topic_events(msg, topic_cache, chat_id) {
43        return Ok(false);
44    }
45    let mut ok = config.allowed_user_ids.contains(&from_id);
46
47    if !ok && from_id != 0 && config.allowed_user_ids.is_empty() {
48        println!("🤖 [tuner] First-time owner auto-registered! Telegram User ID: {}. Restarting...", from_id);
49        let _ = bot.send_message(msg.chat.id, "🤖 Owner registered successfully! Restarting tuner daemon...").await;
50        auto_register_owner(from_id);
51        std::process::exit(0);
52    }
53
54    let is_group = msg.chat.is_group() || msg.chat.is_supergroup();
55    if ok && is_group && !config.allowed_group_ids.contains(&chat_id) {
56        eprintln!("⚠️ [tuner] Unauthorized group ID: {}", chat_id);
57    }
58    ok = ok && (!is_group || config.allowed_group_ids.contains(&chat_id));
59    Ok(ok)
60}
