18/// Formats a remaining duration in seconds into a concise reset countdown string.
19pub fn format_seconds_countdown(secs: i64) -> String {
20    if secs <= 0 {
21        return "Refreshes soon".to_string();
22    }
23    let days = secs / 86_400;
24    let hours = (secs % 86_400) / 3_600;
25    let mins = (secs % 3_600) / 60;
26    if days > 0 {
27        format!("Refreshes in {}d {}h", days, hours)
28    } else if hours > 0 {
29        format!("Refreshes in {}h {}m", hours, mins)
30    } else {
31        format!("Refreshes in {}m", mins.max(1))
32    }
33}

35/// Formats an RFC3339 timestamp or Unix epoch seconds into a human-readable reset countdown.
36pub fn format_reset_timestamp(raw: &str) -> String {
37    if let Ok(dt) = DateTime::parse_from_rfc3339(raw) {
38        let secs = dt.with_timezone(&Utc).signed_duration_since(Utc::now()).num_seconds();
39        return format_seconds_countdown(secs);
40    }
41    if let Ok(epoch) = raw.parse::<i64>() {
42        return format_seconds_countdown(epoch - Utc::now().timestamp());
43    }
44    raw.to_string()
45}

47/// Parses Claude Code `/api/oauth/usage` JSON into a [`ModelQuotaGroup`].
48pub fn parse_claude_oauth_usage_json(val: &Value) -> Option<ModelQuotaGroup> {
49    let parse_win = |w: &Value| -> Option<QuotaLimit> {
50        let used_raw = w.get("utilization").and_then(Value::as_f64)?;
51        let used_pct = if used_raw > 0.0
52            && used_raw <= 1.0
53            && w.get("is_fraction").and_then(Value::as_bool).unwrap_or(false)
54        {
55            used_raw * 100.0
56        } else {
57            used_raw.clamp(0.0, 100.0)
58        };
59        let rem_pct = (100.0 - used_pct).clamp(0.0, 100.0);
60        let reset_str = match w.get("resets_at") {
61            Some(Value::String(s)) => format_reset_timestamp(s),
62            Some(Value::Number(n)) => n
63                .as_i64()
64                .map(|ts| format_seconds_countdown(ts - Utc::now().timestamp()))
65                .unwrap_or_default(),
66            _ => String::new(),
67        };
68        let detail = if reset_str.is_empty() {
69            format!("{:.0}% used", used_pct)
70        } else {
71            format!("{:.0}% used · {}", used_pct, reset_str)
72        };
73        Some(QuotaLimit { pct: rem_pct, detail })
74    };
75    let five_hour = val.get("five_hour").and_then(parse_win);
76    let weekly = val.get("seven_day").and_then(parse_win);
77    (weekly.is_some() || five_hour.is_some()).then_some(ModelQuotaGroup { weekly, five_hour })
78}

80fn read_claude_session_stats(session_id: &str) -> (u64, u64, u64, u64, usize) {
81    if session_id.is_empty() {
82        return (0, 0, 0, 0, 0);
83    }
84    let home = std::env::var("HOME").unwrap_or_else(|_| "/root".to_string());
85    let projects_dir = PathBuf::from(home).join(".claude/projects");
86    let Ok(entries) = std::fs::read_dir(projects_dir) else {
87        return (0, 0, 0, 0, 0);
88    };
89    let target = format!("{}.jsonl", session_id);
90    for entry in entries.flatten() {
91        let candidate = entry.path().join(&target);
92        if let Ok(content) = std::fs::read_to_string(&candidate) {
93            let (mut inp, mut outp, mut cread, mut cwrite, mut turns) = (0u64, 0u64, 0u64, 0u64, 0usize);
94            for line in content.lines() {
95                let Ok(v) = serde_json::from_str::<Value>(line) else { continue };
96                if v.get("type").and_then(Value::as_str) == Some("assistant")
97                    && let Some(u) = v.get("message").and_then(|m| m.get("usage"))
98                {
99                    turns += 1;
100                    inp += u.get("input_tokens").and_then(Value::as_u64).unwrap_or(0);
101                    outp += u.get("output_tokens").and_then(Value::as_u64).unwrap_or(0);
102                    cread += u.get("cache_read_input_tokens").and_then(Value::as_u64).unwrap_or(0);
103                    cwrite += u.get("cache_creation_input_tokens").and_then(Value::as_u64).unwrap_or(0);
104                }
105            }
106            return (inp, outp, cread, cwrite, turns);
107        }
108    }
109    (0, 0, 0, 0, 0)
110}

112async fn fetch_claude_oauth_quota(creds_path: &Path, base_url: &str) -> (Option<String>, Option<ModelQuotaGroup>) {
113    let Ok(raw) = tokio::fs::read_to_string(creds_path).await else { return (None, None) };
114    let Ok(doc) = serde_json::from_str::<Value>(&raw) else { return (None, None) };
115    let Some(oauth) = doc.get("claudeAiOauth") else { return (None, None) };
116    let sub_type = oauth.get("subscriptionType").and_then(Value::as_str).map(String::from);
117    let Some(tok) = oauth.get("accessToken").and_then(Value::as_str).filter(|s| !s.is_empty()) else {
118        return (sub_type, None);
119    };
120    let Ok(client) = reqwest::Client::builder().timeout(Duration::from_secs(4)).build() else {
121        return (sub_type, None);
122    };
123    let url = format!("{}/api/oauth/usage", base_url.trim_end_matches('/'));
124    let Ok(resp) = client.get(&url).bearer_auth(tok).send().await else {
125        return (sub_type, None);
126    };
127    let Ok(val) = resp.json::<Value>().await else {
128        return (sub_type, None);
129    };
130    (sub_type, parse_claude_oauth_usage_json(&val))
131}

133/// Queries Claude Code usage (OAuth `/api/oauth/usage` quota + session JSONL telemetry).
134pub async fn query_claude_usage_html(
135    sess: &SessionData,
136    creds_override: Option<&Path>,
137    base_url_override: Option<&str>,
138) -> String {
139    let home = std::env::var("HOME").unwrap_or_else(|_| "/root".to_string());
140    let default_creds = PathBuf::from(&home).join(".claude/.credentials.json");
141    let creds_path = creds_override.unwrap_or(&default_creds);
142    let base_url = base_url_override
143        .map(String::from)
144        .or_else(|| std::env::var("ANTHROPIC_BASE_URL").ok())
145        .unwrap_or_else(|| "https://api.anthropic.com".to_string());
146
147    let (sub_type, oauth_group) = fetch_claude_oauth_quota(creds_path, &base_url).await;
148    let sid = sess.get_session_id("claude");
149    let (inp, outp, cread, cwrite, turns) = read_claude_session_stats(&sid);
150    let model = if sess.model.is_empty() { "claude-default" } else { &sess.model };
151    let auth_label = sub_type
152        .as_deref()
153        .map(|s| format!("Claude OAuth ({})", s))
154        .unwrap_or_else(|| "API Key / Local Bridge".to_string());
155
156    let mut html = String::from("📊 <b>[tuner] Claude Code Quota & Usage</b>\n");
157    html.push_str(&format!("🔑 <b>Auth</b>: <code>{}</code>\n", html_escape::encode_safe(&auth_label)));
158    html.push_str(&format!("🤖 <b>Model</b>: <code>{}</code>\n", html_escape::encode_safe(model)));
159    if let Some(ref grp) = oauth_group {
160        html.push_str(&render_group_markdown("🧠", "CLAUDE SUBSCRIPTION QUOTA", grp));
161    }
162    html.push_str("\n📈 <b>SESSION TELEMETRY</b>\n");
163    html.push_str(&format!("├ 🔹 Assistant Turns: <code>{}</code>\n", turns));
164    html.push_str(&format!("├ 🔹 Input / Output: <code>{}</code> in / <code>{}</code> out\n", inp, outp));
165    html.push_str(&format!("└ 🔹 Prompt Cache: <code>{}</code> read / <code>{}</code> write\n", cread, cwrite));
166    html
167}
