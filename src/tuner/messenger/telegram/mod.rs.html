1//! # Telegram Bot Interface Module
2//!
3//! ## Overview
4//! Handles Telegram updates, slash commands, PTY streaming previews, and interactive UI menus.
5//!
6//! ## Submodules
7//! - [`formatting`] (`formatting/mod.rs`): Markdown-to-HTML and chunk splitting.
8//! - [`reply`] (`reply.rs`): Reply context extraction and mention filtering.
9//! - [`reply_directives`] (`reply_directives.rs`): `@model` directives and session model display.
10//! - [`session_init`] (`session_init.rs`): Lazy session boot and topic seed synthesis.
11//! - [`history`] (`history.rs`): Persistent JSONL message logging.
12//! - [`handover`] (`handover.rs`): Handover summary across `/new` resets.
13//! - [`handover_helpers`] (`handover_helpers.rs`): Transcript helpers for handover.
14//! - [`commands`] (`commands.rs`): Primary slash command router.
15//! - [`cron_selector`] (`cron_selector.rs`): Inline keyboard selector for cron tasks.
16//! - [`cron_formatter`] (`cron_formatter.rs`): Human-readable cron formatting.
17//! - [`stream`] (`stream.rs`): Debounced PTY streaming and progress updates.
18//! - [`transport`] (`transport.rs`): Adapts bus envelopes into Teloxide API calls.
19//! - [`lang`] (`lang.rs`): Language menu and locale persistence.
20//! - [`callbacks`] (`callbacks.rs`): Callback query dispatcher.
21//! - [`ask_callbacks`] (`ask_callbacks.rs`): Callback handlers for `ask_question`.
22//! - [`ask_helpers`] (`ask_helpers.rs`): Active session prompt feeding helpers.
23//! - [`ask_process`] (`ask_process.rs`): Multi-step `ask_question` state machine.
24//! - [`multi_select`] (`multi_select.rs`): Multi-option keyboard toggle state.
25//! - [`runner`] (`runner.rs`): Teloxide dispatcher setup and polling loop.
26//! - [`attachments`] (`attachments.rs`): Outbound file delivery.
27//! - [`attachments_dl`] (`attachments_dl.rs`): Attachment ZIP/file downloader.
28//! - [`review`] (`review.rs`): Web viewer session manager.
29//! - [`review_server`] (`review_server.rs`): HTTP server for file review.
30//! - [`review_store`] (`review_store.rs`): Reviewable file token store.
31//! - [`review_callback`] (`review_callback.rs`): Review preview callbacks.
32//! - [`media_group`] (`media_group.rs`): Album upload aggregator.
33//! - [`upgrade`] (`upgrade.rs`): Self-upgrade command workflow.
34//! - [`async_observer`] (`async_observer.rs`): Async turn completion watcher.
35//! - [`commands_usage`] (`commands_usage.rs`): `/usage` dispatcher.
36//! - [`commands_usage_agy`] (`commands_usage_agy.rs`): CloudCode quota for `agy`.
37//! - [`commands_usage_claude`] (`commands_usage_claude.rs`): Quota/JSONL usage for `claude`.
38//! - [`commands_usage_codex`] (`commands_usage_codex.rs`): RPC/JSONL usage for `codex`.
39//! - [`commands_session`] (`commands_session.rs`): `/new`, `/stop`, `/abort` handlers.
40//! - [`commands_remote_hub`] (`commands_remote_hub.rs`): Remote service spawner.
41//! - [`commands_remote`] (`commands_remote.rs`): `/remote` dashboard controller.
42//! - [`typing`] (`typing.rs`): Typing indicator guard.
43//! - [`rate_limiter`] (`rate_limiter.rs`): Per-chat rate limiter.
44//! - [`callbacks_effort`] (`callbacks_effort.rs`): Model/effort callbacks.
45//! - [`commands_provider`] (`commands_provider.rs`): `/provider` handler.
46//! - [`media`] (`media.rs`): Media download and prompt injection.
47//! - [`commands_model`] (`commands_model.rs`): `/model` and `/effort` handlers.
48//! - [`commands_registry`] (`commands_registry.rs`): Bot command registry.
49//! - [`auth`] (`auth.rs`): Auth and topic migration.
50//! - [`message_session`] (`message_session.rs`): Turn-gated session & directive resolver.
51//! - [`handler`] (`handler.rs`): Update handler and concurrency gate.
52//! - [`vault`] (`vault.rs`): Web Vault CRUD endpoints.
53//!

165pub(crate) async fn run_cli_stream(
166    bot: &Bot,
167    msg: &Message,
168    prompt: &str,
169    sid: &str,
170    cli: &AntigravityCli,
171    sess_ctx: (&SessionManager, SessionData),
172    config: &CliConfig,
173) -> Result<(), teloxide::RequestError> {
174    let (sessions, sess) = sess_ctx;
175    let _fg = crate::upgrade::begin_foreground_run();
176    let _turn_g = cli.begin_topic_turn(msg.chat.id.0, sess.topic_id.or_else(|| get_topic_id(msg)));
177    let tok = std::env::var("TELEGRAM_TOKEN").unwrap_or_else(|_| config.telegram_token.clone());
178    let _g = typing::TelegramTypingGuard::new(bot.clone(), tok, msg).await;
179    let tid = sess.topic_id.or_else(|| get_topic_id(msg));
180    let cc = session_init::configure_session_cli(cli, &sess, msg.chat.id.0, tid).await;
181    match cc.send_streaming(prompt, (!sid.is_empty()).then_some(sid), false, config.working_dir.clone()).await {
182        Ok(s) => stream::consume_stream(bot, (msg.chat.id, msg.thread_id.map(|t| t.0.0)), s, sessions, sess, &cc.config, cli).await?,
183        Err(e) => {
184            eprintln!("CLI ERROR: {:?}", e);
185            let err_msg = crate::cli::antigravity::error_parser::parse_cli_error(&e, 1, cc.config.model.as_deref());
186            let _ = commands::send_reply(bot, msg, err_msg).await;
187        }
188    }
189    Ok(())
190}

194async fn try_feed_open_ask_message(
195    bot: &Bot,
196    msg: &Message,
197    payload: (&str, &[String]),
198    config: &CliConfig,
199    sessions: &std::sync::Arc<SessionManager>,
200    cli: &AntigravityCli,
201) -> Result<bool, teloxide::RequestError> {
202    let (text, files) = payload;
203    if auth::has_leading_directive(msg, text, config, sessions).await {
204        return Ok(false);
205    }
206    let Some(active_sess) = auth::get_live_open_ask_session(msg, config, sessions, cli).await else {
207        return Ok(false);
208    };
209    let escaped = commands_registry::escape_non_workflow_slash(text.trim());
210    let p = SessionData::effective_provider(&active_sess.provider, &config.provider);
211    let sid = active_sess.get_session_id(p);
212    history::log_telegram_message(&config.working_dir, &sid, (active_sess.topic_id, active_sess.topic_name.as_deref()), "user", Some(msg.id.0), text, (true, None));
213    crate::messenger::web::events_bridge::tg_user(&active_sess, text).await;
214    if ask_helpers::feed_active_session_if_running(bot, msg, &sid, &escaped, cli, (sessions, active_sess), config).await? {
215        queue_pending_files(msg, files, config, sessions).await;
216        return Ok(true);
217    }
218    Ok(false)
219}

221async fn execute_resolved_turn(
222    bot: &Bot,
223    msg: &Message,
224    payload: (&str, &str, &[String]),
225    mut sess: SessionData,
226    config: &CliConfig,
227    sessions: &std::sync::Arc<SessionManager>,
228    cli: &AntigravityCli,
229) -> Result<(), teloxide::RequestError> {
230    let (raw_text, current_text, files) = payload;
231    let escaped = commands_registry::escape_non_workflow_slash(current_text);
232    let mut sid = session_init::ensure_session_ready_for_turn(bot, msg, sessions, &mut sess, cli, config).await?;
233    if sid.is_empty() {
234        queue_pending_files(msg, files, config, sessions).await;
235        return Ok(());
236    }
237    history::log_telegram_message(&config.working_dir, &sid, (sess.topic_id, sess.topic_name.as_deref()), "user", Some(msg.id.0), raw_text, (true, None));
238    crate::messenger::web::events_bridge::tg_user(&sess, raw_text).await;
239    if ask_helpers::feed_active_session_if_running(bot, msg, &sid, &escaped, cli, (sessions, sess.clone()), config).await? {
240        queue_pending_files(msg, files, config, sessions).await;
241        return Ok(());
242    }
243    if let Ok(Some(latest)) = sessions.get_active(&sess.session_key()).await {
244        sess = latest;
245    }
246    let cur_prov = SessionData::effective_provider(&sess.provider, &config.provider);
247    if sess.get_session_id(cur_prov) != sid {
248        sid = session_init::ensure_session_ready_for_turn(bot, msg, sessions, &mut sess, cli, config).await?;
249        if sid.is_empty() {
250            queue_pending_files(msg, files, config, sessions).await;
251            return Ok(());
252        }
253    }
254    let prompt = build_prompt_with_attachments(bot, msg, &escaped, files, config, &mut sess, sessions).await;
255    run_cli_stream(bot, msg, &prompt, &sid, cli, (sessions, sess), config).await
256}

258pub(crate) async fn process_text_with_files(
259    bot: &Bot,
260    msg: &Message,
261    payload: (&str, &[String]),
262    config: &CliConfig,
263    sessions: &std::sync::Arc<SessionManager>,
264    cli: &AntigravityCli,
265    aux: (&crate::cron::manager::CronManager, &TopicNameCache),
266) -> Result<(), teloxide::RequestError> {
267    let (text, files) = payload;
268    if text.trim().is_empty() || commands::handle_commands(bot, msg, text, config, sessions, cli, aux).await? {
269        queue_pending_files(msg, files, config, sessions).await;
270        return Ok(());
271    }
272    if try_feed_open_ask_message(bot, msg, payload, config, sessions, cli).await? {
273        return Ok(());
274    }
275    let Some((sess, current_text)) = auth::resolve_message_session(bot, msg, text, config, sessions, cli).await? else {
276        queue_pending_files(msg, files, config, sessions).await;
277        return Ok(());
278    };
279    if try_feed_open_ask_message(bot, msg, (current_text, files), config, sessions, cli).await? {
280        return Ok(());
281    }
282    execute_resolved_turn(bot, msg, (text, current_text, files), sess, config, sessions, cli).await
283}
