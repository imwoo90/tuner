18pub(crate) async fn handle_stream_result(
19    bot: &Bot,
20    chat_id: ChatId,
21    msg_id: Option<MessageId>,
22    thread_id: Option<i32>,
23    topic: (Option<i64>, Option<&str>),
24    resp: crate::cli::CliResponse,
25    config: &CliConfig,
26) -> Result<(Option<String>, Option<MessageId>), teloxide::RequestError> {
27    let last_sid = resp.session_id.clone();
28    if resp.stderr == "Interrupted by user (/stop)" {
29        return Ok((last_sid, msg_id));
30    }
31    let raw_text = if resp.is_error {
32        let err_msg = if !resp.stderr.is_empty() { &resp.stderr } else { &resp.result };
33        crate::cli::antigravity::error_parser::parse_cli_error(err_msg, resp.returncode.unwrap_or(1), config.model.as_deref())
34    } else {
35        resp.result
36    };
37    let chunks = formatting::split_html_message(&formatting::markdown_to_telegram_html(&raw_text), 4000);
38    let (ok, err, sent_id) = send_chunks_to_telegram(bot, chat_id, msg_id, thread_id, &chunks).await;
39    let _ = super::attachments::send_file_attachments(bot, chat_id, thread_id, &raw_text, config).await;
40    if let Some(ref sid) = last_sid {
41        super::history::log_telegram_message(&config.working_dir, sid, topic, "bot", sent_id, &raw_text, (ok, err.as_deref()));
42    }
43    Ok((last_sid, sent_id.map(MessageId)))
44}

46fn truncate_streaming_preview(delta: &str) -> String {
47    if delta.encode_utf16().count() > 3500 {
48        let mut units = 0;
49        let tail: Vec<char> = delta.chars().rev().take_while(|c| {
50            units += c.len_utf16();
51            units <= 3400
52        }).collect();
53        let tail_str: String = tail.into_iter().rev().collect();
54        let aligned = tail_str.find('\n').map(|i| &tail_str[i + 1..]).unwrap_or(&tail_str);
55        format!("...[중간 생략]...\n{}", aligned)
56    } else {
57        delta.to_string()
58    }
59}

61pub(crate) fn render_streaming_preview_html(delta: &str) -> String {
62    let full_html = formatting::markdown_to_telegram_html(&truncate_streaming_preview(delta));
63    formatting::split_html_message(&full_html, 3800).into_iter().last().unwrap_or(full_html)
64}

66pub(crate) async fn handle_text_delta(
67    bot: &Bot,
68    chat_id: ChatId,
69    thread_id: Option<i32>,
70    delta: &str,
71    last_text: &mut String,
72    pub_msg_id: &mut Option<MessageId>,
73    last_edit: &mut Instant,
74) -> Result<(), teloxide::RequestError> {
75    *last_text = delta.to_string();
76    if pub_msg_id.is_some() && last_edit.elapsed() < Duration::from_secs(2) {
77        return Ok(());
78    }
79    let html = render_streaming_preview_html(delta);
80    match send_single_chunk(bot, chat_id, *pub_msg_id, thread_id, &html, true).await {
81        Ok(mid) => {
82            *pub_msg_id = Some(mid);
83            *last_edit = Instant::now();
84        }
85        Err(e) if !is_message_not_modified(&e) => {
86            eprintln!("❌ [tuner] Stream msg failed ({}, {:?}): {:?}", chat_id, *pub_msg_id, e);
87        }
88        _ => {}
89    }
90    Ok(())
91}

99pub(crate) fn should_send_fresh_completion_message(
100    resp: &crate::cli::CliResponse,
101    elapsed: Duration,
102) -> bool {
103    let has_content = !resp.result.trim().is_empty() || resp.is_error;
104    has_content
105        && resp.stderr != "Interrupted by user (/stop)"
106        && elapsed >= Duration::from_secs(60)
107}

109async fn handle_result_event(
110    bot: &Bot,
111    target: (ChatId, Option<i32>),
112    sdata: &crate::session::data::SessionData,
113    config: &CliConfig,
114    elapsed: Duration,
115    state: &mut StreamLoopState,
116    resp: crate::cli::CliResponse,
117) {
118    let (chat_id, thread_id) = target;
119    if !resp.result.is_empty() {
120        state.last_text = resp.result.clone();
121    }
122    let target_mid = if should_send_fresh_completion_message(&resp, elapsed) {
123        if let Some(prev_mid) = state.pub_id.take() {
124            let _ = bot.delete_message(chat_id, prev_mid).await;
125        }
126        None
127    } else {
128        state.pub_id
129    };
130    let topic = (sdata.topic_id, sdata.topic_name.as_deref());
131    if let Ok((sid, sent_mid)) = handle_stream_result(bot, chat_id, target_mid, thread_id, topic, resp, config).await {
132        if let Some(sid) = sid {
133            state.last_sid = Some(sid);
134        }
135        if let Some(mid) = sent_mid {
136            state.pub_id = Some(mid);
137        }
138    }
139}

141async fn process_stream_events(
142    bot: &Bot,
143    chat_id: ChatId,
144    thread_id: Option<i32>,
145    mut stream: futures::stream::BoxStream<'_, StreamEvent>,
146    sdata: &crate::session::data::SessionData,
147    ctx: (&CliConfig, &crate::cli::antigravity::AntigravityCli),
148    state: &mut StreamLoopState,
149) -> Result<(), teloxide::RequestError> {
150    let (config, cli) = ctx;
151    let (stream_start, mut last_edit, mut pending) = (Instant::now(), Instant::now(), false);
152    let mut tick = tokio::time::interval(Duration::from_millis(700));
153    use futures::StreamExt;
154    loop {
155        tokio::select! {
156            ev = stream.next() => match ev {
157                Some(StreamEvent::TextDelta(delta)) => {
158                    let prev_edit = last_edit;
159                    handle_text_delta(bot, chat_id, thread_id, &delta, &mut state.last_text, &mut state.pub_id, &mut last_edit).await?;
160                    pending = state.pub_id.is_some() && last_edit == prev_edit;
161                }
162                Some(StreamEvent::AskQuestion(ask)) => {
163                    super::ask_process::handle_ask_question_event(bot, chat_id, thread_id, ask, sdata, config, cli).await?;
164                }
165                Some(StreamEvent::Result(resp)) => {
166                    pending = false;
167                    handle_result_event(bot, (chat_id, thread_id), sdata, config, stream_start.elapsed(), state, resp).await;
168                }
169                None => break,
170            },
171            _ = tick.tick(), if pending => if last_edit.elapsed() >= Duration::from_secs(2) {
172                let current_text = state.last_text.clone();
173                handle_text_delta(bot, chat_id, thread_id, &current_text, &mut state.last_text, &mut state.pub_id, &mut last_edit).await?;
174                pending = false;
175            }
176        }
177    }
178    Ok(())
179}

181pub(crate) async fn consume_stream(
182    bot: &Bot,
183    target: (ChatId, Option<i32>),
184    stream: futures::stream::BoxStream<'_, StreamEvent>,
185    sessions: &crate::session::manager::SessionManager,
186    session_data: crate::session::data::SessionData,
187    config: &CliConfig,
188    cli: &crate::cli::antigravity::AntigravityCli,
189) -> Result<(), teloxide::RequestError> {
190    let (chat_id, thread_id) = target;
191    let mut state = StreamLoopState {
192        last_text: String::new(),
193        pub_id: None,
194        last_sid: None,
195    };
196    let mut updated = session_data.clone();
197    let mut cleared = false;
198    if let Some(last_mid) = session_data.last_progress_msg_id {
199        let tok = std::env::var("TELEGRAM_TOKEN").unwrap_or_else(|_| config.telegram_token.clone());
200        super::typing::clear_old_progress_reaction(last_mid, chat_id, tok);
201        updated.last_progress_msg_id = None;
202        cleared = true;
203    }
204
205    process_stream_events(bot, chat_id, thread_id, stream, &session_data, (config, cli), &mut state).await?;
206
207    if let Some(mid) = state.pub_id
208        && (state.last_text.contains("🛠️ **Tool Calls:") || state.last_text.contains("<!-- Waiting for")) {
209            let tok = std::env::var("TELEGRAM_TOKEN").unwrap_or_else(|_| config.telegram_token.clone());
210            super::typing::set_progress_reaction(mid.0, chat_id, tok);
211            updated.last_progress_msg_id = Some(mid.0);
212        }
213
214    if state.last_sid.is_some() || cleared || updated.last_progress_msg_id.is_some() {
215        if let Some(sid) = state.last_sid { updated.set_session_id(&config.provider, &sid); }
216        let _ = sessions.update_session(&updated).await;
217    }
218    Ok(())
219}
