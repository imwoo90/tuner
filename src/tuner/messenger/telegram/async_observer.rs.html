1//! # Session Async Event Observer
2//!
3//! ## Overview
4//! Monitors active Antigravity CLI session transcript logs for asynchronous
5//! turn events (subagents, timers, background asks) while idle, dispatching
6//! updates to Telegram and/or the Web Dashboard event hub.
7//!
8//! ## Search Tags
9//! #async-observer, #transcript-watcher, #background-turn

26static WATCHED_SESSIONS: LazyLock<Arc<Mutex<HashSet<String>>>> = LazyLock::new(|| Arc::new(Mutex::new(HashSet::new())));

28struct ObserverCtx {
29    bot: Option<Bot>,
30    transport: String,
31    chat_id: ChatId,
32    thread_id: Option<i32>,
33    session_id: String,
34    cli: AntigravityCli,
35    sessions: Arc<SessionManager>,
36    config: CliConfig,
37}

39impl ObserverCtx {
40    fn key(&self) -> SessionKey {
41        SessionKey { transport: self.transport.clone(), chat_id: self.chat_id.0, topic_id: self.thread_id.map(|t| t as i64) }
42    }
43    fn tid(&self) -> Option<i64> {
44        self.thread_id.map(|t| t as i64)
45    }
46}

48pub(crate) fn spawn_session_async_observer_target(
49    bot: Option<Bot>,
50    target: (i64, Option<i64>, String),
51    session_id: String,
52    cli: AntigravityCli,
53    sessions: Arc<SessionManager>,
54    config: CliConfig,
55) {
56    let (cid, tid, tr) = target;
57    let key = format!("{tr}:{session_id}:{cid}:{tid:?}");
58    if !WATCHED_SESSIONS.lock().unwrap_or_else(|e| e.into_inner()).insert(key.clone()) { return; }
59    let bot = bot.filter(|_| tr == "tg" && cid != 0);
60    let ctx = ObserverCtx { bot, transport: tr, chat_id: ChatId(cid), thread_id: tid.map(|t| t as i32), session_id, cli, sessions, config };
61    tokio::spawn(async move {
62        run_observer_loop(ctx).await;
63        WATCHED_SESSIONS.lock().unwrap_or_else(|e| e.into_inner()).remove(&key);
64    });
65}

67pub(crate) fn spawn_session_async_observer(
68    bot: Bot,
69    msg: &Message,
70    session_id: String,
71    cli: AntigravityCli,
72    sessions: Arc<SessionManager>,
73    config: CliConfig,
74) {
75    spawn_session_async_observer_target(
76        Some(bot), (msg.chat.id.0, msg.thread_id.map(|t| t.0.0 as i64), "tg".into()), session_id, cli, sessions, config,

86async fn handle_async_turn_output(ctx: &ObserverCtx, msg_id: Option<MessageId>, txt: &str) {
87    let (tid64, key) = (ctx.tid(), ctx.key());
88    let (ok, err, sent_id) = if let Some(ref bot) = ctx.bot {
89        let chunks = super::formatting::split_html_message(&super::formatting::markdown_to_telegram_html(txt), 4000);
90        let res = super::stream::send_chunks_to_telegram(bot, ctx.chat_id, msg_id, ctx.thread_id, &chunks).await;
91        let _ = super::attachments::send_file_attachments(bot, ctx.chat_id, ctx.thread_id, txt, &ctx.config).await;
92        res
93    } else {
94        (true, None, None)
95    };
96    let tname = persist_async_antigravity_turn(&ctx.sessions, &key, &ctx.session_id, &ctx.config.provider).await;
97    super::history::log_telegram_message(&ctx.config.working_dir, &ctx.session_id, (tid64, tname.as_deref()), "bot", sent_id, txt, (ok, err.as_deref()));
98    wev::emit_turn_res(&ctx.transport, ctx.chat_id.0, tid64, Some(ctx.session_id.clone()), txt, !ok).await;
99}

101fn create_brain_dir_watcher(brain_dir: &Path, fs_tx: tokio::sync::mpsc::UnboundedSender<()>) -> Option<notify::RecommendedWatcher> {
102    if let Ok(mut w) = notify::recommended_watcher(move |res: Result<notify::Event, _>| {
103        if let Ok(e) = res && matches!(e.kind, EventKind::Modify(_) | EventKind::Create(_)) {
104            let _ = fs_tx.send(());
105        }
106    }) && w.watch(brain_dir, RecursiveMode::Recursive).is_ok() {
107        return Some(w);
108    }
109    None
110}

121async fn dispatch_terminal_event(ctx: &ObserverCtx, delta: ParsedLogDelta, active_turn: &mut Option<ActiveAsyncTurn>) {
122    let tid64 = ctx.tid();
123    if let Some(ref txt) = delta.final_content {
124        let pub_msg_id = active_turn.as_ref().and_then(|t| t.pub_msg_id);
125        if let Some(t) = active_turn.as_ref().and_then(|x| x.typing.as_ref()) { t.abort(); }
126        let lock_arc = ctx.sessions.lock_pool.get((ctx.chat_id.0, tid64));
127        let _fb = if active_turn.is_none() { Some(lock_arc.lock().await) } else { None };
128        handle_async_turn_output(ctx, pub_msg_id, txt).await;
129        *active_turn = None;
130    }
131    if let Some(ask) = delta.ask_question {
132        if let Some(t) = active_turn.as_ref().and_then(|x| x.typing.as_ref()) { t.abort(); }
133        if let Some(ref bot) = ctx.bot {
134            let mut sd = build_observer_ask_session(get_active_session(ctx).await, ctx.chat_id.0, tid64, &ctx.session_id);
135            sd.transport = ctx.transport.clone();
136            let _ = super::ask_process::handle_ask_question_event(bot, ctx.chat_id, ctx.thread_id, ask, &sd, (&ctx.config, &ctx.cli, &ctx.sessions)).await;
137        } else {
138            wev::web_obs_ask(&ctx.transport, ctx.chat_id.0, tid64, &ctx.session_id, ask).await;
139        }
140        *active_turn = None;
141    }
142}

144async fn ensure_active_turn(ctx: &ObserverCtx, active_turn: &mut Option<ActiveAsyncTurn>) {
145    if active_turn.is_none() {
146        let lock = ctx.sessions.lock_pool.get((ctx.chat_id.0, ctx.tid()));
147        if let Ok(guard) = lock.try_lock_owned() {
148            ctx.cli.sessions.set_async_running(&ctx.session_id, true).await;
149            let typing = ctx.bot.as_ref().map(|b| super::typing::AsyncTypingHandle::new(b.clone(), ctx.chat_id, ctx.thread_id));
150            *active_turn = Some(ActiveAsyncTurn { _guard: guard, typing, last_activity: Instant::now(), pub_msg_id: None, last_edit: Instant::now(), last_text: String::new() });
151        }
152    } else if let Some(turn) = active_turn {
153        turn.last_activity = Instant::now();
154    }
155}

157async fn check_and_dispatch_delta(
158    ctx: &ObserverCtx,
159    transcript_path: &Path,
160    last_size: &mut u64,
161    parser: &mut AntigravityLogParser,
162    active_turn: &mut Option<ActiveAsyncTurn>,
163) {
164    let Ok(meta) = std::fs::metadata(transcript_path) else { return };
165    if meta.len() <= *last_size { return; }
166    let delta = parser.parse_log_delta_structured(transcript_path, Some(*last_size));
167    *last_size = delta.new_size;
168    *parser = AntigravityLogParser::new();
169    if !delta.has_activity { return; }
170    crate::upgrade::record_user_activity();
171    ensure_active_turn(ctx, active_turn).await;
172    if delta.final_content.is_some() || delta.ask_question.is_some() {
173        ctx.cli.sessions.set_async_running(&ctx.session_id, false).await;
174        dispatch_terminal_event(ctx, delta, active_turn).await;
175    } else if let (Some(pt), Some(turn)) = (delta.formatted.as_deref(), active_turn.as_mut()) {
176        wev::obs_delta(&ctx.transport, ctx.chat_id.0, ctx.tid(), &ctx.session_id, pt).await;
177        if let Some(ref bot) = ctx.bot {
178            let _ = super::stream::handle_text_delta(bot, ctx.chat_id, ctx.thread_id, pt, &mut turn.last_text, &mut turn.pub_msg_id, &mut turn.last_edit).await;
179        }
180    }
181}

183async fn handle_running_state_check(
184    cli: &AntigravityCli,
185    session_id: &str,
186    transcript_path: &Path,
187    was_running: &mut bool,
188    last_size: &mut u64,
189    parser: &mut AntigravityLogParser,
190) -> bool {
191    let is_running = cli.sessions.is_running(session_id).await;
192    let recent_int = cli.sessions.was_recently_interrupted(session_id, Duration::from_secs(6)).await;
193    if is_running || recent_int || *was_running {
194        *was_running = is_running;
195        cli.sessions.set_async_running(session_id, false).await;
196        *last_size = std::fs::metadata(transcript_path).map(|m| m.len()).unwrap_or(*last_size);
197        *parser = AntigravityLogParser::new();
198        return true;
199    }
200    false
201}

203async fn run_observer_loop(ctx: ObserverCtx) {
204    let bd = crate::cli::antigravity::events::agy_state_root(Some(&ctx.cli.build_env())).join("brain").join(&ctx.session_id);
205    let tp = bd.join(".system_generated").join("logs").join("transcript_full.jsonl");
206    let mut last_size = std::fs::metadata(&tp).map(|m| m.len()).unwrap_or(0);
207    let mut parser = AntigravityLogParser::new();
208    let (fs_tx, mut fs_rx) = tokio::sync::mpsc::unbounded_channel();
209    let watcher = create_brain_dir_watcher(&bd, fs_tx);
210    let mut fs_closed = watcher.is_none();
211    let mut tick = tokio::time::interval(Duration::from_secs(2));
212    let (mut was_running, mut active_turn) = (false, None::<ActiveAsyncTurn>);
213    loop {
214        tokio::select! {
215            _ = ctx.cli.sessions.interrupt_notify.notified() => {}
216            _ = tick.tick() => {}
217            res = fs_rx.recv(), if !fs_closed => if res.is_none() { fs_closed = true; },
218        }
219        let still_current = get_active_session(&ctx).await.is_some_and(|s| s.get_session_id("antigravity") == ctx.session_id);
220        if !ctx.cli.sessions.is_active(&ctx.session_id).await && (!still_current || !tp.exists()) { break; }
221        if handle_running_state_check(&ctx.cli, &ctx.session_id, &tp, &mut was_running, &mut last_size, &mut parser).await {
222            active_turn = None;
223            continue;
224        }
225        if active_turn.as_ref().is_some_and(|t| t.last_activity.elapsed() > Duration::from_secs(300)) {
226            ctx.cli.sessions.set_async_running(&ctx.session_id, false).await;
227            active_turn = None;
228        }
229        check_and_dispatch_delta(&ctx, &tp, &mut last_size, &mut parser, &mut active_turn).await;
230    }
231    ctx.cli.sessions.set_async_running(&ctx.session_id, false).await;
232    drop(watcher);
233}
