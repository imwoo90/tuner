19pub async fn ensure_server_running(server_port: &Arc<Mutex<Option<u16>>>) -> u16 {
20    let mut port_guard = server_port.lock().await;
21    if let Some(port) = *port_guard {
22        if tokio::net::TcpStream::connect(("127.0.0.1", port)).await.is_ok() {
23            return port;
24        }
25        *port_guard = None;
26    }
27
28    let bind_addr = if cfg!(test) {
29        "0.0.0.0:0"
30    } else {
31        "0.0.0.0:8743"
32    };
33
34    let listener = match tokio::net::TcpListener::bind(bind_addr).await {
35        Ok(l) => l,
36        Err(_) => tokio::net::TcpListener::bind("0.0.0.0:0")
37            .await
38            .expect("Failed to bind ephemeral port"),
39    };
40
41    let port = listener.local_addr().map(|a| a.port()).unwrap_or(8743);
42    *port_guard = Some(port);
43
44    let app = Router::new()
45        .route("/review/:token", get(handle_review_page))
46        .route("/review/:token/json", get(handle_review_json))
47        .route("/review/:token/media/:filename", get(handle_media))
48        .route("/review/:token/download/:filename", get(handle_download))
49        .route("/health", get(handle_health));
50
51    tokio::spawn(async move {
52        axum::serve(listener, app).await.ok();
53    });
54
55    port
56}

58pub async fn warmup(server_port: &Arc<Mutex<Option<u16>>>, tunnel_url: &Arc<Mutex<Option<String>>>) {
59    let port = ensure_server_running(server_port).await;
60    let mut tunnel_guard = tunnel_url.lock().await;
61    if tunnel_guard.is_none() {
62        if let Some(url) = spawn_quick_tunnel(port).await {
63            *tunnel_guard = Some(url);
64        }
65    }
66}

68pub async fn get_review_url(
69    server_port: &Arc<Mutex<Option<u16>>>,
70    tunnel_url: &Arc<Mutex<Option<String>>>,
71    token: &str,
72) -> String {
73    let port = ensure_server_running(server_port).await;
74    let mut tunnel_guard = tunnel_url.lock().await;
75    if let Some(ref base) = *tunnel_guard {
76        return format!("{}/review/{}", base, token);
77    }
78
79    if let Some(url) = spawn_quick_tunnel(port).await {
80        *tunnel_guard = Some(url.clone());
81        format!("{}/review/{}", url, token)
82    } else {
83        format!("http://127.0.0.1:{}/review/{}", port, token)
84    }
85}

87fn find_cloudflared_bin() -> Option<PathBuf> {
88    let home = std::env::var("HOME").unwrap_or_else(|_| ".".to_string());
89    let bin_candidates = [
90        PathBuf::from(&home).join(".tuner/bin/cloudflared"),
91        PathBuf::from("/usr/local/bin/cloudflared"),
92        PathBuf::from("/usr/bin/cloudflared"),
93    ];
94    bin_candidates.into_iter().find(|p| p.is_file())
95}

97fn parse_tunnel_url(line: &str) -> Option<String> {
98    if line.contains("api.trycloudflare.com") {
99        return None;
100    }
101    let pos = line.find("https://")?;
102    let sub = &line[pos..];
103    let end = sub.find(".trycloudflare.com")?;
104    let url = &sub[..end + ".trycloudflare.com".len()];
105    let trimmed = url.trim().to_string();
106    if trimmed.contains("api.trycloudflare.com") || trimmed == "https://trycloudflare.com" {
107        None
108    } else {
109        Some(trimmed)
110    }
111}

113pub async fn spawn_quick_tunnel(port: u16) -> Option<String> {
114    let bin = find_cloudflared_bin()?;
115    let mut cmd = tokio::process::Command::new(bin);
116    cmd.args(["tunnel", "--url", &format!("http://127.0.0.1:{}", port)])
117        .stdout(std::process::Stdio::piped())
118        .stderr(std::process::Stdio::piped());
119
120    let mut child = cmd.spawn().ok()?;
121    let stderr = child.stderr.take()?;
122    let mut reader = tokio::io::BufReader::new(stderr);
123    let mut line = String::new();
124
125    let start = Instant::now();
126    while start.elapsed() < Duration::from_secs(10) {
127        use tokio::io::AsyncBufReadExt;
128        line.clear();
129        if reader.read_line(&mut line).await.unwrap_or(0) == 0 {
130            break;
131        }
132        if let Some(url) = parse_tunnel_url(&line) {
133            tokio::spawn(async move {
134                use tokio::io::AsyncBufReadExt;
135                let mut discard = String::new();
136                while let Ok(n) = reader.read_line(&mut discard).await {
137                    if n == 0 {
138                        break;
139                    }
140                    discard.clear();
141                }
142                let _ = child.wait().await;
143            });
144            return Some(url);
145        }
146    }
147    None
148}

150async fn handle_review_page(Path(token): Path<String>) -> Response {
151    let mgr = super::review::global_review_manager();
152    if let Some(html) = mgr.get_html(&token).await {
153        Html(html).into_response()
154    } else {
155        (StatusCode::NOT_FOUND, "Review session not found or expired").into_response()
156    }
157}

159async fn handle_review_json(Path(token): Path<String>) -> Response {
160    let mgr = super::review::global_review_manager();
161    if let Some(files) = mgr.get_files(&token).await {
162        Json(files).into_response()
163    } else {
164        (StatusCode::NOT_FOUND, "Session not found").into_response()
165    }
166}

168async fn handle_media(
169    Path((token, filename)): Path<(String, String)>,
170    req: axum::extract::Request,
171) -> Response {
172    let mgr = super::review::global_review_manager();
173    let file_path = match mgr.get_file_path(&token, &filename).await {
174        Some(p) => p,
175        None => return (StatusCode::NOT_FOUND, "File not found").into_response(),
176    };
177
178    let serve = tower_http::services::ServeFile::new(&file_path);
179    match serve.oneshot(req).await {
180        Ok(res) => res.into_response(),
181        Err(err) => (
182            StatusCode::INTERNAL_SERVER_ERROR,
183            format!("File error: {}", err),
184        )
185            .into_response(),
186    }
187}

189async fn handle_download(
190    Path((token, filename)): Path<(String, String)>,
191    req: axum::extract::Request,
192) -> Response {
193    let mgr = super::review::global_review_manager();
194    let file_path = match mgr.get_file_path(&token, &filename).await {
195        Some(p) => p,
196        None => return (StatusCode::NOT_FOUND, "File not found").into_response(),
197    };
198
199    let serve = tower_http::services::ServeFile::new(&file_path);
200    match serve.oneshot(req).await {
201        Ok(res) => {
202            let mut response = res.into_response();
203            let disp = format!("attachment; filename=\"{}\"", filename);
204            if let Ok(hv) = header::HeaderValue::from_str(&disp) {
205                response.headers_mut().insert(header::CONTENT_DISPOSITION, hv);
206            }
207            response
208        }
209        Err(err) => (
210            StatusCode::INTERNAL_SERVER_ERROR,
211            format!("File error: {}", err),
212        )
213            .into_response(),
214    }
215}
