17/// Scans text for `file://` URLs, validates that they are files within allowed roots,
18/// and returns a list of unique safe file paths.
19pub fn extract_file_paths(text: &str, allowed_roots: &[PathBuf]) -> Vec<PathBuf> {
20    let re = match regex::Regex::new(r"file://[^\s\)]+") {
21        Ok(r) => r,
22        Err(_) => return Vec::new(),
23    };
24    let mut paths = Vec::new();
25    for cap in re.captures_iter(text) {
26        let url_str = cap[0].to_string();
27        if let Ok(parsed_url) = url::Url::parse(&url_str) {
28            if let Ok(path) = parsed_url.to_file_path() {
29                if path.is_file() && crate::security::paths::is_path_safe(&path, allowed_roots) {
30                    if !paths.contains(&path) {
31                        paths.push(path);
32                    }
33                }
34            }
35        }
36    }
37    paths
38}

40fn is_blacklisted_file(path: &std::path::Path) -> bool {
41    let name = path.file_name().and_then(|n| n.to_str()).unwrap_or("");
42    if name == "Cargo.lock" || name == "package-lock.json" || name == "yarn.lock" {
43        return true;
44    }
45    let ext = path.extension().and_then(|e| e.to_str()).unwrap_or("").to_lowercase();
46    if ext == "log" || ext == "jsonl" || ext == "lock" {
47        return true;
48    }
49    let p_str = path.to_string_lossy();
50    if p_str.contains("/target/") || p_str.contains("/.git/") || p_str.contains("/node_modules/") {
51        return true;
52    }
53    false
54}

56fn is_image_file(path: &std::path::Path) -> bool {
57    let ext = path.extension().and_then(|e| e.to_str()).unwrap_or("").to_lowercase();
58    matches!(ext.as_str(), "png" | "jpg" | "jpeg" | "gif" | "webp")
59}

61/// Helper task to send matching file links as In-App review buttons or photos.
62pub(crate) async fn send_file_attachments(
63    bot: &Bot,
64    chat_id: ChatId,
65    thread_id: Option<i32>,
66    raw_text: &str,
67    config: &CliConfig,
68) {
69    let home = std::env::var("HOME").unwrap_or_else(|_| ".".to_string());
70    let home_path = PathBuf::from(&home);
71    let allowed_roots = vec![
72        config.working_dir.clone(),
73        home_path.join(".tuner"),
74        home_path.join("tuner"),
75        home_path.join(".gemini/antigravity-cli"),
76    ];
77    let file_paths = extract_file_paths(raw_text, &allowed_roots);
78    let mut images = Vec::new();
79    let mut code_files = Vec::new();
80
81    for path in file_paths {
82        if is_blacklisted_file(&path) {
83            continue;
84        }
85        if is_image_file(&path) {
86            images.push(path);
87        } else {
88            code_files.push(path);
89        }
90    }
91
92    for img in images {
93        let mut req = bot.send_photo(chat_id, teloxide::types::InputFile::file(&img));
94        if let Some(tid) = thread_id {
95            req = req.message_thread_id(tid);
96        }
97        let _ = req.await;
98    }
99
100    if !code_files.is_empty() {
101        let mgr = super::review::global_review_manager();
102        if let Some((token, count)) = mgr.create_session(&code_files).await {
103            let review_url = mgr.get_review_url(&token).await;
104            send_review_button(bot, chat_id, thread_id, &token, count, &review_url).await;
105        }
106    }
107}

109async fn send_review_button(
110    bot: &Bot,
111    chat_id: ChatId,
112    thread_id: Option<i32>,
113    token: &str,
114    count: usize,
115    review_url: &str,
116) {
117    use teloxide::types::{InlineKeyboardButton, InlineKeyboardMarkup, WebAppInfo};
118    let btn_text = format!("🔍 참조 파일 {}개 검토 (In-App)", count);
119    let review_btn = if let Ok(parsed_url) = review_url.parse() {
120        if review_url.starts_with("https://") {
121            InlineKeyboardButton::web_app(btn_text, WebAppInfo { url: parsed_url })
122        } else {
123            InlineKeyboardButton::url(btn_text, parsed_url)
124        }
125    } else {
126        InlineKeyboardButton::callback(btn_text, format!("dl_files:{}", token))
127    };
128
129    let download_btn = InlineKeyboardButton::callback("📥 직접 받기", format!("dl_files:{}", token));
130    let keyboard = InlineKeyboardMarkup::new(vec![vec![review_btn, download_btn]]);
131
132    let mut msg_req = bot.send_message(chat_id, format!("📁 <b>답변에서 {}개의 파일이 참조되었습니다.</b>", count))
133        .parse_mode(teloxide::types::ParseMode::Html)
134        .reply_markup(keyboard);
135    if let Some(tid) = thread_id {
136        msg_req = msg_req.message_thread_id(tid);
137    }
138    let _ = msg_req.await;
139}

141#[cfg(test)]
142mod tests {
143    use super::*;
144    use tempfile::tempdir;
145
146    #[test]
147    fn test_extract_file_paths_valid_and_invalid() {
148        let dir = tempdir().unwrap();
149        let file1 = dir.path().join("test_file_1.rs");
150        std::fs::write(&file1, "content").unwrap();
151
152        let allowed_roots = vec![dir.path().to_path_buf()];
153
154        // Test absolute file:// link
155        let text = format!("Please review [test_file_1.rs](file://{}) and some invalid [nonexistent](file://{})", file1.to_string_lossy(), dir.path().join("nonexistent.rs").to_string_lossy());
156        let paths = extract_file_paths(&text, &allowed_roots);
157        assert_eq!(paths.len(), 1);
158        assert_eq!(paths[0], file1);
159
160        // Test file:// link with line numbers fragment
161        let text_frag = format!("Error at [line](file://{}#L10-L20)", file1.to_string_lossy());
162        let paths_frag = extract_file_paths(&text_frag, &allowed_roots);
163        assert_eq!(paths_frag.len(), 1);
164        assert_eq!(paths_frag[0], file1);
165    }
