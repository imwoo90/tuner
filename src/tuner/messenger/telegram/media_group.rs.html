40/// Represents `MediaGroupManager` (media group manager).
41pub struct MediaGroupManager {
42    buffers: Arc<Mutex<HashMap<String, MediaGroupBuffer>>>,
43    next_seq: AtomicU64,
44}

46type TargetTopicResult = Result<Option<i64>, String>;
47type DrainedMediaGroup = (Vec<Message>, Option<i64>, TargetTopicResult);
48type GroupLockContext = (i64, Option<i64>, TargetTopicResult);
49
50struct MediaGroupBuffer {
51    seq: u64,
52    messages: Vec<Message>,
53    cancel_tx: mpsc::Sender<()>,
54}

62impl MediaGroupManager {
63    /// Creates a new instance.
64    pub fn new() -> Self {
65        Self {
66            buffers: Arc::new(Mutex::new(HashMap::new())),
67            next_seq: AtomicU64::new(0),
68        }
69    }

71    /// Executes `add_message` (add message).
72    pub async fn add_message(
73        &self,
74        bot: Bot,
75        msg: Message,
76        group_id: String,
77        deps: MediaGroupDeps,
78    ) {
79        let buffers = self.buffers.clone();
80        let mut tx_to_notify = None;
81
82        {
83            let mut guard = buffers.lock().unwrap_or_else(|e| e.into_inner());
84            if let Some(buf) = guard.get_mut(&group_id) {
85                buf.messages.push(msg);
86                tx_to_notify = Some(buf.cancel_tx.clone());
87            } else {
88                let (tx, rx) = mpsc::channel(10);
89                let seq = self.next_seq.fetch_add(1, Ordering::Relaxed);
90                guard.insert(
91                    group_id.clone(),
92                    MediaGroupBuffer {
93                        seq,
94                        messages: vec![msg],
95                        cancel_tx: tx,
96                    },
97                );
98                spawn_timer_task(group_id, buffers.clone(), rx, bot, deps);
99            }
100        }
101
102        if let Some(tx) = tx_to_notify {
103            let _ = tx.send(()).await;
104        }
105    }

107    /// Synchronously flushes any buffered media groups targeting `(chat_id, held_lock_tid)`
108    /// (excluding `exclude_group_id`) while the caller already holds that topic's `lock_pool` mutex.
109    pub async fn flush_other_groups_for_topic(
110        &self,
111        bot: &Bot,
112        chat_id: i64,
113        held_lock_tid: Option<i64>,
114        exclude_group_id: Option<&str>,
115        deps: &MediaGroupDeps,
116    ) -> Result<(), teloxide::RequestError> {
117        let ready = self.drain_matching_groups(chat_id, held_lock_tid, exclude_group_id, deps);
118        for (messages, msg_tid, target_res) in ready {
119            let active_lang = super::handler::resolve_active_lang(&deps.sessions, &deps.config, chat_id, msg_tid).await;
120            let fut = super::handler::LOCKED_TOPIC_ID.scope(
121                target_res,
122                process_media_group(bot.clone(), messages, deps.clone()),
123            );
124            crate::i18n::TASK_ACTIVE_LANG.scope(active_lang, fut).await?;
125            crate::upgrade::record_user_activity();
126        }
127        Ok(())
128    }
129
130    fn drain_matching_groups(
131        &self,
132        chat_id: i64,
133        held_lock_tid: Option<i64>,
134        exclude_group_id: Option<&str>,
135        deps: &MediaGroupDeps,
136    ) -> Vec<DrainedMediaGroup> {
137        let mut guard = self.buffers.lock().unwrap_or_else(|e| e.into_inner());
138        let mut matching: Vec<(u64, String)> = guard
139            .iter()
140            .filter_map(|(gid, buf)| {
141                if exclude_group_id == Some(gid.as_str()) {
142                    return None;
143                }
144                let (cid, msg_tid, target_res) = resolve_group_lock_context(&buf.messages, deps)?;
145                let lock_tid = *target_res.as_ref().unwrap_or(&msg_tid);
146                ((cid, lock_tid) == (chat_id, held_lock_tid)).then(|| (buf.seq, gid.clone()))
147            })
148            .collect();
149        matching.sort_by_key(|(seq, _)| *seq);
150        let mut drained = Vec::with_capacity(matching.len());
151        for (_, gid) in matching {
152            if let Some(buf) = guard.remove(&gid)
153                && let Some((_, msg_tid, target_res)) = resolve_group_lock_context(&buf.messages, deps)
154            {
155                drained.push((buf.messages, msg_tid, target_res));
156            }
157        }
158        drained
159    }
160}

162fn resolve_group_lock_context(
163    messages: &[Message],
164    deps: &MediaGroupDeps,
165) -> Option<GroupLockContext> {
166    let mut sorted: Vec<&Message> = messages.iter().collect();
167    sorted.sort_by_key(|m| m.id.0);
168    let mut caption = String::new();
169    for msg in &sorted {
170        let txt = msg.text().or(msg.caption()).unwrap_or("").trim();
171        if !txt.is_empty() && caption.is_empty() {
172            caption = super::handler::normalize_command_text(txt, deps.bot_username.as_deref());
173        }
174    }
175    let trigger = sorted
176        .iter()
177        .copied()
178        .find(|m| m.reply_to_message().is_some())
179        .or_else(|| sorted.iter().copied().find(|m| !m.text().or(m.caption()).unwrap_or("").trim().is_empty()))
180        .or_else(|| sorted.first().copied())?;
181    let msg_tid = super::get_topic_id(trigger);
182    let target_res = super::handler::resolve_command_target_topic(trigger, &caption, &deps.topic_cache, msg_tid);
183    Some((trigger.chat.id.0, msg_tid, target_res))
184}

186async fn flush_group_on_timeout(
187    group_id: &str,
188    buffers: &Arc<Mutex<HashMap<String, MediaGroupBuffer>>>,
189    bot: &Bot,
190    deps: &MediaGroupDeps,
191) {
192    loop {
193        let peeked = {
194            let guard = buffers.lock().unwrap_or_else(|e| e.into_inner());
195            guard.get(group_id).and_then(|b| {
196                let (cid, msg_tid, target_res) = resolve_group_lock_context(&b.messages, deps)?;
197                Some((cid, *target_res.as_ref().unwrap_or(&msg_tid)))
198            })
199        };
200        let Some((chat_id, want_lock_tid)) = peeked else { return; };
201        let topic_lock = deps.sessions.lock_pool.get((chat_id, want_lock_tid));
202        let _topic_guard = topic_lock.lock().await;
203        let taken = {
204            let mut guard = buffers.lock().unwrap_or_else(|e| e.into_inner());
205            let Some(cur_ctx) = guard.get(group_id).and_then(|b| resolve_group_lock_context(&b.messages, deps)) else {
206                return;
207            };
208            let cur_lock_tid = *cur_ctx.2.as_ref().unwrap_or(&cur_ctx.1);
209            if (cur_ctx.0, cur_lock_tid) != (chat_id, want_lock_tid) {
210                None
211            } else {
212                guard.remove(group_id).map(|b| (b.messages, cur_ctx.1, cur_ctx.2))
213            }
214        };
215        let Some((messages, msg_tid, target_res)) = taken else { continue; };
216        let active_lang = super::handler::resolve_active_lang(&deps.sessions, &deps.config, chat_id, msg_tid).await;
217        let fut = super::handler::LOCKED_TOPIC_ID.scope(
218            target_res,
219            process_media_group(bot.clone(), messages, deps.clone()),
220        );
221        if let Err(e) = crate::i18n::TASK_ACTIVE_LANG.scope(active_lang, fut).await {
222            eprintln!("❌ [tuner] process_media_group failed: {:?}", e);
223        }
224        crate::upgrade::record_user_activity();
225        return;
226    }
227}

229fn spawn_timer_task(
230    group_id: String,
231    buffers: Arc<Mutex<HashMap<String, MediaGroupBuffer>>>,
232    mut rx: mpsc::Receiver<()>,
233    bot: Bot,
234    deps: MediaGroupDeps,
235) {
236    tokio::spawn(async move {
237        loop {
238            tokio::select! {
239                msg = rx.recv() => {
240                    if msg.is_none() {
241                        break;
242                    }
243                }
244                _ = tokio::time::sleep(Duration::from_millis(1500)) => {
245                    flush_group_on_timeout(&group_id, &buffers, &bot, &deps).await;
246                    break;
247                }
248            }
249        }
250    });
251}

253pub(crate) async fn process_media_group(
254    bot: Bot,
255    messages: Vec<Message>,
256    deps: MediaGroupDeps,
257) -> Result<(), teloxide::RequestError> {
258    let mut downloaded_files = Vec::new();
259    let dest_dir = deps.config.working_dir.join("telegram_files");
260    
261    // Sort messages by ID to have deterministic order of files
262    let mut sorted_messages = messages;
263    sorted_messages.sort_by_key(|m| m.id.0);
264
265    // Find the caption text (first non-empty text/caption in the group) and normalize mentions/commands
266    let mut caption = String::new();
267    for msg in &sorted_messages {
268        let txt = msg.text().or(msg.caption()).unwrap_or("").trim();
269        if !txt.is_empty() && caption.is_empty() {
270            caption = super::handler::normalize_command_text(txt, deps.bot_username.as_deref());
271        }
272    }
273
274    for msg in &sorted_messages {
275        match download_telegram_media(&bot, msg, &dest_dir).await {
276            Ok(Some(relative_path)) => {
277                downloaded_files.push(relative_path);
278            }
279            Ok(None) => {}
280            Err(e) => {
281                eprintln!("Error downloading media in group: {:?}", e);
282            }
283        }
284    }
285
286    // Prefer the message carrying a reply quote or caption as the triggering context
287    let trigger_msg = sorted_messages
288        .iter()
289        .find(|m| m.reply_to_message().is_some())
290        .or_else(|| sorted_messages.iter().find(|m| !m.text().or(m.caption()).unwrap_or("").trim().is_empty()))
291        .or_else(|| sorted_messages.first());
292    if let Some(trigger_msg) = trigger_msg {
293        crate::telegram::process_text_with_files(
294            &bot,
295            trigger_msg,
296            (&caption, &downloaded_files),
297            &deps.config,
298            &deps.sessions,
299            &deps.cli,
300            (&deps.cron_manager, &deps.topic_cache),
301        ).await?;
302    }
303
304    Ok(())
305}
