60pub(crate) fn split_slash_command(text: &str) -> (&str, &str) {
61    let trimmed = text.trim();
62    if !trimmed.starts_with('/') {
63        return ("", trimmed);
64    }
65    let first_token = trimmed.split_whitespace().next().unwrap_or("");
66    let cmd = first_token.split('@').next().unwrap_or(first_token);
67    let args = trimmed[first_token.len()..].trim();
68    (cmd, args)
69}

71pub(crate) fn is_lock_free_command(text: &str) -> bool {
72    let (cmd, args) = split_slash_command(text);
73    args.is_empty()
74        && matches!(
75            cmd,
76            "/status"
77                | "/usage"
78                | "/help"

85                | "/stop"
86                | "/stop_all"
87                | "/abort"
88                | "/remote"
89                | "/vault"

91}

101pub(crate) fn get_bot_commands_for_provider(provider: &str) -> Vec<teloxide::types::BotCommand> {
102    COMMON_COMMANDS
103        .iter()
104        .chain(workflow_commands_for_provider(provider).iter())
105        .map(|(c, d)| teloxide::types::BotCommand {
106            command: (*c).to_string(),
107            description: (*d).to_string(),
108        })
109        .collect()
110}

112pub(crate) fn merge_claude_bot_commands(dynamic: Vec<(String, String)>) -> Vec<teloxide::types::BotCommand> {
113    if dynamic.is_empty() {
114        return get_bot_commands_for_provider("claude");
115    }
116    let mut cmds: Vec<teloxide::types::BotCommand> = COMMON_COMMANDS
117        .iter()
118        .map(|(c, d)| teloxide::types::BotCommand { command: (*c).to_string(), description: (*d).to_string() })
119        .collect();
120    for (c, d) in dynamic.into_iter().chain(CLAUDE_WORKFLOW_COMMANDS.iter().map(|(c, d)| ((*c).to_string(), (*d).to_string()))) {
121        if cmds.len() >= 100 { break; }
122        if !cmds.iter().any(|b| b.command == c) {
123            cmds.push(teloxide::types::BotCommand { command: c, description: d });
124        }
125    }
126    cmds
127}

136pub(crate) async fn update_chat_commands_for_provider(bot: &Bot, chat_id: ChatId, provider: &str) {
137    if let Ok(mut map) = LAST_SYNCED_CHAT_PROVIDER.lock() {
138        map.insert(chat_id.0, provider.to_string());
139    }
140    let cmds = if provider == "claude" && !cfg!(test) {
141        let env: std::collections::HashMap<_, _> = std::env::vars().collect();
142        merge_claude_bot_commands(crate::cli::claude::discover_commands_live(&env).await)
143    } else {
144        get_bot_commands_for_provider(provider)
145    };
146    let _ = bot.set_my_commands(cmds).scope(teloxide::types::BotCommandScope::Chat { chat_id: chat_id.into() }).await;
147}

149pub(crate) async fn sync_chat_commands_for_topic(
150    bot: &Bot,
151    msg: &Message,
152    sessions: &crate::session::manager::SessionManager,
153    config: &crate::config::CliConfig,
154) {
155    if cfg!(test) { return; }
156    let key = crate::session::key::SessionKey::telegram(msg.chat.id.0, crate::telegram::get_topic_id(msg));
157    let prov = sessions.get_active(&key).await.ok().flatten().map(|s| s.provider).filter(|p| !p.is_empty()).unwrap_or_else(|| config.provider.clone());
158    if prov.is_empty() { return; }
159    if LAST_SYNCED_CHAT_PROVIDER.lock().ok().is_some_and(|m| m.get(&msg.chat.id.0).map(String::as_str) != Some(prov.as_str())) {
160        let (b, cid) = (bot.clone(), msg.chat.id);
161        tokio::spawn(async move { update_chat_commands_for_provider(&b, cid, &prov).await; });
162    }
163}

165pub(crate) fn is_workflow_slash_command(text: &str) -> bool {
166    let (cmd, _) = split_slash_command(text);
167    matches!(
168        cmd,
169        "/goal" | "/plan" | "/learn" | "/grill_me" | "/grill-me" | "/teamwork" | "/teamwork_preview"
170        | "/teamwork-preview" | "/browser" | "/boost" | "/schedule" | "/compact" | "/review"
171        | "/security_review" | "/security-review" | "/verify" | "/debug" | "/simplify"
172        | "/deep_research" | "/deep-research" | "/code_review" | "/code-review" | "/init" | "/clear"
173    )
174}

176pub(crate) fn escape_non_workflow_slash(text: &str) -> String {
177    let trimmed = text.trim_start();
178    if trimmed.starts_with('/') && !is_workflow_slash_command(trimmed) { format!(" {}", text) } else { text.to_string() }
179}

181pub(crate) async fn register_commands(bot: &Bot) -> Result<(), teloxide::RequestError> {
182    let cmds = get_bot_commands();
183    for s in [None, Some(teloxide::types::BotCommandScope::AllPrivateChats), Some(teloxide::types::BotCommandScope::AllGroupChats)] {
184        if let Some(sc) = s.clone() { let _ = bot.delete_my_commands().scope(sc).await; } else { let _ = bot.delete_my_commands().await; }
185        if let Some(sc) = s { let _ = bot.set_my_commands(cmds.clone()).scope(sc).await; } else { let _ = bot.set_my_commands(cmds.clone()).await; }
186    }
187    Ok(())
188}

190pub(crate) fn format_help_text_for_provider(provider: &str) -> String {
191    let mut text = String::from("🤖 <b>[tuner] Command Reference:</b>\n\n");
192    for (cmd, desc) in COMMON_COMMANDS {
193        text.push_str(&format!("• /{} - {}\n", cmd, desc));
194    }
195    text.push_str(&format!("\n⚡ <b>Workflow Commands ({}):</b>\n", provider));
196    for (cmd, desc) in workflow_commands_for_provider(provider) {
197        text.push_str(&format!("• /{} - {}\n", cmd, desc));
198    }
199    text
200}

202pub(crate) async fn handle_help_command_for_provider(bot: &Bot, msg: &Message, provider: &str) -> Result<(), teloxide::RequestError> {
203    let mut req = bot.send_message(msg.chat.id, format_help_text_for_provider(provider)).parse_mode(teloxide::types::ParseMode::Html);
204    if let Some(tid) = msg.thread_id { req = req.message_thread_id(tid); }
205    let _ = req.await;
206    Ok(())
207}
