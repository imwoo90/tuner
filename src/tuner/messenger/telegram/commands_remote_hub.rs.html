6use std::net::TcpStream;

11#[derive(Clone, Debug, PartialEq)]
12pub enum ServiceStatus {
13    Active,
14    Inactive,
15    NotInstalled,
16}

18#[derive(Clone, Debug, PartialEq)]
19pub struct RemoteHubStatus {
20    pub profile: String,
21    pub remote_workspace: PathBuf,
22    pub antigravity: ServiceStatus,
23    pub antigravity_url: Option<String>,
24    pub antigravity_uptime: Option<String>,
25    pub antigravity_pid: Option<u32>,
26    pub tailscale: ServiceStatus,
27    pub tailscale_ip: Option<String>,
28    pub ssh: ServiceStatus,
29}

31pub fn resolve_profile_info(working_dir: &Path) -> (String, PathBuf, String, String) {
32    let profile = working_dir
33        .parent()
34        .and_then(|p| p.file_name())
35        .and_then(|n| n.to_str())
36        .filter(|n| *n != "profiles" && !n.is_empty())
37        .unwrap_or("default")
38        .to_string();
39    let remote_ws = working_dir
40        .parent()
41        .map(|p| p.join("remote_workspace"))
42        .unwrap_or_else(|| working_dir.join("remote_workspace"));
43    let screen_name = format!("tuner-agy-remote-{}", profile);
44    let instance_name = format!("{}-remote", profile);
45    (profile, remote_ws, screen_name, instance_name)
46}

48pub fn binary_exists(name: &str) -> bool {
49    if let Ok(out) = Command::new("which").arg(name).output() {
50        if out.status.success() && !out.stdout.is_empty() {
51            return true;
52        }
53    }
54    let home = std::env::var("HOME").unwrap_or_default();
55    if !home.is_empty() {
56        for p in [".local/bin", ".gemini/antigravity-cli/bin"] {
57            if Path::new(&format!("{}/{}/{}", home, p, name)).exists() {
58                return true;
59            }
60        }
61    }
62    false
63}

65pub fn find_antigravity_process(instance_name: &str) -> Option<(u32, String)> {
66    let out = Command::new("pgrep").args(["-f", "agy --remote-control"]).output().ok()?;
67    if !out.status.success() {
68        return None;
69    }
70    for line in String::from_utf8_lossy(&out.stdout).lines() {
71        if let Ok(pid) = line.trim().parse::<u32>() {
72            let cmd_path = format!("/proc/{}/cmdline", pid);
73            if let Ok(bytes) = std::fs::read(&cmd_path) {
74                let s = String::from_utf8_lossy(&bytes);
75                if s.contains("--remote-control") && !s.contains("SCREEN") {
76                    let is_match = s.contains(instance_name)
77                        || (instance_name.starts_with("default") && (s.contains("wimvm-dev") || !s.contains("-remote")));
78                    if is_match {
79                        let mut uptime = "Active".to_string();
80                        if let Ok(meta) = std::fs::metadata(format!("/proc/{}", pid)) {
81                            if let Ok(mtime) = meta.created().or_else(|_| meta.modified()) {
82                                if let Ok(el) = std::time::SystemTime::now().duration_since(mtime) {
83                                    let m = el.as_secs() / 60;
84                                    uptime = if m >= 60 { format!("{}h {}m", m / 60, m % 60) } else { format!("{}m", m) };
85                                }
86                            }
87                        }
88                        return Some((pid, uptime));
89                    }
90                }
91            }
92        }
93    }
94    None
95}

97pub fn extract_antigravity_url(profile: &str, screen_name: &str) -> Option<String> {
98    let cache_file = format!("/tmp/tuner_agy_url_{}.txt", profile);
99    if let Ok(c) = std::fs::read_to_string(&cache_file) {
100        let t = c.trim();
101        if t.starts_with("https://antigravity.google.com/r/") {
102            let norm = if t.ends_with("-v1") { format!("{}-v2", &t[..t.len() - 3]) } else { t.to_string() };
103            return Some(norm);
104        }
105    }
106
107    let tmp_screen = format!("/tmp/screen_{}.txt", screen_name);
108    let _ = Command::new("screen").args(["-S", screen_name, "-X", "hardcopy", &tmp_screen]).output();
109    if let Ok(bytes) = std::fs::read(&tmp_screen) {
110        let text = String::from_utf8_lossy(&bytes);
111        if let Some(pos) = text.find("https://antigravity.google.com/r/") {
112            let raw: String = text[pos..].chars().take_while(|c| !c.is_whitespace() && *c != '\0').collect();
113            if !raw.is_empty() {
114                let url = if raw.ends_with("-v1") { format!("{}-v2", &raw[..raw.len() - 3]) } else { raw };
115                let _ = std::fs::write(&cache_file, &url);
116                return Some(url);
117            }
118        }
119    }
120    None
121}

123pub fn check_antigravity_status(instance_name: &str, profile: &str, screen_name: &str) -> (ServiceStatus, Option<String>, Option<String>, Option<u32>) {
124    if !binary_exists("agy") {
125        return (ServiceStatus::NotInstalled, None, None, None);
126    }
127    if let Some((pid, uptime)) = find_antigravity_process(instance_name) {
128        (ServiceStatus::Active, extract_antigravity_url(profile, screen_name), Some(uptime), Some(pid))
129    } else {
130        (ServiceStatus::Inactive, None, None, None)
131    }
132}

134pub fn check_tailscale_status() -> (ServiceStatus, Option<String>) {
135    if !binary_exists("tailscale") {
136        return (ServiceStatus::NotInstalled, None);
137    }
138    if let Ok(out) = Command::new("tailscale").args(["ip", "-4"]).output() {
139        if out.status.success() {
140            let ip = String::from_utf8_lossy(&out.stdout).trim().to_string();
141            if !ip.is_empty() && ip.starts_with("100.") {
142                return (ServiceStatus::Active, Some(ip));
143            }
144        }
145    }
146    (ServiceStatus::Inactive, None)
147}

149pub fn check_ssh_status() -> ServiceStatus {
150    if let Ok(addr) = "127.0.0.1:22".parse() {
151        if TcpStream::connect_timeout(&addr, Duration::from_millis(350)).is_ok() {
152            return ServiceStatus::Active;
153        }
154    }
155    if binary_exists("sshd") || binary_exists("ssh") {
156        ServiceStatus::Inactive
157    } else {
158        ServiceStatus::NotInstalled
159    }
160}

162pub fn query_remote_hub_status(working_dir: &Path) -> RemoteHubStatus {
163    let (profile, remote_ws, screen_name, instance_name) = resolve_profile_info(working_dir);
164    let (agy_status, agy_url, agy_uptime, agy_pid) = check_antigravity_status(&instance_name, &profile, &screen_name);
165    let (ts_status, ts_ip) = check_tailscale_status();
166    RemoteHubStatus {
167        profile,
168        remote_workspace: remote_ws,
169        antigravity: agy_status,
170        antigravity_url: agy_url,
171        antigravity_uptime: agy_uptime,
172        antigravity_pid: agy_pid,
173        tailscale: ts_status,
174        tailscale_ip: ts_ip,
175        ssh: check_ssh_status(),
176    }
177}

179pub async fn start_antigravity_remote(working_dir: &Path) -> Result<String, String> {
180    let (profile, remote_ws, screen_name, instance_name) = resolve_profile_info(working_dir);
181    let _ = stop_antigravity_remote(working_dir).await;
182    let _ = std::fs::create_dir_all(&remote_ws);
183
184    crate::workspace::sync_layout::sync_remote_workspace_rules(working_dir, &remote_ws);
185
186    let cmd = format!("agy --remote-control --remote-control-name {} --add-dir {}", instance_name, remote_ws.display());
187    let status = Command::new("screen")
188        .args(["-dmS", &screen_name, "bash", "-c", &cmd])
189        .current_dir(&remote_ws)
190        .status()
191        .map_err(|e| format!("Spawn error: {}", e))?;
192
193    if !status.success() {
194        return Err("Screen spawn failed".to_string());
195    }
196
197    for _ in 0..8 {
198        tokio::time::sleep(Duration::from_millis(500)).await;
199        if let Some(url) = extract_antigravity_url(&profile, &screen_name) {
200            return Ok(url);
201        }
202    }
203    Ok("https://antigravity.google.com".to_string())
204}

206pub async fn stop_antigravity_remote(working_dir: &Path) -> Result<(), String> {
207    let (profile, _remote_ws, screen_name, instance_name) = resolve_profile_info(working_dir);
208    let _ = Command::new("screen").args(["-S", &screen_name, "-X", "quit"]).output();
209    if profile == "default" {
210        let _ = Command::new("screen").args(["-S", "tuner-agy-remote", "-X", "quit"]).output();
211        let _ = Command::new("screen").args(["-S", "agy-tuner", "-X", "quit"]).output();
212        let _ = Command::new("pkill").args(["-15", "-f", "wimvm-dev"]).output();
213    }
214    let _ = Command::new("pkill").args(["-15", "-f", &format!("--remote-control-name {}", instance_name)]).output();
215    let _ = std::fs::remove_file(format!("/tmp/tuner_agy_url_{}.txt", profile));
216    let _ = std::fs::remove_file(format!("/tmp/screen_{}.txt", screen_name));
217
218    for _ in 0..10 {
219        tokio::time::sleep(Duration::from_millis(200)).await;
220        if find_antigravity_process(&instance_name).is_none() {
221            return Ok(());
222        }
223    }
224
225    let _ = Command::new("pkill").args(["-9", "-f", &format!("--remote-control-name {}", instance_name)]).output();
226    tokio::time::sleep(Duration::from_millis(150)).await;
227    Ok(())
