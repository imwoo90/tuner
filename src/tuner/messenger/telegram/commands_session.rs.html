15pub(crate) fn parse_new_command_args<'a>(
16    msg: &Message,
17    args: &'a str,
18    topic_cache: &super::TopicNameCache,
19    default_tid: Option<i64>,
20) -> (Option<i64>, Option<&'a str>) {
21    let trimmed = args.trim();
22    let raw_cmd = trimmed.split_whitespace().next().unwrap_or("");
23    let rest = trimmed[raw_cmd.len()..].trim();
24    if rest.is_empty() {
25        return (default_tid, None);
26    }
27    if let Some(resolved_tid) = topic_cache.find_by_name(msg.chat.id.0, rest) {
28        return (Some(resolved_tid), None);
29    }
30    (default_tid, Some(rest))
31}

33async fn prepare_topic_reset(
34    config: &CliConfig,
35    sessions: &crate::session::manager::SessionManager,
36    cli: &AntigravityCli,
37    chat_id: i64,
38    topic_id: Option<i64>,
39) -> String {
40    let key = crate::session::key::SessionKey::telegram(chat_id, topic_id);
41    let default_model = config.model.as_deref().unwrap_or("antigravity-default");
42    let mut preserved_model = default_model.to_string();
43
44    if let Ok(Some(existing_sess)) = sessions.get_active(&key).await {
45        if !existing_sess.model.is_empty() && existing_sess.model != "antigravity-default" {
46            preserved_model = existing_sess.model.clone();
47        }
48        let old_sid = existing_sess.get_session_id(&config.provider);
49        if !old_sid.is_empty() {
50            cli.sessions.terminate(&old_sid).await;
51        }
52    }
53    preserved_model
54}

56pub(crate) async fn handle_new_command(
57    bot: &Bot,
58    msg: &Message,
59    args: &str,
60    config: &CliConfig,
61    sessions: &crate::session::manager::SessionManager,
62    cli: &AntigravityCli,
63    topic_cache: &super::TopicNameCache,
64) -> Result<bool, teloxide::RequestError> {
65    let topic_id = crate::telegram::get_topic_id(msg);
66    let (target_topic, inline_prompt) = parse_new_command_args(msg, args, topic_cache, topic_id);
67
68    let preserved_model = prepare_topic_reset(
69        config, sessions, cli, msg.chat.id.0, target_topic,
70    ).await;
71
72    let key = crate::session::key::SessionKey::telegram(msg.chat.id.0, target_topic);
73    let Ok(mut sess) = sessions.reset_provider_session(&key, &config.provider, &preserved_model).await else {
74        return Ok(true);
75    };
76
77    if sess.topic_name.is_none()
78        && let Some(tid) = target_topic
79        && let Some(name) = topic_cache.find_by_id(msg.chat.id.0, tid)
80    {
81        sess.topic_name = Some(name);
82        let _ = sessions.preserve_session_identity(&sess).await;
83    }
84
85    if inline_prompt.is_some() {
86        return Ok(false);
87    }
88
89    let _ = crate::telegram::session_init::initialize_session_with_prompt(
90        bot, msg, sessions, &mut sess, cli, config, None,
91    ).await;
92    Ok(true)
93}

95pub(crate) async fn handle_session_control_commands(
96    bot: &Bot,
97    msg: &Message,
98    text: &str,
99    config: &CliConfig,
100    sessions: &crate::session::manager::SessionManager,
101    cli: &AntigravityCli,
102    topic_cache: &super::TopicNameCache,
103) -> Result<bool, teloxide::RequestError> {
104    let args = text.trim();
105    let raw_cmd = args.split_whitespace().next().unwrap_or("");
106    let cmd = raw_cmd.split('@').next().unwrap_or(raw_cmd);
107    let topic_id = crate::telegram::get_topic_id(msg);
108
109    if cmd == "/new" || cmd == "/reset" {
110        return handle_new_command(bot, msg, args, config, sessions, cli, topic_cache).await;
111    }
112    if cmd == "/stop" {
113        let interrupted = cli.sessions.interrupt(msg.chat.id.0, topic_id).await;
114        let reply_text = if interrupted {
115            t!("bot.stop_interrupted")
116        } else {
117            t!("bot.stop_none")
118        };
119        let _ = send_reply(bot, msg, reply_text).await;
120        return Ok(true);
121    }
122    if cmd == "/stop_all" || cmd == "/abort" {
123        cli.sessions.terminate_all().await;
124        let _ = send_reply(bot, msg, t!("bot.stop_all_success")).await;
125        return Ok(true);
126    }
127    Ok(false)
128}
