14fn resolve_target_topic(
15    msg: &Message,
16    args: &str,
17    topic_cache: &super::TopicNameCache,
18    default_tid: Option<i64>,
19) -> Result<Option<i64>, String> {
20    let parts: Vec<&str> = args.split_whitespace().collect();
21    if parts.len() > 1 {
22        let name = parts[1];
23        if let Some(resolved_tid) = topic_cache.find_by_name(msg.chat.id.0, name) {
24            Ok(Some(resolved_tid))
25        } else {
26            Err(name.to_string())
27        }
28    } else {
29        Ok(default_tid)
30    }
31}

33async fn collect_handover_context(
34    config: &CliConfig,
35    sessions: &crate::session::manager::SessionManager,
36    cli: &AntigravityCli,
37    topic_cache: &super::TopicNameCache,
38    chat_id: i64,
39    topic_id: Option<i64>,
40) -> Option<super::handover::TopicHandoverContext> {
41    let key = crate::session::key::SessionKey::telegram(chat_id, topic_id);
42    let mut old_sid = String::new();
43    let mut topic_name = None;
44    let mut prior_summary = None;
45    let mut past_ids = Vec::new();
46
47    if let Ok(Some(existing_sess)) = sessions.get_active(&key).await {
48        old_sid = existing_sess.get_session_id(&config.provider);
49        past_ids = existing_sess.get_past_session_ids(&config.provider);
50        if old_sid.is_empty() {
51            if let Some(last_id) = past_ids.last() {
52                old_sid = last_id.clone();
53            }
54        }
55        topic_name = existing_sess.topic_name.clone();
56        prior_summary = existing_sess.last_handover_summary.clone();
57        if !old_sid.is_empty() {
58            cli.sessions.terminate(&old_sid).await;
59        }
60    }
61
62    if topic_id.is_none() {
63        return None;
64    }
65
66    if topic_name.is_none() {
67        if let Some(tid) = topic_id {
68            topic_name = topic_cache.find_by_id(chat_id, tid);
69        }
70    }
71
72    super::handover::build_topic_handover(
73        &config.working_dir,
74        &old_sid,
75        topic_id,
76        topic_name,
77        prior_summary.as_deref(),
78        &past_ids,
79    )
80}

82pub(crate) async fn handle_new_command(
83    bot: &Bot,
84    msg: &Message,
85    args: &str,
86    topic_id: Option<i64>,
87    config: &CliConfig,
88    sessions: &crate::session::manager::SessionManager,
89    cli: &AntigravityCli,
90    topic_cache: &super::TopicNameCache,
91) -> Result<(), teloxide::RequestError> {
92    let target_topic = match resolve_target_topic(msg, args, topic_cache, topic_id) {
93        Ok(tid) => tid,
94        Err(name) => {
95            let _ = send_reply(bot, msg, t!("bot.unknown_topic", name = name)).await;
96            return Ok(());
97        }
98    };
99
100    let handover_context = collect_handover_context(
101        config, sessions, cli, topic_cache, msg.chat.id.0, target_topic,
102    ).await;
103
104    let key = crate::session::key::SessionKey::telegram(msg.chat.id.0, target_topic);
105    let model = config.model.as_deref().unwrap_or("antigravity-default");
106    let mut sess = sessions.reset_provider_session(&key, &config.provider, model).await.unwrap();
107
108    let startup_prompt = if let Some(ref handover) = handover_context {
109        sess.last_handover_summary = Some(format!(
110            "Objective: {}\nDecisions: {}\nProgress: {}",
111            handover.topic_objective,
112            handover.agreed_decisions.join("; "),
113            handover.current_progress
114        ));
115        let _ = sessions.preserve_session_identity(&sess).await;
116        Some(handover.handover_prompt.as_str())
117    } else {
118        None
119    };
120
121    let _ = crate::telegram::session_init::initialize_session_with_prompt(
122        bot, msg, sessions, &mut sess, cli, config, startup_prompt,
123    ).await;
124    Ok(())
125}

127pub(crate) async fn handle_session_control_commands(
128    bot: &Bot,
129    msg: &Message,
130    text: &str,
131    config: &CliConfig,
132    sessions: &crate::session::manager::SessionManager,
133    cli: &AntigravityCli,
134    topic_cache: &super::TopicNameCache,
135) -> Result<bool, teloxide::RequestError> {
136    let args = text.trim();
137    let raw_cmd = args.split_whitespace().next().unwrap_or("");
138    let cmd = raw_cmd.split('@').next().unwrap_or(raw_cmd);
139    let topic_id = crate::telegram::get_topic_id(msg);
140
141    if cmd == "/new" || cmd == "/reset" {
142        handle_new_command(bot, msg, args, topic_id, config, sessions, cli, topic_cache).await?;
143        return Ok(true);
144    }
145    if cmd == "/stop" {
146        let interrupted = cli.sessions.interrupt(msg.chat.id.0, topic_id).await;
147        let reply_text = if interrupted {
148            t!("bot.stop_interrupted")
149        } else {
150            t!("bot.stop_none")
151        };
152        let _ = send_reply(bot, msg, reply_text).await;
153        return Ok(true);
154    }
155    if cmd == "/stop_all" || cmd == "/abort" {
156        cli.sessions.terminate_all().await;
157        let _ = send_reply(bot, msg, t!("bot.stop_all_success")).await;
158        return Ok(true);
159    }
160    Ok(false)
161}
