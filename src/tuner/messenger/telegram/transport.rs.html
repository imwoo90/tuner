1//! # Telegram Message Transport Adapter
2//!
3//! ## Overview
4//! Bridges standard background task output envelopes to Teloxide API calls. Supports text splitting,
5//! quiet hour queuing, and forum topic routing.
6//!
7//! ## Collaboration Graph
8//! - Registers as an observer on the central [`MessageBus`](crate::bus::bus::MessageBus).
9//! - Feeds output strings through [`splitting::split_html_message`](super::formatting::splitting::split_html_message).
10//!
11//! ## Search Tags
12//! #transport-adapter, #message-bus, #quiet-hour-queue, #topic-routing

29#[async_trait::async_trait]
30impl Transport for TelegramTransport {
31    fn transport_name(&self) -> &str {
32        "tg"
33    }
34
35    async fn deliver(&self, envelope: &Envelope) -> Result<(), String> {
36        let chat_id = envelope.chat_id;
37        let topic_id = envelope.topic_id;
38        let html_text = crate::telegram::formatting::markdown_to_telegram_html(&envelope.result_text);
39
40        let mut req = self.bot.send_message(teloxide::types::ChatId(chat_id), html_text)
41            .parse_mode(teloxide::types::ParseMode::Html);
42        if let Some(tid) = topic_id {
43            req = req.message_thread_id(tid as i32);
44        }
45        req.await.map(|_| ()).map_err(|e| e.to_string())
46    }
47
48    async fn deliver_broadcast(&self, envelope: &Envelope) -> Result<(), String> {
49        self.deliver(envelope).await
50    }
51}

69#[async_trait::async_trait]
70impl crate::bus::bus::PromptInjector for TelegramPromptInjector {
71    async fn inject_prompt(
72        &self,
73        prompt: &str,
74        chat_id: i64,
75        _label: &str,
76        topic_id: Option<i64>,
77        transport: &str,
78    ) -> Result<String, String> {
79        let key = crate::session::key::SessionKey::for_transport(transport, chat_id, topic_id);
80        let model = self.cfg.model.as_deref().unwrap_or("gemini-3.8-flash");
81        let (session, _) = self.sess.resolve_session(&key, &self.cfg.provider, model).await?;
82        let sid = session.get_session_id(&self.cfg.provider);
83        let opt_sid = if sid.is_empty() { None } else { Some(&sid[..]) };
84        let resp = self.cli.send(prompt, opt_sid, false, self.cfg.working_dir.clone()).await?;
85        Ok(resp.result)
86    }
