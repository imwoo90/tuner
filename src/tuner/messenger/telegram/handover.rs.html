1//! # Telegram Topic Session Handover Manager
2//!
3//! ## Overview
4//! Extracts, distills, and formats topic-specific context across `/new` and `/reset` commands.
5//! Preserves topic objective, agreed decisions, and current progress without context leakage.
6//!
7//! ## Collaboration Graph
8//! - Invoked by [`super::commands_session::handle_new_command`] when resetting a topic session.
9//! - Injected into [`super::session_init::initialize_session_with_prompt`] as the initial boot prompt.
10//! - Delegates history distillation algorithms to [`super::handover_helpers`].
11//!
12//! ## Search Tags
13//! #topic-handover, #session-reset, #context-compression, #forum-topics

19/// Structured representation of compressed topic context for handover.

21pub struct TopicHandoverContext {

23    pub topic_name: Option<String>,
24    pub previous_session_id: String,
25    pub topic_objective: String,
26    pub agreed_decisions: Vec<String>,
27    pub current_progress: String,

31pub fn parse_prior_summary(summary: &str) -> (Option<String>, Vec<String>, Option<String>) {
32    let mut obj = None;
33    let mut decs = Vec::new();
34    let mut prog = None;
35    let mut current_section = 0;
36    let mut obj_lines = Vec::new();
37    let mut prog_lines = Vec::new();
38
39    for line in summary.lines() {
40        let trimmed = line.trim();
41        if let Some(rest) = trimmed.strip_prefix("Objective:") {
42            current_section = 1;
43            let o = rest.trim();
44            if !o.is_empty() { obj_lines.push(o.to_string()); }
45        } else if let Some(rest) = trimmed.strip_prefix("Decisions:") {
46            current_section = 2;
47            let d = rest.trim();
48            if !d.is_empty() {
49                for item in d.split(';') {
50                    let c = item.trim();
51                    if !c.is_empty() && !c.starts_with("No explicit technical decisions") {
52                        decs.push(c.to_string());
53                    }
54                }
55            }
56        } else if let Some(rest) = trimmed.strip_prefix("Progress:") {
57            current_section = 3;
58            let p = rest.trim();
59            if !p.is_empty() { prog_lines.push(p.to_string()); }
60        } else if current_section == 1 && !trimmed.is_empty() {
61            obj_lines.push(trimmed.to_string());
62        } else if current_section == 2 && !trimmed.is_empty() {
63            let c = trimmed.trim_start_matches(|ch| ch == '-' || ch == '*' || ch == '•').trim();
64            if !c.is_empty() && !c.starts_with("No explicit technical decisions") {
65                decs.push(c.to_string());
66            }
67        } else if current_section == 3 && !trimmed.is_empty() {
68            prog_lines.push(trimmed.to_string());
69        }
70    }
71
72    if !obj_lines.is_empty() { obj = Some(obj_lines.join(" ")); }
73    if !prog_lines.is_empty() { prog = Some(prog_lines.join("\n")); }
74    (obj, decs, prog)
75}

77pub fn format_recent_excerpts(entries: &[TelegramHistoryEntry], max_count: usize) -> Vec<String> {
78    let clean_entries: Vec<&TelegramHistoryEntry> = entries.iter()
79        .filter(|e| !is_meta_cmd(&e.text) && !is_boilerplate_or_header(&e.text))
80        .collect();
81    let start = clean_entries.len().saturating_sub(max_count);
82    clean_entries[start..].iter().map(|e| {
83        let role = if e.sender.eq_ignore_ascii_case("user") { "User" } else { "Assistant" };
84        let t = e.text.trim();
85        let clean = if t.len() > 250 { format!("{}... [truncated]", &t[..250]) } else { t.to_string() };
86        format!("[{}]: {}", role, clean.replace('\n', " "))
87    }).collect()
88}

90fn format_handover_prompt(sid: &str, label: &str, obj: &str, decs: &[String], prog: &str, excs: &[String]) -> String {
91    let dec_str = decs.iter().map(|d| format!("- {}\n", d)).collect::<String>();
92    let exc_str = excs.iter().map(|e| format!("{}\n", e)).collect::<String>();
93    format!(
94r#"[TOPIC CONTEXT HANDOVER]
95This session replaces previous session '{sid}' in Telegram forum topic "{label}".
96Previous context has been compressed into this handover summary:
97
98### 1. Topic Objective
99{obj}
100
101### 2. Key Decisions & Agreed Specifications
102{dec_str}
103### 3. Current Progress & State
104{prog}
105
106### 4. Recent Dialogue Excerpt
107{exc_str}
108---
109INSTRUCTIONS FOR CONTINUING THIS SESSION:
1101. Seamlessly continue the work in this topic without losing context.
1112. For your very first response:
112   - Respond in the language used in the topic dialogue (e.g. Korean if previous dialogue was Korean).
113   - Acknowledge that this topic session was refreshed and you have inherited the context.
114   - Mention the topic objective and current state in 1-2 concise sentences.
115   - Ask the user what next step or task they would like to focus on now."#
116    )
117}

119fn resolve_topic_entries(working_dir: &Path, session_id: &str, past_session_ids: &[String]) -> Vec<TelegramHistoryEntry> {
120    let mut entries = extract_topic_history(working_dir, session_id);
121    if entries.is_empty() {
122        for past_sid in past_session_ids.iter().rev() {
123            if past_sid != session_id {
124                let past_entries = extract_topic_history(working_dir, past_sid);
125                if !past_entries.is_empty() {
126                    entries = past_entries;
127                    break;
128                }
129            }
130        }
131    }
132    entries
133}

135fn resolve_topic_name(working_dir: &Path, session_id: &str, mut topic_name: Option<String>, entries: &[TelegramHistoryEntry]) -> Option<String> {
136    if topic_name.is_none() && !session_id.is_empty() {
137        let meta_file = working_dir.join("brain").join(session_id).join("topic.json");
138        if let Ok(c) = std::fs::read_to_string(&meta_file) {
139            if let Ok(m) = serde_json::from_str::<TopicMetadata>(&c) { topic_name = m.topic_name; }
140        }
141    }
142    if topic_name.is_none() {
143        topic_name = entries.iter().find_map(|e| e.topic_name.clone().filter(|s| !s.trim().is_empty()));
144    }
145    topic_name
146}

148pub fn build_topic_handover(
149    working_dir: &Path,
150    session_id: &str,
151    topic_id: Option<i64>,
152    topic_name: Option<String>,
153    prior_summary: Option<&str>,
154    past_session_ids: &[String],
155) -> Option<TopicHandoverContext> {
156    let (prior_obj, prior_decs, prior_prog) = prior_summary.map(parse_prior_summary).unwrap_or((None, Vec::new(), None));
157    let entries = resolve_topic_entries(working_dir, session_id, past_session_ids);
158
159    if entries.is_empty() && prior_summary.is_none() {
160        return None;
161    }
162
163    let resolved_name = resolve_topic_name(working_dir, session_id, topic_name, &entries);
164    let label = resolved_name.clone().unwrap_or_else(|| topic_id.map(|t| format!("Topic #{}", t)).unwrap_or_else(|| "General Topic".to_string()));
165    let obj = extract_topic_objective(&entries, resolved_name.as_deref(), prior_obj.as_deref());
166    let decs = extract_agreed_decisions(&entries, &prior_decs);
167    let prog = extract_current_progress(&entries, prior_prog.as_deref());
168    let excs = format_recent_excerpts(&entries, 4);
169    let sid_ref = if !session_id.is_empty() { session_id } else { "prior_session" };
170    let prompt = format_handover_prompt(sid_ref, &label, &obj, &decs, &prog, &excs);
171
172    Some(TopicHandoverContext {
173        topic_id,
174        topic_name: resolved_name,
175        previous_session_id: session_id.to_string(),
176        topic_objective: obj,
177        agreed_decisions: decs,
178        current_progress: prog,
179        handover_prompt: prompt,
180    })
181}
