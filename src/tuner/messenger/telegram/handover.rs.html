14/// Structured representation of compressed topic context for handover.
15#[derive(Clone, Debug, PartialEq)]
16pub struct TopicHandoverContext {
17    /// Topic objective.
18    pub topic_objective: String,
19    /// Agreed decisions.
20    pub agreed_decisions: Vec<String>,
21    /// Current progress.
22    pub current_progress: String,
23    /// Handover prompt.
24    pub handover_prompt: String,
25}

27/// Executes `parse_prior_summary` (parse prior summary).
28pub fn parse_prior_summary(summary: &str) -> (Option<String>, Vec<String>, Option<String>) {
29    let mut obj = None;
30    let mut decs = Vec::new();
31    let mut prog = None;
32    let mut current_section = 0;
33    let mut obj_lines = Vec::new();
34    let mut prog_lines = Vec::new();
35
36    for line in summary.lines() {
37        let trimmed = line.trim();
38        if let Some(rest) = trimmed.strip_prefix("Objective:") {
39            current_section = 1;
40            let o = rest.trim();
41            if !o.is_empty() { obj_lines.push(o.to_string()); }
42        } else if let Some(rest) = trimmed.strip_prefix("Decisions:") {
43            current_section = 2;
44            let d = rest.trim();
45            if !d.is_empty() {
46                for item in d.split(';') {
47                    let c = item.trim();
48                    if !c.is_empty() && !c.starts_with("No explicit technical decisions") {
49                        decs.push(c.to_string());
50                    }
51                }
52            }
53        } else if let Some(rest) = trimmed.strip_prefix("Progress:") {
54            current_section = 3;
55            let p = rest.trim();
56            if !p.is_empty() { prog_lines.push(p.to_string()); }
57        } else if current_section == 1 && !trimmed.is_empty() {
58            obj_lines.push(trimmed.to_string());
59        } else if current_section == 2 && !trimmed.is_empty() {
60            let c = trimmed.trim_start_matches(['-', '*', '•']).trim();
61            if !c.is_empty() && !c.starts_with("No explicit technical decisions") {
62                decs.push(c.to_string());
63            }
64        } else if current_section == 3 && !trimmed.is_empty() {
65            prog_lines.push(trimmed.to_string());
66        }
67    }
68
69    if !obj_lines.is_empty() { obj = Some(obj_lines.join(" ")); }
70    if !prog_lines.is_empty() { prog = Some(prog_lines.join("\n")); }
71    (obj, decs, prog)
72}

74/// Executes `format_recent_excerpts` (format recent excerpts).
75pub fn format_recent_excerpts(entries: &[TelegramHistoryEntry], max_count: usize) -> Vec<String> {
76    let clean_entries: Vec<&TelegramHistoryEntry> = entries.iter()
77        .filter(|e| !is_meta_cmd(&e.text) && !is_boilerplate_or_header(&e.text))
78        .collect();
79    let start = clean_entries.len().saturating_sub(max_count);
80    clean_entries[start..].iter().map(|e| {
81        let role = if e.sender.eq_ignore_ascii_case("user") { "User" } else { "Assistant" };
82        let t = e.text.trim();
83        let clean = if t.chars().count() > 250 { format!("{} [truncated]", truncate_chars(t, 250)) } else { t.to_string() };
84        format!("[{}]: {}", role, clean.replace('\n', " "))
85    }).collect()
86}

88fn format_handover_prompt(sid: &str, label: &str, obj: &str, decs: &[String], prog: &str, excs: &[String]) -> String {
89    let dec_str = decs.iter().map(|d| format!("- {}\n", d)).collect::<String>();
90    let exc_str = excs.iter().map(|e| format!("{}\n", e)).collect::<String>();
91    format!(
92r#"[TOPIC CONTEXT HANDOVER]
93This session replaces previous session '{sid}' in Telegram forum topic "{label}".
94Previous context has been compressed into this handover summary:
95
96### 1. Topic Objective
97{obj}
98
99### 2. Key Decisions & Agreed Specifications
100{dec_str}
101### 3. Current Progress & State
102{prog}
103
104### 4. Recent Dialogue Excerpt
105{exc_str}
106---
107INSTRUCTIONS FOR CONTINUING THIS SESSION:
1081. Seamlessly continue the work in this topic without losing context.
1092. For your very first response:
110   - Respond in the language used in the topic dialogue (e.g. Korean if previous dialogue was Korean).
111   - Acknowledge that this topic session was refreshed and you have inherited the context.
112   - Mention the topic objective and current state in 1-2 concise sentences.
113   - Ask the user what next step or task they would like to focus on now."#
114    )
115}

117fn resolve_topic_entries(working_dir: &Path, session_id: &str, past_session_ids: &[String]) -> Vec<TelegramHistoryEntry> {
118    let mut entries = extract_topic_history(working_dir, session_id);
119    if entries.is_empty() {
120        for past_sid in past_session_ids.iter().rev() {
121            if past_sid != session_id {
122                let past_entries = extract_topic_history(working_dir, past_sid);
123                if !past_entries.is_empty() {
124                    entries = past_entries;
125                    break;
126                }
127            }
128        }
129    }
130    entries
131}

133fn resolve_topic_name(working_dir: &Path, session_id: &str, mut topic_name: Option<String>, entries: &[TelegramHistoryEntry]) -> Option<String> {
134    if topic_name.is_none() && !session_id.is_empty() {
135        let meta_file = working_dir.join("brain").join(session_id).join("topic.json");
136        if let Ok(c) = std::fs::read_to_string(&meta_file)
137            && let Ok(m) = serde_json::from_str::<TopicMetadata>(&c) { topic_name = m.topic_name; }
138    }
139    if topic_name.is_none() {
140        topic_name = entries.iter().find_map(|e| e.topic_name.clone().filter(|s| !s.trim().is_empty()));
141    }
142    topic_name
143}

145/// Executes `build_topic_handover` (build topic handover).
146pub fn build_topic_handover(
147    working_dir: &Path,
148    session_id: &str,
149    topic_id: Option<i64>,
150    topic_name: Option<String>,
151    prior_summary: Option<&str>,
152    past_session_ids: &[String],
153) -> Option<TopicHandoverContext> {
154    let (prior_obj, prior_decs, prior_prog) = prior_summary.map(parse_prior_summary).unwrap_or((None, Vec::new(), None));
155    let entries = resolve_topic_entries(working_dir, session_id, past_session_ids);
156
157    if entries.is_empty() && prior_summary.is_none() {
158        return None;
159    }
160
161    let resolved_name = resolve_topic_name(working_dir, session_id, topic_name, &entries);
162    let label = resolved_name.clone().unwrap_or_else(|| topic_id.map(|t| format!("Topic #{}", t)).unwrap_or_else(|| "General Topic".to_string()));
163    let obj = extract_topic_objective(&entries, resolved_name.as_deref(), prior_obj.as_deref());
164    let decs = extract_agreed_decisions(&entries, &prior_decs);
165    let prog = extract_current_progress(&entries, prior_prog.as_deref());
166    let excs = format_recent_excerpts(&entries, 4);
167    let sid_ref = if !session_id.is_empty() { session_id } else { "prior_session" };
168    let prompt = format_handover_prompt(sid_ref, &label, &obj, &decs, &prog, &excs);
169
170    Some(TopicHandoverContext {
171        topic_objective: obj,
172        agreed_decisions: decs,
173        current_progress: prog,
174        handover_prompt: prompt,
175    })
