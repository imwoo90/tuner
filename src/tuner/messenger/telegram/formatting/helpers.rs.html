14/// Executes `placeholder` (placeholder).
15pub fn placeholder(kind: &str, idx: usize) -> String {
16    format!("{}{}{}{}", SENTINEL, kind, idx, SENTINEL)
17}

19/// Executes `markdown_to_telegram_html` (markdown to telegram html).
20pub fn markdown_to_telegram_html(text: &str) -> String {
21    let cleaned_text = super::buttons::strip_button_syntax(text);
22
23    let mut code_blocks: Vec<(String, String, bool)> = Vec::new();
24    let mut inline_codes: Vec<String> = Vec::new();
25    let mut links: Vec<(String, String)> = Vec::new();
26    let mut table_blocks: Vec<String> = Vec::new();
27
28    let text_buf = extract_entities(&cleaned_text, &mut code_blocks, &mut inline_codes, &mut links, &mut table_blocks);
29    let escaped = html_escape::encode_safe(&text_buf).to_string();
30    let formatted = apply_markdown_formatting(&escaped);
31
32    restore_entities(formatted, &code_blocks, &inline_codes, &links, &table_blocks)
33}

35fn sanitize_code_block(code: &str, is_quoted: bool) -> String {
36    let raw_lines: Vec<&str> = code.split('\n').collect();
37    if raw_lines.is_empty() {
38        return code.to_string();
39    }
40
41    let has_quote_prefix = raw_lines.iter().any(|l| {
42        let t = l.trim_start();
43        t.starts_with(">! ") || t.starts_with(">!") || (is_quoted && (t.starts_with("> ") || t.starts_with(">")))
44    }) && raw_lines.iter().all(|l| {
45        let t = l.trim_start();
46        t.is_empty()
47            || t.starts_with(">! ")
48            || t.starts_with(">!")
49            || (is_quoted && (t.starts_with("> ") || t.starts_with(">")))
50    });
51
52    if !has_quote_prefix {
53        return code.to_string();
54    }
55
56    let mut cleaned_lines = Vec::new();
57    let total = raw_lines.len();
58    for (i, line) in raw_lines.iter().enumerate() {
59        let trimmed = line.trim();
60        if i == total - 1 && (trimmed == ">!" || trimmed == ">" || trimmed.is_empty()) {
61            continue;
62        }
63
64        if let Some(rest) = line.strip_prefix(">! ") {
65            cleaned_lines.push(rest);
66        } else if let Some(rest) = line.strip_prefix("> ") {
67            cleaned_lines.push(rest);
68        } else if let Some(rest) = line.strip_prefix(">!") {
69            cleaned_lines.push(rest);
70        } else if let Some(rest) = line.strip_prefix(">") {
71            cleaned_lines.push(rest);
72        } else if trimmed == ">!" || trimmed == ">" {
73            cleaned_lines.push("");
74        } else {
75            cleaned_lines.push(*line);
76        }
77    }
78
79    let mut result = cleaned_lines.join("\n");
80    if !result.ends_with('\n') {
81        result.push('\n');
82    }
83    result
84}

86fn unwrap_carousels(text: &str) -> String {
87    let unwrapped = if let Ok(car_re) = Regex::new(r"(?s)`{4,}carousel[ \t]*\n(.*?)`{4,}") {
88        car_re.replace_all(text, "$1").to_string()
89    } else {
90        text.to_string()
91    };
92    if let Ok(slide_re) = Regex::new(r"(?m)^[ \t]*<!--\s*slide\s*-->[ \t]*\n?") {
93        slide_re.replace_all(&unwrapped, "").to_string()
94    } else {
95        unwrapped
96    }
97}

99fn replace_markdown_images(text: &str, links: &mut Vec<(String, String)>) -> String {
100    let Ok(img_re) = Regex::new(r"!\[([^\]]*)\]\(([^)]+)\)") else {
101        return text.to_string();
102    };
103    img_re.replace_all(text, |caps: &Captures| {
104        let caption = caps.get(1).map_or("", |m| m.as_str()).trim();
105        let url = caps.get(2).map_or("", |m| m.as_str()).trim();
106        if url.starts_with('/') || url.starts_with("file://") {
107            if caption.is_empty() {
108                "🖼️".to_string()
109            } else {
110                format!("🖼️ **{}**", caption)
111            }
112        } else {
113            let idx = links.len();
114            let label = if caption.is_empty() { "image".to_string() } else { format!("🖼️ {}", caption) };
115            links.push((label, url.to_string()));
116            placeholder("LK", idx)
117        }
118    }).to_string()
119}

121fn extract_entities(
122    text: &str,
123    code_blocks: &mut Vec<(String, String, bool)>,
124    inline_codes: &mut Vec<String>,
125    links: &mut Vec<(String, String)>,
126    table_blocks: &mut Vec<String>,
127) -> String {
128    let cleaned = unwrap_carousels(text);
129    let mut text_buf = if let Ok(cb_re) = Regex::new(r"(?s)([ \t]*>[!]?[ \t]*)?`{3,}(\w*)\n(.*?)`{3,}") {
130        cb_re.replace_all(&cleaned, |caps: &Captures| {
131            let prefix_quote = caps.get(1).map_or("", |m| m.as_str());
132            let lang = caps.get(2).map_or("", |m| m.as_str()).to_string();
133            let code = caps.get(3).map_or("", |m| m.as_str());
134            let is_quoted = !prefix_quote.trim().is_empty();
135            let sanitized_code = sanitize_code_block(code, is_quoted);
136            let idx = code_blocks.len();
137            code_blocks.push((lang, sanitized_code, is_quoted));
138            format!("{}{}", prefix_quote, placeholder("CB", idx))
139        }).to_string()
140    } else {
141        cleaned
142    };
143
144    text_buf = super::table::extract_tables(&text_buf, table_blocks);
145
146    if let Ok(ic_re) = Regex::new(r"`([^`\n]+)`") {
147        text_buf = ic_re.replace_all(&text_buf, |caps: &Captures| {
148            let code = caps.get(1).map_or("", |m| m.as_str()).to_string();
149            let idx = inline_codes.len();
150            inline_codes.push(code);
151            placeholder("IC", idx)
152        }).to_string();
153    }
154
155    text_buf = replace_markdown_images(&text_buf, links);
156
157    if let Ok(link_re) = Regex::new(r"\[([^\]]+)\]\(([^)]+)\)") {
158        link_re.replace_all(&text_buf, |caps: &Captures| {
159            let link_text = caps.get(1).map_or("", |m| m.as_str()).to_string();
160            let link_url = caps.get(2).map_or("", |m| m.as_str()).to_string();
161            let idx = links.len();
162            links.push((link_text, link_url));
163            placeholder("LK", idx)
164        }).to_string()
165    } else {
166        text_buf
167    }
168}

170fn format_restored_link(l_text: &str, l_url: &str) -> String {
171    let escaped_text = html_escape::encode_safe(l_text).to_string();
172    let formatted_text = apply_markdown_formatting(&escaped_text);
173    if l_url.starts_with("file://") || (l_url.starts_with('/') && !l_url.starts_with("//")) {
174        if formatted_text.contains("\x00IC") || formatted_text.contains("<code>") {
175            formatted_text
176        } else {
177            format!("<code>{}</code>", escaped_text)
178        }
179    } else {
180        let escaped_url = html_escape::encode_double_quoted_attribute(l_url).to_string();
181        format!("<a href=\"{}\">{}</a>", escaped_url, formatted_text)
182    }
183}

185fn restore_entities(
186    mut text: String,
187    code_blocks: &[(String, String, bool)],
188    inline_codes: &[String],
189    links: &[(String, String)],
190    table_blocks: &[String],
191) -> String {
192    for (i, table_text) in table_blocks.iter().enumerate() {
193        let p = placeholder("TB", i);
194        let escaped_table = html_escape::encode_safe(table_text).to_string();
195        text = text.replace(&p, &format!("<pre>{}</pre>", escaped_table));
196    }
197
198    for (i, (l_text, l_url)) in links.iter().enumerate() {
199        let p = placeholder("LK", i);
200        text = text.replace(&p, &format_restored_link(l_text, l_url));
201    }
202
203    for (i, code) in inline_codes.iter().enumerate() {
204        let p = placeholder("IC", i);
205        let escaped_code = html_escape::encode_safe(code).to_string();
206        text = text.replace(&p, &format!("<code>{}</code>", escaped_code));
207    }
208
209    for (i, (lang, code, is_quoted)) in code_blocks.iter().enumerate() {
210        let p = placeholder("CB", i);
211        let escaped_code = html_escape::encode_safe(code).to_string();
212        let raw_block = if !lang.is_empty() {
213            format!("<pre><code class=\"language-{}\">{}</code></pre>", html_escape::encode_safe(lang), escaped_code)
214        } else {
215            format!("<pre>{}</pre>", escaped_code)
216        };
217        let block = if *is_quoted {
218            raw_block
219        } else {
220            format!("<blockquote expandable>{}</blockquote>", raw_block)
221        };
222        text = text.replace(&p, &block);
223    }
224
225    text
226}

228/// Executes `apply_markdown_formatting` (apply markdown formatting).
229pub fn apply_markdown_formatting(escaped: &str) -> String {
230    let mut out = escaped.to_string();
231    let rules = [
232        (r"(?m)^#{1,6}\s+(.+)$", "<b>$1</b>"),
233        (r"\*\*\*([^\*\n]+?)\*\*\*", "<b><i>$1</i></b>"),
234        (r"(?s)\*\*(.+?\*?)\*\*", "<b>$1</b>"),
235    ];
236    for (pat, rep) in rules {
237        if let Ok(re) = Regex::new(pat) {
238            out = re.replace_all(&out, rep).to_string();
239        }
240    }
241
242    out = super::italic::format_italic_delim(&out, '*', "<i>", "</i>");
243    out = super::italic::format_italic_delim(&out, '_', "<i>", "</i>");
244
245    for (pat, rep) in [
246        (r"~~(.+?)~~", "<s>$1</s>"),
247        (r"\|\|(.+?)\|\|", "<tg-spoiler>$1</tg-spoiler>"),
248    ] {
249        if let Ok(re) = Regex::new(pat) {
250            out = re.replace_all(&out, rep).to_string();
251        }
252    }
253
254    out = super::blockquotes::convert_blockquotes(&out);
255
256    for (pat, rep) in [
257        (r"(?m)^[-*]{3,}$", "———"),
258        (r"(?m)^[-*]\s+", "• "),
259    ] {
260        if let Ok(re) = Regex::new(pat) {
261            out = re.replace_all(&out, rep).to_string();
262        }
263    }
264
265    out
