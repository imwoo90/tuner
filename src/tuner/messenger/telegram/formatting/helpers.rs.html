14pub fn placeholder(kind: &str, idx: usize) -> String {
15    format!("{}{}{}{}", SENTINEL, kind, idx, SENTINEL)

18pub fn markdown_to_telegram_html(text: &str) -> String {
19    let cleaned_text = super::buttons::strip_button_syntax(text);
20
21    let mut code_blocks: Vec<(String, String)> = Vec::new();
22    let mut inline_codes: Vec<String> = Vec::new();
23    let mut links: Vec<(String, String)> = Vec::new();
24    let mut table_blocks: Vec<String> = Vec::new();
25
26    let text_buf = extract_entities(&cleaned_text, &mut code_blocks, &mut inline_codes, &mut links, &mut table_blocks);
27    let escaped = html_escape::encode_safe(&text_buf).to_string();
28    let formatted = apply_markdown_formatting(&escaped);
29
30    restore_entities(formatted, &code_blocks, &inline_codes, &links, &table_blocks)
31}

33fn sanitize_code_block(code: &str, is_quoted: bool) -> String {
34    let raw_lines: Vec<&str> = code.split('\n').collect();
35    if raw_lines.is_empty() {
36        return code.to_string();
37    }
38
39    let has_quote_prefix = raw_lines.iter().any(|l| {
40        let t = l.trim_start();
41        t.starts_with(">! ") || t.starts_with(">!") || (is_quoted && (t.starts_with("> ") || t.starts_with(">")))
42    }) && raw_lines.iter().all(|l| {
43        let t = l.trim_start();
44        t.is_empty()
45            || t.starts_with(">! ")
46            || t.starts_with(">!")
47            || (is_quoted && (t.starts_with("> ") || t.starts_with(">")))
48    });
49
50    if !has_quote_prefix {
51        return code.to_string();
52    }
53
54    let mut cleaned_lines = Vec::new();
55    let total = raw_lines.len();
56    for (i, line) in raw_lines.iter().enumerate() {
57        let trimmed = line.trim();
58        if i == total - 1 && (trimmed == ">!" || trimmed == ">" || trimmed.is_empty()) {
59            continue;
60        }
61
62        if let Some(rest) = line.strip_prefix(">! ") {
63            cleaned_lines.push(rest);
64        } else if let Some(rest) = line.strip_prefix("> ") {
65            cleaned_lines.push(rest);
66        } else if let Some(rest) = line.strip_prefix(">!") {
67            cleaned_lines.push(rest);
68        } else if let Some(rest) = line.strip_prefix(">") {
69            cleaned_lines.push(rest);
70        } else if trimmed == ">!" || trimmed == ">" {
71            cleaned_lines.push("");
72        } else {
73            cleaned_lines.push(*line);
74        }
75    }
76
77    let mut result = cleaned_lines.join("\n");
78    if !result.ends_with('\n') {
79        result.push('\n');
80    }
81    result
82}

84fn extract_entities(
85    text: &str,
86    code_blocks: &mut Vec<(String, String)>,
87    inline_codes: &mut Vec<String>,
88    links: &mut Vec<(String, String)>,
89    table_blocks: &mut Vec<String>,
90) -> String {
91    let cb_re = Regex::new(r"(?s)([ \t]*>[!]?[ \t]*)?```(\w*)\n(.*?)```").unwrap();
92    let mut text_buf = cb_re.replace_all(text, |caps: &Captures| {
93        let prefix_quote = caps.get(1).map_or("", |m| m.as_str());
94        let lang = caps.get(2).map_or("", |m| m.as_str()).to_string();
95        let code = caps.get(3).map_or("", |m| m.as_str());
96        let is_quoted = !prefix_quote.trim().is_empty();
97        let sanitized_code = sanitize_code_block(code, is_quoted);
98        let idx = code_blocks.len();
99        code_blocks.push((lang, sanitized_code));
100        format!("{}{}", prefix_quote, placeholder("CB", idx))
101    }).to_string();
102
103    text_buf = super::table::extract_tables(&text_buf, table_blocks);
104
105    let ic_re = Regex::new(r"`([^`\n]+)`").unwrap();
106    text_buf = ic_re.replace_all(&text_buf, |caps: &Captures| {
107        let code = caps.get(1).map_or("", |m| m.as_str()).to_string();
108        let idx = inline_codes.len();
109        inline_codes.push(code);
110        placeholder("IC", idx)
111    }).to_string();
112
113    let link_re = Regex::new(r"\[([^\]]+)\]\(([^)]+)\)").unwrap();
114    link_re.replace_all(&text_buf, |caps: &Captures| {
115        let link_text = caps.get(1).map_or("", |m| m.as_str()).to_string();
116        let link_url = caps.get(2).map_or("", |m| m.as_str()).to_string();
117        let idx = links.len();
118        links.push((link_text, link_url));
119        placeholder("LK", idx)
120    }).to_string()
121}

123fn restore_entities(
124    mut text: String,
125    code_blocks: &[(String, String)],
126    inline_codes: &[String],
127    links: &[(String, String)],
128    table_blocks: &[String],
129) -> String {
130    for (i, table_text) in table_blocks.iter().enumerate() {
131        let p = placeholder("TB", i);
132        let escaped_table = html_escape::encode_safe(table_text).to_string();
133        text = text.replace(&p, &format!("<pre>{}</pre>", escaped_table));
134    }
135
136    for (i, (l_text, l_url)) in links.iter().enumerate() {
137        let p = placeholder("LK", i);
138        let escaped_text = html_escape::encode_safe(l_text).to_string();
139        let formatted_text = apply_markdown_formatting(&escaped_text);
140        let escaped_url = html_escape::encode_double_quoted_attribute(l_url).to_string();
141        text = text.replace(&p, &format!("<a href=\"{}\">{}</a>", escaped_url, formatted_text));
142    }
143
144    for (i, code) in inline_codes.iter().enumerate() {
145        let p = placeholder("IC", i);
146        let escaped_code = html_escape::encode_safe(code).to_string();
147        text = text.replace(&p, &format!("<code>{}</code>", escaped_code));
148    }
149
150    for (i, (lang, code)) in code_blocks.iter().enumerate() {
151        let p = placeholder("CB", i);
152        let escaped_code = html_escape::encode_safe(code).to_string();
153        let block = if !lang.is_empty() {
154            format!("<pre><code class=\"language-{}\">{}</code></pre>", html_escape::encode_safe(lang), escaped_code)
155        } else {
156            format!("<pre>{}</pre>", escaped_code)
157        };
158        text = text.replace(&p, &block);
159    }
160
161    text
162}

164pub fn apply_markdown_formatting(escaped: &str) -> String {
165    let h_re = Regex::new(r"(?m)^#{1,6}\s+(.+)$").unwrap();
166    let mut out = h_re.replace_all(escaped, "<b>$1</b>").to_string();
167
168    let bold_re = Regex::new(r"(?s)\*\*(.+?)\*\*").unwrap();
169    out = bold_re.replace_all(&out, "<b>$1</b>").to_string();
170
171    out = super::italic::format_italic_delim(&out, '*', "<i>", "</i>");
172    out = super::italic::format_italic_delim(&out, '_', "<i>", "</i>");
173
174    let strike_re = Regex::new(r"~~(.+?)~~").unwrap();
175    out = strike_re.replace_all(&out, "<s>$1</s>").to_string();
176
177    let spoiler_re = Regex::new(r"\|\|(.+?)\|\|").unwrap();
178    out = spoiler_re.replace_all(&out, "<tg-spoiler>$1</tg-spoiler>").to_string();
179
180    out = super::blockquotes::convert_blockquotes(&out);
181
182    let hr_re = Regex::new(r"(?m)^[-*]{3,}$").unwrap();
183    out = hr_re.replace_all(&out, "———").to_string();
184
185    let bullet_re = Regex::new(r"(?m)^[-*]\s+").unwrap();
186    out = bullet_re.replace_all(&out, "• ").to_string();
187
188    out
