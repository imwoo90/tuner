1//! # Telegram HTML Italic Normalizer
2//!
3//! ## Overview
4//! Normalizes markdown italic syntax (asterisks and underscores) to prevent unbalanced tags.
5//! Standardizes them into HTML elements compatible with Teloxide output formatting rules.
6//!
7//! ## Search Tags
8//! #italic
9
10fn is_whitespace_or_punct(c: char, delim: char) -> bool {
11    c.is_whitespace() || (c.is_ascii_punctuation() && c != delim)
12}
13
14/// Executes `format_italic_delim` (format italic delim).
15pub fn format_italic_delim(text: &str, delim: char, tag_open: &str, tag_close: &str) -> String {
16    let chars: Vec<char> = text.chars().collect();
17    let mut open_idx = None;
18    let mut pairs = std::collections::HashMap::new();
19    let mut in_html_tag = false;
20
21    let mut idx = 0;
22    while idx < chars.len() {
23        let ch = chars[idx];
24        if ch == '<' {
25            in_html_tag = true;
26            open_idx = None;
27            idx += 1;
28            continue;
29        }
30        if ch == '>' {
31            in_html_tag = false;
32            open_idx = None;
33            idx += 1;
34            continue;
35        }
36        if ch == '\n' {
37            open_idx = None;
38            idx += 1;
39            continue;
40        }
41        if !in_html_tag && ch == delim {
42            let is_left = (idx == 0 || is_whitespace_or_punct(chars[idx - 1], delim))
43                && (idx + 1 < chars.len() && !chars[idx + 1].is_whitespace() && chars[idx + 1] != delim);
44            let is_right = (idx > 0 && !chars[idx - 1].is_whitespace() && chars[idx - 1] != delim)
45                && (idx + 1 == chars.len() || is_whitespace_or_punct(chars[idx + 1], delim));
46
47            if is_right && open_idx.is_some() {
48                if let Some(left) = open_idx.take() {
49                    pairs.insert(left, idx);
50                }
51            } else if is_left && !is_right {
52                open_idx = Some(idx);
53            }
54        }
55        idx += 1;
56    }
57
58    let mut result = String::new();
59    for (idx, &ch) in chars.iter().enumerate() {
60        if ch == delim {
61            if pairs.contains_key(&idx) {
62                result.push_str(tag_open);
63            } else if pairs.values().any(|&right| right == idx) {
64                result.push_str(tag_close);
65            } else {
66                result.push(ch);
67            }
68        } else {
69            result.push(ch);
70        }
71    }
72    result
73}
