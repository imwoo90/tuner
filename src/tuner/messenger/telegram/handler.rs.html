20async fn handle_media_group_message(
21    bot: Bot,
22    msg: Message,

24    deps: MediaGroupDeps,
25    media_group_manager: Arc<MediaGroupManager>,

31fn default_media_prompt(is_video_note: bool) -> &'static str {
32    if is_video_note {
33        "[Video note attached. Process or inspect the attached video file (e.g., via `view_file` or `tools/media_tools/process_video.py`) and respond directly to the user's request.]"
34    } else {
35        "[Voice message attached. Listen to or transcribe the attached audio file (e.g., via `view_file` or `tools/media_tools/transcribe_audio.py`) and respond directly to the user's spoken request.]"
36    }
37}

39async fn handle_single_media_message(
40    bot: Bot,
41    msg: Message,
42    text: String,
43    deps: &MediaGroupDeps,
44) -> Result<(), teloxide::RequestError> {
45    let dest_dir = deps.config.working_dir.join("telegram_files");
46    let downloaded = reply::download_telegram_media(&bot, &msg, &dest_dir).await.ok().flatten();
47    if text.is_empty() && downloaded.is_none() {
48        return Ok(());
49    }
50    let files: Vec<String> = downloaded.into_iter().collect();
51    let is_voice_or_video_note = msg.voice().is_some() || msg.video_note().is_some();
52    let effective_text = if text.is_empty() && is_voice_or_video_note {
53        default_media_prompt(msg.video_note().is_some())
54    } else {
55        text.as_str()
56    };
57    process_text_with_files(
58        &bot,
59        &msg,
60        (effective_text, &files),
61        &deps.config,
62        &deps.sessions,
63        &deps.cli,
64        (&deps.cron_manager, &deps.topic_cache),
65    ).await
66}

85async fn route_media_message(
86    bot: Bot,
87    msg: Message,
88    text: String,
89    deps: MediaGroupDeps,
90    media_group_manager: Arc<MediaGroupManager>,
91) -> Result<(), teloxide::RequestError> {
92    let media_group_id = msg.media_group_id().map(|s| s.to_string());
93    if let Some(group_id) = media_group_id {
94        return handle_media_group_message(bot, msg, group_id, deps, media_group_manager).await;
95    }
96
97    if reply::has_media(&msg) {
98        handle_single_media_message(bot, msg, text, &deps).await
99    } else if !text.is_empty() {
100        handle_pure_text_message(bot, msg, text, &deps).await
101    } else {
102        Ok(())
103    }
104}

106async fn resolve_active_lang(
107    sessions: &SessionManager,
108    config: &CliConfig,
109    chat_id: i64,
110    topic_id: Option<i64>,
111) -> String {
112    let key = crate::session::key::SessionKey::telegram(chat_id, topic_id);
113    sessions.get_active(&key).await
114        .ok().flatten()
115        .and_then(|s| s.language)
116        .or_else(|| config.language.clone())
117        .unwrap_or_else(|| "en".to_string())
118}

120fn normalize_command_text(raw_text: &str, bot_username: Option<&str>) -> String {
121    reply::strip_mention(raw_text, bot_username)
122        .replace("/teamwork_preview", "/teamwork-preview")
123        .replace("/grill_me", "/grill-me")

126async fn dispatch_lock_free(
127    bot: Bot,
128    msg: Message,
129    text: String,
130    deps: MediaGroupDeps,
131) {
132    if let Err(e) = super::commands::handle_commands(
133        &bot,
134        &msg,
135        &text,
136        &deps.config,
137        &deps.sessions,
138        &deps.cli,
139        (&deps.cron_manager, &deps.topic_cache),
140    ).await {
141        eprintln!("❌ [tuner] Error handling lock-free command '{}' (chat: {}, thread: {:?}): {:?}", text, msg.chat.id, msg.thread_id.map(|t| t.0.0), e);
142    }
143}

145pub(crate) async fn handle_message(
146    bot: Bot,
147    msg: Message,
148    config: Arc<CliConfig>,
149    sessions: Arc<SessionManager>,
150    cli: Arc<AntigravityCli>,
151    aux: (Arc<CronManager>, Arc<TopicNameCache>),
152    meta: (Arc<BotInfo>, Arc<MediaGroupManager>),
153) -> Result<(), teloxide::RequestError> {
154    let (cron_manager, topic_cache) = aux;
155    let (bot_info, media_group_manager) = meta;
156    println!("🤖 [tuner] handle_message: received update from chat {}, text: {:?}", msg.chat.id, msg.text().or(msg.caption()));
157    if !super::auth::validate_and_auth_message(&bot, &msg, &config, &sessions, &topic_cache).await? {
158        return Ok(());
159    }
160
161    let topic_id = get_topic_id(&msg);
162    let active_lang = resolve_active_lang(&sessions, &config, msg.chat.id.0, topic_id).await;
163
164    let raw_text = msg.text().or(msg.caption()).unwrap_or("");
165    let text = normalize_command_text(raw_text, bot_info.username.as_deref());
166    let deps = MediaGroupDeps {
167        config,
168        sessions: sessions.clone(),
169        cli,
170        cron_manager,
171        topic_cache,
172    };
173
174    if super::commands_registry::is_lock_free_command(&text) {
175        let fut = crate::i18n::TASK_ACTIVE_LANG.scope(active_lang, async move {
176            dispatch_lock_free(bot, msg, text, deps).await;
177        });
178        if cfg!(test) { fut.await; } else { tokio::spawn(fut); }
179        return Ok(());
180    }
181
182    let lock_pool = sessions.lock_pool.clone();
183    let fut = crate::i18n::TASK_ACTIVE_LANG.scope(active_lang, async move {
184        let lock = lock_pool.get((msg.chat.id.0, topic_id));
185        let _guard = lock.lock().await;
186        route_media_message(bot, msg, text, deps, media_group_manager).await
187    });
188    if cfg!(test) { fut.await } else { tokio::spawn(fut); Ok(()) }
189}
