15pub fn extract_topic_history(working_dir: &Path, session_id: &str) -> Vec<TelegramHistoryEntry> {
16    if session_id.trim().is_empty() { return Vec::new(); }
17    let path = working_dir.join("brain").join(session_id).join("telegram_history.jsonl");
18    let file = match File::open(&path) {
19        Ok(f) => f,
20        Err(_) => return Vec::new(),
21    };
22
23    let mut reader = BufReader::new(file);
24    let mut entries = Vec::new();
25    let mut raw_line = Vec::new();
26
27    while let Ok(n) = reader.read_until(b'\n', &mut raw_line) {
28        if n == 0 { break; }
29        let line = String::from_utf8_lossy(&raw_line);
30        let trimmed = line.trim();
31        if !trimmed.is_empty() {
32            if let Ok(entry) = serde_json::from_str::<TelegramHistoryEntry>(trimmed) {
33                let text = entry.text.trim();
34                if entry.is_success && entry.error.is_none() && !text.is_empty() && !is_meta_cmd(text) {
35                    entries.push(entry);
36                }
37            }
38        }
39        raw_line.clear();
40    }
41    entries
42}

44pub fn is_meta_cmd(text: &str) -> bool {
45    let lower = text.trim().to_lowercase();
46    if lower.starts_with("[topic context handover]") || lower.contains("instructions for continuing this session") {
47        return true;
48    }
49    let raw_cmd = lower.split_whitespace().next().unwrap_or("");
50    let cmd = raw_cmd.split('@').next().unwrap_or(raw_cmd);
51    matches!(cmd, "/new" | "/reset" | "/stop" | "/abort" | "/stop_all" | "/model" | "/status" | "/usage" | "/memory" | "/help" | "/diagnose" | "/restart" | "/upgrade")
52}

54pub fn is_greeting(line: &str) -> bool {
55    let l = line.trim().to_lowercase();
56    matches!(l.as_str(), "hi" | "hello" | "hey" | "안녕" | "안녕하세요" | "시작" | "start" | "hi!" | "hello!" | "안녕하세요!" | "hi tuner" | "hello tuner")
57}

59pub fn extract_topic_objective(entries: &[TelegramHistoryEntry], topic_name: Option<&str>, prior_obj: Option<&str>) -> String {
60    if let Some(po) = prior_obj {
61        let clean_po = po.trim();
62        if !clean_po.is_empty() && clean_po != "Ongoing topic tasks and technical discussions" {
63            return clean_po.to_string();
64        }
65    }
66
67    let user_msg = entries.iter()
68        .find(|e| e.sender.eq_ignore_ascii_case("user"))
69        .map(|e| {
70            let non_greetings: Vec<&str> = e.text.lines()
71                .map(|l| l.trim())
72                .filter(|l| !l.is_empty() && !is_greeting(l))
73                .collect();
74            let combined = if !non_greetings.is_empty() {
75                non_greetings.join(" ")
76            } else {
77                e.text.trim().to_string()
78            };
79            if combined.len() > 180 { format!("{}...", &combined[..180]) } else { combined }
80        });
81
82    match (topic_name, user_msg) {
83        (Some(n), Some(u)) if !n.is_empty() => format!("[Topic: {}] {}", n, u),
84        (Some(n), None) if !n.is_empty() => format!("Topic focus: {}", n),
85        (_, Some(u)) => u,
86        _ => "Ongoing topic tasks and technical discussions".to_string(),
87    }
88}

90pub fn is_boilerplate_or_header(line: &str) -> bool {
91    let l = line.trim();
92    l.starts_with('#')
93        || l.starts_with("```")
94        || l.starts_with("---")
95        || l.starts_with("[TOPIC")
96        || l.contains("Seamlessly continue")
97        || l.contains("very first response")
98        || l.contains("Acknowledge that this topic")
99        || l.contains("INSTRUCTIONS FOR")
100}

102pub fn is_decision_line(line: &str) -> bool {
103    if is_boilerplate_or_header(line) { return false; }
104    let l = line.to_lowercase();
105    let kw = [
106        "결정", "확정", "합의", "선택", "채택", "적용", "완료", "하기로", "방식으로",
107        "decided", "agreed", "chosen", "will use", "adopted", "plan is", "confirmed", "implemented"
108    ];
109    kw.iter().any(|k| l.contains(k))
110}

112fn merge_prior_decisions(decisions: Vec<String>, prior_decisions: &[String]) -> Vec<String> {
113    if prior_decisions.is_empty() { return decisions; }
114    let mut merged = Vec::new();
115    for pd in prior_decisions {
116        let clean = pd.trim();
117        if !clean.is_empty()
118            && !clean.starts_with("No explicit technical decisions")
119            && !merged.contains(&clean.to_string())
120            && !decisions.contains(&clean.to_string())
121        {
122            merged.push(clean.to_string());
123        }
124    }
125    merged.extend(decisions);
126    if merged.len() > 5 {
127        let start = merged.len() - 5;
128        merged[start..].to_vec()
129    } else {
130        merged
131    }
132}

134fn extract_fallback_steps(entries: &[TelegramHistoryEntry]) -> Vec<String> {
135    let mut steps = Vec::new();
136    for entry in entries.iter().rev().filter(|e| !e.sender.eq_ignore_ascii_case("user")) {
137        for line in entry.text.lines().rev() {
138            let clean = line.trim().trim_start_matches(|c| c == '-' || c == '*' || c == '•').trim();
139            if !is_boilerplate_or_header(clean)
140                && (clean.starts_with("1.") || clean.starts_with("2.") || clean.contains("Step") || clean.contains("단계"))
141                && clean.len() > 10
142            {
143                let item = if clean.len() > 150 { format!("{}...", &clean[..150]) } else { clean.to_string() };
144                if !steps.contains(&item) { steps.push(item); }
145            }
146            if steps.len() >= 3 { break; }
147        }
148        if steps.len() >= 3 { break; }
149    }
150    steps.reverse();
151    steps
152}

154pub fn extract_agreed_decisions(entries: &[TelegramHistoryEntry], prior_decisions: &[String]) -> Vec<String> {
155    let mut decisions = Vec::new();
156    for entry in entries.iter().rev() {
157        for line in entry.text.lines().rev() {
158            let clean = line.trim().trim_start_matches(|c| c == '-' || c == '*' || c == '•').trim();
159            if clean.len() >= 6 && is_decision_line(clean) {
160                let item = if clean.len() > 150 { format!("{}...", &clean[..150]) } else { clean.to_string() };
161                if !decisions.contains(&item) { decisions.push(item); }
162            }
163            if decisions.len() >= 5 { break; }
164        }
165        if decisions.len() >= 5 { break; }
166    }
167    decisions.reverse();
168    decisions = merge_prior_decisions(decisions, prior_decisions);
169
170    if decisions.is_empty() {
171        decisions = extract_fallback_steps(entries);
172    }
173    if decisions.is_empty() {
174        decisions.push("No explicit technical decisions recorded in previous session; continuing with baseline.".to_string());
175    }
176    decisions
177}

179pub fn extract_current_progress(entries: &[TelegramHistoryEntry], prior_prog: Option<&str>) -> String {
180    let last_bot = entries.iter().rev().find(|e| !e.sender.eq_ignore_ascii_case("user") && !is_boilerplate_or_header(&e.text));
181    let last_user = entries.iter().rev().find(|e| e.sender.eq_ignore_ascii_case("user") && !is_meta_cmd(&e.text));
182    let mut parts = Vec::new();
183
184    if let Some(bot) = last_bot {
185        let lines: Vec<&str> = bot.text.lines()
186            .map(|l| l.trim())
187            .filter(|l| !l.is_empty() && !is_boilerplate_or_header(l))
188            .take(3)
189            .collect();
190        let text = lines.join(" ");
191        let snip = if text.len() > 200 { format!("{}...", &text[..200]) } else { text };
192        if !snip.is_empty() { parts.push(format!("Last assistant status: {}", snip)); }
193    }
194    if let Some(usr) = last_user {
195        let lines: Vec<&str> = usr.text.lines()
196            .map(|l| l.trim())
197            .filter(|l| !l.is_empty() && !is_greeting(l))
198            .take(3)
199            .collect();
200        let text = lines.join(" ");
201        let snip = if text.len() > 200 { format!("{}...", &text[..200]) } else { text };
202        if !snip.is_empty() { parts.push(format!("Latest user request: {}", snip)); }
203    }
204
205    if parts.is_empty() {
206        if let Some(pp) = prior_prog {
207            let clean_pp = pp.trim();
208            if !clean_pp.is_empty() { return clean_pp.to_string(); }
209        }
210        "Initial session discussion in progress.".to_string()
211    } else {
212        parts.join("\n")
213    }
