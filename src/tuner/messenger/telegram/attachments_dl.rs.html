1//! # Telegram Attachment Direct Downloader
2//!
3//! Handles direct downloading of referenced files via Telegram inline callback buttons.
4//! Packages multiple referenced files into a single compressed ZIP archive to avoid chat flooding,
5//! and ensures documents are delivered to the originating forum topic rather than the General topic.
6//!
7//! ## Collaboration Graph
8//! - [`handle_callback_query`](super::callbacks::handle_callback_query): Dispatches `dl_files:<token>` to this module.
9//! - [`ReviewManager`](super::review::ReviewManager): Provides cached file metadata by token.
10//!
11//! ## Search Tags
12//! #attachments, #zip-compression, #direct-download, #topic-routing

15use std::io::Write;

17use teloxide::prelude::*;
18use super::review::{global_review_manager, ReviewFile};

20/// Compresses a slice of `ReviewFile` items into an in-memory ZIP archive.
21/// Deduplicates entry names if multiple files share identical basenames.
22pub(crate) fn create_zip_archive(files: &[ReviewFile]) -> Result<Vec<u8>, String> {
23    let mut buf = std::io::Cursor::new(Vec::new());
24    let mut zip = zip::ZipWriter::new(&mut buf);
25    let options = zip::write::SimpleFileOptions::default()
26        .compression_method(zip::CompressionMethod::Deflated);
27
28    let mut used_names = HashSet::new();
29    for f in files {
30        let p = Path::new(&f.path);
31        let content = std::fs::read(p).unwrap_or_else(|_| f.content.as_bytes().to_vec());
32
33        let mut entry_name = f.filename.clone();
34        if entry_name.is_empty() {
35            entry_name = p
36                .file_name()
37                .map(|n| n.to_string_lossy().to_string())
38                .unwrap_or_else(|| "file".to_string());
39        }
40
41        if used_names.contains(&entry_name) {
42            let stem = Path::new(&entry_name)
43                .file_stem()
44                .map(|s| s.to_string_lossy().to_string())
45                .unwrap_or_else(|| "file".to_string());
46            let ext = Path::new(&entry_name)
47                .extension()
48                .map(|e| format!(".{}", e.to_string_lossy()))
49                .unwrap_or_default();
50            let mut counter = 2;
51            loop {
52                let candidate = format!("{}_{}{}", stem, counter, ext);
53                if !used_names.contains(&candidate) {
54                    entry_name = candidate;
55                    break;
56                }
57                counter += 1;
58            }
59        }
60        used_names.insert(entry_name.clone());
61
62        zip.start_file(&entry_name, options)
63            .map_err(|e| format!("Failed to create zip entry {}: {}", entry_name, e))?;
64        zip.write_all(&content)
65            .map_err(|e| format!("Failed to write zip content for {}: {}", entry_name, e))?;
66    }
67
68    zip.finish().map_err(|e| format!("Failed to finalize zip archive: {}", e))?;
69    Ok(buf.into_inner())
70}

72/// Handles the `dl_files:<token>` inline button callback.
73/// Delivers files directly to the calling forum topic (`msg.thread_id`).
74pub(crate) async fn handle_dl_files_callback(
75    bot: &teloxide::Bot,
76    msg: &teloxide::types::Message,
77    token: &str,
78) {
79    let mgr = global_review_manager();
80    let files = match mgr.get_files(token).await {
81        Some(f) if !f.is_empty() => f,
82        _ => return,
83    };
84
85    let valid: Vec<_> = files
86        .into_iter()
87        .filter(|f| Path::new(&f.path).is_file())
88        .collect();
89
90    if valid.is_empty() {
91        return;
92    }
93
94    if valid.len() == 1 {
95        let path = PathBuf::from(&valid[0].path);
96        let mut req = bot.send_document(msg.chat.id, teloxide::types::InputFile::file(&path));
97        if let Some(tid) = msg.thread_id {
98            req = req.message_thread_id(tid);
99        }
100        let _ = req.await;
101        return;
102    }
103
104    match create_zip_archive(&valid) {
105        Ok(zip_bytes) => {
106            let input_file = teloxide::types::InputFile::memory(zip_bytes).file_name("attachments.zip");
107            let mut req = bot.send_document(msg.chat.id, input_file);
108            if let Some(tid) = msg.thread_id {
109                req = req.message_thread_id(tid);
110            }
111            let _ = req.await;
112        }
113        Err(err) => {
114            eprintln!("❌ Failed to create zip for download: {}", err);
115            for f in valid {
116                let path = PathBuf::from(&f.path);
117                let mut req = bot.send_document(msg.chat.id, teloxide::types::InputFile::file(&path));
118                if let Some(tid) = msg.thread_id {
119                    req = req.message_thread_id(tid);
120                }
121                let _ = req.await;
122            }
123        }
124    }
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130    use tempfile::tempdir;

132    #[test]
133    fn test_create_zip_archive_success() {
134        let dir = tempdir().unwrap();
135        let file1 = dir.path().join("first.txt");
136        let file2 = dir.path().join("second.txt");
137        std::fs::write(&file1, b"hello from first").unwrap();
138        std::fs::write(&file2, b"hello from second").unwrap();
139
140        let review_files = vec![
141            ReviewFile {
142                filename: "first.txt".to_string(),
143                path: file1.to_string_lossy().to_string(),
144                content: "hello from first".to_string(),
145                size_bytes: 16,
146                language: "text".to_string(),
147            },
148            ReviewFile {
149                filename: "second.txt".to_string(),
150                path: file2.to_string_lossy().to_string(),
151                content: "hello from second".to_string(),
152                size_bytes: 17,
153                language: "text".to_string(),
154            },
155        ];
156
157        let zip_bytes = create_zip_archive(&review_files).expect("zip creation should succeed");
158        assert!(!zip_bytes.is_empty());
159
160        let reader = std::io::Cursor::new(zip_bytes);
161        let mut archive = zip::ZipArchive::new(reader).expect("valid zip archive");
162        assert_eq!(archive.len(), 2);
163
164        {
165            let mut entry1 = archive.by_name("first.txt").expect("first.txt exists");
166            let mut content1 = String::new();
167            std::io::Read::read_to_string(&mut entry1, &mut content1).unwrap();
168            assert_eq!(content1, "hello from first");
169        }
170
171        {
172            let mut entry2 = archive.by_name("second.txt").expect("second.txt exists");
173            let mut content2 = String::new();
174            std::io::Read::read_to_string(&mut entry2, &mut content2).unwrap();
175            assert_eq!(content2, "hello from second");
176        }
177    }

183        let file2 = dir.path().join("sub2").join("common.txt");

185        std::fs::create_dir_all(file2.parent().unwrap()).unwrap();
186        std::fs::write(&file1, b"common 1").unwrap();
187        std::fs::write(&file2, b"common 2").unwrap();
188
189        let review_files = vec![
190            ReviewFile {
191                filename: "common.txt".to_string(),
192                path: file1.to_string_lossy().to_string(),
193                content: "common 1".to_string(),
194                size_bytes: 8,
195                language: "text".to_string(),
196            },
197            ReviewFile {
198                filename: "common.txt".to_string(),
199                path: file2.to_string_lossy().to_string(),
200                content: "common 2".to_string(),
201                size_bytes: 8,
202                language: "text".to_string(),
203            },
204        ];
205

207        let reader = std::io::Cursor::new(zip_bytes);
