29fn parse_agy_group_entries(g: &Value) -> Option<ModelQuotaGroup> {
30    let (mut weekly, mut five_hour) = (None, None);
31    let items_opt = g.get("buckets").or_else(|| g.get("entries")).and_then(Value::as_array);
32    if let Some(items) = items_opt {
33        for item in items {
34            let bid = item.get("bucketId").or_else(|| item.get("quotaId")).and_then(Value::as_str).unwrap_or("");
35            let win = item.get("window").and_then(Value::as_str).unwrap_or("");
36            let label = item.get("displayName").or_else(|| item.get("label")).and_then(Value::as_str).unwrap_or("");
37            let frac = item.get("remainingFraction").and_then(Value::as_f64);
38            let detail = item.get("resetTime").and_then(Value::as_str).map(format_reset_timestamp).unwrap_or_default();
39            let pct = (frac.unwrap_or(0.0) * 100.0).clamp(0.0, 100.0);
40            let lim = QuotaLimit { pct, detail };
41            if win == "weekly" || bid.ends_with("weekly") || label.contains("Weekly") {
42                weekly = Some(lim);
43            } else if win == "5h" || bid.ends_with("5h") || label.contains("5-Hour") || label.contains("Five Hour") {
44                five_hour = Some(lim);
45            }
46        }
47    }
48    if weekly.as_ref().is_some_and(|w| w.pct == 0.0)
49        && let Some(ref mut f) = five_hour
50        && f.pct == 0.0
51        && f.detail.is_empty()
52    {
53        f.detail = "Disabled: Weekly limit hit.".to_string();
54    }
55    (weekly.is_some() || five_hour.is_some()).then_some(ModelQuotaGroup { weekly, five_hour })
56}

58/// Parses the Google CloudCode `retrieveUserQuotaSummary` JSON response into a [`QuotaReport`].
59pub fn parse_antigravity_quota_json(
60    quota_json: &Value,
61    email: Option<&str>,
62    tier: Option<&str>,
63) -> Result<QuotaReport, String> {
64    let groups = quota_json
65        .get("groups")
66        .and_then(Value::as_array)
67        .ok_or_else(|| "Missing 'groups' array in quota summary".to_string())?;
68
69    let account = match (email.filter(|s| !s.is_empty()), tier.filter(|s| !s.is_empty())) {
70        (Some(e), Some(t)) => Some(format!("{} ({})", e, t)),
71        (Some(e), None) => Some(e.to_string()),
72        (None, Some(t)) => Some(t.to_string()),
73        (None, None) => None,
74    };
75
76    let (mut gemini, mut claude) = (None, None);
77    for g in groups {
78        let lower = g
79            .get("displayName")
80            .or_else(|| g.get("groupName"))
81            .and_then(Value::as_str)
82            .unwrap_or("")
83            .to_ascii_lowercase();
84        if let Some(group) = parse_agy_group_entries(g) {
85            if lower.contains("gemini") {
86                gemini = Some(group);
87            } else if lower.contains("claude") || lower.contains("gpt") {
88                claude = Some(group);
89            }
90        }
91    }
92    if gemini.is_none() && claude.is_none() {
93        return Err("No recognized quota groups in response".to_string());
94    }
95    Ok(QuotaReport { account, gemini, claude, raw_fallback: String::new() })
96}

98async fn refresh_agy_oauth_token(
99    client: &reqwest::Client,
100    token_path: &Path,
101    token_url: &str,
102    refresh: &str,
103    mut doc: Value,
104) -> Result<String, String> {
105    let client_id = std::env::var("AGY_OAUTH_CLIENT_ID").unwrap_or_else(|_| "antigravity-cli".to_string());
106    let client_secret = std::env::var("AGY_OAUTH_CLIENT_SECRET").unwrap_or_default();
107    let params = [
108        ("client_id", client_id.as_str()),
109        ("client_secret", client_secret.as_str()),
110        ("refresh_token", refresh),
111        ("grant_type", "refresh_token"),
112    ];
113    let resp: Value = client
114        .post(token_url)
115        .form(&params)
116        .send()
117        .await
118        .map_err(|e| e.to_string())?
119        .json()
120        .await
121        .map_err(|e| e.to_string())?;
122    let new_access = resp
123        .get("access_token")
124        .and_then(Value::as_str)
125        .ok_or_else(|| "Missing access_token".to_string())?
126        .to_string();
127    let expires_in = resp.get("expires_in").and_then(Value::as_i64).unwrap_or(3600);
128    if let Some(t) = doc.get_mut("token").and_then(Value::as_object_mut) {
129        t.insert("access_token".to_string(), Value::String(new_access.clone()));
130        let exp = (Utc::now() + chrono::Duration::seconds(expires_in)).to_rfc3339();
131        t.insert("expiry".to_string(), Value::String(exp));
132        if let Ok(s) = serde_json::to_string(&doc) {
133            let _ = tokio::fs::write(token_path, s).await;
134        }
135    }
136    Ok(new_access)
137}

139async fn load_or_refresh_agy_token(
140    client: &reqwest::Client,
141    token_path: &Path,
142    token_url: &str,
143) -> Result<(String, Option<String>), String> {
144    let raw = tokio::fs::read_to_string(token_path).await.map_err(|e| e.to_string())?;
145    let doc: Value = serde_json::from_str(&raw).map_err(|e| e.to_string())?;
146    let email = doc.get("id_token_email").and_then(Value::as_str).map(String::from);
147    let tok_obj = doc.get("token").ok_or_else(|| "Missing 'token' field".to_string())?;
148    let access = tok_obj.get("access_token").and_then(Value::as_str).unwrap_or("").to_string();
149    let refresh = tok_obj.get("refresh_token").and_then(Value::as_str).unwrap_or("").to_string();
150    let expiry_str = tok_obj.get("expiry").and_then(Value::as_str).unwrap_or("");
151
152    let needs_refresh = DateTime::parse_from_rfc3339(expiry_str)
153        .map(|dt| dt.with_timezone(&Utc).signed_duration_since(Utc::now()).num_seconds() < 60)
154        .unwrap_or(access.is_empty());
155
156    if !needs_refresh && !access.is_empty() {
157        return Ok((access, email));
158    }
159    if refresh.is_empty() {
160        return Err("OAuth access token expired and no refresh_token".to_string());
161    }
162    let new_access = refresh_agy_oauth_token(client, token_path, token_url, &refresh, doc).await?;
163    Ok((new_access, email))
164}

166/// Queries Antigravity quota directly via the CloudCode HTTP API using the given endpoint URLs.
167pub async fn query_antigravity_quota_http_with_urls(
168    token_path: &Path,
169    base_url: &str,
170    token_url: &str,
171) -> Result<QuotaReport, String> {
172    let client = reqwest::Client::builder().timeout(Duration::from_secs(4)).build().map_err(|e| e.to_string())?;
173    let (access_token, email) = load_or_refresh_agy_token(&client, token_path, token_url).await?;
174    let base = base_url.trim_end_matches('/');
175
176    let assist_val: Value = client
177        .post(format!("{}/v1internal:loadCodeAssist", base))
178        .bearer_auth(&access_token)
179        .header(reqwest::header::USER_AGENT, AGY_USER_AGENT)
180        .json(&json!({"metadata": {"ideType": "ANTIGRAVITY"}}))
181        .send()
182        .await
183        .map_err(|e| e.to_string())?
184        .json()
185        .await
186        .unwrap_or_else(|_| json!({}));
187
188    let project = assist_val.get("cloudaicompanionProject").and_then(Value::as_str).unwrap_or("aicode-consumers");
189    let tier = assist_val.get("paidTier").and_then(|t| t.get("name")).and_then(Value::as_str);
190
191    let quota_val: Value = client
192        .post(format!("{}/v1internal:retrieveUserQuotaSummary", base))
193        .bearer_auth(&access_token)
194        .header(reqwest::header::USER_AGENT, AGY_USER_AGENT)
195        .json(&json!({"project": project}))
196        .send()
197        .await
198        .map_err(|e| e.to_string())?
199        .json()
200        .await
201        .map_err(|e| e.to_string())?;
202
203    parse_antigravity_quota_json(&quota_val, email.as_deref(), tier)
204}

206/// Queries Antigravity quota directly from Google's CloudCode endpoint using the default token file.
207pub async fn query_antigravity_quota_http() -> Result<QuotaReport, String> {
208    let home = std::env::var("HOME").unwrap_or_else(|_| "/root".to_string());
209    let token_path = PathBuf::from(home).join(".gemini/antigravity-cli/antigravity-oauth-token");
210    query_antigravity_quota_http_with_urls(&token_path, AGY_CLOUDCODE_BASE, GOOGLE_TOKEN_URL).await
211}

213/// Queries Antigravity quota via direct HTTP with automatic fallback to PTY modal scraping.
214pub(crate) async fn query_antigravity_usage_html(
215    bot: &Bot,
216    msg: &Message,
217    config: &CliConfig,
218    sessions: &SessionManager,
219    cli: &AntigravityCli,
220    mut sess: SessionData,
221) -> Option<String> {
222    if let Ok(report) = query_antigravity_quota_http().await {
223        return Some(render_quota_message(&report));
224    }
225    let sid = match crate::telegram::session_init::initialize_session_if_needed(
226        bot, msg, sessions, &mut sess, cli, config,
227    )
228    .await
229    {
230        Ok(s) if !s.is_empty() => s,
231        Ok(_) => {
232            let _ = send_reply(bot, msg, "❌ Could not initialize session for quota check").await;
233            return None;
234        }
235        Err(e) => {
236            let _ = send_reply(bot, msg, format!("❌ Error initializing session: {}", e)).await;
237            return None;
238        }
239    };
240    let mut s_cli = cli.clone();
241    s_cli.config.chat_id = msg.chat.id.0;
242    s_cli.config.topic_id = sess.topic_id.or_else(|| msg.thread_id.map(|t| t.0.0 as i64));
243    if !sess.model.is_empty() && sess.model != "antigravity-default" {
244        s_cli.config.model = Some(sess.model.clone());
245    }
246    if sess.effort.is_some() {
247        s_cli.config.effort = sess.effort.clone();
248    }
249    match query_quota_from_pty(&s_cli, &sid).await {
250        Ok(r) => Some(render_quota_message(&r)),
251        Err(e) => {
252            let _ = send_reply(bot, msg, format!("⚠️ Could not retrieve quota from session: {}", e)).await;
253            None
254        }
255    }
