28async fn resolve_auto_handover(
29    sess: &mut SessionData,
30    config: &CliConfig,
31    sessions: &SessionManager,
32) -> Option<String> {
33    let past_ids = sess.get_past_session_ids(&config.provider);
34    if past_ids.is_empty() && sess.last_handover_summary.is_none() {
35        return None;
36    }
37    let last_sid = past_ids.last().map(String::as_str).unwrap_or("");
38    let handover = super::handover::build_topic_handover(
39        &config.working_dir,
40        last_sid,
41        sess.topic_id,
42        sess.topic_name.clone(),
43        sess.last_handover_summary.as_deref(),
44        &past_ids,
45    )?;
46    sess.last_handover_summary = Some(format!(
47        "Objective: {}\nDecisions: {}\nProgress: {}",
48        handover.topic_objective,
49        handover.agreed_decisions.join("; "),
50        handover.current_progress
51    ));
52    let _ = sessions.preserve_session_identity(sess).await;
53    Some(handover.handover_prompt)
54}

56pub(crate) fn has_alive_session(
57    sess: &SessionData,
58    cli: &AntigravityCli,
59    provider: &str,
60) -> bool {
61    let sid = sess.get_session_id(provider);
62    !sid.is_empty() && (sid.starts_with("mock-") || cli.is_session_alive(&sid))
63}

65pub(crate) fn combine_handover_and_prompt(handover: &str, user_prompt: &str) -> String {
66    let header = handover.split("\n---\n").next().unwrap_or(handover);
67    format!(
68        "{}\n---\nApply the restored topic context above and directly respond to the user's message below without a separate session greeting:\n\n{}",
69        header, user_prompt
70    )
71}

73pub(crate) async fn initialize_session_with_prompt(
74    bot: &Bot,
75    msg: &Message,
76    sessions: &SessionManager,
77    sess: &mut SessionData,
78    cli: &AntigravityCli,
79    config: &CliConfig,
80    startup_prompt: Option<(&str, &str)>,
81) -> Result<String, teloxide::RequestError> {
82    let provider = &config.provider;
83    let session_id = sess.get_session_id(provider);
84    if !session_id.is_empty() {
85        if session_id.starts_with("mock-") || cli.is_session_alive(&session_id) {
86            return Ok(session_id);
87        }
88        eprintln!("⚠️ [tuner] Session {} for topic {:?} is expired. Resetting...", session_id, sess.topic_name);
89        sess.set_session_id(provider, "");
90        let _ = sessions.preserve_session_identity(sess).await;
91    }
92
93    let auto_handover = resolve_auto_handover(sess, config, sessions).await;
94    let user_log_text = startup_prompt.map(|(u, _)| u);
95    let raw_prompt = startup_prompt.map(|(_, p)| p);
96    let combined;
97    let effective_prompt = match (raw_prompt, auto_handover.as_deref()) {
98        (Some(user_p), Some(ho)) => {
99            combined = combine_handover_and_prompt(ho, user_p);
100            Some(combined.as_str())
101        }
102        (Some(user_p), None) => Some(user_p),
103        (None, ho) => ho,
104    };
105
106    if cfg!(test) {
107        let mock_sid = "mock-session-123".to_string();
108        sess.set_session_id(provider, &mock_sid);
109        let _ = sessions.preserve_session_identity(sess).await;
110        return Ok(mock_sid);
111    }
112
113    let tok = std::env::var("TELEGRAM_TOKEN").unwrap_or_else(|_| config.telegram_token.clone());
114    let _g = super::typing::TelegramTypingGuard::new(bot.clone(), tok, msg).await;
115    boot_fresh_session(bot, msg, sessions, sess, cli, (user_log_text, effective_prompt)).await
116}

118async fn handle_boot_success(
119    bot: &Bot,
120    msg: &Message,
121    sessions: &SessionManager,
122    sess: &mut SessionData,
123    user_log_text: Option<&str>,
124    mut res: crate::cli::CliResponse,
125    config: &CliConfig,
126) -> Result<String, teloxide::RequestError> {
127    let thread_id = msg.thread_id.map(|t| t.0.0);
128    if res.is_error {
129        let topic = (sess.topic_id, sess.topic_name.as_deref());
130        let _ = super::stream::handle_stream_result(bot, msg.chat.id, None, thread_id, topic, res, config).await;
131        return Ok(String::new());
132    }
133    let Some(new_sid) = res.session_id.clone() else {
134        eprintln!("Initialization oneshot did not return a session_id");
135        return Ok(String::new());
136    };
137    sess.set_session_id(&config.provider, &new_sid);
138    let _ = sessions.preserve_session_identity(sess).await;
139    if res.result.trim().is_empty() {
140        res.result = crate::t!("bot.new_session");
141    }
142    let topic = (sess.topic_id, sess.topic_name.as_deref());
143    if let Some(user_txt) = user_log_text.filter(|s| !s.is_empty()) {
144        super::history::log_telegram_message(&config.working_dir, &new_sid, topic, "user", Some(msg.id.0), user_txt, (true, None));
145    }
146    let _ = super::stream::handle_stream_result(bot, msg.chat.id, None, thread_id, topic, res, config).await;
147    Ok(new_sid)
148}

150async fn handle_boot_response(
151    bot: &Bot,
152    msg: &Message,
153    sessions: &SessionManager,
154    sess: &mut SessionData,
155    user_log_text: Option<&str>,
156    send_res: Result<crate::cli::CliResponse, String>,
157    config: &CliConfig,
158) -> Result<String, teloxide::RequestError> {
159    match send_res {
160        Ok(res) => handle_boot_success(bot, msg, sessions, sess, user_log_text, res, config).await,
161        Err(e) => {
162            eprintln!("Initialization oneshot failed: {:?}", e);
163            let err_msg = crate::cli::antigravity::error_parser::parse_cli_error(&e, 1, config.model.as_deref());
164            let html = super::formatting::markdown_to_telegram_html(&err_msg);
165            let chunks = super::formatting::split_html_message(&html, 4000);
166            let _ = super::stream::send_chunks_to_telegram(bot, msg.chat.id, None, msg.thread_id.map(|t| t.0.0), &chunks).await;
167            Ok(String::new())
168        }
169    }
170}

172async fn boot_fresh_session(
173    bot: &Bot,
174    msg: &Message,
175    sessions: &SessionManager,
176    sess: &mut SessionData,
177    cli: &AntigravityCli,
178    prompts: (Option<&str>, Option<&str>),
179) -> Result<String, teloxide::RequestError> {
180    let (user_log_text, startup_prompt) = prompts;
181    let default_prompt = crate::t!("bot.session_init_prompt");
182    let initial_prompt = startup_prompt.unwrap_or(&default_prompt);
183    let ws = cli.agy_workspace();
184    let mut session_cli = cli.clone();
185    let (chat_id, topic_id) = (
186        msg.chat.id.0,
187        sess.topic_id.or_else(|| msg.thread_id.map(|t| t.0.0 as i64)),
188    );
189    session_cli.config.chat_id = chat_id;
190    session_cli.config.topic_id = topic_id;
191    if !sess.model.is_empty() && sess.model != "antigravity-default" {
192        session_cli.config.model = Some(sess.model.clone());
193    }
194    if sess.effort.is_some() {
195        session_cli.config.effort = sess.effort.clone();
196    }
197
198    let mut cancel_rx = cli.sessions.register_boot(chat_id, topic_id).await;
199    let send_fut = session_cli.send(initial_prompt, None, false, ws);
200    tokio::pin!(send_fut);
201
202    let send_res = tokio::select! {
203        res = &mut send_fut => {
204            cli.sessions.unregister_boot(chat_id, topic_id).await;
205            res
206        }
207        Ok(()) = &mut cancel_rx => {
208            cli.sessions.unregister_boot(chat_id, topic_id).await;
209            eprintln!("⚠️ [tuner] Session boot cancelled by user (/stop) in chat {}, topic {:?}", chat_id, topic_id);
210            return Ok(String::new());
211        }
212    };
213
214    handle_boot_response(bot, msg, sessions, sess, user_log_text, send_res, &session_cli.config).await
215}
