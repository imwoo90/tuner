26pub(crate) fn build_sessions(
27    path: std::path::PathBuf,
28    cache: Arc<TopicNameCache>,
29    config: &CliConfig,
30) -> SessionManager {
31    let tz = config.user_timezone.clone().unwrap_or_else(|| "UTC".to_string());
32    SessionManager::new(
33        path,
34        config.idle_timeout_minutes,
35        config.daily_reset_hour,
36        config.daily_reset_enabled,
37        tz,
38        config.max_session_messages,
39    )
40    .with_topic_resolver(Arc::new(move |c, t| cache.find_by_id(c, t)))
41}

52async fn init_core_runtime(config: CliConfig, paths: &TunerPaths) -> CoreRuntime {
53    let cfg = Arc::new(config);
54    spawn_restart_watcher(paths.tuner_home.clone());
55    let topic_cache = Arc::new(TopicNameCache::new());
56    let sessions = Arc::new(build_sessions(paths.sessions_path(), topic_cache.clone(), &cfg));
57    reply::load_sessions_cache(&sessions, &topic_cache).await;
58    let cli = Arc::new(AntigravityCli::new((*cfg).clone()));
59    let bus = Arc::new(MessageBus::with_lock_pool(sessions.lock_pool.clone()));
60    let cron = Arc::new(CronManager::new(paths.cron_jobs_path()));
61    let vault_dir = paths.vault_dir();
62    let profile = paths.profile.clone().unwrap_or_else(|| "default".to_string());
63    let web_ctx = Arc::new(WebAppCtx {
64        profile,
65        tuner_home: paths.tuner_home.clone(),
66        vault_dir: vault_dir.clone(),
67        working_dir: paths.workspace().clone(),
68        config: cfg.clone(),
69        sessions: sessions.clone(),
70        cli: cli.clone(),
71        cron_manager: cron.clone(),
72        topic_cache: topic_cache.clone(),
73        bus: bus.clone(),
74        bot: RwLock::new(None),
75        events: Arc::new(WebEventHub::new()),
76        auth: Arc::new(WebAuthStore::new(vault_dir)),
77    });
78    init_global_web_ctx(web_ctx.clone());
79    bus.register_transport(Arc::new(WebTransport::new(web_ctx.clone())));
80    CoreRuntime { cfg, sessions, cli, bus, cron, web_ctx }
81}

83fn start_background_schedulers(rt: &CoreRuntime, paths: &TunerPaths) {
84    Arc::new(crate::heartbeat::scheduler::HeartbeatScheduler::new(
85        rt.cfg.clone(),
86        rt.sessions.clone(),
87        rt.cli.clone(),
88        rt.bus.clone(),
89    ))
90    .start();
91    crate::upgrade::spawn_auto_upgrade_scheduler(rt.cfg.clone(), rt.cli.clone());
92    Arc::new(crate::cron::scheduler::CronScheduler::new(
93        rt.cfg.clone(),
94        rt.cron.clone(),
95        rt.bus.clone(),
96    ))
97    .start();
98    let clean = Arc::new(crate::cleanup::observer::CleanupObserver::new(
99        rt.cfg.cleanup.clone(),
100        paths.telegram_files_dir(),
101        paths.output_to_user_dir(),
102    ));
103    tokio::spawn(async move { clean.start().await; });
104    let skill_paths = paths.clone();
105    tokio::spawn(async move {
106        let _ = crate::workspace::skills::watch_skill_sync(&skill_paths, 30_000).await;
107    });
108    tokio::spawn(async {
109        crate::cli::antigravity::model_spec::ensure_live_catalog_fresh().await;
110    });
111    spawn_web_warmup(paths);
112}

114async fn try_init_telegram_bot(tok: String, profile: Option<String>) -> Option<(Bot, Arc<BotInfo>)> {
115    let prof_name = profile.as_deref().unwrap_or("default");
116    if tok.trim().is_empty() || tok == "YOUR_BOT_TOKEN_HERE" || tok.starts_with("YOUR_") {
117        eprintln!("ℹ️ [tuner] Profile '{prof_name}' has no Telegram token; starting Web-Standalone mode.");
118        return None;
119    }
120    let bot = Bot::new(tok);
121    match tokio::time::timeout(tokio::time::Duration::from_secs(15), bot.get_me()).await {
122        Ok(Ok(me)) => Some((bot, Arc::new(BotInfo { username: me.user.username }))),
123        Ok(Err(e)) => {
124            eprintln!("⚠️ [tuner] Profile '{prof_name}' failed Telegram token validation ({e}); starting Web-Standalone mode.");
125            None
126        }
127        Err(_) => {
128            eprintln!("⚠️ [tuner] Profile '{prof_name}' Telegram API probe timed out; starting Web-Standalone mode.");
129            None
130        }
131    }
132}

134fn spawn_restart_watcher(tuner_home: std::path::PathBuf) {
135    tokio::spawn(async move {
136        let marker = tuner_home.join("restart-requested");
137        let mut interval = tokio::time::interval(tokio::time::Duration::from_secs(2));
138        loop {
139            interval.tick().await;
140            if marker.exists() {
141                let _ = std::fs::remove_file(&marker);
142                println!("🤖 [tuner] Restart requested via marker. Exiting...");
143                std::process::exit(42);
144            }
145        }
146    });
147}

149fn spawn_web_warmup(paths: &TunerPaths) {
150    let review_path = paths.review_sessions_path();
151    let vault_prof = paths.profile.clone().unwrap_or_else(|| "default".to_string());
152    let vault_dir = paths.vault_dir();
153    tokio::spawn(async move {
154        super::vault::global_vault_manager()
155            .init_storage(vault_prof, vault_dir.clone())
156            .await;
157        let mgr = super::review::global_review_manager();
158        mgr.init_storage(review_path).await;
159        let port = mgr.ensure_server_running().await;
160        crate::security::vault_auth::proxy::publish_tunnel_endpoint(&vault_dir, port).await;
161        mgr.warmup().await;
162    });
163}

165/// Spawns the live Telegram long-polling dispatcher and attaches `bot` to `web_ctx`.
166pub fn spawn_telegram_dispatcher(
167    web_ctx: Arc<WebAppCtx>,
168    cfg: Arc<CliConfig>,
169    bot: Bot,
170    bot_info: Arc<BotInfo>,
171) -> tokio::task::JoinHandle<()> {
172    tokio::spawn(async move {
173        web_ctx.set_bot(bot.clone()).await;
174        let targets = if !cfg.allowed_group_ids.is_empty() {
175            cfg.allowed_group_ids.clone()
176        } else {
177            cfg.allowed_user_ids.clone()
178        };
179        web_ctx.bus.register_transport(Arc::new(
180            super::transport::TelegramTransport::with_targets(bot.clone(), targets)
181                .with_context(cfg.clone(), web_ctx.sessions.clone()),
182        ));
183        let _ = commands::register_commands(&bot).await;
184        let (b, s) = (bot.clone(), web_ctx.sessions.clone());
185        tokio::spawn(async move {
186            reply::send_startup_notification(b, s).await;
187        });
188
189        let media_group_manager = Arc::new(super::media_group::MediaGroupManager::new());
190        let handler = dptree::entry()
191            .branch(Update::filter_message().endpoint(handle_message))
192            .branch(Update::filter_callback_query().endpoint(callbacks::handle_callback_query));
193
194        let aux_deps = (web_ctx.cron_manager.clone(), web_ctx.topic_cache.clone());
195        let meta_deps = (bot_info.clone(), media_group_manager.clone());
196        Dispatcher::builder(bot, handler)
197            .dependencies(dptree::deps![
198                cfg,
199                web_ctx.sessions.clone(),
200                web_ctx.cli.clone(),
201                web_ctx.cron_manager.clone(),
202                web_ctx.topic_cache.clone(),
203                bot_info,
204                media_group_manager,
205                aux_deps,
206                meta_deps
207            ])
208            .build()
209            .dispatch()
210            .await;
211    })
212}

214async fn run_telegram_dispatcher(rt: CoreRuntime, bot: Bot, bot_info: Arc<BotInfo>) {
215    let _ = spawn_telegram_dispatcher(rt.web_ctx, rt.cfg, bot, bot_info).await;
216}

218/// Executes `run_bot` (starting Web/control-plane services and optional Telegram dispatcher).
219pub async fn run_bot(mut config: CliConfig, paths: TunerPaths) -> Result<(), String> {
220    let tok = std::env::var("TELEGRAM_TOKEN").unwrap_or_else(|_| config.telegram_token.clone());
221    if tok.trim().is_empty() || tok == "YOUR_BOT_TOKEN_HERE" || tok.starts_with("YOUR_") {
222        config.transport = "web".to_string();
223    }
224    let rt = init_core_runtime(config, &paths).await;
225    start_background_schedulers(&rt, &paths);
226    if let Some((bot, bot_info)) = try_init_telegram_bot(tok, paths.profile.clone()).await {
227        run_telegram_dispatcher(rt, bot, bot_info).await;
228    } else {
229        let port = std::env::var("TUNER_WEB_PORT")
230            .ok()
231            .and_then(|s| s.trim().parse::<u16>().ok())
232            .filter(|&p| p > 0)
233            .unwrap_or(8743);
234        if !rt.web_ctx.auth.has_pin().await
235            && let Ok(magic) = rt.web_ctx.auth.issue_magic_token().await
236        {
237            println!("🔑 [tuner] First-Run Setup Magic Link (10m TTL): http://127.0.0.1:{port}/dashboard/magic/{magic}");
238        }
239        println!("🖥️ [tuner] Web-Standalone Dashboard active on http://127.0.0.1:{port}/dashboard (run 'tuner --dashboard' for Magic Link).");
240        std::future::pending::<()>().await;
241    }
242    Ok(())
