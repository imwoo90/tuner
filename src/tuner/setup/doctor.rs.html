1//! # System Dependency Doctor Diagnostics
2//!
3//! ## Overview
4//! Inspects all runtime dependencies required by Tuner across three tiers:
5//! 1. Core runtime & companion binaries (`agy`, `agy-vault`, `cloudflared`, `_home_defaults`).
6//! 2. OS & media utilities (`curl`, `git`, `screen`, `pgrep`, `ffmpeg`, `node`).
7//! 3. Python runtime, packages (`requests`, `bs4`, `httpx`, `pypdf`, `PIL`, `pptx`, `openpyxl`, `playwright`), and headless Chromium CDP readiness.
8//!
9//! ## Search Tags
10//! #system-doctor, #dependency-checks, #cloudflared-check, #cdp-browser-check

16/// Status of an individual dependency diagnostic check.
17#[derive(Clone, Copy, Debug, PartialEq, Eq)]
18pub enum CheckStatus {
19    /// Dependency is present and healthy.
20    Ok,
21    /// Optional dependency or non-fatal configuration warning.
22    Warn,
23    /// Required dependency is missing or broken.
24    Missing,
25}
26
27/// Result of a single dependency diagnostic item.
28#[derive(Clone, Debug)]
29pub struct DoctorCheck {
30    /// Tier category label.
31    pub tier: &'static str,
32    /// Dependency identifier.
33    pub name: &'static str,
34    /// Diagnostic status.
35    pub status: CheckStatus,
36    /// Version or path detail string.
37    pub detail: String,
38    /// Actionable remediation hint when not `Ok`.
39    pub hint: String,
40}
41
42/// Aggregated report returned by [`collect_doctor_checks`].
43#[derive(Clone, Debug)]
44pub struct DoctorReport {
45    /// Ordered list of diagnostic check items.
46    pub checks: Vec<DoctorCheck>,
47}
48
49impl DoctorReport {
50    /// Returns the number of checks with [`CheckStatus::Missing`].
51    pub fn missing_count(&self) -> usize {
52        self.checks
53            .iter()
54            .filter(|c| c.status == CheckStatus::Missing)
55            .count()
56    }
57
58    /// Returns the number of checks with [`CheckStatus::Warn`].
59    pub fn warn_count(&self) -> usize {
60        self.checks
61            .iter()
62            .filter(|c| c.status == CheckStatus::Warn)
63            .count()
64    }
65}

67/// Locates `cloudflared` in user/tuner/system binary directories.
68pub fn find_cloudflared_bin(home: &str) -> Option<PathBuf> {
69    let mut candidates = vec![
70        PathBuf::from(home).join(".tuner/bin/cloudflared"),
71        PathBuf::from(home).join(".local/bin/cloudflared"),
72    ];
73    if let Ok(exe) = std::env::current_exe()
74        && let Some(parent) = exe.parent()
75    {
76        candidates.push(parent.join("cloudflared"));
77    }
78    candidates.push(PathBuf::from("/usr/local/bin/cloudflared"));
79    candidates.push(PathBuf::from("/usr/bin/cloudflared"));
80    candidates.into_iter().find(|p| p.is_file())
81}

83/// Locates the `agy` CLI binary in standard Antigravity or PATH locations.
84pub fn find_agy_bin(home: &str) -> Option<PathBuf> {
85    let candidates = [
86        PathBuf::from(home).join(".gemini/antigravity-cli/bin/agy"),
87        PathBuf::from(home).join(".local/bin/agy"),
88        PathBuf::from("/usr/local/bin/agy"),
89        PathBuf::from("/usr/bin/agy"),
90    ];
91    if let Some(found) = candidates.into_iter().find(|p| p.is_file()) {
92        return Some(found);
93    }
94    which_bin("agy")
95}

97/// Resolves a binary name using `which` or standard user paths.
98pub fn which_bin(name: &str) -> Option<PathBuf> {
99    if let Ok(out) = Command::new("which").arg(name).output()
100        && out.status.success()
101    {
102        let s = String::from_utf8_lossy(&out.stdout).trim().to_string();
103        if !s.is_empty() {
104            return Some(PathBuf::from(s));
105        }
106    }
107    let home = std::env::var("HOME").unwrap_or_else(|_| ".".to_string());
108    for sub in [".tuner/bin", ".local/bin", ".gemini/antigravity-cli/bin"] {
109        let p = PathBuf::from(&home).join(sub).join(name);
110        if p.is_file() {
111            return Some(p);
112        }
113    }
114    None
115}

117fn check_tier1_agy(home: &str, checks: &mut Vec<DoctorCheck>) {
118    let verified_agy = env!("TUNER_VERIFIED_AGY_VERSION");
119    match find_agy_bin(home) {
120        Some(path) => {
121            let ver = Command::new(&path)
122                .arg("--version")
123                .output()
124                .ok()
125                .map(|o| String::from_utf8_lossy(&o.stdout).trim().to_string())
126                .unwrap_or_else(|| "unknown".to_string());
127            checks.push(DoctorCheck {
128                tier: "Tier 1: Core Runtime",
129                name: "agy (Antigravity CLI)",
130                status: CheckStatus::Ok,
131                detail: format!("{} (verified: {}, path: {})", ver, verified_agy, path.display()),
132                hint: String::new(),
133            });
134        }
135        None => checks.push(DoctorCheck {
136            tier: "Tier 1: Core Runtime",
137            name: "agy (Antigravity CLI)",
138            status: CheckStatus::Warn,
139            detail: "Not found in ~/.gemini/antigravity-cli/bin/agy".to_string(),
140            hint: "Install Antigravity CLI if using the 'antigravity' provider.".to_string(),
141        }),
142    }
143
144    let oauth_tok = PathBuf::from(home).join(".gemini/antigravity-cli/antigravity-oauth-token");
145    if oauth_tok.is_file() {
146        checks.push(DoctorCheck {
147            tier: "Tier 1: Core Runtime",
148            name: "agy OAuth Token",
149            status: CheckStatus::Ok,
150            detail: format!("Signed in ({})", oauth_tok.display()),
151            hint: String::new(),
152        });
153    } else {
154        checks.push(DoctorCheck {
155            tier: "Tier 1: Core Runtime",
156            name: "agy OAuth Token",
157            status: CheckStatus::Warn,
158            detail: "Not signed in".to_string(),
159            hint: "Run `agy` interactively to complete Google OAuth sign-in.".to_string(),
160        });
161    }
162}

164fn push_tier1(
165    checks: &mut Vec<DoctorCheck>,
166    name: &'static str,
167    status: CheckStatus,
168    detail: String,

171    checks.push(DoctorCheck {

180fn check_tier1_companions(home: &str, checks: &mut Vec<DoctorCheck>) {
181    match super::find_agy_vault_bin(home) {
182        Some(path) => push_tier1(
183            checks,
184            "agy-vault (Session Daemon)",
185            CheckStatus::Ok,
186            path.display().to_string(),
187            "",
188        ),
189        None => push_tier1(
190            checks,
191            "agy-vault (Session Daemon)",
192            CheckStatus::Missing,
193            "Missing companion binary".to_string(),
194            "Run `tuner --doctor --fix` to auto-download agy-vault.",
195        ),
196    }
197
198    match find_cloudflared_bin(home) {
199        Some(path) => push_tier1(
200            checks,
201            "cloudflared (Web Tunnel)",
202            CheckStatus::Ok,
203            path.display().to_string(),
204            "",
205        ),
206        None => push_tier1(
207            checks,
208            "cloudflared (Web Tunnel)",
209            CheckStatus::Missing,
210            "Missing (required for /vault & /review external URLs)".to_string(),
211            "Run `tuner --doctor --fix` to auto-download cloudflared into ~/.tuner/bin.",
212        ),
213    }
214
215    let defaults_ok = PathBuf::from(home).join(".tuner/bin/_home_defaults").is_dir()
216        || Path::new("_home_defaults").is_dir();
217    push_tier1(
218        checks,
219        "_home_defaults (Templates)",
220        if defaults_ok { CheckStatus::Ok } else { CheckStatus::Warn },
221        if defaults_ok {
222            "Present".to_string()
223        } else {
224            "Not in ~/.tuner/bin/_home_defaults".to_string()
225        },
226        if defaults_ok {
227            ""
228        } else {
229            "Re-run install.sh or copy _home_defaults to ~/.tuner/bin."
230        },
231    );
232}

234/// Runs all dependency diagnostic checks and returns a [`DoctorReport`].
235pub fn collect_doctor_checks() -> DoctorReport {
236    let home = std::env::var("HOME").unwrap_or_else(|_| ".".to_string());
237    let mut checks = Vec::new();
238    check_tier1_agy(&home, &mut checks);
239    check_tier1_companions(&home, &mut checks);
240    check_tier2_os_tools(&mut checks);
241    if let Some(py_path) = check_tier3_python(&mut checks) {
242        check_tier3_cdp(&py_path, &mut checks);
243    }
244    DoctorReport { checks }
245}

247/// Prints a formatted human-readable [`DoctorReport`] to stdout.
248pub fn print_doctor_report(report: &DoctorReport) {
249    println!("🩺 [tuner --doctor] System Dependency & Runtime Diagnostics");
250    println!("========================================================================");
251    let mut current_tier = "";
252    for item in &report.checks {
253        if item.tier != current_tier {
254            current_tier = item.tier;
255            println!("\n[{}]", current_tier);
256        }
257        let badge = match item.status {
258            CheckStatus::Ok => "✅ [OK]     ",
259            CheckStatus::Warn => "⚠️  [WARN]   ",
260            CheckStatus::Missing => "❌ [MISSING]",
261        };
262        println!("  {} {:<30} — {}", badge, item.name, item.detail);
263        if !item.hint.is_empty() {
264            println!("               💡 {}", item.hint);
265        }
266    }
267    println!("\n========================================================================");
268    println!(
269        "Summary: {} OK, {} warnings, {} missing.",
270        report
271            .checks
272            .iter()
273            .filter(|c| c.status == CheckStatus::Ok)
274            .count(),
275        report.warn_count(),
276        report.missing_count()
277    );
