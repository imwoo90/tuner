1//! # System Dependency Doctor Auto-Installer (`--fix`)
2//!
3//! ## Overview
4//! Implements `tuner --doctor [--fix]`, automatically downloading user-space companion
5//! binaries (`cloudflared`, `agy-vault`), installing missing OS packages (`screen`, `ffmpeg`,
6//! `procps`, `nodejs`, `python3-pip`, Chromium CDP shared libraries) when root/passwordless
7//! `sudo` is available, and installing Python packages plus Playwright Chromium (`CDP`).
8//!
9//! ## Search Tags
10//! #doctor-fix, #dependency-installer, #cloudflared-installer, #playwright-cdp
11

16/// Entrypoint for `tuner --doctor [--fix]`.
17pub fn run_doctor_cli(args: &[String]) -> Result<(), String> {
18    let fix = args.iter().any(|a| a == "--fix");
19    let initial = collect_doctor_checks();
20    print_doctor_report(&initial);
21
22    if !fix {
23        if initial.missing_count() > 0 {
24            println!(
25                "\n💡 Found {} missing dependencies. Run `tuner --doctor --fix` to install them automatically.",
26                initial.missing_count()
27            );
28        } else {
29            println!("\n🎉 All required Tuner dependencies are installed and verified!");
30        }
31        return Ok(());
32    }
33
34    println!("\n🔧 [tuner --doctor --fix] Starting automated dependency installation & repair...");
35    let home = std::env::var("HOME").unwrap_or_else(|_| ".".to_string());
36    let tuner_bin_dir = PathBuf::from(&home).join(".tuner/bin");
37    std::fs::create_dir_all(&tuner_bin_dir).map_err(|e| e.to_string())?;
38
39    if find_cloudflared_bin(&home).is_none() {
40        install_cloudflared(&tuner_bin_dir)?;
41    }
42    if super::find_agy_vault_bin(&home).is_none() {
43        install_agy_vault(&home, &tuner_bin_dir);
44    }
45
46    install_os_packages_if_needed();
47    install_python_and_cdp_if_needed(&home);
48
49    println!("\n🔄 Re-running diagnostics after `--fix`...\n");
50    let final_report = collect_doctor_checks();
51    print_doctor_report(&final_report);
52
53    if final_report.missing_count() == 0 {
54        println!("\n🎉 All required dependencies are now installed and verified!");
55        Ok(())
56    } else {
57        Err(format!(
58            "{} dependencies still require attention (see hints above).",
59            final_report.missing_count()
60        ))
61    }
62}

64fn install_cloudflared(tuner_bin_dir: &Path) -> Result<(), String> {
65    let arch = if std::env::consts::ARCH == "aarch64" { "arm64" } else { "amd64" };
66    let url = format!(
67        "https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-linux-{}",
68        arch
69    );
70    let dest = tuner_bin_dir.join("cloudflared");
71    let tmp = tuner_bin_dir.join(".cloudflared.tmp");
72    println!("📥 [doctor] Downloading cloudflared from {}...", url);
73    let status = Command::new("curl")
74        .args(["-fsSL", "-o"])
75        .arg(&tmp)
76        .arg(&url)
77        .status()
78        .map_err(|e| format!("Failed to invoke curl for cloudflared: {}", e))?;
79    if !status.success() {
80        let _ = std::fs::remove_file(&tmp);
81        return Err("Failed to download cloudflared binary".to_string());
82    }
83    set_executable(&tmp);
84    std::fs::rename(&tmp, &dest).map_err(|e| format!("Failed to move cloudflared: {}", e))?;
85    println!("✅ [doctor] Installed cloudflared to {}", dest.display());
86    Ok(())
87}

89fn install_agy_vault(home: &str, tuner_bin_dir: &Path) {
90    println!("📥 [doctor] Installing companion daemon agy-vault...");
91    let tb = tuner_bin_dir.display();
92    let script = format!(
93        "set -e; TMP=$(mktemp -d); \
94         REL=$(curl -fsSL https://api.github.com/repos/imwoo90/agy-vault/releases/latest); \
95         TAG=$(echo \"$REL\" | grep -m 1 '\"tag_name\":' | awk -F '\"' '{{print $4}}'); \
96         if [ -n \"$TAG\" ]; then \
97           curl -fsSL -o \"$TMP/v.tgz\" \"https://github.com/imwoo90/agy-vault/releases/download/$TAG/agy-vault-linux-amd64.tar.gz\"; \
98           tar -xzf \"$TMP/v.tgz\" -C \"$TMP\"; \
99           cp \"$TMP/agy-vault-linux-amd64/agy-vault\" \"{tb}/agy-vault\"; chmod +x \"{tb}/agy-vault\"; \
100           mkdir -p \"{home}/.local/bin\" 2>/dev/null && [ -w \"{home}/.local/bin\" ] && ln -sf \"{tb}/agy-vault\" \"{home}/.local/bin/agy-vault\" || true; \
101           \"{tb}/agy-vault\" patch-binary 2>/dev/null || true; \
102         fi; rm -rf \"$TMP\""
103    );
104    if Command::new("bash").arg("-c").arg(&script).status().map(|s| s.success()).unwrap_or(false) {
105        println!("✅ [doctor] Installed agy-vault companion daemon.");
106    } else {
107        eprintln!("⚠️  [doctor] Could not auto-install agy-vault (check network).");
108    }
109}

111fn can_run_privileged_apt() -> Option<&'static str> {
112    if Command::new("id").arg("-u").output().map(|o| String::from_utf8_lossy(&o.stdout).trim() == "0").unwrap_or(false) {
113        return Some("");
114    }
115    if Command::new("sudo").args(["-n", "true"]).output().map(|o| o.status.success()).unwrap_or(false) {
116        return Some("sudo -n");
117    }
118    None
119}

121fn install_os_packages_if_needed() {
122    let missing_bin = ["curl", "git", "screen", "pgrep", "ffmpeg", "node"]
123        .iter()
124        .any(|b| which_bin(b).is_none());
125    let pip_ok = Command::new("python3").args(["-m", "pip", "--version"]).output().map(|o| o.status.success()).unwrap_or(false);
126    if !missing_bin && pip_ok {
127        return;
128    }
129    let pkgs = "curl git screen procps ffmpeg nodejs python3-pip python3-requests python3-bs4 python3-pil python3-lxml libnss3 libnspr4 libatk1.0-0 libatk-bridge2.0-0 libcups2 libdrm2 libxkbcommon0 libxcomposite1 libxdamage1 libxfixes3 libxrandr2 libgbm1 libpango-1.0-0 libcairo2";
130    if let Some(p) = can_run_privileged_apt() {
131        println!("📦 [doctor] Installing missing OS & media packages via apt-get...");
132        let env_p = if p.is_empty() { "env DEBIAN_FRONTEND=noninteractive" } else { "sudo -n env DEBIAN_FRONTEND=noninteractive" };
133        let cmd = format!(
134            "{p} apt-get update -qq && {env_p} apt-get install -y -qq {pkgs} && ({env_p} apt-get install -y -qq libasound2t64 || {env_p} apt-get install -y -qq libasound2 || true)"
135        );
136        let _ = Command::new("bash").arg("-c").arg(&cmd).status();
137    } else {
138        println!("⚠️  [doctor] Root privileges required for OS packages. Run:\n    sudo apt-get update && sudo apt-get install -y {pkgs} libasound2t64");
139    }
140}

142fn is_dir_writable(dir: &Path) -> bool {
143    let _ = std::fs::create_dir_all(dir);
144    let probe = dir.join(".tuner_write_probe");
145    if std::fs::write(&probe, b"ok").is_ok() {
146        let _ = std::fs::remove_file(&probe);
147        true
148    } else {
149        false
150    }
151}

153fn ensure_user_pip(py_path: &Path, home: &str) -> bool {
154    if Command::new(py_path).args(["-m", "pip", "--version"]).output().map(|o| o.status.success()).unwrap_or(false) {
155        return true;
156    }
157    println!("📥 [doctor] Bootstrapping user-space pip via get-pip.py...");
158    let py = py_path.display();
159    let script = if is_dir_writable(&PathBuf::from(home).join(".local/bin")) {
160        format!("curl -fsSL https://bootstrap.pypa.io/get-pip.py | {py} - --user --break-system-packages -q")
161    } else {
162        format!("SITE=$({py} -c 'import site; print(site.getusersitepackages())') && mkdir -p \"$SITE\" && curl -fsSL https://bootstrap.pypa.io/get-pip.py | {py} - --target \"$SITE\" -q")
163    };
164    let _ = Command::new("bash").arg("-c").arg(&script).status();
165    Command::new(py_path).args(["-m", "pip", "--version"]).output().map(|o| o.status.success()).unwrap_or(false)
166}

168fn install_python_packages(py_path: &Path, home: &str) {
169    println!("🐍 [doctor] Ensuring Python core & Vault automation packages...");
170    let py_pkgs = ["requests", "beautifulsoup4", "httpx", "pypdf", "pillow", "python-pptx", "openpyxl", "playwright"];
171    if is_dir_writable(&PathBuf::from(home).join(".local/bin")) {
172        let mut args = vec!["-m", "pip", "install", "--user", "--break-system-packages", "-q"];
173        args.extend_from_slice(&py_pkgs);
174        if Command::new(py_path).args(&args).status().map(|s| !s.success()).unwrap_or(true) {
175            let mut fb = vec!["-m", "pip", "install", "--user", "-q"];
176            fb.extend_from_slice(&py_pkgs);
177            let _ = Command::new(py_path).args(&fb).status();
178        }
179    } else if let Ok(out) = Command::new(py_path).args(["-c", "import site; print(site.getusersitepackages())"]).output() {
180        let site_dir = String::from_utf8_lossy(&out.stdout).trim().to_string();
181        if !site_dir.is_empty() {
182            let _ = std::fs::create_dir_all(&site_dir);
183            let mut args = vec!["-m", "pip", "install", "--upgrade", "--target", &site_dir, "-q"];
184            args.extend_from_slice(&py_pkgs);
185            let _ = Command::new(py_path).args(&args).status();
186        }
187    }
188}

190fn install_python_and_cdp_if_needed(home: &str) {
191    let Some(py_path) = which_bin("python3") else { return };
192    if !ensure_user_pip(&py_path, home) {
193        return;
194    }
195    install_python_packages(&py_path, home);
196    println!("🌐 [doctor] Ensuring Playwright Chromium headless browser (CDP)...");
197    if let Some(p) = can_run_privileged_apt() {
198        let cmd = format!("{p} {} -m playwright install-deps chromium", py_path.display());
199        let _ = Command::new("bash").arg("-c").arg(cmd.trim()).status();
200    }
201    let _ = Command::new(&py_path).args(["-m", "playwright", "install", "chromium"]).status();
202    fix_home_ownership_if_root(home);
203}

205fn fix_home_ownership_if_root(home: &str) {
206    #[cfg(unix)]
207    {
208        use std::os::unix::fs::MetadataExt;
209        if Command::new("id").arg("-u").output().map(|o| String::from_utf8_lossy(&o.stdout).trim() == "0").unwrap_or(false)
210            && let Ok(meta) = std::fs::metadata(home)
211            && meta.uid() != 0
212        {
213            let cmd = format!("chown -R {}:{} \"{home}/.tuner/bin\" \"{home}/.local/lib\" \"{home}/.cache\" 2>/dev/null || true", meta.uid(), meta.gid());
214            let _ = Command::new("bash").arg("-c").arg(&cmd).status();
215        }
216    }
217}

219fn set_executable(path: &Path) {
220    #[cfg(unix)]
221    {
222        use std::os::unix::fs::PermissionsExt;
223        if let Ok(meta) = std::fs::metadata(path) {
224            let mut perms = meta.permissions();
225            perms.set_mode(0o755);
226            let _ = std::fs::set_permissions(path, perms);
227        }
228    }
