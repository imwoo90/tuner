1//! # Antigravity Transcript Step Parser
2//!
3//! ## Overview
4//! Parses individual JSONL transcript lines from Antigravity `transcript.jsonl` /
5//! `transcript_full.jsonl` into compact `step_index`-keyed records (`CompactStep`),
6//! normalizing embedded-quote scalar arguments, compound timer durations, and quiet heartbeats.
7//!
8//! ## Search Tags
9//! #transcript-parser, #step-index, #schedule-timer, #tool-args

11const HEARTBEAT_PROMPT_PREFIX: &str =
12    "System self-check: are there any outstanding issues or background alerts to report?";
13
14fn clean_json_scalar(val: Option<&serde_json::Value>) -> Option<String> {
15    let v = val?;
16    let raw = match v {
17        serde_json::Value::String(s) => s.clone(),
18        serde_json::Value::Number(n) => n.to_string(),
19        serde_json::Value::Bool(b) => b.to_string(),
20        _ => return None,
21    };
22    let mut s = raw.trim();
23    while s.len() >= 2 && s.starts_with('"') && s.ends_with('"') {
24        s = s[1..s.len() - 1].trim();
25    }
26    (!s.is_empty()).then(|| s.to_string())
27}
28
29fn clean_arg_bool(args: &serde_json::Value, key: &str) -> bool {
30    clean_json_scalar(args.get(key)).is_some_and(|s| s.eq_ignore_ascii_case("true"))
31}

33/// Sums all `<num><unit>` segments in `Task Description: Timer: 1h 30m 10s, Prompt: ...`.
34pub fn parse_timer_desc_secs(content: &str) -> Option<u64> {
35    let after = content.split("Task Description:").nth(1)?.trim_start();
36    let timer = after.strip_prefix("Timer:")?;
37    let header = timer
38        .lines()
39        .next()
40        .unwrap_or(timer)
41        .split(',')
42        .next()
43        .unwrap_or("");
44    let mut rest = header.trim();
45    let mut total = 0_u64;
46    let mut found = false;
47    while !rest.is_empty() {
48        rest = rest.trim_start();
49        let dlen = rest.chars().take_while(|c| c.is_ascii_digit()).count();
50        if dlen == 0 {
51            break;
52        }
53        let Ok(num) = rest[..dlen].parse::<u64>() else {
54            break;
55        };
56        rest = rest[dlen..].trim_start();
57        let (mult, ulen) = match rest.chars().next() {
58            Some('s') => (1_u64, 1),
59            Some('m') => (60, 1),
60            Some('h') => (3600, 1),
61            Some('d') => (86400, 1),
62            _ => (1, 0),
63        };
64        rest = &rest[ulen..];
65        total = total.saturating_add(num.saturating_mul(mult));
66        found = true;
67    }
68    found.then_some(total)
69}

72pub(crate) enum PendingCall {
73    ManageTask {
74        kill_tid: Option<String>,
75    },
76    Schedule {
77        dur: Option<u64>,
78        cond: String,
79        is_daemon: bool,
80    },
81    RunCommand {
82        is_daemon: bool,
83    },
84    Other,
85}

87pub(crate) struct CompactStep {
88    pub(crate) stype: String,
89    pub(crate) status: String,
90    pub(crate) step_ts: u64,
91    pub(crate) has_tools: bool,
92    pub(crate) is_hb_prompt: bool,
93    pub(crate) is_hb_ack: bool,
94    pub(crate) calls: Vec<PendingCall>,
95    pub(crate) content: Option<String>,
96}

98fn is_relevant_content(stype: &str, status: &str, c: &str) -> bool {
99    status == "RUNNING"
100        || stype == "SYSTEM_MESSAGE"
101        || (stype == "GENERIC"
102            && status == "DONE"
103            && (c.contains("No background tasks are currently running.")
104                || c.contains("background task(s) currently running:")
105                || c.contains(" cancelled.")
106                || (c.contains("Task:") && c.contains("Status:"))))
107}

109fn extract_calls(val: &serde_json::Value) -> Vec<PendingCall> {
110    let Some(arr) = val.get("tool_calls").and_then(|v| v.as_array()) else {
111        return Vec::new();
112    };
113    let mut out = Vec::with_capacity(arr.len());
114    for tc in arr {
115        let name = tc.get("name").and_then(|v| v.as_str()).unwrap_or("");
116        let args = tc.get("args").or_else(|| tc.get("arguments"));
117        let item = match name {
118            "manage_task" => {
119                let kill_tid = args.and_then(|a| {
120                    (clean_json_scalar(a.get("Action")).as_deref() == Some("kill"))
121                        .then(|| clean_json_scalar(a.get("TaskId")))
122                        .flatten()
123                });
124                PendingCall::ManageTask { kill_tid }
125            }
126            "schedule" => {
127                let (dur, cond, is_daemon) = if let Some(a) = args {
128                    (
129                        clean_json_scalar(a.get("DurationSeconds"))
130                            .and_then(|s| s.parse::<u64>().ok()),
131                        clean_json_scalar(a.get("TimerCondition"))
132                            .unwrap_or_else(|| "never".into()),
133                        clean_arg_bool(a, "IsDaemon"),
134                    )
135                } else {
136                    (None, "never".into(), false)
137                };
138                PendingCall::Schedule {
139                    dur,
140                    cond,
141                    is_daemon,
142                }
143            }
144            "run_command" => PendingCall::RunCommand {
145                is_daemon: args.is_some_and(|a| clean_arg_bool(a, "IsDaemon")),
146            },
147            _ => PendingCall::Other,
148        };
149        out.push(item);
150    }
151    out
152}

154pub(crate) fn parse_compact_step_with_ack(
155    line: &str,
156    seq: u64,
157    fallback_epoch: u64,
158    ack_token: &str,
159) -> Option<((u8, u64), CompactStep)> {
160    let val = serde_json::from_str::<serde_json::Value>(line).ok()?;
161    let key = match val.get("step_index").and_then(|v| v.as_u64()) {
162        Some(idx) => (0_u8, idx),
163        None => (1_u8, seq),
164    };
165    let stype = val.get("type").and_then(|v| v.as_str()).unwrap_or("").to_string();
166    let status = val.get("status").and_then(|v| v.as_str()).unwrap_or("").to_string();
167    let step_ts = val
168        .get("created_at")
169        .and_then(|v| v.as_str())
170        .and_then(|s| chrono::DateTime::parse_from_rfc3339(s).ok())
171        .map(|dt| dt.timestamp().max(0) as u64)
172        .unwrap_or(fallback_epoch);
173    let calls = if stype == "PLANNER_RESPONSE" { extract_calls(&val) } else { Vec::new() };
174    let has_tools = val
175        .get("tool_calls")
176        .and_then(|v| v.as_array())
177        .is_some_and(|a| !a.is_empty());
178    let raw_content = val.get("content").and_then(|v| v.as_str());
179    let is_hb_prompt = stype == "USER_INPUT"
180        && raw_content.is_some_and(|c| c.contains(HEARTBEAT_PROMPT_PREFIX));
181    let clean_ack = ack_token.trim();
182    let is_hb_ack = stype == "PLANNER_RESPONSE"
183        && !has_tools
184        && raw_content.is_some_and(|c| {
185            let t = c.trim();
186            t == "HEARTBEAT_OK" || (!clean_ack.is_empty() && t == clean_ack)
187        });
188    let content = raw_content
189        .filter(|c| is_relevant_content(&stype, &status, c))
190        .map(|c| c.to_string());
191    Some((
192        key,
193        CompactStep { stype, status, step_ts, has_tools, is_hb_prompt, is_hb_ack, calls, content },
194    ))
195}
